Systematic Reed-Solomon Erasure Coding over GF(2^8) (poly 0x11d) — Diagnosis & Fix

Root-cause analysis (failure modes in this problem class)

# Failure mode Fix applied
1 Wrong field / wrong primitive polynomial (e.g. 0x11b) → silent byte corruption Reduce with 0x11d, primitive element 2; exp table extended to 512 so log/exp arithmetic never wraps
2 Treating subtraction as arithmetic (must be XOR in char 2) gfAdd === gfSub === (a ^ b)
3 Non-systematic generator / top block not identity E = V · inv(V_top) ⇒ E_top = I, first k shards equal input
4 Naive Vandermonde that isn't MDS Distinct points x_i = i+1 ⇒ every k×k minor nonzero; right-multiply by invertible inv(V_top) preserves MDS
5 Zero pivot / no pivoting Gauss-Jordan with partial pivoting
6 Singular survivor sub-matrix returns garbage Detect missing pivot and throw
7 Missing input validation Reject < k shards, duplicates, out-of-range indices, unequal lengths, bad k/n
8 Encode/decode build different matrices One this.matrix built in the constructor, used by both

The fix — reed-solomon.js

'use strict';

const PRIMITIVE_POLY = 0x11d;

const EXP = new Uint8Array(512); // EXP[i] = 2^i, extended to avoid reductions
const LOG = new Uint8Array(256); // LOG[EXP[i]] = i, LOG[0] is unused

(function buildTables() {
  let x = 1;
  for (let i = 0; i < 255; i++) {
    EXP[i] = x;
    LOG[x] = i;
    x <<= 1;
    if (x & 0x100) x ^= PRIMITIVE_POLY;
  }
  for (let i = 255; i < 512; i++) EXP[i] = EXP[i - 255];
})();

function gfAdd(a, b) {
  return (a ^ b) & 0xff; // addition == subtraction in characteristic 2
}
const gfSub = gfAdd;

function gfMul(a, b) {
  if (a === 0 || b === 0) return 0;
  return EXP[LOG[a] + LOG[b]]; // max index 254 + 254 = 508 < 512
}

function gfInv(a) {
  if (a === 0) throw new Error('GF(2^8): inverse of zero');
  return EXP[255 - LOG[a]];
}

function gfDiv(a, b) {
  if (b === 0) throw new Error('GF(2^8): division by zero');
  if (a === 0) return 0;
  return EXP[LOG[a] - LOG[b] + 255];
}

function gfPow(a, e) {
  if (e === 0) return 1;
  if (a === 0) return 0;
  e = ((e % 255) + 255) % 255;
  return EXP[(LOG[a] * e) % 255];
}

// Invert a k x k matrix (array of Uint8Array rows). Throws if singular.
function invertMatrix(A, k) {
  const M = new Array(k);
  for (let i = 0; i < k; i++) {
    const row = new Uint8Array(2 * k);
    row.set(A[i], 0);
    row[k + i] = 1;
    M[i] = row;
  }

  for (let col = 0; col < k; col++) {
    let piv = -1;
    for (let r = col; r < k; r++) {
      if (M[r][col] !== 0) { piv = r; break; }
    }
    if (piv === -1) throw new Error('GF(2^8): singular matrix (not invertible)');
    if (piv !== col) { const t = M[piv]; M[piv] = M[col]; M[col] = t; }
    const inv = gfInv(M[col][col]);
    if (inv !== 1) {
      for (let c = col; c < 2 * k; c++) M[col][c] = gfMul(M[col][c], inv);
    }
    for (let r = 0; r < k; r++) {
      if (r === col) continue;
      const f = M[r][col];
      if (f === 0) continue;
      for (let c = col; c < 2 * k; c++) M[r][c] ^= gfMul(f, M[col][c]);
    }
  }

  const out = new Array(k);
  for (let i = 0; i < k; i++) out[i] = M[i].slice(k);
  return out;
}

/*
 * Solve A * X = B where A is k x k and B is k x L (each row a Uint8Array).
 * Returns X as k x L. Throws when A is singular.
 * Gauss-Jordan elimination with partial pivoting over GF(2^8).
 */
function solveLinearSystem(A, B, k, L) {
  const M = new Array(k);
  for (let i = 0; i < k; i++) {
    const row = new Uint8Array(k + L);
    row.set(A[i], 0);
    row.set(B[i], k);
    M[i] = row;
  }

  for (let col = 0; col < k; col++) {
    let piv = -1;
    for (let r = col; r < k; r++) {
      if (M[r][col] !== 0) { piv = r; break; }
    }
    if (piv === -1) throw new Error('GF(2^8): singular sub-matrix; cannot reconstruct');
    if (piv !== col) { const t = M[piv]; M[piv] = M[col]; M[col] = t; }
    const inv = gfInv(M[col][col]);
    if (inv !== 1) {
      for (let c = col; c < k + L; c++) M[col][c] = gfMul(M[col][c], inv);
    }
    for (let r = 0; r < k; r++) {
      if (r === col) continue;
      const f = M[r][col];
      if (f === 0) continue;
      for (let c = col; c < k + L; c++) M[r][c] ^= gfMul(f, M[col][c]);
    }
  }

  const X = new Array(k);
  for (let i = 0; i < k; i++) X[i] = M[i].slice(k);
  return X;
}

function assertInt(name, v) {
  if (!Number.isInteger(v)) throw new Error(`${name} must be an integer`);
}

class ReedSolomon {
  constructor(k, n) {
    assertInt('k', k);
    assertInt('n', n);
    if (k < 1) throw new Error('k must be >= 1');
    if (n < k) throw new Error('n must be >= k');
    if (n > 255) throw new Error('n must be <= 255 (GF(2^8) has 256 elements, one is zero)');
    this.k = k;
    this.n = n;
    this.matrix = buildEncodeMatrix(k, n);
  }

  encode(dataShards) {
    const { k, n, matrix } = this;
    if (!Array.isArray(dataShards) || dataShards.length !== k) {
      throw new Error(`encode expects exactly ${k} data shards`);
    }
    for (let i = 0; i < k; i++) {
      if (!(dataShards[i] instanceof Uint8Array)) {
        throw new Error(`data shard ${i} must be a Uint8Array`);
      }
    }
    const L = dataShards[0].length;
    for (let i = 1; i < k; i++) {
      if (dataShards[i].length !== L) {
        throw new Error('data shards must all have the same length');
      }
    }

    const out = new Array(n);
    for (let i = 0; i < n; i++) {
      const row = matrix[i];
      const shard = new Uint8Array(L);
      for (let j = 0; j < k; j++) {
        const coeff = row[j];
        if (coeff === 0) continue;
        const src = dataShards[j];
        if (coeff === 1) {
          for (let p = 0; p < L; p++) shard[p] ^= src[p];
        } else {
          for (let p = 0; p < L; p++) shard[p] ^= gfMul(coeff, src[p]);
        }
      }
      out[i] = shard;
    }
    return out;
  }

  reconstruct(shards) {
    const { k, n, matrix } = this;

    if (!Array.isArray(shards)) throw new Error('shards must be an array');
    if (shards.length < k) {
      throw new Error(`cannot reconstruct: need at least ${k} shards, got ${shards.length}`);
    }

    const L = shards[0].data ? shards[0].data.length : -1;
    const seen = new Set();
    const used = [];

    for (let s = 0; s < shards.length; s++) {
      const entry = shards[s];
      if (entry == null || !Number.isInteger(entry.index)) {
        throw new Error(`shard ${s} has an invalid index`);
      }
      const idx = entry.index;
      if (idx < 0 || idx >= n) {
        throw new Error(`shard index ${idx} out of range [0, ${n - 1}]`);
      }
      if (seen.has(idx)) throw new Error(`duplicate shard index ${idx}`);
      seen.add(idx);
      if (!(entry.data instanceof Uint8Array)) {
        throw new Error(`shard ${s} data must be a Uint8Array`);
      }
      if (entry.data.length !== L) {
        throw new Error('surviving shards must all have the same length');
      }
      if (used.length < k) used.push(entry);
    }

    const A = new Array(k);
    const B = new Array(k);
    for (let r = 0; r < k; r++) {
      A[r] = matrix[used[r].index];
      B[r] = used[r].data;
    }

    return solveLinearSystem(A, B, k, L);
  }
}

function buildEncodeMatrix(k, n) {
  // V is n x k: V[i][j] = x_i^j with distinct non-zero points x_i = i + 1.
  const V = new Array(n);
  for (let i = 0; i < n; i++) {
    const x = i + 1;
    const row = new Uint8Array(k);
    let p = 1;
    for (let j = 0; j < k; j++) { row[j] = p; p = gfMul(p, x); }
    V[i] = row;
  }

  const invTop = invertMatrix(V.slice(0, k), k);

  // E = V * inv(Top); top k rows become the identity (systematic).
  const E = new Array(n);
  for (let i = 0; i < n; i++) {
    const row = new Uint8Array(k);
    for (let j = 0; j < k; j++) {
      let acc = 0;
      for (let t = 0; t < k; t++) {
        const a = V[i][t], b = invTop[t][j];
        if (a !== 0 && b !== 0) acc ^= gfMul(a, b);
      }
      row[j] = acc;
    }
    E[i] = row;
  }
  return E;
}

module.exports = {
  ReedSolomon, gfAdd, gfSub, gfMul, gfDiv, gfInv, gfPow,
  buildEncodeMatrix, solveLinearSystem, invertMatrix,
  EXP, LOG, PRIMITIVE_POLY,
};

Usage

const { ReedSolomon } = require('./reed-solomon');
const rs = new ReedSolomon(4, 7);
const data = [
  Uint8Array.from([1,2,3,4]), Uint8Array.from([5,6,7,8]),
  Uint8Array.from([9,10,11,12]), Uint8Array.from([13,14,15,16]),
];
const shards = rs.encode(data);                     // length 7, shards[0..3] === data
const survivors = [0,3,4,6].map(i => ({ index: i, data: shards[i] }));
const rebuilt = rs.reconstruct(survivors);          // byte-identical to data
