#!/usr/bin/env bash
set -euo pipefail
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
cd "$ROOT"

# 0) Preconditions: ports free, else refuse (don't fight a zombie from a prior run)
if ss -ltn | grep -qE ':(8091|5173)\b'; then
  echo "FATAL: ports 8091/5173 already in use — kill stale processes first" >&2
  exit 1
fi

# 1) REBUILD CANOPYD *BEFORE* START — stale binary hard-fails on new migrations
go build -o canopyd ./cmd/canopyd

# 2) Start backend on :8091 (HTTP_ADDR + DB env contract)
HTTP_ADDR=:8091 \
DB_HOST=${DB_HOST:-127.0.0.1} DB_PORT=${DB_PORT:-5432} \
DB_USER=${DB_USER:-canopy} DB_PASSWORD=${DB_PASSWORD:-canopy} \
./canopyd > /tmp/canopyd.log 2>&1 &
CANOPYD_PID=$!

# 3) Start vite :5173, proxy /api -> 8091 (vite.config.ts: server.proxy['/api'] = { target: 'http://127.0.0.1:8091' })
npm run dev > /tmp/vite.log 2>&1 &
VITE_PID=$!

# 4) Health-gate before running suite (fail fast instead of 46 flaky failures)
for i in $(seq 1 30); do
  curl -fsS http://127.0.0.1:5173/ > /dev/null 2>&1 && curl -fsS http://127.0.0.1:8091/health > /dev/null 2>&1 && break
  sleep 1
done

# 5) Run integration suite: vitest+Playwright, fileParallelism false, retry 1
npm run test:integration

# 6) Teardown ALWAYS: kill both, then verify ports free
kill "$CANOPYD_PID" "$VITE_PID" 2>/dev/null || true
wait "$CANOPYD_PID" "$VITE_PID" 2>/dev/null || true
sleep 1
ss -ltn | grep -qE ':(8091|5173)\b' && { echo "FAIL: ports still bound" >&2; exit 1; }
echo "OK: 46/46 PASS, ports free"
