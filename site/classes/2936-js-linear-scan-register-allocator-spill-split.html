Linear-Scan Register Allocator with Loop-Aware Spill Splitting

Root-Cause Analysis

A naive linear-scan allocator fails on this problem for three independent reasons:

  1. Single-pass liveness is unsound on a CFG. Walking blocks in layout order under-approximates values that are live across a back edge. A value used at the top of a loop and defined at the bottom appears dead in a single pass, so it is assigned a register whose live range does not actually cover the back edge. The fix is to compute liveness as an iterative dataflow fixpoint over the CFG (liveIn[b] = use[b] ∪ (liveOut[b] − def[b]), liveOut[b] = ⋃ liveIn[succ]), and only then build live intervals from the exact per-instruction live sets.

  2. Spilling without a real split leaves a value in a clobbered register across the loop back edge. If a value is simply "spilled" but its live interval still claims a register at the loop header, a later definition of a different value in that register silently destroys it. The fix is to split the interval: the spilled value lives in a stack slot, and every use gets a reload into a fresh temporary whose live range is only the use. Definitions of the spilled value store to the slot. Because the slot is written at every definition, it is valid at every block boundary, including non-dominated boundaries; no separate boundary store is required.

  3. Blind reloads create redundant loads and hide the "was it actually clobbered?" question. After physical register assignment, several reload temporaries may land in the same register. Reloading again when the register still holds the value is redundant. The fix is a forward must-dataflow over the physical IR: facts are (register → spilled value), killed by any write to the register, and intersected at joins. A reload is kept only when the incoming facts do not prove that the register already holds the value, i.e. only when some path actually clobbered it. This also gives the correct loop behavior: a value kept in a register around a back edge is reloaded only if the loop body clobbered that register.

The spill candidate is chosen by maximizing next-use-distance / spill-cost (reloads weighted higher than stores), matching "furthest next use weighted against spill cost".

Exact Fix

Create allocator.js with the following self-contained implementation and run it:

node allocator.js 1000

'use strict';
/*
 * linear-scan-register-allocator-spill-split
 * ------------------------------------------
 * A self-contained linear-scan register allocator for a three-address IR
 * over a control-flow graph with loops.
 *
 *   - liveness is computed by iterative dataflow to a fixpoint
 *   - live intervals are the convex hull of all live/def/use positions
 *   - 4 general registers r0..r3
 *   - spill candidate = max( next-use-distance / spill-cost )  ("furthest next
 *     use, weighted by spill cost")
 *   - the live range is split at the spill point into usable intervals: the
 *     spilled portion lives in a stack slot and is reloaded at uses / stored
 *     at definitions (and at the boundary), so no value is ever left in a
 *     register that was clobbered across a back edge
 */

const PHYS = ['r0', 'r1', 'r2', 'r3'];
const LARGE = 1e9;

// ---------------------------------------------------------------------------
// tiny set helpers
// ---------------------------------------------------------------------------
function union(a, b) { const s = new Set(a); for (const x of b) s.add(x); return s; }
function setdiff(a, b) { const s = new Set(); for (const x of a) if (!b.has(x)) s.add(x); return s; }
function eqset(a, b) { if (a.size !== b.size) return false; for (const x of a) if (!b.has(x)) return false; return true; }
function mapEqual(a, b) { if (a.size !== b.size) return false; for (const [k, v] of a) if (b.get(k) !== v) return false; return true; }

// ---------------------------------------------------------------------------
// instruction accessors (three-address IR)
// ---------------------------------------------------------------------------
function usesOf(ins) {
  switch (ins.op) {
    case 'const': return [];
    case 'mov': return [ins.src];
    case 'add': case 'sub': case 'mul': case 'div':
    case 'lt': case 'gt': case 'eq': return [ins.a, ins.b];
    case 'load': return [ins.addr];
    case 'store': return [ins.addr, ins.src];
    case 'br': return [ins.cond];
    case 'ret': return [ins.src];
    case 'jmp': case 'nop': case 'reload': return [];
    case 'spill': return [ins.src];
    default: throw new Error('usesOf: unknown op ' + ins.op);
  }
}
function defsOf(ins) {
  switch (ins.op) {
    case 'const': case 'mov': case 'load': case 'reload': return [ins.dst];
    case 'add': case 'sub': case 'mul': case 'div':
    case 'lt': case 'gt': case 'eq': return [ins.dst];
    case 'store': case 'br': case 'ret': case 'jmp':
    case 'nop': case 'spill': return [];
    default: throw new Error('defsOf: unknown op ' + ins.op);
  }
}

// ---------------------------------------------------------------------------
// reverse post-order (covers unreachable blocks too)
// ---------------------------------------------------------------------------
function computeRPO(prog) {
  const post = [];
  const seen = new Set();
  function dfs(start) {
    if (seen.has(start)) return;
    seen.add(start);
    const st = [{ id: start, i: 0 }];
    while (st.length) {
      const top = st[st.length - 1];
      const b = prog.blocks[top.id];
      const succ = b ? b.succ : [];
      if (top.i < succ.length) {
        const s = succ[top.i++];
        if (!seen.has(s)) { seen.add(s); st.push({ id: s, i: 0 }); }
      } else {
        post.push(top.id);
        st.pop();
      }
    }
  }
  dfs(prog.entry);
  for (const id in prog.blocks) if (!seen.has(id)) dfs(id);
  return post.reverse();
}

// ---------------------------------------------------------------------------
// iterative dataflow liveness + per-instruction live sets + positions
// ---------------------------------------------------------------------------
function analyze(prog) {
  const rpo = computeRPO(prog);
  const positions = {};
  let g = 0;
  for (const id of rpo) {
    const b = prog.blocks[id];
    positions[id] = b.instrs.map(() => g++);
  }

  // block local use/def
  const use = {}, def = {};
  for (const id in prog.blocks) {
    const b = prog.blocks[id];
    const U = new Set(), D = new Set();
    for (const ins of b.instrs) {
      for (const u of usesOf(ins)) if (!D.has(u)) U.add(u);
      for (const d of defsOf(ins)) D.add(d);
    }
    use[id] = U; def[id] = D;
  }

  // fixpoint: liveIn[b] = use[b] U (liveOut[b] - def[b])
  //           liveOut[b] = U liveIn[succ]
  const liveIn = {}, liveOut = {};
  for (const id in prog.blocks) { liveIn[id] = new Set(); liveOut[id] = new Set(); }
  let changed = true;
  while (changed) {
    changed = false;
    for (let ri = rpo.length - 1; ri >= 0; ri--) {
      const id = rpo[ri];
      const b = prog.blocks[id];
      const out = new Set();
      for (const s of b.succ) for (const x of liveIn[s]) out.add(x);
      const inn = union(use[id], setdiff(out, def[id]));
      if (!eqset(out, liveOut[id]) || !eqset(inn, liveIn[id])) {
        liveOut[id] = out; liveIn[id] = inn; changed = true;
      }
    }
  }

  // per-instruction live sets (exact, given the fixpoint)
  const liveBefore = {}, liveAfter = {};
  for (const id in prog.blocks) {
    const b = prog.blocks[id];
    let after = new Set(liveOut[id]);
    for (let k = b.instrs.length - 1; k >= 0; k--) {
      const ins = b.instrs[k];
      const pos = positions[id][k];
      liveAfter[pos] = after;
      const before = union(usesOf(ins), setdiff(after, new Set(defsOf(ins))));
      liveBefore[pos] = before;
      after = before;
    }
  }

  // collect positions per var
  const varPos = {}, usePos = {}, defPos = {};
  const touch = (v, p) => { (varPos[v] = varPos[v] || []).push(p); };
  for (const id in prog.blocks) {
    const b = prog.blocks[id];
    b.instrs.forEach((ins, k) => {
      const pos = positions[id][k];
      for (const u of usesOf(ins)) { touch(u, pos); (usePos[u] = usePos[u] || []).push(pos); }
      for (const d of defsOf(ins)) { touch(d, pos); (defPos[d] = defPos[d] || []).push(pos); }
    });
  }
  for (const pos in liveBefore) for (const v of liveBefore[pos]) touch(v, +pos);
  for (const pos in liveAfter) for (const v of liveAfter[pos]) touch(v, +pos);

  const allVars = Object.keys(varPos);
  return { rpo, positions, use, def, liveIn, liveOut,
           liveBefore, liveAfter, varPos, usePos, defPos, allVars };
}

// ---------------------------------------------------------------------------
// live intervals: convex hull of every position where the value is live
// ---------------------------------------------------------------------------
function buildIntervals(an) {
  const intervals = [];
  for (const v of an.allVars) {
    const ps = an.varPos[v];
    if (!ps || ps.length === 0) continue;
    let mn = Infinity, mx = -Infinity;
    for (const p of ps) { if (p < mn) mn = p; if (p > mx) mx = p; }
    intervals.push({ vr: v, start: mn, end: mx + 1 });
  }
  return intervals;
}

// ---------------------------------------------------------------------------
// linear scan.  Returns either {ok:true, regOf} or {ok:false, candidate}
// ---------------------------------------------------------------------------
function tryAllocate(intervals, K, usePos, defPos, tempSet) {
  const sorted = intervals.slice().sort((a, b) => a.start - b.start || a.end - b.end);
  const active = [];
  const free = [];
  for (let i = 0; i < K; i++) free.push(i);
  const regOf = {};

  function nextUse(v, p) {
    const arr = usePos[v];
    if (!arr || arr.length === 0) return LARGE;
    let lo = 0, hi = arr.length;
    while (lo < hi) { const mid = (lo + hi) >> 1; if (arr[mid] < p) lo = mid + 1; else hi = mid; }
    return lo < arr.length ? arr[lo] : LARGE;
  }
  function spillCost(v) {
    const u = (usePos[v] || []).length;
    const d = (defPos[v] || []).length;
    return 1 + 2 * u + d;             // reloads cost more than stores
  }

  for (const I of sorted) {
    // expire intervals that ended at or before I.start
    for (let j = active.length - 1; j >= 0; j--) {
      if (active[j].end <= I.start) { free.push(active[j].reg); active.splice(j, 1); }
    }
    if (free.length > 0) {
      const r = free.shift();
      regOf[I.vr] = r;
      active.push({ vr: I.vr, start: I.start, end: I.end, reg: r });
      continue;
    }
    // pressure: choose spill candidate = max(next-use distance / cost)
    let cand = null, best = -1;
    const consider = (C) => {
      if (tempSet.has(C.vr)) return;          // reload/store fragments must stay in registers
      const score = nextUse(C.vr, I.start) / spillCost(C.vr);
      if (score > best) { best = score; cand = C; }
    };
    for (const C of active) consider(C);
    consider(I);
    if (!cand) return { ok: false, candidate: null, error: 'no spillable candidate' };
    return { ok: false, candidate: cand.vr };
  }
  return { ok: true, regOf };
}

// ---------------------------------------------------------------------------
// spill rewrite: turn a spilled virtual register into reload/store fragments
// (this is the live-range split -- the value is materialised to a slot and
//  only re-materialised into a register at the exact point of use)
// ---------------------------------------------------------------------------
function rewrite(prog, spilled, counter) {
  const slots = {};
  [...spilled].sort().forEach((v, i) => { slots[v] = i; });
  const tempSet = new Set();
  const blocks = {};
  for (const id in prog.blocks) {
    const b = prog.blocks[id];
    const out = [];
    const fresh = () => { const t = '$t' + (counter.n++); tempSet.add(t); return t; };
    const reload = (v) => {
      const t = fresh();
      out.push({ op: 'reload', dst: t, slot: slots[v], from: v });
      return t;
    };
    for (const ins of b.instrs) {
      switch (ins.op) {
        case 'const': {
          if (spilled.has(ins.dst)) {
            const t = fresh();
            out.push({ op: 'const', dst: t, value: ins.value });
            out.push({ op: 'spill', slot: slots[ins.dst], src: t, to: ins.dst });
          } else out.push({ ...ins });
          break;
        }
        case 'mov': {
          let src = ins.src; if (spilled.has(src)) src = reload(src);
          if (spilled.has(ins.dst)) {
            const t = fresh();
            out.push({ op: 'mov', dst: t, src });
            out.push({ op: 'spill', slot: slots[ins.dst], src: t, to: ins.dst });
          } else out.push({ op: 'mov', dst: ins.dst, src });
          break;
        }
        case 'add': case 'sub': case 'mul': case 'div':
        case 'lt': case 'gt': case 'eq': {
          let a = ins.a; if (spilled.has(a)) a = reload(a);
          let b2 = ins.b; if (spilled.has(b2)) b2 = reload(b2);
          if (spilled.has(ins.dst)) {
            const t = fresh();
            out.push({ op: ins.op, dst: t, a, b: b2 });
            out.push({ op: 'spill', slot: slots[ins.dst], src: t, to: ins.dst });
          } else out.push({ op: ins.op, dst: ins.dst, a, b: b2 });
          break;
        }
        case 'load': {
          let addr = ins.addr; if (spilled.has(addr)) addr = reload(addr);
          if (spilled.has(ins.dst)) {
            const t = fresh();
            out.push({ op: 'load', dst: t, addr });
            out.push({ op: 'spill', slot: slots[ins.dst], src: t, to: ins.dst });
          } else out.push({ op: 'load', dst: ins.dst, addr });
          break;
        }
        case 'store': {
          let addr = ins.addr; if (spilled.has(addr)) addr = reload(addr);
          let src = ins.src; if (spilled.has(src)) src = reload(src);
          out.push({ op: 'store', addr, src });
          break;
        }
        case 'br': {
          let cond = ins.cond; if (spilled.has(cond)) cond = reload(cond);
          out.push({ op: 'br', cond, target: ins.target, fallthrough: ins.fallthrough });
          break;
        }
        case 'ret': {
          let src = ins.src; if (spilled.has(src)) src = reload(src);
          out.push({ op: 'ret', src });
          break;
        }
        case 'jmp': case 'nop':
          out.push({ ...ins });
          break;
        default:
          throw new Error('rewrite: unexpected op ' + ins.op);
      }
    }
    blocks[id] = { id, instrs: out, succ: b.succ.slice() };
  }
  return { program: { entry: prog.entry, blocks }, slots, tempSet };
}

// ---------------------------------------------------------------------------
// redundant-load elimination on the physical IR.
//
// A reload is redundant when the destination physical register is already
// known to hold the spilled value on *every* incoming path.  That is exactly
// the "only reload when the register was clobbered" rule: facts are killed by
// any write to the register, so a load survives only where a path lost the
// value.  The analysis is a forward must-dataflow with intersection meet.
// ---------------------------------------------------------------------------
function eliminateRedundantReloads(prog) {
  const preds = {};
  for (const id in prog.blocks) preds[id] = [];
  for (const id in prog.blocks) {
    for (const s of prog.blocks[id].succ) {
      if (!preds[s]) preds[s] = [];
      preds[s].push(id);
    }
  }
  const inFacts = {}, outFacts = {};
  for (const id in prog.blocks) { inFacts[id] = new Map(); outFacts[id] = new Map(); }

  const transfer = (inMap, b) => {
    const f = new Map(inMap);
    for (const ins of b.instrs) {
      if (ins.op === 'reload') {
        f.set(ins.dst, ins.from);
      } else if (ins.op === 'spill') {
        f.set(ins.src, ins.to);            // a spill leaves the register intact
      } else if (typeof ins.dst === 'string') {
        f.delete(ins.dst);                 // a definition clobbers the register
      }
    }
    return f;
  };

  let changed = true;
  while (changed) {
    changed = false;
    for (const id in prog.blocks) {
      let inn;
      const ps = preds[id] || [];
      if (id === prog.entry || ps.length === 0) inn = new Map();
      else {
        inn = new Map(outFacts[ps[0]]);
        for (let i = 1; i < ps.length; i++) {
          const o = outFacts[ps[i]];
          for (const [r, v] of [...inn]) if (o.get(r) !== v) inn.delete(r);
        }
      }
      const out = transfer(inn, prog.blocks[id]);
      if (!mapEqual(inn, inFacts[id]) || !mapEqual(out, outFacts[id])) {
        inFacts[id] = inn; outFacts[id] = out; changed = true;
      }
    }
  }

  const redundant = new Set();
  for (const id in prog.blocks) {
    let f = new Map(inFacts[id]);
    const b = prog.blocks[id];
    b.instrs.forEach((ins, index) => {
      if (ins.op === 'reload') {
        if (f.get(ins.dst) === ins.from) redundant.add(`${id}#${index}`);
        f.set(ins.dst, ins.from);
      } else if (ins.op === 'spill') {
        f.set(ins.src, ins.to);
      } else if (typeof ins.dst === 'string') {
        f.delete(ins.dst);
      }
    });
  }

  const blocks = {};
  for (const id in prog.blocks) {
    const b = prog.blocks[id];
    const instrs = b.instrs.filter((ins, index) => !(ins.op === 'reload' && redundant.has(`${id}#${index}`)));
    blocks[id] = { id, instrs, succ: b.succ.slice() };
  }
  return { program: { entry: prog.entry, blocks }, removed: redundant.size };
}

// ---------------------------------------------------------------------------
// driver: keep spilling until everything fits in K registers
// ---------------------------------------------------------------------------
function allocate(prog, K = 4) {
  const spilled = new Set();
  const counter = { n: 0 };
  let guard = 0;
  for (;;) {
    if (++guard > 100000) throw new Error('allocation did not converge');
    const rw = rewrite(prog, spilled, counter);
    const an = analyze(rw.program);
    const intervals = buildIntervals(an);
    const res = tryAllocate(intervals, K, an.usePos, an.defPos, rw.tempSet);
    if (res.ok) {
      const phys0 = toPhysical(rw.program, res.regOf);
      const elim = eliminateRedundantReloads(phys0);
      const phys = elim.program;
      const events = collectSpillEvents(phys);
      return {
        program: rw.program,
        physical: phys,
        reloadsRemoved: elim.removed,
        regOf: res.regOf,
        slots: rw.slots,
        spilled: new Set(spilled),
        tempSet: rw.tempSet,
        intervals,
        analysis: an,
        events,
      };
    }
    if (!res.candidate) throw new Error(res.error);
    spilled.add(res.candidate);
  }
}

function toPhysical(prog, regOf) {
  const m = (x) => (typeof x === 'string' && regOf[x] !== undefined) ? PHYS[regOf[x]] : x;
  const blocks = {};
  for (const id in prog.blocks) {
    const b = prog.blocks[id];
    const instrs = b.instrs.map((ins) => {
      const n = { ...ins };
      if (typeof n.dst === 'string') n.dst = m(n.dst);
      if (typeof n.src === 'string') n.src = m(n.src);
      if (typeof n.a === 'string') n.a = m(n.a);
      if (typeof n.b === 'string') n.b = m(n.b);
      if (typeof n.addr === 'string') n.addr = m(n.addr);
      if (typeof n.cond === 'string') n.cond = m(n.cond);
      return n;
    });
    blocks[id] = { id, instrs, succ: b.succ.slice() };
  }
  return { entry: prog.entry, blocks };
}

// ---------------------------------------------------------------------------
// public allocation view: {intervalId, reg, spills:[{at, kind}]}
// ---------------------------------------------------------------------------
function collectSpillEvents(prog, positions) {
  const out = [];
  for (const id in prog.blocks) {
    const b = prog.blocks[id];
    b.instrs.forEach((ins, index) => {
      if (ins.op === 'reload') out.push({ at: `${id}#${index}`, kind: 'reload' });
      if (ins.op === 'spill') out.push({ at: `${id}#${index}`, kind: 'store' });
    });
  }
  return out;
}

function allocationReport(alloc) {
  const { physical: program, regOf, slots, spilled } = alloc;
  const perVar = new Map();
  const ev = [];
  for (const id in program.blocks) {
    const b = program.blocks[id];
    b.instrs.forEach((ins, index) => {
      if (ins.op === 'reload') ev.push({ vr: ins.from, at: `${id}#${index}`, kind: 'reload' });
      if (ins.op === 'spill') ev.push({ vr: ins.to, at: `${id}#${index}`, kind: 'store' });
    });
  }
  const vars = new Set([...Object.keys(regOf), ...spilled]);
  for (const v of vars) {
    const spills = ev.filter((e) => e.vr === v).map((e) => ({ at: e.at, kind: e.kind }));
    perVar.set(v, {
      intervalId: v,
      reg: regOf[v] !== undefined ? PHYS[regOf[v]] : null,
      spilled: spilled.has(v),
      slot: slots[v] !== undefined ? slots[v] : null,
      spills,
    });
  }
  return [...perVar.values()];
}

// ---------------------------------------------------------------------------
// interpreter (runs both the virtual and the physical IR)
// ---------------------------------------------------------------------------
function interpret(prog, initialMem, maxSteps = 2000000) {
  const regs = new Map();
  const mem = new Map();
  for (const [k, v] of initialMem) mem.set(k, v);
  const stack = [];
  const get = (x) => { const v = regs.get(x); return v === undefined ? 0 : v; };
  let blockId = prog.entry, idx = 0, steps = 0;
  while (steps++ < maxSteps) {
    const b = prog.blocks[blockId];
    if (!b) throw new Error('interp: bad block ' + blockId);
    if (idx >= b.instrs.length) throw new Error('interp: fell off block ' + blockId);
    const ins = b.instrs[idx];
    switch (ins.op) {
      case 'const': regs.set(ins.dst, ins.value); idx++; break;
      case 'mov': regs.set(ins.dst, get(ins.src)); idx++; break;
      case 'add': regs.set(ins.dst, get(ins.a) + get(ins.b)); idx++; break;
      case 'sub': regs.set(ins.dst, get(ins.a) - get(ins.b)); idx++; break;
      case 'mul': regs.set(ins.dst, get(ins.a) * get(ins.b)); idx++; break;
      case 'div': { const d = get(ins.b); regs.set(ins.dst, d === 0 ? 0 : Math.trunc(get(ins.a) / d)); idx++; break; }
      case 'lt': regs.set(ins.dst, get(ins.a) < get(ins.b) ? 1 : 0); idx++; break;
      case 'gt': regs.set(ins.dst, get(ins.a) > get(ins.b) ? 1 : 0); idx++; break;
      case 'eq': regs.set(ins.dst, get(ins.a) === get(ins.b) ? 1 : 0); idx++; break;
      case 'load': regs.set(ins.dst, mem.get(get(ins.addr)) || 0); idx++; break;
      case 'store': mem.set(get(ins.addr), get(ins.src)); idx++; break;
      case 'jmp': blockId = ins.target; idx = 0; break;
      case 'br': blockId = get(ins.cond) ? ins.target : ins.fallthrough; idx = 0; break;
      case 'ret': return { ret: get(ins.src), mem };
      case 'reload': regs.set(ins.dst, stack[ins.slot] || 0); idx++; break;
      case 'spill': stack[ins.slot] = get(ins.src); idx++; break;
      case 'nop': idx++; break;
      default: throw new Error('interp: unknown op ' + ins.op);
    }
  }
  throw new Error('interp: step limit exceeded');
}

function memEqual(a, b) {
  const keys = new Set([...a.keys(), ...b.keys()]);
  for (const k of keys) if ((a.get(k) || 0) !== (b.get(k) || 0)) return false;
  return true;
}

// ---------------------------------------------------------------------------
// random test programs: a loop that keeps several values live across the back
// edge, so spilling and loop-correct reloading are exercised.
// ---------------------------------------------------------------------------
function lcg(seed) {
  let s = seed >>> 0;
  return () => { s = (Math.imul(s, 1664525) + 1013904223) >>> 0; return s / 4294967296; };
}

function makeProgram(rng) {
  const nIters = 2 + Math.floor(rng() * 6);
  const nExtra = Math.floor(rng() * 7);           // 0..6 extra loop-carried values
  const blocks = {};
  const entry = [];
  entry.push({ op: 'const', dst: 'n', value: nIters });
  entry.push({ op: 'const', dst: 'i', value: 0 });
  entry.push({ op: 'const', dst: 'acc', value: 0 });
  entry.push({ op: 'const', dst: 'one', value: 1 });
  const extras = [];
  for (let k = 0; k < nExtra; k++) {
    entry.push({ op: 'const', dst: 'p' + k, value: k });
    entry.push({ op: 'load', dst: 'x' + k, addr: 'p' + k });
    extras.push('x' + k);
  }
  entry.push({ op: 'jmp', target: 'header' });
  blocks.entry = { id: 'entry', instrs: entry, succ: ['header'] };

  blocks.header = {
    id: 'header',
    instrs: [
      { op: 'lt', dst: 'c', a: 'i', b: 'n' },
      { op: 'br', cond: 'c', target: 'body', fallthrough: 'exit' },
    ],
    succ: ['body', 'exit'],
  };

  const body = [];
  // random short-lived temporaries
  const nTemp = Math.floor(rng() * 5);
  for (let t = 0; t < nTemp; t++) {
    body.push({ op: 'const', dst: 'tmp' + t, value: Math.floor(rng() * 9) });
    body.push({ op: 'add', dst: 'tmp' + t, a: 'tmp' + t, b: 'i' });
  }
  // keep loop-carried values alive and mutate them
  for (const x of extras) {
    const op = rng() < 0.5 ? 'add' : 'mul';
    body.push({ op, dst: x, a: x, b: 'i' });
  }
  if (extras.length > 0) body.push({ op: 'add', dst: 'acc', a: 'acc', b: extras[0] });
  else body.push({ op: 'add', dst: 'acc', a: 'acc', b: 'i' });
  body.push({ op: 'add', dst: 'i', a: 'i', b: 'one' });
  body.push({ op: 'jmp', target: 'header' });
  blocks.body = { id: 'body', instrs: body, succ: ['header'] };

  const outAddr = 100 + Math.floor(rng() * 20);
  blocks.exit = {
    id: 'exit',
    instrs: [
      { op: 'const', dst: 'outp', value: outAddr },
      { op: 'store', addr: 'outp', src: 'acc' },
      { op: 'ret', src: 'acc' },
    ],
    succ: [],
  };
  return { entry: 'entry', blocks };
}

// ---------------------------------------------------------------------------
// differential test: interpret the source IR and the allocated IR on the same
// randomized inputs and require identical observable results.
// ---------------------------------------------------------------------------
function differential(numSeeds) {
  let spilledPrograms = 0;
  for (let seed = 1; seed <= numSeeds; seed++) {
    const rng = lcg(Math.imul(seed, 2654435761) >>> 0);
    const prog = makeProgram(rng);
    const alloc = allocate(prog, 4);
    if (alloc.spilled.size > 0) spilledPrograms++;

    const phys = alloc.physical;
    const mem0 = new Map();
    for (let j = 0; j < 32; j++) mem0.set(j, Math.floor(rng() * 41) - 20);

    let a, b;
    try { a = interpret(prog, new Map(mem0)); }
    catch (e) { throw new Error(`seed ${seed}: source interpretation failed: ${e.message}`); }
    try { b = interpret(phys, new Map(mem0)); }
    catch (e) { throw new Error(`seed ${seed}: allocated interpretation failed: ${e.message}`); }

    if (a.ret !== b.ret || !memEqual(a.mem, b.mem)) {
      throw new Error(`seed ${seed}: MISMATCH ret ${a.ret} vs ${b.ret}, mem ${JSON.stringify([...a.mem])} vs ${JSON.stringify([...b.mem])}`);
    }
  }
  return { seeds: numSeeds, spilledPrograms };
}

Both reloads are eliminated because the spill in A leaves r0 holding v, and no path clobbers r0 before the reloads. If any instruction wrote r0 between those points, the facts would be killed and the reload would be retained.

Allocation report shape

allocationReport(alloc) returns one entry per virtual register in the requested format:

{
  intervalId: "x0",     // virtual register / interval id
  reg: null,            // physical register, or null when spilled
  spilled: true,
  slot: 2,              // stack slot when spilled
  spills: [             // [{at, kind}]
    { at: "entry#8",  kind: "store"  },
    { at: "body#6",   kind: "reload" },
    { at: "body#8",   kind: "store"  },
    { at: "body#24",  kind: "reload" }
  ]
}

Non-spilled intervals get reg: "r0".."r3" and an empty spills array. The rewritten physical IR is available as alloc.physical and is what the differential test interprets.
