BT-020: Closing the Section 5 Judge Gap in the boardctl render HTML Report

Repository: coding-hermes/boardctl Feature commit: ff72fcb2aa54faf60e72e63129ee485880d262d1 Fix commit: bd9f52a533d735b4e61c045bc1c4a8fe497aa472 Judge evidence: .gitreins/history/2026-09-12/1249e112/verdict.json (FAIL) → .gitreins/history/2026-09-12/b1a3e017/verdict.json (PASS) Normative source: docs/specs/board-analytics-report.md §2.4, §3.9, §5.1–5.7, §7.1

1. Symptom

All 10 explicit §7.1 acceptance criteria passed with hard evidence, yet GitReins Tier 2 returned:

Four concrete gaps in the required §5 UX section:

2. Root-cause analysis

Proximate cause: implementation and tests targeted the 10 numbered §7.1 end-to-end probes. The task criterion actually requires spec sections 2, 3, 5, 6. Section 5 adds normative UX behaviors outside those probes, and four shipped missing.

Why each escaped the AC suite: 1. Counts panel — §7.1 never names it; the payload already had derived.status_counts/priority_counts, so payload checks passed while the client never rendered them. 2. Detail ladder — §7.1 only checks <script>/image escaping; the Go ladder existed but the client detailBody() only handled typeof v === "object". 3. Zero-streak wording — the synthetic streak AC only covers positive/live/ended values. 4. Error state — §2.7/§7.1 only test a torn line among valid rows; the stronger "not JSONL at all" case had no loader threshold.

Process cause: no requirement-to-test matrix for the full normative UX section. The durable fix is code + a matrix (§3) + contract tests asserting each UI/data path.

4.4 Client detail ladder (template.go)

Replace object/array-only detailBody with the full §2.4 ladder — never throws, textContent only:

function isPlainObject(v){ return v !== null && typeof v === "object" && !Array.isArray(v); }
function trimS(s){ return String(s).replace(/^[\s\uFEFF\xA0]+/, "").replace(/[\s\uFEFF\xA0]+$/, ""); }
function parseJSONContainerStr(s){
  if (typeof s !== "string") return null;
  var t = trimS(s);
  if (!t || (t.charAt(0) !== "{" && t.charAt(0) !== "[")) return null;
  try { var v = JSON.parse(t); if (isPlainObject(v) || Array.isArray(v)) return v; } catch (e) {}
  return null;
}
function strictBase64ToJSON(s){
  if (typeof s !== "string") return null;
  var t = trimS(s); if (!t) return null; var txt = null;
  try {
    var bin = atob(t); if (!bin) return null;
    var bytes = new Uint8Array(bin.length);
    for (var i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    if (typeof TextDecoder === "function") txt = new TextDecoder("utf-8").decode(bytes);
    else txt = decodeURIComponent(escape(bin));
  } catch (e) { return null; }
  if (txt === null) return null;
  return parseJSONContainerStr(txt);
}
function detailLadder(v){
  if (isPlainObject(v) || Array.isArray(v)) return v;   // 1
  if (typeof v === "string") {
    var p = parseJSONContainerStr(v);                   // 2
    if (p !== null) return p;
    var b = strictBase64ToJSON(v);                      // 3
    if (b !== null) return b;
  }
  return v;                                             // 4
}
function detailBody(e){
  var pre = el("pre");
  var v = e.detail;
  if (v === undefined || v === null) { pre.textContent = "(no detail)"; return pre; }
  var out;
  try { out = detailLadder(v); } catch (err) { out = v; }
  if (isPlainObject(out) || Array.isArray(out)) {
    var txt = null;
    try { txt = JSON.stringify(out, null, 2); } catch (err2) { txt = null; }
    if (txt !== null) { pre.textContent = txt; return pre; }
  }
  pre.textContent = String(out);
  return pre;
}

4.5 V11 counts panel (template.go)

CSS adds .countspanel, .countsbox, .errorcard. Board page gets a host div #counts-panel filled by renderCountsPanel(bd); recompute from raw rows with ComputeStats semantics:

function countsFromRaw(bd, fixtures){
  var sc = {}, pc = {};
  boardTasks(bd).forEach(function(r){
    var isFix = isFixtureRow(bd, r);
    if (!fixtures && isFix) return;
    var s = taskStatus(r); sc[s] = (sc[s] || 0) + 1;
    var p = priorityLabelOf(r); if (p !== null) pc[p] = (pc[p] || 0) + 1;
  });
  return {status: sc, priority: pc};
}
function priorityLabelOf(r){
  var v = r ? r.priority : undefined;
  if (v === undefined || v === null || v === "") return null;
  if (typeof v === "number") { if (!isFinite(v)) return null; return String(v); }
  if (typeof v === "string") return v;
  return null;
}

renderCountsPanel builds "Status composition"/"Priority composition" boxes (sorted counts, chip(k)), with empty states "no tasks yet" / "no priorities set", and a fixture-count note when the toggle is on. Wire the toggle:

fx.addEventListener("change", function(){ state.showFixtures = fx.checked; renderCountsPanel(bd); renderTable(bd); });
