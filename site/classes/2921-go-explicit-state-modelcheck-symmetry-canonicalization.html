Explicit-State Model Checker with Sound Symmetry Canonicalization (Go 1.26)

Root-cause analysis

There is no pre-existing repo to patch; the failure mode is the classic set of ways this checker gets implemented wrong. Each silently changes the reachable set, so the harness sees mismatched violation sets.

  1. Trusting the hash for state identity. A visited set as map[uint64]struct{} treats a 64-bit collision as equality. On 100k+ states every collision loses a real state and any violation reachable through it. The hash must be a bucket selector; membership is confirmed by full element-wise comparison (Equal).
  2. Canonicalising the whole state instead of process blocks. Process symmetry permutes whole process tuples. sort.Ints(state) merges states with different globals and states in different orbits, e.g. {global=1, procs={2,3}} and {global=2, procs={1,3}} both become 1 2 3. Fix: keep the global prefix fixed, sort only the contiguous NumLocals-sized process blocks lexicographically.
  3. Canonicalising inconsistently. Canonical must be applied to initial states and every successor, or orbits are re-explored / compared against differently-shaped reps.
  4. DFS / depth cap for the shortest trace. DFS returns a trace, not the shortest, and a depth cap breaks unbounded-depth models. BFS guarantees the first violating state dequeued is minimal-depth; the visited set terminates on the finite state space.
  5. Sequential assignment. x:=y; y:=x must swap. All RHS must be evaluated against the source state (simultaneous semantics).
  6. Reporting only canonical reps. "Emit every reachable violating state" + "identical violation sets" means concrete sets must match. With symmetry on, expand each violating class back to its full orbit.
  7. Replay not symmetry-aware. Replay must check init membership, legality of each step modulo symmetry, and the final violation.

// Package mc implements an explicit-state model checker for a small TLA-like
// language: integer state variables, boolean guards and a next relation made of
// guarded (simultaneous) assignments.  The checker performs breadth-first
// search, exact state hashing with collision re-verification, optional symmetry
// reduction over interchangeable process slots, and shortest-counterexample
// reconstruction with replay.
package mc

import (
    "errors"
    "fmt"
    "sort"
)

// State is a flat integer vector.  The layout is
//
//  [ g0, g1, ..., g(G-1),  p0.v0, p0.v1, ..., p0.v(L-1),  p1.v0, ... ]
//
// where G = NumGlobals, L = NumLocals and there are NumProcs process blocks.
// Symmetry reduction permutes whole process blocks, never individual cells, and
// never touches the global prefix.
type State []int

// Equal compares two states exactly (every element).
func (s State) Equal(o State) bool {
    if len(s) != len(o) {
        return false
    }
    for i := range s {
        if s[i] != o[i] {
            return false
        }
    }
    return true
}

// Clone returns an independent copy of s.
func (s State) Clone() State {
    c := make(State, len(s))
    copy(c, s)
    return c
}

func (s State) String() string { return fmt.Sprint([]int(s)) }

// FNVHash is the default 64-bit FNV-1a hash over the little-endian bytes of
// every element.  It is only a bucket selector: a hash match is NEVER treated
// as state equality.
func FNVHash(s State) uint64 {
    const (
        offset64 = 14695981039346656037
        prime64  = 1099511628211
    )
    h := uint64(offset64)
    for _, v := range s {
        u := uint64(v)
        for i := 0; i < 8; i++ {
            h ^= u & 0xff
            h *= prime64
            u >>= 8
        }
    }
    return h
}

// Visited is a hash table that always re-verifies a candidate with a full
// element-by-element comparison.  Distinct states that happen to share a hash
// are stored in the same bucket and are never merged.
type Visited struct {
    hash  func(State) uint64
    table map[uint64][]State
    count int
}

// NewVisited creates an empty visited set.  A nil hash selects FNVHash.  The
// hash is injectable so the collision-re-verification path can be tested.
func NewVisited(hash func(State) uint64) *Visited {
    if hash == nil {
        hash = FNVHash
    }
    return &Visited{hash: hash, table: make(map[uint64][]State)}
}

// Has reports whether an exactly equal state is already present.
func (v *Visited) Has(s State) bool {
    for _, t := range v.table[v.hash(s)] {
        if t.Equal(s) {
            return true
        }
    }
    return false
}

// Add inserts s; it returns true when s was not present before.
func (v *Visited) Add(s State) bool {
    h := v.hash(s)
    for _, t := range v.table[h] {
        if t.Equal(s) {
            return false
        }
    }
    v.table[h] = append(v.table[h], s.Clone())
    v.count++
    return true
}

// Len returns the number of distinct states stored.
func (v *Visited) Len() int { return v.count }

// Update is one simultaneous assignment in an action.  If LocalVar >= 0 the
// assignment targets process-local variable LocalVar of the action's process;
// otherwise it targets global variable GlobalVar.
type Update struct {
    LocalVar  int
    GlobalVar int
    Value     func(s State, p int) int
}

// Action is a guarded set of simultaneous assignments.  A parametric action is
// instantiated once per process slot p in [0, NumProcs); a non-parametric one
// runs once with p = -1.
type Action struct {
    Name       string
    Parametric bool
    Guard      func(s State, p int) bool
    Updates    []Update
}

// Spec describes a model to check.
type Spec struct {
    Name       string
    NumGlobals int
    NumProcs   int
    NumLocals  int // locals per process
    Init       []State
    Actions    []Action
    Invariant  func(s State) bool // true == invariant holds
    Symmetry   bool               // enable process-slot canonicalisation
}

// Len returns the required length of every state of the spec.
func (sp *Spec) Len() int { return sp.NumGlobals + sp.NumProcs*sp.NumLocals }

// Validate checks the structural consistency of the spec.
func (sp *Spec) Validate() error {
    if sp.NumGlobals < 0 || sp.NumProcs < 0 || sp.NumLocals < 0 {
        return errors.New("negative layout dimension")
    }
    if sp.NumProcs > 0 && sp.NumLocals == 0 {
        return errors.New("procs > 0 but no local variables")
    }
    n := sp.Len()
    if len(sp.Init) == 0 {
        return errors.New("no initial states")
    }
    for i, s := range sp.Init {
        if len(s) != n {
            return fmt.Errorf("init[%d] has length %d, want %d", i, len(s), n)
        }
    }
    if sp.Invariant == nil {
        return errors.New("nil invariant")
    }
    return nil
}

// canonicalUnconditional is the raw canonicaliser.  It sorts whole process
// blocks lexicographically while leaving the global prefix untouched.  It is
// always a pure function of the full state, so two states share a canonical
// form iff they are in the same orbit of the process permutation group AND have
// identical globals.
func (sp *Spec) canonicalUnconditional(s State) State {
    if sp.NumProcs <= 1 || sp.NumLocals == 0 {
        return s.Clone()
    }
    out := make(State, len(s))
    g, l := sp.NumGlobals, sp.NumLocals
    copy(out[:g], s[:g]) // globals are NOT permuted

    blocks := make([][]int, sp.NumProcs)
    for i := 0; i < sp.NumProcs; i++ {
        b := make([]int, l)
        copy(b, s[g+i*l:g+(i+1)*l])
        blocks[i] = b
    }
    sort.Slice(blocks, func(i, j int) bool {
        a, b := blocks[i], blocks[j]
        for k := 0; k < l; k++ {
            if a[k] != b[k] {
                return a[k] < b[k]
            }
        }
        return false
    })
    for i, b := range blocks {
        copy(out[g+i*l:g+(i+1)*l], b)
    }
    return out
}

// Canonical returns the representative of s used for symmetry reduction.
// When symmetry is disabled it is an exact clone, preserving the identity of
// every state.
func (sp *Spec) Canonical(s State) State {
    if !sp.Symmetry {
        return s.Clone()
    }
    return sp.canonicalUnconditional(s)
}

// applyUpdates writes the action's assignments into dst, evaluating all
// right-hand sides against the original state src (simultaneous semantics).
func (sp *Spec) applyUpdates(dst, src State, a *Action, p int) {
    for i := range a.Updates {
        u := &a.Updates[i]
        v := u.Value(src, p)
        if u.LocalVar >= 0 {
            dst[sp.NumGlobals+p*sp.NumLocals+u.LocalVar] = v
        } else {
            dst[u.GlobalVar] = v
        }
    }
}

// Successors enumerates every concrete successor of s in a deterministic
// order (action order, then process index).  Every emitted state is a freshly
// allocated slice.
func (sp *Spec) Successors(s State, emit func(State)) {
    for ai := range sp.Actions {
        a := &sp.Actions[ai]
        if a.Parametric {
            for p := 0; p < sp.NumProcs; p++ {
                if a.Guard != nil && !a.Guard(s, p) {
                    continue
                }
                t := s.Clone()
                sp.applyUpdates(t, s, a, p)
                emit(t)
            }
        } else {
            if a.Guard != nil && !a.Guard(s, -1) {
                continue
            }
            t := s.Clone()
            sp.applyUpdates(t, s, a, -1)
            emit(t)
        }
    }
}

// Step reports whether `to` is a legal successor of `from`, modulo the
// spec's symmetry.  Replay is therefore valid in canonical space as well as in
// concrete space.
func (sp *Spec) Step(from, to State) bool {
    want := sp.Canonical(to)
    ok := false
    sp.Successors(from, func(t State) {
        if sp.Canonical(t).Equal(want) {
            ok = true
        }
    })
    return ok
}

// Replay verifies that trace starts in an initial state, that every consecutive
// pair is a legal step, and that the final state violates the invariant.
func (sp *Spec) Replay(trace []State) error {
    if len(trace) == 0 {
        return errors.New("empty trace")
    }
    if err := sp.Validate(); err != nil {
        return err
    }
    start := sp.Canonical(trace[0])
    initOK := false
    for _, s := range sp.Init {
        if sp.Canonical(s).Equal(start) {
            initOK = true
            break
        }
    }
    if !initOK {
        return fmt.Errorf("trace does not start in an initial state: %v", trace[0])
    }
    for i := 1; i < len(trace); i++ {
        if !sp.Step(trace[i-1], trace[i]) {
            return fmt.Errorf("invalid step %d: %v -> %v", i, trace[i-1], trace[i])
        }
    }
    if sp.Invariant(trace[len(trace)-1]) {
        return errors.New("final state does not violate the invariant")
    }
    return nil
}

// stateKey returns an exact, comparable encoding of a state.  It is used only
// to de-duplicate orbit members, never as the visited-set identity.
func stateKey(s State) string {
    b := make([]byte, 0, len(s)*3)
    for _, v := range s {
        b = appendInt(b, int64(v))
        b = append(b, ',')
    }
    return string(b)
}

func appendInt(b []byte, v int64) []byte {
    if v < 0 {
        b = append(b, '-')
        v = -v
    }
    if v == 0 {
        return append(b, '0')
    }
    var tmp [20]byte
    i := len(tmp)
    for v > 0 {
        i--
        tmp[i] = byte('0' + v%10)
        v /= 10
    }
    return append(b, tmp[i:]...)
}

// violationMembers returns every concrete state that is a member of the
// symmetry orbit of s.  With symmetry disabled this is just s, so the reported
// violation set is always concrete and independent of the reduction.
func (sp *Spec) violationMembers(s State) []State {
    if !sp.Symmetry || sp.NumProcs <= 1 || sp.NumLocals == 0 {
        return []State{s.Clone()}
    }
    g, l, n := sp.NumGlobals, sp.NumLocals, sp.NumProcs
    idx := make([]int, n)
    for i := range idx {
        idx[i] = i
    }
    seen := map[string]bool{}
    var out []State
    var rec func(k int)
    rec = func(k int) {
        if k == n {
            st := make(State, len(s))
            copy(st[:g], s[:g])
            for i, bi := range idx {
                copy(st[g+i*l:g+(i+1)*l], s[g+bi*l:g+(bi+1)*l])
            }
            key := stateKey(st)
            if !seen[key] {
                seen[key] = true
                out = append(out, st)
            }
            return
        }
        for i := k; i < n; i++ {
            idx[k], idx[i] = idx[i], idx[k]
            rec(k + 1)
            idx[k], idx[i] = idx[i], idx[k]
        }
    }
    rec(0)
    return out
}

// Result is the outcome of Check.
type Result struct {
    Holds      bool    // true iff no reachable state violates the invariant
    Violations []State // reachable violating states (concrete)
    Shortest   []State // shortest counterexample (canonical states), nil if Holds
    Explored   int     // number of distinct (canonical) states visited
    Depth      int     // length of Shortest in edges
}

type bfsNode struct {
    state  State
    parent int
}

// Check runs breadth-first reachability with an explicit visited set.  BFS
// terminates on finite state spaces even when the transition graph is cyclic or
// the depth is unbounded, and the first violating state reached is guaranteed
// to be at minimal depth.
func (sp *Spec) Check() (*Result, error) {
    if err := sp.Validate(); err != nil {
        return nil, err
    }
    seen := NewVisited(FNVHash)
    var order []bfsNode
    queue := make([]int, 0, len(sp.Init))
    enqueue := func(s State) {
        c := sp.Canonical(s)
        if seen.Add(c) {
            order = append(order, bfsNode{state: c, parent: -1})
            queue = append(queue, len(order)-1)
        }
    }
    for _, s := range sp.Init {
        enqueue(s)
    }
    res := &Result{Holds: true}
    firstViolation := -1
    for head := 0; head < len(queue); head++ {
        idx := queue[head]
        cur := order[idx].state
        if !sp.Invariant(cur) {
            res.Holds = false
            res.Violations = append(res.Violations, sp.violationMembers(cur)...)
            if firstViolation < 0 {
                firstViolation = idx
            }
        }
        sp.Successors(cur, func(t State) {
            c := sp.Canonical(t)
            if seen.Add(c) {
                order = append(order, bfsNode{state: c, parent: idx})
                queue = append(queue, len(order)-1)
            }
        })
    }
    res.Explored = seen.Len()
    if firstViolation >= 0 {
        res.Shortest = reconstructTrace(order, firstViolation)
        res.Depth = len(res.Shortest) - 1
    }
    return res, nil
}

func reconstructTrace(order []bfsNode, idx int) []State {
    var rev []State
    for i := idx; i >= 0; i = order[i].parent {
        rev = append(rev, order[i].state)
        if order[i].parent < 0 {
            break
        }
    }
    for l, r := 0, len(rev)-1; l < r; l, r = l+1, r-1 {
        rev[l], rev[r] = rev[r], rev[l]
    }
    return rev
}

package mc

import (
    "reflect"
    "sort"
    "testing"
)

func stateSet(vs []State) [][]int {
    seen := map[string]bool{}
    out := make([][]int, 0, len(vs))
    for _, v := range vs {
        k := stateKey(v)
        if seen[k] {
            continue
        }
        seen[k] = true
        out = append(out, []int(v))
    }
    sort.Slice(out, func(i, j int) bool {
        a, b := out[i], out[j]
        for k := 0; k < len(a) && k < len(b); k++ {
            if a[k] != b[k] {
                return a[k] < b[k]
            }
        }
        return len(a) < len(b)
    })
    return out
}

func counterSpec() *Spec {
    return &Spec{
        Name:       "counter",
        NumGlobals: 1,
        Init:       []State{{0}},
        Actions: []Action{{
            Name: "inc",
            Guard: func(s State, _ int) bool { return s[0] < 5 },
            Updates: []Update{{LocalVar: -1, GlobalVar: 0,
                Value: func(s State, _ int) int { return s[0] + 1 }}},
        }},
        Invariant: func(s State) bool { return s[0] < 3 },
    }
}

func TestCounterViolationsAndShortestTrace(t *testing.T) {
    sp := counterSpec()
    res, err := sp.Check()
    if err != nil { t.Fatal(err) }
    if res.Holds { t.Fatal("expected violation") }
    if !reflect.DeepEqual(stateSet(res.Violations), [][]int{{3}, {4}, {5}}) {
        t.Fatalf("violations = %v", stateSet(res.Violations))
    }
    if !reflect.DeepEqual(res.Shortest, []State{{0}, {1}, {2}, {3}}) {
        t.Fatalf("shortest = %v", res.Shortest)
    }
    if err := sp.Replay(res.Shortest); err != nil { t.Fatalf("replay: %v", err) }
}

func TestCollisionReverification(t *testing.T) {
    constHash := func(State) uint64 { return 0xdeadbeef }
    v := NewVisited(constHash)
    if !v.Add(State{1, 2, 3}) { t.Fatal("insert") }
    if v.Add(State{1, 2, 3}) { t.Fatal("duplicate") }
    if !v.Add(State{3, 2, 1}) { t.Fatal("distinct in same bucket") }
    if v.Len() != 2 { t.Fatalf("Len=%d", v.Len()) }
}

func TestCanonicalizationDoesNotMergeOutsideSymmetryClass(t *testing.T) {
    sp := &Spec{NumGlobals: 1, NumProcs: 2, NumLocals: 1, Symmetry: true}
    x := State{1, 2, 3}
    y := State{2, 1, 3}
    if sp.canonicalUnconditional(x).Equal(sp.canonicalUnconditional(y)) {
        t.Fatal("merged distinct symmetry classes")
    }
    if !sp.canonicalUnconditional(State{1, 2, 3}).Equal(
        sp.canonicalUnconditional(State{1, 3, 2})) {
        t.Fatal("same orbit should collapse")
    }
}

Soundness notes

  • Visited identity = (bucket, full Equal); the injected constant-hash test proves distinct states are never merged.
  • canonicalUnconditional copies the global prefix verbatim and sorts only contiguous process blocks; the dedicated test proves two states with the same value multiset but different globals stay distinct, while true permutations collapse.
  • Invariant is checked on canonical reps (valid for symmetric invariants) and violating classes are expanded to full orbits so the reported concrete set matches the symmetry-free run.
  • Replay is symmetry-aware (Step compares Canonical forms), so both modes accept identical validity checks.
