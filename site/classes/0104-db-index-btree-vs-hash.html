B-tree point lookup (O(log n)):

def _lookup(self, node: BTreeNode, key: int) -> Optional[str]:
    i = 0
    while i < len(node.keys) and key > node.keys[i]:
        i += 1
    if i < len(node.keys) and key == node.keys[i]:
        return node.values[i]
    if node.leaf:
        return None
    return self._lookup(node.children[i], key)

B-tree range scan (O(log n + k)):

def _range_scan(self, node, lo, hi, results):
    i = 0
    while i < len(node.keys) and node.keys[i] < lo:
        i += 1
    if not node.leaf:
        self._range_scan(node.children[i], lo, hi, results)
    while i < len(node.keys) and node.keys[i] <= hi:
        if lo <= node.keys[i] <= hi:
            results.append((node.keys[i], node.values[i]))
        if not node.leaf:
            self._range_scan(node.children[i+1], lo, hi, results)
        i += 1

Hash index point lookup (O(1)):

def lookup(self, key: int) -> Optional[str]:
    idx = self._hash(key)
    for k, v in self.buckets[idx]:
        if k == key:
            return v
    return None

Hash index range scan (O(n) — must scan everything):

def range_scan(self, lo: int, hi: int) -> List[Tuple[int, str]]:
    results = []
    for bucket in self.buckets:
        for k, v in bucket:
            if lo <= k <= hi:
                results.append((k, v))
    return sorted(results, key=lambda x: x[0])
