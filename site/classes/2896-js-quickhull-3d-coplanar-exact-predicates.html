Robust 3D Convex Hull with Exact Adaptive Predicates

1. Root‑cause analysis

A naïve 3D hull dies on degenerate data for five independent reasons:

  1. Floating‑point orientation/incircle predicates. orient3d/orient2d compute a determinant whose terms are ~O(M³) while the true value can be 0 or ±1 (near‑coplanar/near‑collinear or large coordinates). The rounding error is ~eps·permanent, which can exceed the exact value, so the sign is wrong. Any hull built on top inherits the wrong sign.

  2. Degenerate initial simplex. Picking the first two/three points can give a zero‑length edge, a zero‑area seed triangle, or a zero‑volume seed tetrahedron. Faces then have zero area and the whole incremental structure is meaningless (“every facet non‑degenerate” fails).

  3. Points exactly on a facet plane. With a blind >= 0 outside test they are inserted and create degenerate/overlapping triangles; with a strict > 0 test they are correctly kept out, but a naïve implementation then leaves a triangulation whose vertices are not the true extreme points and whose coplanar facets are fragmented (Euler/winding can still break).

  4. Duplicates produce zero‑area faces, zero predicates, and divide‑by‑zero style failure in the initial simplex.

  5. Cospherical / coplanar ties. Many points sharing a supporting plane must be merged into one maximal polygonal facet; otherwise the triangulation is non‑manifold or has extra interior vertices.

The fix is: exact sign predicates with a filtered fast path, a coplanarity‑aware initial simplex, strict outside tests, and a coplanar‑triangle merge into maximal facets, followed by machine‑checked topological/geometric invariants.

2. The fix

2.1 Key ideas

  • Dyadic exact arithmetic. Every double is exactly m · 2^e with m: BigInt, e: int (extracted from IEEE‑754 bits, sub‑normals included). All four predicates have a stage‑1 double evaluation with a conservative forward‑error bound and a stage‑2 exact dyadic evaluation used whenever the filter is inconclusive. This makes the sign exact for any finite double input, not just integers.
  • Initial simplex. p0 arbitrary → p1 farthest from p0 → p2 farthest from line p0p1 chosen among points that are exactly non‑collinear → p3 farthest from plane p0p1p2 chosen among points exactly non‑coplanar. If p2 fails the hull degrades to a segment; if p3 fails it degrades to a 2D convex polygon (doubled into two opposite facets so V−E+F=2 holds).
  • Horizon insertion. From p, visible faces are exactly orient3d(face, p) > 0. The horizon is the set of directed edges of visible faces whose neighbour is not visible; new faces are (u,v,p) using the horizon edge direction (proved outward), so no fragile interior point is needed for winding.
  • Facet merge. Coplanar edge‑adjacent triangles (checked by an exact orient3d == 0) are union‑find‑merged; the boundary is walked and exactly collinear boundary vertices are removed. Non‑extremal points that were transiently added now disappear from the facet boundary.
  • Invariants. checkInvariants verifies Euler, edge–manifoldness/opposite winding, strict convexity of each facet, exact plane membership, completeness (every input point on a facet plane is inside that facet polygon ⇔ no missed extreme vertex), the half‑space property, and outward orientation w.r.t. an exact dyadic interior point.

 *
 * Predicates (orient2d, orient3d, incircle, insphere):
 *   - stage 1: plain double evaluation + a conservative forward-error bound.
 *   - stage 2: exact dyadic (mantissa * 2^exp) BigInt arithmetic.
 *
 * Hull:
 *   - dedup of exact duplicates,
 *   - initial simplex rejects collinear / coplanar seeds,

const assert = require('assert');

/* =====================================================================
 * Exact dyadic arithmetic.  A value is {m: BigInt, e: int} = m * 2^e.
 * ===================================================================== */
const DV = new DataView(new ArrayBuffer(8));
const TWO52 = 1n << 52n;

function toDyadic(x) {
  if (!Number.isFinite(x)) throw new Error('non-finite coordinate: ' + x);
  if (x === 0) return { m: 0n, e: 0 };
  DV.setFloat64(0, x, false);
  const hi = DV.getUint32(0, false);
  const lo = DV.getUint32(4, false);
  const sign = (hi >>> 31) ? -1n : 1n;
  const expField = (hi >>> 20) & 0x7ff;
  const mant = (BigInt(hi & 0xfffff) << 32n) | BigInt(lo);
  if (expField === 0) return { m: sign * mant, e: -1074 }; // subnormal
  return { m: sign * (mant | TWO52), e: expField - 1023 - 52 };
}
function dSub(a, b) {
  const e = a.e < b.e ? a.e : b.e;
  return { m: (a.m << BigInt(a.e - e)) - (b.m << BigInt(b.e - e)), e };
}
function dAdd(a, b) {
  const e = a.e < b.e ? a.e : b.e;
  return { m: (a.m << BigInt(a.e - e)) + (b.m << BigInt(b.e - e)), e };
}
function dMul(a, b) { return { m: a.m * b.m, e: a.e + b.e }; }
function dNeg(a) { return { m: -a.m, e: a.e }; }
function dSign(a) { return a.m > 0n ? 1 : (a.m < 0n ? -1 : 0); }
function dIsZero(a) { return a.m === 0n; }
function dLift2(x, y) { return dAdd(dMul(x, x), dMul(y, y)); }
function dLift3(x, y, z) { return dAdd(dLift2(x, y), dMul(z, z)); }

function toDyadicPoint(p) {
  return { x: toDyadic(p.x), y: toDyadic(p.y), z: toDyadic(p.z) };
}

/* dyadic 3x3 determinant, row-major */
function det3Dy(a, b, c, d, e, f, g, h, i) {
  return dAdd(
    dSub(dMul(a, dSub(dMul(e, i), dMul(f, h))),
         dMul(b, dSub(dMul(d, i), dMul(f, g)))),
    dMul(c, dSub(dMul(d, h), dMul(e, g))));
}
/* dyadic 4x4 determinant, m[r][c] */
function det4Dy(m) {
  const a = m[0][0], b = m[0][1], c = m[0][2], d = m[0][3];
  const A = det3Dy(m[1][1], m[1][2], m[1][3], m[2][1], m[2][2], m[2][3], m[3][1], m[3][2], m[3][3]);
  const B = det3Dy(m[1][0], m[1][2], m[1][3], m[2][0], m[2][2], m[2][3], m[3][0], m[3][2], m[3][3]);
  const C = det3Dy(m[1][0], m[1][1], m[1][3], m[2][0], m[2][1], m[2][3], m[3][0], m[3][1], m[3][3]);
  const D = det3Dy(m[1][0], m[1][1], m[1][2], m[2][0], m[2][1], m[2][2], m[3][0], m[3][1], m[3][2]);
  return dSub(dAdd(dMul(a, A), dMul(c, C)), dAdd(dMul(b, B), dMul(d, D)));
}

 * Predicates

const EPS = Math.pow(2, -53);
const O2D_ERR = (3 + 16 * EPS) * EPS;
const O3D_ERR = (7 + 56 * EPS) * EPS;

/* ---------- orient2d ---------- */
function orient2dFloat(pa, pb, pc) {
  const ax = pa.x - pc.x, ay = pa.y - pc.y;
  const bx = pb.x - pc.x, by = pb.y - pc.y;
  return ax * by - ay * bx;
}
function orient2dPerm(pa, pb, pc) {
  const ax = pa.x - pc.x, ay = pa.y - pc.y;
  const bx = pb.x - pc.x, by = pb.y - pc.y;
  return Math.abs(ax * by) + Math.abs(ay * bx);
}
function orient2dExact(pa, pb, pc) {
  const ax = dSub(toDyadic(pa.x), toDyadic(pc.x));
  const ay = dSub(toDyadic(pa.y), toDyadic(pc.y));
  const bx = dSub(toDyadic(pb.x), toDyadic(pc.x));
  const by = dSub(toDyadic(pb.y), toDyadic(pc.y));
  return dSign(dSub(dMul(ax, by), dMul(ay, bx)));
}
function orient2d(pa, pb, pc) {
  const det = orient2dFloat(pa, pb, pc);
  const err = O2D_ERR * orient2dPerm(pa, pb, pc);
  if (det > err || -det > err) return det > 0 ? 1 : -1;
  return orient2dExact(pa, pb, pc);
}

/* ---------- orient3d ---------- */
function orient3dFloat(pa, pb, pc, pd) {
  const adx = pa.x - pd.x, ady = pa.y - pd.y, adz = pa.z - pd.z;
  const bdx = pb.x - pd.x, bdy = pb.y - pd.y, bdz = pb.z - pd.z;
  const cdx = pc.x - pd.x, cdy = pc.y - pd.y, cdz = pc.z - pd.z;
  return adx * (bdy * cdz - bdz * cdy)
       - ady * (bdx * cdz - bdz * cdx)
       + adz * (bdx * cdy - bdy * cdx);
}
function orient3dPerm(pa, pb, pc, pd) {
  const adx = Math.abs(pa.x - pd.x), ady = Math.abs(pa.y - pd.y), adz = Math.abs(pa.z - pd.z);
  const bdx = Math.abs(pb.x - pd.x), bdy = Math.abs(pb.y - pd.y), bdz = Math.abs(pb.z - pd.z);
  const cdx = Math.abs(pc.x - pd.x), cdy = Math.abs(pc.y - pd.y), cdz = Math.abs(pc.z - pd.z);
  return adx * (bdy * cdz + bdz * cdy)
       + ady * (bdx * cdz + bdz * cdx)
       + adz * (bdx * cdy + bdy * cdx);
}
function orient3dDyadicExact(A, B, C, D) {
  const adx = dSub(A.x, D.x), ady = dSub(A.y, D.y), adz = dSub(A.z, D.z);
  const bdx = dSub(B.x, D.x), bdy = dSub(B.y, D.y), bdz = dSub(B.z, D.z);
  const cdx = dSub(C.x, D.x), cdy = dSub(C.y, D.y), cdz = dSub(C.z, D.z);
  return dSign(dSub(
    dAdd(dMul(adx, dSub(dMul(bdy, cdz), dMul(bdz, cdy))),
         dMul(adz, dSub(dMul(bdx, cdy), dMul(bdy, cdx)))),
    dMul(ady, dSub(dMul(bdx, cdz), dMul(bdz, cdx)))));
}
function orient3dExact(pa, pb, pc, pd) {
  return orient3dDyadicExact(toDyadicPoint(pa), toDyadicPoint(pb), toDyadicPoint(pc), toDyadicPoint(pd));
}
function orient3d(pa, pb, pc, pd) {
  const det = orient3dFloat(pa, pb, pc, pd);
  const err = O3D_ERR * orient3dPerm(pa, pb, pc, pd);
  if (det > err || -det > err) return det > 0 ? 1 : -1;
  return orient3dExact(pa, pb, pc, pd);
}

/* ---------- incircle ---------- */
function incircleExact(pa, pb, pc, pd) {
  const ax = dSub(toDyadic(pa.x), toDyadic(pd.x)), ay = dSub(toDyadic(pa.y), toDyadic(pd.y));
  const bx = dSub(toDyadic(pb.x), toDyadic(pd.x)), by = dSub(toDyadic(pb.y), toDyadic(pd.y));
  const cx = dSub(toDyadic(pc.x), toDyadic(pd.x)), cy = dSub(toDyadic(pc.y), toDyadic(pd.y));
  return dSign(det3Dy(ax, ay, dLift2(ax, ay), bx, by, dLift2(bx, by), cx, cy, dLift2(cx, cy)));
}
function incircle(pa, pb, pc, pd) {
  const ax = pa.x - pd.x, ay = pa.y - pd.y, al = ax * ax + ay * ay;
  const bx = pb.x - pd.x, by = pb.y - pd.y, bl = bx * bx + by * by;
  const cx = pc.x - pd.x, cy = pc.y - pd.y, cl = cx * cx + cy * cy;
  const det = ax * (by * cl - bl * cy) - ay * (bx * cl - bl * cx) + al * (bx * cy - by * cx);
  const s = Math.abs(ax) + Math.abs(ay) + Math.abs(al) + Math.abs(bx) + Math.abs(by) + Math.abs(bl)
          + Math.abs(cx) + Math.abs(cy) + Math.abs(cl);
  const err = 1e-13 * s * s * s;
  if (det > err || -det > err) return det > 0 ? 1 : -1;
  return incircleExact(pa, pb, pc, pd);
}

/* ---------- insphere ---------- */
function insphereExact(pa, pb, pc, pd, pe) {
  const A = toDyadicPoint(pa), B = toDyadicPoint(pb), C = toDyadicPoint(pc), D = toDyadicPoint(pd), E = toDyadicPoint(pe);
  const ax = dSub(A.x, E.x), ay = dSub(A.y, E.y), az = dSub(A.z, E.z);
  const bx = dSub(B.x, E.x), by = dSub(B.y, E.y), bz = dSub(B.z, E.z);
  const cx = dSub(C.x, E.x), cy = dSub(C.y, E.y), cz = dSub(C.z, E.z);
  const dx = dSub(D.x, E.x), dy = dSub(D.y, E.y), dz = dSub(D.z, E.z);
  return dSign(det4Dy([
    [ax, ay, az, dLift3(ax, ay, az)],
    [bx, by, bz, dLift3(bx, by, bz)],
    [cx, cy, cz, dLift3(cx, cy, cz)],
    [dx, dy, dz, dLift3(dx, dy, dz)],
  ]));
}
function insphere(pa, pb, pc, pd, pe) {
  const mk = (p) => {
    const x = p.x - pe.x, y = p.y - pe.y, z = p.z - pe.z;
    return [x, y, z, x * x + y * y + z * z];
  };
  const m = [mk(pa), mk(pb), mk(pc), mk(pd)];
  const det = det4F(m);
  let s = 0;
  for (let i = 0; i < 4; i++) for (let j = 0; j < 4; j++) s += Math.abs(m[i][j]);
  const err = 1e-13 * s * s * s * s;
  if (det > err || -det > err) return det > 0 ? 1 : -1;
  return insphereExact(pa, pb, pc, pd, pe);
}
function det3F(a, b, c, d, e, f, g, h, i) {
  return a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g);
}
function det4F(m) {
  const a = m[0][0], b = m[0][1], c = m[0][2], d = m[0][3];
  const A = det3F(m[1][1], m[1][2], m[1][3], m[2][1], m[2][2], m[2][3], m[3][1], m[3][2], m[3][3]);
  const B = det3F(m[1][0], m[1][2], m[1][3], m[2][0], m[2][2], m[2][3], m[3][0], m[3][2], m[3][3]);
  const C = det3F(m[1][0], m[1][1], m[1][3], m[2][0], m[2][1], m[2][3], m[3][0], m[3][1], m[3][3]);
  const D = det3F(m[1][0], m[1][1], m[1][2], m[2][0], m[2][1], m[2][2], m[3][0], m[3][1], m[3][2]);
  return a * A - b * B + c * C - d * D;
}

function dist2(a, b) {
  const dx = a.x - b.x, dy = a.y - b.y, dz = a.z - b.z;

}
function crossMag2(a, b, c) {
  const ux = b.x - a.x, uy = b.y - a.y, uz = b.z - a.z;
  const vx = c.x - a.x, vy = c.y - a.y, vz = c.z - a.z;
  const nx = uy * vz - uz * vy, ny = uz * vx - ux * vz, nz = ux * vy - uy * vx;
  return nx * nx + ny * ny + nz * nz;
}
function collinearExact(a, b, c) {
  const ux = dSub(toDyadic(b.x), toDyadic(a.x));
  const uy = dSub(toDyadic(b.y), toDyadic(a.y));
  const uz = dSub(toDyadic(b.z), toDyadic(a.z));
  const vx = dSub(toDyadic(c.x), toDyadic(a.x));
  const vy = dSub(toDyadic(c.y), toDyadic(a.y));
  const vz = dSub(toDyadic(c.z), toDyadic(a.z));
  const cx = dSub(dMul(uy, vz), dMul(uz, vy));
  const cy = dSub(dMul(uz, vx), dMul(ux, vz));
  const cz = dSub(dMul(ux, vy), dMul(uy, vx));
  return dIsZero(cx) && dIsZero(cy) && dIsZero(cz);
}
function faceNonDegenerate(a, b, c) { return !collinearExact(a, b, c); }

/* dyadic plane normal  n = (b-a) x (c-a)  */
function planeNormalDyadic(a, b, c) {
  const ux = dSub(toDyadic(b.x), toDyadic(a.x));
  const uy = dSub(toDyadic(b.y), toDyadic(a.y));
  const uz = dSub(toDyadic(b.z), toDyadic(a.z));
  const vx = dSub(toDyadic(c.x), toDyadic(a.x));
  const vy = dSub(toDyadic(c.y), toDyadic(a.y));
  const vz = dSub(toDyadic(c.z), toDyadic(a.z));
  return {
    x: dSub(dMul(uy, vz), dMul(uz, vy)),
    y: dSub(dMul(uz, vx), dMul(ux, vz)),
    z: dSub(dMul(ux, vy), dMul(uy, vx)),
  };
}

/* exact sign of dot(N, cross(b-a, c-a)) : in-plane orientation */
function inPlaneOrient(pa, pb, pc, N) {
  const ux = dSub(toDyadic(pb.x), toDyadic(pa.x));
  const uy = dSub(toDyadic(pb.y), toDyadic(pa.y));
  const uz = dSub(toDyadic(pb.z), toDyadic(pa.z));
  const vx = dSub(toDyadic(pc.x), toDyadic(pa.x));
  const vy = dSub(toDyadic(pc.y), toDyadic(pa.y));
  const vz = dSub(toDyadic(pc.z), toDyadic(pa.z));
  const cx = dSub(dMul(uy, vz), dMul(uz, vy));
  const cy = dSub(dMul(uz, vx), dMul(ux, vz));
  const cz = dSub(dMul(ux, vy), dMul(uy, vx));
  return dSign(dAdd(dAdd(dMul(N.x, cx), dMul(N.y, cy)), dMul(N.z, cz)));
}

/*
 * Merge coplanar, edge-adjacent triangles into maximal polygonal facets.
 * Two faces are coplanar iff the opposite vertex of one lies on the exact
 * plane of the other (they already share an edge).
 */
function mergeFacets(tris, pts) {
  const n = tris.length;
  const parent = new Array(n);
  for (let i = 0; i < n; i++) parent[i] = i;
  const find = (x) => { while (parent[x] !== x) { parent[x] = parent[parent[x]]; x = parent[x]; } return x; };
  const union = (a, b) => { a = find(a); b = find(b); if (a !== b) parent[a] = b; };

  const emap = new Map();
  tris.forEach((f, idx) => {
    const [a, b, c] = f.v;
    for (const [u, v] of [[a, b], [b, c], [c, a]]) {
      const k = u < v ? u + '_' + v : v + '_' + u;
      let arr = emap.get(k);
      if (!arr) { arr = []; emap.set(k, arr); }
      arr.push(idx);
    }
  });
  for (const arr of emap.values()) {
    if (arr.length !== 2) continue;
    const f = tris[arr[0]], g = tris[arr[1]];
    const setF = new Set(f.v);
    const opp = g.v.find((v) => !setF.has(v));
    if (opp === undefined) continue;
    if (orient3d(pts[f.v[0]], pts[f.v[1]], pts[f.v[2]], pts[opp]) === 0) union(arr[0], arr[1]);
  }

  const groups = new Map();
  tris.forEach((f, idx) => {
    const r = find(idx);
    let g = groups.get(r);
    if (!g) { g = []; groups.set(r, g); }
    g.push(idx);
  });

  const facets = [];
  for (const idxs of groups.values()) {
    const dirSet = new Set();
    for (const idx of idxs) {
      const [a, b, c] = tris[idx].v;
      dirSet.add(a + '_' + b); dirSet.add(b + '_' + c); dirSet.add(c + '_' + a);
    }
    const boundary = [];
    for (const e of dirSet) {
      const [u, v] = e.split('_').map(Number);
      if (!dirSet.has(v + '_' + u)) boundary.push([u, v]);
    }
    if (boundary.length === 0) continue;
    const next = new Map();
    for (const [u, v] of boundary) next.set(u, v);
    const poly = [];
    let cur = boundary[0][0];
    const seen = new Set();
    while (cur !== undefined && !seen.has(cur)) {
      seen.add(cur); poly.push(cur); cur = next.get(cur);
    }
    // remove exactly collinear boundary vertices
    let changed = true;
    while (changed && poly.length > 3) {
      changed = false;
      for (let i = 0; i < poly.length; i++) {
        const a = poly[(i - 1 + poly.length) % poly.length];
        const b = poly[i];
        const c = poly[(i + 1) % poly.length];
        if (collinearExact(pts[a], pts[b], pts[c])) { poly.splice(i, 1); changed = true; break; }
      }
    }
    facets.push({ v: poly });
  }
  return facets;
}

function dedupe(points) {
  const map = new Map();
  const pts = [];
  const orig = [];
  for (let i = 0; i < points.length; i++) {
    const p = points[i];
    const key = p.x + ',' + p.y + ',' + p.z;
    if (!map.has(key)) { map.set(key, pts.length); pts.push(p); orig.push(i); }
  }
  return { pts, orig };
}

/* =====================================================================
 * 2D convex hull (used when the whole set is coplanar).
 * Returns { poly: [indices into pts, CCW wrt +normal] }.
 * ===================================================================== */
function buildHull2D(pts, seed) {
  const [i0, i1, i2] = seed;
  const P0 = pts[i0], P1 = pts[i1], P2 = pts[i2];
  const ux = P1.x - P0.x, uy = P1.y - P0.y, uz = P1.z - P0.z;
  const vx = P2.x - P0.x, vy = P2.y - P0.y, vz = P2.z - P0.z;
  const nx = uy * vz - uz * vy;
  const ny = uz * vx - ux * vz;
  const nz = ux * vy - uy * vx;
  const ax = Math.abs(nx), ay = Math.abs(ny), az = Math.abs(nz);
  let drop = 2;
  if (ax >= ay && ax >= az) drop = 0;
  else if (ay >= az) drop = 1;

  // projected 2D coordinates (exact copies of two coordinates)
  const proj = [];
  for (let i = 0; i < pts.length; i++) {
    const p = pts[i];
    proj.push(drop === 0 ? { x: p.y, y: p.z } : drop === 1 ? { x: p.x, y: p.z } : { x: p.x, y: p.y });
  }
  const idx = [];
  for (let i = 0; i < pts.length; i++) idx.push(i);
  idx.sort((i, j) => (proj[i].x - proj[j].x) || (proj[i].y - proj[j].y));

  const lower = [];
  for (const i of idx) {
    while (lower.length >= 2 &&
           orient2d(proj[lower[lower.length - 2]], proj[lower[lower.length - 1]], proj[i]) <= 0) lower.pop();
    lower.push(i);
  }
  const upper = [];
  for (let k = idx.length - 1; k >= 0; k--) {
    const i = idx[k];
    while (upper.length >= 2 &&
           orient2d(proj[upper[upper.length - 2]], proj[upper[upper.length - 1]], proj[i]) <= 0) upper.pop();
    upper.push(i);
  }
  lower.pop(); upper.pop();
  let poly = lower.concat(upper);
  const nDrop = drop === 0 ? nx : drop === 1 ? ny : nz;
  if (nDrop < 0) poly = poly.slice().reverse();
  return { poly };
}

/* =====================================================================
 * 3D convex hull (incremental, horizon based).
 * ===================================================================== */
function buildHull(rawPoints) {
  const { pts, orig } = dedupe(rawPoints);
  const n = pts.length;
  const base = (dim) => ({
    dimension: dim, points: rawPoints, uniquePoints: pts, origIndex: orig,
    vertices: [], faces: [], facets: [], interior: null, polygon: null,
  });

  if (n === 0) return base(-1);
  if (n === 1) { const r = base(0); r.vertices = [0]; return r; }

  // --- p0: arbitrary, p1: farthest from p0 ---
  const p0 = 0;
  let p1 = -1, bestD = -1;
  for (let i = 0; i < n; i++) {
    if (i === p0) continue;
    const d = dist2(pts[p0], pts[i]);
    if (d > bestD) { bestD = d; p1 = i; }
  }
  if (p1 < 0 || bestD <= 0) throw new Error('unreachable: identical points survived dedup');

  // --- p2: farthest from line p0p1 among exactly non-collinear points ---
  let p2 = -1, bestA = -1;
  for (let i = 0; i < n; i++) {
    if (i === p0 || i === p1) continue;
    if (collinearExact(pts[p0], pts[p1], pts[i])) continue;
    const a = crossMag2(pts[p0], pts[p1], pts[i]);
    if (a > bestA) { bestA = a; p2 = i; }
  }
  if (p2 < 0) {
    // all points collinear: the hull is the segment between the true extremes.
    // Ordering along the dominant component of the line direction is exact.
    const r = base(1);
    const dx = Math.abs(pts[p1].x - pts[p0].x);
    const dy = Math.abs(pts[p1].y - pts[p0].y);
    const dz = Math.abs(pts[p1].z - pts[p0].z);
    const k = dx >= dy && dx >= dz ? 'x' : (dy >= dz ? 'y' : 'z');
    let imin = 0, imax = 0;
    for (let i = 1; i < n; i++) {
      if (pts[i][k] < pts[imin][k]) imin = i;
      if (pts[i][k] > pts[imax][k]) imax = i;
    }
    r.vertices = [imin, imax];
    return r;
  }

  // --- p3: farthest from plane p0p1p2, or coplanar ---
  let p3 = -1, bestV = -1;
  for (let i = 0; i < n; i++) {
    if (i === p0 || i === p1 || i === p2) continue;
    if (orient3d(pts[p0], pts[p1], pts[p2], pts[i]) === 0) continue;
    const v = Math.abs(orient3dFloat(pts[p0], pts[p1], pts[p2], pts[i]));
    if (v > bestV) { bestV = v; p3 = i; }
  }
  if (p3 < 0) {
    // all coplanar: 2D polygon hull, doubled into two facets (front/back)
    const { poly } = buildHull2D(pts, [p0, p1, p2]);
    if (poly.length < 3) {
      // degenerate (collinear) safety net
      const r = base(1);
      r.vertices = [poly[0], poly[poly.length - 1]];
      return r;
    }
    const r = base(2);
    r.vertices = poly.slice();
    r.polygon = poly;
    r.facets = [{ v: poly.slice(), n: [0, 0, 0] }, { v: poly.slice().reverse(), n: [0, 0, 0] }];
    r.faces = r.facets.slice();
    return r;
  }

  // --- initial tetrahedron, each face oriented outward (opposite vertex behind) ---
  const tet = [p0, p1, p2, p3];
  const faces = [];
  const triples = [[p0, p1, p2], [p0, p1, p3], [p0, p2, p3], [p1, p2, p3]];
  for (const [a, b, c] of triples) {
    const opp = tet.find((v) => v !== a && v !== b && v !== c);
    if (orient3d(pts[a], pts[b], pts[c], pts[opp]) > 0) faces.push({ v: [a, c, b], alive: true });
    else faces.push({ v: [a, b, c], alive: true });
  }

  // exact interior point = centroid of the initial simplex (dyadic exact)
  const interior = (() => {
    const P = tet.map((i) => toDyadicPoint(pts[i]));
    const q = (k) => dMul(dAdd(dAdd(P[0][k], P[1][k]), dAdd(P[2][k], P[3][k])), { m: 1n, e: -2 });
    return { x: q('x'), y: q('y'), z: q('z') };
  })();

  // --- incremental insertion ---
  const inTet = new Set(tet);
  for (let pi = 0; pi < n; pi++) {
    if (inTet.has(pi)) continue;
    const P = pts[pi];

    // visible faces: strictly outside (orient3d > 0)
    const vis = [];
    for (const f of faces) {
      if (f.alive && orient3d(pts[f.v[0]], pts[f.v[1]], pts[f.v[2]], P) > 0) vis.push(f);
    }
    if (vis.length === 0) continue; // inside, on the boundary, or coplanar

    // adjacency of currently alive faces
    const emap = new Map();
    for (const f of faces) {
      if (!f.alive) continue;
      const [a, b, c] = f.v;
      const push = (u, v) => {
        const k = u < v ? u + '_' + v : v + '_' + u;
        let arr = emap.get(k);
        if (!arr) { arr = []; emap.set(k, arr); }
        arr.push(f);
      };
      push(a, b); push(b, c); push(c, a);
    }
    const visSet = new Set(vis);

    // horizon = edge of a visible face whose neighbour is not visible
    const horizon = [];
    const horizonSeen = new Set();
    for (const f of vis) {
      const [a, b, c] = f.v;
      for (const [u, v] of [[a, b], [b, c], [c, a]]) {
        const k = u < v ? u + '_' + v : v + '_' + u;
        const arr = emap.get(k) || [];
        const other = arr.find((g) => g !== f);
        if (!other || !visSet.has(other)) {
          if (!horizonSeen.has(k)) { horizonSeen.add(k); horizon.push([u, v]); }
        }
      }
    }

    for (const f of vis) f.alive = false;
    for (const [u, v] of horizon) faces.push({ v: [u, v, pi], alive: true });
  }

  const alive = faces.filter((f) => f.alive);
  const tris = alive.map((f) => ({ v: f.v }));
  const facets = mergeFacets(tris, pts);
  const used = new Set();
  for (const f of facets) for (const v of f.v) used.add(v);

  const r = base(3);
  r.vertices = [...used].sort((a, b) => a - b);
  r.faces = tris;
  r.facets = facets;
  r.interior = interior;
  return r;
}

/* =====================================================================
 * Invariant checker
 * ===================================================================== */
function checkInvariants(hull) {
  const pts = hull.uniquePoints;
  const report = { ok: true, errors: [], stats: {} };

  if (hull.dimension === 0) {
    report.stats = { dimension: 0, V: hull.vertices.length, E: 0, F: 0 };
    if (pts.length !== 1 || hull.vertices.length !== 1) { report.ok = false; report.errors.push('dim0 hull should contain exactly one point'); }
    return report;
  }
  if (hull.dimension === 1) {
    report.stats = { dimension: 1, V: hull.vertices.length, E: 1, F: 0 };
    if (hull.vertices.length !== 2) { report.ok = false; report.errors.push('dim1 hull should have two vertices'); return report; }
    const [ia, ib] = hull.vertices;
    const A = pts[ia], B = pts[ib];
    const dx = Math.abs(B.x - A.x), dy = Math.abs(B.y - A.y), dz = Math.abs(B.z - A.z);
    const k = dx >= dy && dx >= dz ? 'x' : (dy >= dz ? 'y' : 'z');
    const lo = Math.min(A[k], B[k]), hi = Math.max(A[k], B[k]);
    for (const q of pts) {
      if (!collinearExact(A, B, q)) { report.ok = false; report.errors.push('dim1 point not collinear'); break; }
      if (q[k] < lo || q[k] > hi) { report.ok = false; report.errors.push('dim1 point outside segment'); break; }
    }
    return report;
  }

  if (hull.dimension === 2) {
    const poly = hull.polygon;
    const V = poly.length, E = poly.length, F = 2;
    report.stats = { dimension: 2, V, E, F, facets: 2 };
    if (V - E + F !== 2) { report.ok = false; report.errors.push(`Euler V-E+F=${V - E + F} != 2`); }
    // winding: the two facets must be reverse of each other
    const a = hull.facets[0].v, b = hull.facets[1].v;
    if (a.length !== b.length || a.some((v, i) => v !== b[b.length - 1 - i])) {
      report.ok = false; report.errors.push('dim2 facets are not opposite windings');
    }
    if (poly.length < 3) { report.ok = false; report.errors.push('degenerate 2D hull polygon'); return report; }
    const N = planeNormalDyadic(pts[poly[0]], pts[poly[1]], pts[poly[2]]);
    // convexity of the boundary, and no exactly-collinear corner
    let turnSign = 0, convex = true;
    for (let i = 0; i < poly.length; i++) {
      const p = pts[poly[i]], q = pts[poly[(i + 1) % poly.length]], r = pts[poly[(i + 2) % poly.length]];
      const s = inPlaneOrient(p, q, r, N);
      if (s === 0) { convex = false; break; }
      if (turnSign === 0) turnSign = s; else if (s !== turnSign) { convex = false; break; }
    }
    if (!convex) { report.ok = false; report.errors.push('2D hull polygon is not strictly convex'); }
    // every input point lies on the plane and inside/on the polygon
    for (let qi = 0; qi < pts.length; qi++) {
      if (orient3d(pts[poly[0]], pts[poly[1]], pts[poly[2]], pts[qi]) !== 0) {
        report.ok = false; report.errors.push(`dim2 point ${qi} not coplanar`);
        continue;
      }
      for (let i = 0; i < poly.length; i++) {
        const p = pts[poly[i]], r = pts[poly[(i + 1) % poly.length]];
        if (turnSign * inPlaneOrient(p, r, pts[qi], N) < 0) {
          report.ok = false; report.errors.push(`dim2 point ${qi} outside polygon`); break;
        }
      }
    }
    return report;
  }

  // dimension 3: check on the merged polygonal facets
  const facets = hull.facets;
  const V = hull.vertices.length;
  const F = facets.length;
  const edgeCount = new Map();
  for (const f of facets) {
    const vs = f.v;
    for (let i = 0; i < vs.length; i++) {
      const u = vs[i], v = vs[(i + 1) % vs.length];
      const k = u < v ? u + '_' + v : v + '_' + u;
      let rec = edgeCount.get(k);
      if (!rec) { rec = { pos: 0, neg: 0 }; edgeCount.set(k, rec); }
      if (u < v) rec.pos++; else rec.neg++;
    }
  }
  const E = edgeCount.size;
  report.stats = { dimension: 3, V, E, F, facets: F };

  if (V - E + F !== 2) { report.ok = false; report.errors.push(`Euler V-E+F=${V - E + F} != 2 (V=${V},E=${E},F=${F})`); }
  for (const [k, rec] of edgeCount) {
    if (rec.pos !== 1 || rec.neg !== 1) {
      report.ok = false;
      report.errors.push(`edge ${k} has pos=${rec.pos}, neg=${rec.neg} (non-manifold / inconsistent winding)`);
    }
  }

  const vertexSet = new Set(hull.vertices);
  const seenVertex = new Set();
  for (const f of facets) {
    const vs = f.v;
    if (vs.length < 3) {
      report.ok = false; report.errors.push(`facet with < 3 vertices: ${vs}`);
      continue;
    }
    const [a, b, c] = vs;
    if (!faceNonDegenerate(pts[a], pts[b], pts[c])) {
      report.ok = false; report.errors.push(`degenerate facet ${vs}`);
      continue;
    }
    const N = planeNormalDyadic(pts[a], pts[b], pts[c]);
    // (i) convexity: every strict turn has the same in-plane sign
    let turnSign = 0, convex = true;
    for (let i = 0; i < vs.length; i++) {
      const p = pts[vs[i]], q = pts[vs[(i + 1) % vs.length]], r = pts[vs[(i + 2) % vs.length]];
      const s = inPlaneOrient(p, q, r, N);
      if (s === 0) { convex = false; break; }
      if (turnSign === 0) turnSign = s;
      else if (s !== turnSign) { convex = false; break; }
    }
    if (!convex) { report.ok = false; report.errors.push(`facet not convex / has collinear corner ${vs}`); }
    // (ii) all facet vertices lie exactly on the facet plane
    for (const v of vs) {
      if (orient3d(pts[a], pts[b], pts[c], pts[v]) !== 0) {
        report.ok = false; report.errors.push(`facet vertex ${v} off its plane`);
      }
      if (!vertexSet.has(v)) { report.ok = false; report.errors.push(`facet uses vertex ${v} not in hull.vertices`); }
      seenVertex.add(v);
    }
    // (iii) completeness: any input point on this plane must be inside/on the polygon
    for (let qi = 0; qi < pts.length; qi++) {
      const q = pts[qi];
      if (orient3d(pts[a], pts[b], pts[c], q) !== 0) continue;
      for (let i = 0; i < vs.length; i++) {
        const p = pts[vs[i]], r = pts[vs[(i + 1) % vs.length]];
        const s = inPlaneOrient(p, r, q, N);
        if (turnSign * s < 0) {
          report.ok = false;
          report.errors.push(`point ${qi} on facet plane is outside facet ${vs} (missing hull vertex)`);
          break;
        }
      }
    }
  }
  for (const v of vertexSet) if (!seenVertex.has(v)) { report.ok = false; report.errors.push(`hull vertex ${v} is not a corner of any facet`); }

  // (iv) no input point strictly outside any facet plane
  for (const f of facets) {
    const A = pts[f.v[0]], B = pts[f.v[1]], C = pts[f.v[2]];
    for (let qi = 0; qi < pts.length; qi++) {
      if (orient3d(A, B, C, pts[qi]) > 0) {
        report.ok = false;
        report.errors.push(`point ${qi} strictly outside facet ${f.v}`);
        break;
      }
    }
  }
  // (v) every facet normal points away from the exact interior point
  for (const f of facets) {
    const A = toDyadicPoint(pts[f.v[0]]), B = toDyadicPoint(pts[f.v[1]]), C = toDyadicPoint(pts[f.v[2]]);
    if (orient3dDyadicExact(A, B, C, hull.interior) >= 0) {
      report.ok = false;
      report.errors.push(`facet ${f.v} winding does not face away from interior`);
    }
  }
  return report;
}

function mulberry32(a) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function testPredicate100k() {
  const rnd = mulberry32(1234567);
  const RI = (lo, hi) => Math.floor(rnd() * (hi - lo + 1)) + lo;
  let naiveBad = 0, exactBad = 0, filtBad = 0, fallbacks = 0;
  let naiveBad2 = 0, exactBad2 = 0, filtBad2 = 0, fallbacks2 = 0;
  const N = 100000;

  /* --- 100k near-collinear 2D triples (orient2d) --- */
  for (let t = 0; t < N; t++) {
    // Large unimodular triple: det((a-c),(b-c)) = +/-1 while the coordinate
    // products are ~k^2 >> 2^53, so raw float evaluation loses the sign.
    const k = RI(50000000, 200000000);
    let a = { x: k, y: k + 1 };
    let b = { x: k - 1, y: k };
    const c = { x: 0, y: 0 };
    if (rnd() < 0.5) { const tmp = a; a = b; b = tmp; }
    const ref = (() => {
      const ax = BigInt(a.x) - BigInt(c.x), ay = BigInt(a.y) - BigInt(c.y);
      const bx = BigInt(b.x) - BigInt(c.x), by = BigInt(b.y) - BigInt(c.y);
      const det = ax * by - ay * bx;
      return det > 0n ? 1 : (det < 0n ? -1 : 0);
    })();
    const fdet = orient2dFloat(a, b, c);
    const naive = fdet > 0 ? 1 : (fdet < 0 ? -1 : 0);
    if (!(Math.abs(fdet) > O2D_ERR * orient2dPerm(a, b, c))) fallbacks2++;
    if (naive !== ref) naiveBad2++;
    if (orient2dExact(a, b, c) !== ref) exactBad2++;
    if (orient2d(a, b, c) !== ref) filtBad2++;
  }

  /* --- 100k near-coplanar 3D tetrahedra (orient3d) --- */
  for (let t = 0; t < N; t++) {
    // a, b, c random with large integer coordinates; d is nearly on their plane.
    const a = { x: RI(-1000000, 1000000), y: RI(-1000000, 1000000), z: RI(-1000000, 1000000) };
    const b = { x: RI(-1000000, 1000000), y: RI(-1000000, 1000000), z: RI(-1000000, 1000000) };
    const c = { x: RI(-1000000, 1000000), y: RI(-1000000, 1000000), z: RI(-1000000, 1000000) };
    const ux = b.x - a.x, uy = b.y - a.y, uz = b.z - a.z;
    const vx = c.x - a.x, vy = c.y - a.y, vz = c.z - a.z;
    // d = a + u + v + tiny perturbation -> exactly on the plane if perturbation is 0
    const px = RI(-2, 2), py = RI(-2, 2), pz = RI(-2, 2);
    const d = { x: a.x + ux + vx + px, y: a.y + uy + vy + py, z: a.z + uz + vz + pz };

    const ref = (() => {
      const ax = BigInt(a.x) - BigInt(d.x), ay = BigInt(a.y) - BigInt(d.y), az = BigInt(a.z) - BigInt(d.z);
      const bx = BigInt(b.x) - BigInt(d.x), by = BigInt(b.y) - BigInt(d.y), bz = BigInt(b.z) - BigInt(d.z);
      const cx = BigInt(c.x) - BigInt(d.x), cy = BigInt(c.y) - BigInt(d.y), cz = BigInt(c.z) - BigInt(d.z);
      const det = ax * (by * cz - bz * cy) - ay * (bx * cz - bz * cx) + az * (bx * cy - by * cx);
      return det > 0n ? 1 : (det < 0n ? -1 : 0);
    })();

    const fdet = orient3dFloat(a, b, c, d);
    const naive = fdet > 0 ? 1 : (fdet < 0 ? -1 : 0);
    const perm = orient3dPerm(a, b, c, d);
    if (!(Math.abs(fdet) > O3D_ERR * perm)) fallbacks++;
    const filt = orient3d(a, b, c, d);
    const ex = orient3dExact(a, b, c, d);

    if (naive !== ref) naiveBad++;
    if (ex !== ref) exactBad++;
    if (filt !== ref) filtBad++;
  }
  return { N, naiveBad, exactBad, filtBad, fallbacks, naiveBad2, exactBad2, filtBad2, fallbacks2 };
}

function ptsOnSphere(n, R, rnd) {
  const out = [];
  for (let i = 0; i < n; i++) {
    // deterministic spiral on the sphere
    const y = 1 - (2 * i + 1) / n;
    const r = Math.sqrt(Math.max(0, 1 - y * y));
    const th = i * 2.39996322972865332;
    out.push({ x: R * Math.cos(th) * r, y: R * y, z: R * Math.sin(th) * r });
  }
  return out;
}

function big3(a, b, c, d, e, f, g, h, i) {

}
function big4(m) {
  const a = m[0][0], b = m[0][1], c = m[0][2], d = m[0][3];
  const A = big3(m[1][1], m[1][2], m[1][3], m[2][1], m[2][2], m[2][3], m[3][1], m[3][2], m[3][3]);
  const B = big3(m[1][0], m[1][2], m[1][3], m[2][0], m[2][2], m[2][3], m[3][0], m[3][2], m[3][3]);
  const C = big3(m[1][0], m[1][1], m[1][3], m[2][0], m[2][1], m[2][3], m[3][0], m[3][1], m[3][3]);
  const D = big3(m[1][0], m[1][1], m[1][2], m[2][0], m[2][1], m[2][2], m[3][0], m[3][1], m[3][2]);

const sgn = (d) => (d > 0n ? 1 : d < 0n ? -1 : 0);

function incircleRef(a, b, c, d) {
  const ax = BigInt(a.x) - BigInt(d.x), ay = BigInt(a.y) - BigInt(d.y);
  const bx = BigInt(b.x) - BigInt(d.x), by = BigInt(b.y) - BigInt(d.y);
  const cx = BigInt(c.x) - BigInt(d.x), cy = BigInt(c.y) - BigInt(d.y);
  const al = ax * ax + ay * ay, bl = bx * bx + by * by, cl = cx * cx + cy * cy;
  return sgn(big3(ax, ay, al, bx, by, bl, cx, cy, cl));
}
function insphereRef(a, b, c, d, e) {
  const mk = (p) => {
    const x = BigInt(p.x) - BigInt(e.x), y = BigInt(p.y) - BigInt(e.y), z = BigInt(p.z) - BigInt(e.z);
    return [x, y, z, x * x + y * y + z * z];
  };
  return sgn(big4([mk(a), mk(b), mk(c), mk(d)]));
}

/* ---------- brute force boundary points (all points on some supporting plane) ---------- */
function bruteBoundary(pts) {
  const n = pts.length;
  const bd = new Set();
  for (let i = 0; i < n; i++) for (let j = i + 1; j < n; j++) for (let k = j + 1; k < n; k++) {
    const A = pts[i], B = pts[j], C = pts[k];
    if (H.collinearExact(A, B, C)) continue;
    let pos = false, neg = false;
    for (let q = 0; q < n; q++) {
      const s = H.orient3d(A, B, C, pts[q]);
      if (s > 0) pos = true; else if (s < 0) neg = true;
      if (pos && neg) break;
    }
    if (!(pos && neg)) { bd.add(i); bd.add(j); bd.add(k); }
  }
  return bd;
}
