Answer

ESM Top-Level-Await Deadlock Detection

Root cause

Top-level await turns ESM evaluation into an asynchronous dependency graph. A cyclic import is not automatically a deadlock:

  • If module A imports B, evaluation is depth-first. When B imports A again, the spec sees A is already evaluating. If A has not started its body, B runs first and the cycle unwinds. This is an evaluation-order escape.
  • If the target of the cycle has started an async body, every importer of that module must wait for it to finish ([[PendingAsyncDependencies]]). A cycle of such waits is a genuine deadlock.
  • await import("./x.js") adds a hard wait on x's full evaluation, even if x already statically imports back.
  • Awaited live bindings are only ready after their bindings are initialized. Re-export chains and hoisted function / var bindings must be modelled.
  • Cycles are broken by awaits on already-settled values (await null, await Promise.resolve(), non-thenables), by sync modules, and by static back-edges where the ancestor body has not started.

The detector below builds the reachable graph, extracts top-level awaits and import edges, symbolically simulates ESM evaluation, and reports the shortest wait-for cycle with source lines.

Fix

Save this as tla-cycle-detector.mjs and run it with Node 20+ (no npm dependencies; it uses Node's bundled Acorn, and falls back to installed acorn if present).

#!/usr/bin/env node
// Static ESM top-level-await deadlock detector. Node 20+, no npm deps.
// Usage: node tla-cycle-detector.mjs <entry.mjs> [--json]
import { readFileSync } from 'node:fs';
import { pathToFileURL, fileURLToPath } from 'node:url';
import { resolve as resolvePath, relative } from 'node:path';
import { createRequire } from 'node:module';

function acornLoad() {
  const req = createRequire(import.meta.url);
  try { return req('acorn'); } catch {}
  const src = process.binding('natives')['internal/deps/acorn/acorn/dist/acorn'];
  if (!src) throw new Error('acorn not found; run `npm i acorn`');
  const m = { exports: {} };
  new Function('exports','require','module','__filename','__dirname', src)(
    m.exports, (id) => { throw new Error('acorn requires ' + id); }, m, 'acorn.js', '/');
  return m.exports;
}
const acorn = acornLoad();

const isFn = (n) => n && /Function/.test(n.type);
const line = (n) => n && n.loc ? n.loc.start.line : 0;

function konst(n) {
  if (!n) return undefined;
  if (n.type === 'Literal') return { v: n.value };
  if (n.type === 'Identifier') {
    if (n.name === 'undefined') return { v: undefined };
    if (n.name === 'NaN') return { v: NaN };
  }
  if (n.type === 'UnaryExpression') {
    const a = konst(n.argument); if (!a) return;
    if (n.operator === '!') return { v: !a.v };
    if (n.operator === 'void') return { v: undefined };
  }
  if (n.type === 'BinaryExpression') {
    const a = konst(n.left), b = konst(n.right); if (!a || !b) return;
    try {
      switch (n.operator) {
        case '===': return { v: a.v === b.v };
        case '!==': return { v: a.v !== b.v };
        case '==': return { v: a.v == b.v };
        case '!=': return { v: a.v != b.v };
        case '<': return { v: a.v < b.v };
        case '>': return { v: a.v > b.v };
      }
    } catch {}
  }
  return undefined;
}

function walkTop(node, cb) {
  if (!node || typeof node.type !== 'string') return;
  cb(node);
  for (const k of Object.keys(node)) {
    if (k === 'loc' || k === 'start' || k === 'end') continue;
    const c = node[k];
    if (Array.isArray(c)) for (const x of c) if (x && typeof x.type === 'string') {
      if (isFn(x) || /Class/.test(x.type)) continue;
      walkTop(x, cb);
    }
    else if (c && typeof c.type === 'string') {
      if (isFn(c) || /Class/.test(c.type)) continue;
      walkTop(c, cb);
    }
  }
}
function anyRef(node, names) {
  if (!node || typeof node !== 'object') return false;
  if (Array.isArray(node)) return node.some((x) => anyRef(x, names));
  if (node.type === 'Identifier' && names.has(node.name)) return true;
  for (const k of Object.keys(node)) {
    if (k === 'loc' || k === 'start' || k === 'end') continue;
    if (anyRef(node[k], names)) return true;
  }
  return false;
}

const names = (p, out = []) => {
  if (!p) return out;
  if (p.type === 'Identifier') out.push(p.name);
  else if (p.type === 'ObjectPattern') for (const x of p.properties)
    names(x.type === 'Property' ? x.value : x.argument, out);
  else if (p.type === 'ArrayPattern') for (const x of p.elements) names(x, out);
  else if (p.type === 'AssignmentPattern') names(p.left, out);
  else if (p.type === 'RestElement') names(p.argument, out);
  return out;
};

class Module {
  constructor(url) {
    this.url = url;
    this.deps = [];
    this.events = [];
    this.byLocal = new Map();
    this.ns = new Map();
    this.values = new Map();
  }
  init(name, expr, node, hoisted, value) {
    this.events.push({
      type: 'init', start: hoisted ? node.start : node.end, name,
      value: value !== undefined ? value : this.infer(expr),
      hoisted: !!hoisted,
    });
  }
  infer(e) {
    if (!e) return { settled: true };
    switch (e.type) {
      case 'Literal': case 'TemplateLiteral': return { settled: true };
      case 'FunctionExpression': case 'ArrowFunctionExpression': case 'ClassExpression':
        return { settled: true };
      case 'Identifier': {
        const imp = this.byLocal.get(e.name);
        if (imp) return { binding: imp };
        const v = this.values.get(e.name);
        return v ? v : { unknown: true };
      }
      case 'ImportExpression': {
        const t = e.source.type === 'Literal' && typeof e.source.value === 'string'
          ? new URL(e.source.value, this.url).href : null;
        return { dynamic: true, target: t };
      }
      case 'CallExpression':
        if (e.callee.type === 'MemberExpression' && e.callee.object.type === 'Identifier' &&
            e.callee.object.name === 'Promise' && e.callee.property.name === 'resolve') {
          if (e.arguments.length === 0) return { settled: true };
          const v = this.infer(e.arguments[0]);
          return v.settled && !v.binding && !v.unknown ? { settled: true } : { unknown: true };
        }
        return { unknown: true };
      case 'NewExpression':
        if (e.callee.type === 'Identifier' && e.callee.name === 'Promise') {
          const x = e.arguments[0];
          let forever = false;
          if (x && (x.type === 'ArrowFunctionExpression' || x.type === 'FunctionExpression')) {
            const params = new Set(x.params.filter((p) => p.type === 'Identifier').map((p) => p.name));
            forever = params.size === 0 || !anyRef(x.body, params);
          }
          return { foreverPending: forever, unknown: !forever };
        }
        return { unknown: true };
      case 'MemberExpression':
        if (e.object.type === 'Identifier' && this.ns.has(e.object.name)) {
          const n = !e.computed && e.property.type === 'Identifier' ? e.property.name : null;
          if (n) return { binding: { url: this.ns.get(e.object.name), name: n } };
        }
        return { unknown: true };
      default: return { unknown: true };
    }
  }
  awaitKind(node, op) {
    if (op && op.type === 'ImportExpression') {
      const t = op.source.type === 'Literal' && typeof op.source.value === 'string'
        ? new URL(op.source.value, this.url).href : null;
      return { kind: 'dynamic', target: t };
    }
    if (op) {
      const ts = [];
      walkTop(op, (n) => {
        if (n.type === 'ImportExpression' && n.source.type === 'Literal' &&
            typeof n.source.value === 'string') ts.push(new URL(n.source.value, this.url).href);
      });
      const u = [...new Set(ts)];
      if (u.length === 1) return { kind: 'dynamic', target: u[0] };
      if (u.length > 1) return { kind: 'dynamicMulti', targets: u };
    }
    if (op && op.type === 'Identifier') {
      const imp = this.byLocal.get(op.name);
      if (imp) return { kind: 'binding', ...imp };
      const v = this.values.get(op.name);
      if (v) {
        if (v.settled) return { kind: 'ready' };
        if (v.binding) return { kind: 'binding', ...v.binding };
        if (v.dynamic && v.target) return { kind: 'dynamic', target: v.target };
        if (v.foreverPending) return { kind: 'forever' };
      }
      return { kind: 'external' };
    }
    const v = this.infer(op);
    if (v.settled) return { kind: 'ready' };
    if (v.binding) return { kind: 'binding', ...v.binding };
    if (v.foreverPending) return { kind: 'forever' };
    return { kind: 'external' };
  }
  await_(node, op) {
    this.events.push({ type: 'await', start: node.start, line: line(node), await: this.awaitKind(node, op) });
  }
  expr(e) {
    if (!e) return;
    const found = [];
    walkTop(e, (n) => {
      if (n.type === 'AwaitExpression') found.push(n);
      else if (n.type === 'ImportExpression') found.push(n);
    });
    found.sort((a, b) => a.start - b.start);
    for (const f of found) {
      if (f.type === 'AwaitExpression') this.await_(f, f.argument);
      else {
        const t = f.source.type === 'Literal' && typeof f.source.value === 'string'
          ? new URL(f.source.value, this.url).href : null;
        if (t) { if (!this.deps.includes(t)) this.deps.push(t); }
      }
    }
  }
  stmt(s, exported) {
    if (!s) return;
    switch (s.type) {
      case 'ImportDeclaration':
        if (!s.specifiers.length) { const u = new URL(s.source.value, this.url).href; if (!this.deps.includes(u)) this.deps.push(u); }
        for (const sp of s.specifiers) {
          const u = new URL(s.source.value, this.url).href;
          if (!this.deps.includes(u)) this.deps.push(u);
          if (sp.type === 'ImportNamespaceSpecifier') this.ns.set(sp.local.name, u);
          else if (sp.type === 'ImportDefaultSpecifier') this.byLocal.set(sp.local.name, { url: u, name: 'default' });
          else this.byLocal.set(sp.local.name, { url: u, name: sp.imported.type === 'Identifier' ? sp.imported.name : sp.imported.value });
        }
        break;
      case 'ExportNamedDeclaration':
        if (s.declaration) this.decl(s.declaration, true);
        else if (s.source) {
          const u = new URL(s.source.value, this.url).href;
          if (!this.deps.includes(u)) this.deps.push(u);
          for (const sp of s.specifiers) {
            const imp = sp.local.type === 'Identifier' ? sp.local.name : sp.local.value;
            const exp = sp.exported.type === 'Identifier' ? sp.exported.name : sp.exported.value;
            this.init(exp, null, s, true, { binding: { url: u, name: imp } });
          }
        } else for (const sp of s.specifiers) {
          const lo = sp.local.type === 'Identifier' ? sp.local.name : sp.local.value;
          const exp = sp.exported.type === 'Identifier' ? sp.exported.name : sp.exported.value;
          const imp = this.byLocal.get(lo);
          this.init(exp, null, s, true, imp ? { binding: imp } : this.values.get(lo));
        }
        break;
      case 'ExportDefaultDeclaration': {
        const d = s.declaration;
        if (d.type === 'FunctionDeclaration') this.init('default', null, d, true, { settled: true });
        else if (d.type === 'ClassDeclaration') this.init('default', null, d, false, { settled: true });
        else { this.expr(d); this.init('default', d, s, false); }
        break;
      }
      case 'ExportAllDeclaration': {
        const u = new URL(s.source.value, this.url).href;
        if (!this.deps.includes(u)) this.deps.push(u);
        break;
      }
      case 'VariableDeclaration': this.decl(s, exported); break;
      case 'FunctionDeclaration':
        if (s.id) this.init(s.id.name, null, s, true, { settled: true });
        break;
      case 'ClassDeclaration':
        if (s.id) this.init(s.id.name, null, s, false, { settled: true });
        break;
      case 'ExpressionStatement': this.expr(s.expression); break;
      case 'IfStatement': {
        const k = konst(s.test);
        if (k) { if (k.v) this.stmt(s.consequent, exported); else this.stmt(s.alternate, exported); }
        else { this.stmt(s.consequent, exported); this.stmt(s.alternate, exported); }
        break;
      }
      case 'BlockStatement': for (const x of s.body) this.stmt(x, exported); break;
      case 'ForStatement': case 'WhileStatement': case 'DoWhileStatement':
        this.expr(s.test); this.expr(s.update); this.stmt(s.body, exported); break;
      case 'ForInStatement': case 'ForOfStatement':
        if (s.await) this.await_(s, s.right);
        this.stmt(s.body, exported); break;
      case 'TryStatement':
        this.stmt(s.block, exported); this.stmt(s.handler && s.handler.body, exported); this.stmt(s.finalizer, exported); break;
      case 'SwitchStatement': for (const c of s.cases) for (const x of c.consequent) this.stmt(x, exported); break;
      case 'LabeledStatement': this.stmt(s.body, exported); break;
    }
  }
  decl(d, exported) {
    for (const x of d.declarations) {
      this.expr(x.init);
      const ns = names(x.id);
      for (const n of ns) {
        const v = d.kind === 'var' ? { settled: true } : this.infer(x.init);
        this.init(n, x.init, x, d.kind === 'var', v);
        if (!exported) this.values.set(n, v);
      }
    }
  }
}

function extract(url, source) {
  const ast = acorn.parse(source, { ecmaVersion: 'latest', sourceType: 'module', locations: true });
  const m = new Module(url);
  for (const s of ast.body) m.stmt(s, false);
  m.events.sort((a, b) => a.start - b.start);
  return m;
}

function load(entry) {
  const root = pathToFileURL(resolvePath(entry)).href;
  const mods = new Map();
  const q = [root];
  while (q.length) {
    const u = q.shift();
    if (mods.has(u)) continue;
    if (!u.startsWith('file:')) { mods.set(u, { url: u, external: true, deps: [], events: [] }); continue; }
    let m;
    try { m = extract(u, readFileSync(fileURLToPath(u), 'utf8')); }
    catch (e) { m = { url: u, external: true, error: e.message, deps: [], events: [] }; }
    mods.set(u, m);
    for (const d of m.deps) q.push(d);
    for (const e of m.events) if (e.type === 'await' && e.await) {
      if (e.await.kind === 'dynamic' && e.await.target) q.push(e.await.target);
      if (e.await.kind === 'dynamicMulti') for (const t of e.await.targets) q.push(t);
    }
  }
  return { root, mods };
}

function simulate(graph) {
  const S = new Map();
  for (const [u, m] of graph.mods) S.set(u, {
    u, m, status: 'new', started: false, done: false, pc: 0, wait: null,
    pending: new Set(), parents: new Set(), waiters: new Set(), binds: new Map(),
    bwait: new Map(), error: null,
  });
  const mod = (u) => S.get(u) || (S.set(u, { u, m: { url: u, external: true, deps: [], events: [] }, status: 'new', started: false, done: false, pc: 0, wait: null, pending: new Set(), parents: new Set(), waiters: new Set(), binds: new Map(), bwait: new Map(), error: null }), S.get(u));
  const stack = [];
  const errors = [];

  for (const s of S.values()) for (const e of s.m.events || [])
    if (e.type === 'init' && e.hoisted) s.binds.set(e.name, e.value);

  function evaluate(s) {
    if (s.status !== 'new') return;
    if (!s.u.startsWith('file:') || s.m.external) { s.status = 'done'; s.started = s.done = true; return; }
    s.status = 'eval';
    stack.push(s.u);
    for (const durl of s.m.deps || []) {
      const d = mod(durl);
      evaluate(d);
      if (d.done) continue;
      if (d.status !== 'eval') continue;
      const anc = stack.includes(d.u);
      if (anc && !d.started) continue;   // back-edge: this module runs first
      s.pending.add(durl); d.parents.add(s.u);
    }
    stack.pop();
    if (!s.pending.size) start(s);
  }
  function start(s) {
    if (s.started) return;
    s.started = true; s.pc = 0; run(s);
  }
  function complete(s) {
    if (s.done) return;
    s.done = true; s.status = 'done';
    for (const p of s.parents) {
      const x = mod(p); x.pending.delete(s.u);
      if (!x.pending.size && !x.started) start(x);
    }
    for (const w of s.waiters) {
      const x = mod(w);
      if (x.wait && x.wait.type === 'dyn') {
        const all = (x.wait.urls || [x.wait.url]).every((t) => mod(t).done);
        if (!all) continue;
      }
      resume(x);
    }
    s.waiters.clear();
    for (const [n, set] of s.bwait) if (!s.binds.has(n)) for (const w of set) {
      const x = mod(w);
      if (x.wait && x.wait.type === 'bind' && x.wait.url === s.u && x.wait.name === n) {
        x.error = { tdz: true, url: s.u, name: n };
        errors.push({ u: x.u, line: x.wait.line, ...x.error });
      }
    }
    s.bwait.clear();
  }
  function resume(s) { if (s.wait) { s.wait = null; run(s); } }

  function foreverWait(s, e) { s.wait = { type: 'forever', line: e.line }; }

  function run(s) {
    if (s.done || s.error) return;
    const evs = s.m.events || [];
    while (s.pc < evs.length) {
      const e = evs[s.pc];
      if (e.type === 'init') {
        if (!e.hoisted) { s.binds.set(e.name, e.value); notify(s, e.name); }
        s.pc++; continue;
      }
      if (e.type === 'await') {
        let a = e.await;
        if (!a || a.kind === 'ready' || a.kind === 'external') { s.pc++; continue; }
        if (a.kind === 'forever') { foreverWait(s, e); return; }
        if (a.kind === 'dynamic' || a.kind === 'dynamicMulti') {
          const targets = a.kind === 'dynamicMulti' ? a.targets : [a.target];
          let pending = [];
          for (const t of targets) { if (t) { evaluate(mod(t)); if (!mod(t).done) pending.push(t); } }
          if (!pending.length) { s.pc++; continue; }
          for (const t of pending) mod(t).waiters.add(s.u);
          s.wait = { type: 'dyn', url: pending[0], urls: pending, line: e.line };
          return;
        }
        if (a.kind === 'binding') {
          let target = { url: a.url, name: a.name };
          for (let guard = 0; guard < 32; guard++) {
            const d = mod(target.url);
            if (d.binds.has(target.name)) {
              const v = d.binds.get(target.name);
              if (v && v.settled) { s.pc++; break; }
              if (v && v.foreverPending) { foreverWait(s, e); return; }
              if (v && v.binding) { target = v.binding; continue; }
              s.pc++; break;
            }
            if (d.done) { s.error = { tdz: true, url: target.url, name: target.name }; errors.push({ u: s.u, line: e.line, ...s.error }); return; }
            if (d.started) {
              if (!d.bwait.has(target.name)) d.bwait.set(target.name, new Set());
              d.bwait.get(target.name).add(s.u);
              s.wait = { type: 'bind', url: target.url, name: target.name, line: e.line };
              return;
            }
            s.error = { tdz: true, url: target.url, name: target.name }; errors.push({ u: s.u, line: e.line, ...s.error }); return;
          }
          continue;
        }
      }
      s.pc++;
    }
    complete(s);
  }
  function notify(s, name) {
    const set = s.bwait.get(name);
    if (!set) return;
    s.bwait.delete(name);
    for (const w of set) {
      const x = mod(w);
      if (x.wait && x.wait.type === 'bind' && x.wait.url === s.u && x.wait.name === name) resume(x);
    }
  }
  evaluate(mod(graph.root));

  const stuck = [...S.values()].filter((s) => s.u.startsWith('file:') && !s.m.external && !s.done && !s.error);
  if (errors.length) return { kind: 'error', errors };
  if (!stuck.length) return null;
  const edges = new Map();
  const add = (from, to, label, line) => {
    if (!edges.has(from)) edges.set(from, []);
    edges.get(from).push({ to, label, line });
  };
  for (const s of stuck) {
    if (s.wait) {
      if (s.wait.type === 'dyn') for (const t of (s.wait.urls || [s.wait.url])) add(s.u, t, 'await import(' + short(t) + ')', s.wait.line);
      else if (s.wait.type === 'bind') add(s.u, s.wait.url, 'await binding ' + s.wait.name + ' from ' + short(s.wait.url), s.wait.line);
    } else if (!s.started) {
      for (const d of s.pending) add(s.u, d, 'body start waits for async dependency ' + short(d), null);
      if (!s.pending.size) for (const d of s.m.deps || []) if (stuck.some((x) => x.u === d)) add(s.u, d, 'body start waits for static dependency ' + short(d), null);
    }
  }
  let best = null;
  for (const s of stuck) {
    const q = [{ path: [s.u], steps: [] }];
    while (q.length) {
      const { path, steps } = q.shift();
      for (const e of edges.get(path[path.length - 1]) || []) {
        if (e.to === s.u) { const cand = { cycle: path, steps: [...steps, e] }; if (!best || cand.cycle.length < best.cycle.length) best = cand; }
        else if (!path.includes(e.to)) q.push({ path: [...path, e.to], steps: [...steps, e] });
      }
    }
  }
  if (!best) return { kind: 'blocked', modules: stuck.map((s) => s.u) };
  return { kind: 'cycle', cycle: best.cycle, steps: best.steps };
}

function short(u) {
  try { return relative(process.cwd(), fileURLToPath(u)); } catch { return u; }
}

const args = process.argv.slice(2);
const json = args.includes('--json');
const entry = args.find((a) => !a.startsWith('--'));
if (!entry) { console.error('usage: node tla-cycle-detector.mjs <entry.mjs> [--json]'); process.exit(2); }
const graph = load(entry);
const result = simulate(graph);
if (json) {
  console.log(JSON.stringify({ entry: short(graph.root), result }, null, 2));
} else {
  console.log('Entry: ' + short(graph.root));
  if (!result) console.log('no deadlock');
  else if (result.kind === 'error') {
    console.log('evaluation error:');
    for (const e of result.errors) console.log('  ' + short(e.u) + ': TDZ ' + e.name + ' from ' + short(e.url) + (e.line ? ' line ' + e.line : ''));
  } else if (result.kind === 'blocked') {
    console.log('blocked modules:');
    for (const m of result.modules) console.log('  ' + short(m));
  } else {
    console.log('deadlock cycle:');
    for (let i = 0; i < result.cycle.length; i++) {
      const st = result.steps[i];
      console.log('  ' + short(result.cycle[i]) + ' -> ' + (st ? short(st.to) : '?') + '  [' + (st ? st.label : '') + ']' + (st && st.line ? ' line ' + st.line : ''));
    }
  }
}
process.exit(result ? 1 : 0);
