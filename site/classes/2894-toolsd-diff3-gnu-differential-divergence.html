Answer

CHT-144 — Fix diff3 conflict-region accounting to match GNU diff3 -m

Summary

The differential failure is caused by two coupled defects in internal/diff3:

  1. Non-canonical LCS alignment. When duplicate lines make several LCSs equally valid, the Hirschberg split picks the first maximum. For the seed case this represents [tok5 tok5] -> [tok5 mut0] as “delete first tok5, append mut0” instead of GNU diff’s “replace second tok5 with mut0”. That splits one logical change into two regions, so combineRegions produces a conflict only for the first base line.
  2. Replacement-only conflict text. Even when regions overlap correctly, combineRegions concatenates only each region’s repl text. It does not include lines that are unchanged on that side but lie inside the combined base interval. This is why toolsd conflict halves omit context lines that GNU includes.

The fix canonicalizes equivalent LCS matches onto the diagonal within runs of duplicate base lines, then reconstructs each conflict side from the full [sideStart, sideEnd) span of the combined base interval.

Root cause

For the seed case, toolsd’s LCS chooses:

base:  [tok5, tok5]
ours:  [tok5, mut0]
match:  base[1] -> ours[0]

so the computed regions for ours are:

[0,1] -> []       // delete first tok5
[2,2] -> [mut0]   // append mut0

and for theirs:

[0,1] -> [mut1]

combineRegions therefore combines only the first region into a conflict. The mut0 insertion starts at base position 2, after the conflict’s end == 1, so it is emitted as a separate clean edit. GNU diff instead chooses the diagonal match base[0] -> ours[0] and represents the change as [1,2] -> [mut0]; that is adjacent to theirs’ [0,1] -> [mut1], so GNU diff3 forms one conflict over both base lines.

Simply expanding every conflict by one line would be wrong: it would also merge genuinely separate edits, for example:

base:   A B C
ours:   A X B Y C
theirs: A Z B C

GNU keeps X/Z as the conflict and Y as a separate clean insertion. The edit script must first be canonicalized, not blindly widened.

The second defect is independent of LCS choice: once regions [1,1] -> [X], [2,3] -> [Y], and [1,2] -> [Z,Z] overlap into base interval [1,3], the correct ours half is the whole slice ours[1:4] == [X,A,Y]. The old code appends only [X,Y], dropping the matched A.

1. Add side coordinates to region

type region struct {
    start  int // base interval [start,end)
    end    int
    sstart int // side interval [sstart,send)
    send   int
    repl   []string
}

2. Canonicalize equivalent LCS matches

Add this helper near lcsMatches:

func absInt(n int) int {
    if n < 0 {
        return -n
    }
    return n
}

// canonicalMatches snaps equivalent LCS matches onto the main diagonal
// inside runs of duplicate base lines. GNU diff chooses the diagonal
// occurrence; the raw Hirschberg tie-break often chooses an earlier one.
func canonicalMatches(base []string, matches []match) []match {
    out := append([]match(nil), matches...)

    for k := range out {
        i, j := out[k].i, out[k].j

        lo, hi := 0, len(base)-1
        if k > 0 {
            lo = out[k-1].i + 1
        }
        if k+1 < len(out) {
            hi = out[k+1].i - 1
        }
        if lo > hi {
            continue
        }

        l, r := i, i
        for l > lo && base[l-1] == base[i] {
            l--
        }
        for r < hi && base[r+1] == base[i] {
            r++
        }

        best, bestDist := i, absInt(i-j)
        for p := l; p <= r; p++ {
            if d := absInt(p - j); d < bestDist {
                best, bestDist = p, d
            }
        }
        out[k].i = best
    }
    return out
}

3. Replace changedRegions

func changedRegions(base, side []string) []region {
    if equalLines(base, side) {
        return nil
    }

    regions := []region{}
    bi, si := 0, 0
    for _, m := range canonicalMatches(base, lcsMatches(base, side)) {
        if bi == m.i && si == m.j {
            bi, si = m.i+1, m.j+1
            continue
        }
        regions = append(regions, region{
            start:  bi,
            end:    m.i,
            sstart: si,
            send:   m.j,
            repl:   cloneLines(side[si:m.j]),
        })
        bi, si = m.i+1, m.j+1
    }

    if bi < len(base) || si < len(side) {
        regions = append(regions, region{
            start:  bi,
            end:    len(base),
            sstart: si,
            send:   len(side),
            repl:   cloneLines(side[si:]),
        })
    }
    return regions
}

4. Replace combineRegions

The new version reconstructs each side from the original sides, including matched lines between absorbed regions and matched prefix/suffix context.

func combineRegions(ours, theirs []region, oursLines, theirsLines []string) []combinedRegion {
    out := []combinedRegion{}
    oi, ti := 0, 0

    for oi < len(ours) || ti < len(theirs) {
        cur := combinedRegion{}

        var oursBaseEnd, oursSideEnd int
        var theirsBaseEnd, theirsSideEnd int

        addOurs := func(r region) {
            if !cur.oursSet {
                pre := r.start - cur.start
                cur.ours = append(cur.ours,
                    cloneLines(oursLines[r.sstart-pre:r.sstart])...)
            } else {
                gap := r.start - oursBaseEnd
                cur.ours = append(cur.ours,
                    cloneLines(oursLines[oursSideEnd:oursSideEnd+gap])...)
            }
            cur.ours = append(cur.ours,
                cloneLines(oursLines[r.sstart:r.send])...)
            cur.oursSet = true
            if r.end > cur.end {
                cur.end = r.end
            }
            oursBaseEnd, oursSideEnd = r.end, r.send
        }

        addTheirs := func(r region) {
            if !cur.theirsSet {
                pre := r.start - cur.start
                cur.theirs = append(cur.theirs,
                    cloneLines(theirsLines[r.sstart-pre:r.sstart])...)
            } else {
                gap := r.start - theirsBaseEnd
                cur.theirs = append(cur.theirs,
                    cloneLines(theirsLines[theirsSideEnd:theirsSideEnd+gap])...)
            }
            cur.theirs = append(cur.theirs,
                cloneLines(theirsLines[r.sstart:r.send])...)
            cur.theirsSet = true
            if r.end > cur.end {
                cur.end = r.end
            }
            theirsBaseEnd, theirsSideEnd = r.end, r.send
        }

        if oi < len(ours) &&
            (ti >= len(theirs) || ours[oi].start <= theirs[ti].start) {
            cur.start = ours[oi].start
            cur.end = ours[oi].end
            addOurs(ours[oi])
            oi++
        } else {
            cur.start = theirs[ti].start
            cur.end = theirs[ti].end
            addTheirs(theirs[ti])
            ti++
        }

        for {
            absorbed := false

            for oi < len(ours) && ours[oi].start <= cur.end {
                addOurs(ours[oi])
                oi++
                absorbed = true
            }
            for ti < len(theirs) && theirs[ti].start <= cur.end {
                addTheirs(theirs[ti])
                ti++
                absorbed = true
            }

            if !absorbed {
                break
            }
        }

        // Include matched tail context up to the final combined end.
        if cur.oursSet {
            gap := cur.end - oursBaseEnd
            cur.ours = append(cur.ours,
                cloneLines(oursLines[oursSideEnd:oursSideEnd+gap])...)
        }
        if cur.theirsSet {
            gap := cur.end - theirsBaseEnd
            cur.theirs = append(cur.theirs,
                cloneLines(theirsLines[theirsSideEnd:theirsSideEnd+gap])...)
        }

        out = append(out, cur)
    }
    return out
}

package diff3

import (
    "reflect"
    "testing"
)

func TestCHT144ConflictParity(t *testing.T) {
    tests := []struct {
        name                    string
        base, ours, theirs      []string
        wantBase, wantOurs      []string
        wantTheirs              []string
    }{
        {
            name: "seed2case8",
            base:   []string{"tok5", "tok5"},
            ours:   []string{"tok5", "mut0"},
            theirs: []string{"mut1", "tok5"},
            wantBase:   []string{"tok5", "tok5"},
            wantOurs:   []string{"tok5", "mut0"},
            wantTheirs: []string{"mut1", "tok5"},
        },
        {
            name: "matched-context-inside-conflict",
            base:   []string{"A", "A", "B", "C", "C"},
            ours:   []string{"A", "X", "A", "Y", "C", "C"},
            theirs: []string{"A", "Z", "Z", "B", "C", "C"},
            wantBase:   []string{"A", "B"},
            wantOurs:   []string{"X", "A", "Y"},
            wantTheirs: []string{"Z", "Z", "B"},
        },
    }

    for _, tc := range tests {
        t.Run(tc.name, func(t *testing.T) {
            res, err := Merge(tc.base, tc.ours, tc.theirs, Options{
                OursLabel:   "ours",
                TheirsLabel: "theirs",
            })
            if err != nil {
                t.Fatal(err)
            }
            if len(res.Conflicts) != 1 {
                t.Fatalf("got %d conflicts, want 1", len(res.Conflicts))
            }
            c := res.Conflicts[0]
            if !reflect.DeepEqual(c.Base, tc.wantBase) ||
                !reflect.DeepEqual(c.Ours, tc.wantOurs) ||
                !reflect.DeepEqual(c.Theirs, tc.wantTheirs) {
                t.Fatalf("conflict mismatch:\n got base=%v ours=%v theirs=%v\nwant base=%v ours=%v theirs=%v",
                    c.Base, c.Ours, c.Theirs,
                    tc.wantBase, tc.wantOurs, tc.wantTheirs)
            }
        })
    }
}
