OpenSSH Certificate Principal-Scope Audit — Verified Solution

Root cause analysis

The failures this auditor must catch are not visible from a naive text scan of ssh-keygen -L:

  1. OpenSSH certificates are SSH wire-format blobs. ssh-keygen -L prints a convenient view, but the authoritative fields — principal list, critical-option tuples, extension tuples, validity epoch values, and the signing CA key blob — are length-prefixed binary fields. Duplicate critical options and raw unknown option data can be normalized or hidden in the human view.
  2. Wildcard principals are match patterns. A principal of * is not merely a string; sshd may use it to match any requested username. If policy denies root, a certificate containing * escalates scope even though the literal string root never appears in the certificate.
  3. Validity is raw Unix epoch, not a local date string. ssh-keygen -L renders local time. Comparing formatted dates can be off by timezone and cannot represent the 0xffffffffffffffff “forever” sentinel safely with signed shell arithmetic.
  4. CIDR membership is a subset operation. Comparing CIDR strings is insufficient. A certificate source range is valid only when its entire network is contained in one policy-allowed network. Both IPv4 and IPv6 need bit-prefix comparison.
  5. ssh-keygen -L verifies certificate signatures. Hand-built violating fixtures must therefore be structurally correct and actually signed. The fixture builder constructs the SSH wire blob and signs it with a real RSA CA using OpenSSL.
  6. A host certificate is a different certificate type field. Type 2 must be rejected for user authentication even if every other field is policy-clean.

The solution below parses the raw blob as the authoritative source, parses ssh-keygen -L independently, cross-checks both views, and emits one JSON finding per violation.

# OpenSSH user-certificate policy for the fixture set.
cert_type=user

# Only these CA key types may sign user certificates.
ca_key_types=ssh-rsa,ssh-ed25519

cert-audit.sh

#!/usr/bin/env bash
# cert-audit.sh - audit OpenSSH user certificates against a policy.
#
# Usage: cert-audit.sh -p POLICY cert.pub [cert.pub ...]
#
# Emits one JSON object per finding on stdout.  Exit status:
#   0  no findings
#   1  one or more policy violations
#   2  usage, policy, dependency, or hard parse error
#
# The auditor intentionally uses two independent views of every certificate:
#   1. the raw SSH wire-format certificate blob (authoritative), and
#   2. ssh-keygen -L output (cross-check / fallback).

set -u
export LC_ALL=C
export TZ=UTC
umask 077

PROG=${0##*/}
FINDING_COUNT=0
HARD_ERROR=0
POLICY_FILE=
CERTS=()
declare -A POLICY=()
WORKDIR=$(mktemp -d "${TMPDIR:-/tmp}/cert-audit.XXXXXX") || { echo "$PROG: mktemp failed" >&2; exit 2; }
trap 'rm -rf "$WORKDIR"' EXIT

usage() {
  cat <<EOF
Usage: $PROG -p POLICY_FILE CERT [CERT ...]
Audit OpenSSH certificates and print one JSON finding per line.
EOF
}
die() { printf '%s: %s\n' "$PROG" "$*" >&2; exit 2; }

trim() {
  local s=$1
  s=${s#"${s%%[![:space:]]*}"}
  s=${s%"${s##*[![:space:]]}"}
  printf '%s' "$s"
}

json_escape() {
  local s=$1
  s=${s//\\/\\\\}
  s=${s//\"/\\\"}
  s=${s//$'\n'/\\n}
  s=${s//$'\r'/\\r}
  s=${s//$'\t'/\\t}
  printf '%s' "$s"
}

# emit CODE SEVERITY DETAIL [KEY VALUE ...]
emit() {
  local code=$1 severity=$2 detail=$3
  shift 3
  ((FINDING_COUNT++))
  local out
  out=$(printf '{"code":"%s","severity":"%s","cert":"%s","detail":"%s"' \
    "$(json_escape "$code")" "$(json_escape "$severity")" \
    "$(json_escape "${CURRENT_CERT:-}")" "$(json_escape "$detail")")
  while (($# >= 2)); do
    out+=$(printf ',"%s":"%s"' "$(json_escape "$1")" "$(json_escape "$2")")
    shift 2
  done
  out+='}'
  printf '%s\n' "$out"
}

split_csv() {
  SPLIT=()
  local s=$1 item
  local -a parts=()
  [[ -n $s ]] || return 0
  IFS=',' read -r -a parts <<< "$s"
  for item in "${parts[@]}"; do
    item=$(trim "$item")
    [[ -n $item ]] && SPLIT+=("$item")
  done
}

in_list() {
  local needle=$1 x
  shift
  for x in "$@"; do [[ $x == "$needle" ]] && return 0; done
  return 1
}

has_policy() { [[ -n ${POLICY[$1]+x} ]]; }
pol_get() { printf '%s' "${POLICY[$1]-}"; }

parse_time_epoch() {
  local t=$1
  [[ -n $t ]] || return 1
  if [[ $t =~ ^[0-9]+$ ]]; then
    printf '%s' "$t"
    return 0
  fi
  date -u -d "$t" +%s 2>/dev/null
}

# Raw SSH certificate parser
# ---------------------------------------------------------------------------
declare -a BYTES=()
declare -a C_PRINCIPALS=()
declare -a C_CRIT_NAMES=()
declare -a C_CRIT_VALUES=()
declare -a C_CRIT_RAW_LEN=()
declare -a C_CRIT_MALFORMED=()
declare -a C_EXT_NAMES=()
declare -a C_EXT_DATA_LEN=()
C_ALG=''; C_KEY_ID=''; C_CA_TYPE=''
C_TYPE=0; C_SERIAL=0; C_VALID_AFTER=0; C_VALID_BEFORE=0
C_CA_BLOB_START=0; C_CA_BLOB_LEN=0
POS=0; U32=0; U64=0; STR=''; STR_START=0; STR_LEN=0; PUBTYPE=''

u32_at() {
  local o=$1
  (( o + 4 <= ${#BYTES[@]} )) || return 1
  U32=$(( (BYTES[o] << 24) | (BYTES[o+1] << 16) | (BYTES[o+2] << 8) | BYTES[o+3] ))
}

bytes_to_text() {
  local start=$1 len=$2 i oct out=''
  for ((i=start; i<start+len; i++)); do
    printf -v oct '\\%03o' "${BYTES[i]}"
    out+=$oct
  done
  printf -v STR '%b' "$out"
  STR_START=$start
  STR_LEN=$len
}

read_u32() {
  u32_at "$POS" || return 1
  POS=$((POS + 4))
}

read_u64() {
  local hi lo
  read_u32 || return 1; hi=$U32
  read_u32 || return 1; lo=$U32
  # Certificates may use 0xffffffffffffffff as an infinite valid_before.
  # Bash arithmetic is signed 64-bit, so clamp values above INT64_MAX rather
  # than letting them wrap negative (which would hide an over-long validity).
  if (( hi >= 0x80000000 )); then
    U64=9223372036854775807
  else
    U64=$(( (hi << 32) | lo ))
  fi
}

skip_string() {
  read_u32 || return 1
  local n=$U32
  (( POS + n <= ${#BYTES[@]} )) || return 1
  STR_START=$POS; STR_LEN=$n
  POS=$((POS + n))
}

read_string() {
  read_u32 || return 1
  local n=$U32
  (( POS + n <= ${#BYTES[@]} )) || return 1
  bytes_to_text "$POS" "$n"
  POS=$((POS + n))
}

str_at() {
  local off=$1
  u32_at "$off" || return 1
  local n=$U32
  (( off + 4 + n <= ${#BYTES[@]} )) || return 1
  bytes_to_text $((off + 4)) "$n"
}

pubkey_type_at() {
  local off=$1
  u32_at "$off" || return 1
  local n=$U32
  (( off + 4 + n <= ${#BYTES[@]} )) || return 1
  bytes_to_text $((off + 4)) "$n"
  PUBTYPE=$STR
}

# load_cert_blob FILE
# Decodes the first certificate line and parses every field into C_* globals.
load_cert_blob() {
  local file=$1 line tok alg='' b64='' found=0
  local blob="$WORKDIR/blob.$$"
  IFS= read -r line < "$file" || return 1
  for tok in $line; do
    if [[ $tok == *-<email> ]]; then
      alg=$tok; found=1; continue
    fi
    if ((found)) && [[ -z $b64 ]]; then b64=$tok; break; fi
  done
  [[ -n $alg && -n $b64 ]] || return 1
  printf '%s' "$b64" | base64 -d > "$blob" 2>/dev/null || return 1
  [[ -s $blob ]] || return 1

  local -a _od=() _bytes=()
  mapfile -t _od < <(od -An -v -tu1 "$blob")
  local chunk b
  for chunk in "${_od[@]}"; do
    for b in $chunk; do _bytes+=("$b"); done
  done
  BYTES=("${_bytes[@]}")
  POS=0
  C_PRINCIPALS=(); C_CRIT_NAMES=(); C_CRIT_VALUES=()
  C_CRIT_RAW_LEN=(); C_CRIT_MALFORMED=(); C_EXT_NAMES=(); C_EXT_DATA_LEN=()

  read_string || return 1
  C_ALG=$STR
  skip_string || return 1                    # nonce

  # public key fields for the certificate's subject key
  case $C_ALG in
    <email>)
      skip_string || return 1 ;;
    <email>|rsa-sha2-*-<email>)
      skip_string || return 1; skip_string || return 1 ;;
    <email>)
      skip_string || return 1; skip_string || return 1
      skip_string || return 1; skip_string || return 1 ;;
    ecdsa-sha2-*-<email>)
      skip_string || return 1; skip_string || return 1 ;;
    <email>*-<email>)
      skip_string || return 1; skip_string || return 1 ;;
    sk-ecdsa-sha2-*-<email>)
      skip_string || return 1; skip_string || return 1; skip_string || return 1 ;;
    *)
      return 1 ;;
  esac

  read_u64 || return 1; C_SERIAL=$U64
  read_u32 || return 1; C_TYPE=$U32
  read_string || return 1; C_KEY_ID=$STR

  # principals: a string containing a sequence of strings
  read_string || return 1
  local pend=$((STR_START + STR_LEN)) off=$STR_START
  while (( off < pend )); do
    str_at "$off" || return 1
    C_PRINCIPALS+=("$STR")
    off=$((STR_START + STR_LEN))
  done
  (( off == pend )) || return 1

  read_u64 || return 1; C_VALID_AFTER=$U64
  read_u64 || return 1; C_VALID_BEFORE=$U64

  # critical options: string containing (name,data) tuples.  For the two
  # options OpenSSH understands, data is itself a string.
  read_string || return 1
  local cend=$((STR_START + STR_LEN)) coff=$STR_START
  while (( coff < cend )); do
    str_at "$coff" || return 1
    local cname=$STR
    coff=$((STR_START + STR_LEN))
    str_at "$coff" || return 1
    local dstart=$STR_START dlen=$STR_LEN value='' malformed=0
    coff=$((STR_START + STR_LEN))
    case $cname in
      force-command|source-address)
        if str_at "$dstart"; then
          (( STR_LEN + 4 == dlen )) || malformed=1
          value=$STR
        else
          malformed=1
        fi
        ;;
      *)
        value=''
        ;;
    esac
    C_CRIT_NAMES+=("$cname")
    C_CRIT_VALUES+=("$value")
    C_CRIT_RAW_LEN+=("$dlen")
    C_CRIT_MALFORMED+=("$malformed")
  done
  (( coff == cend )) || return 1

  # extensions: string containing (name,data) tuples
  read_string || return 1
  local eend=$((STR_START + STR_LEN)) eoff=$STR_START
  while (( eoff < eend )); do
    str_at "$eoff" || return 1
    local ename=$STR
    eoff=$((STR_START + STR_LEN))
    str_at "$eoff" || return 1
    local edlen=$STR_LEN
    eoff=$((STR_START + STR_LEN))
    C_EXT_NAMES+=("$ename")
    C_EXT_DATA_LEN+=("$edlen")
  done
  (( eoff == eend )) || return 1

  skip_string || return 1                    # reserved
  read_string || return 1                    # signature key (CA public key)
  C_CA_BLOB_START=$STR_START; C_CA_BLOB_LEN=$STR_LEN
  pubkey_type_at "$C_CA_BLOB_START" || return 1
  C_CA_TYPE=$PUBTYPE
  skip_string || return 1                    # signature
  (( POS == ${#BYTES[@]} )) || return 1
  return 0
}

# ---------------------------------------------------------------------------
# ssh-keygen -L parser (human view, cross-check)
# ---------------------------------------------------------------------------
L_TYPE=''; L_KEY_ID=''; L_VALID_FROM=''; L_VALID_TO=''; L_CA=''
declare -a L_PRINCIPALS=() L_CRIT=() L_EXT=()

parse_sshkeygen_L() {
  local file=$1 out line section='' item
  out=$(ssh-keygen -L -f "$file" 2>/dev/null) || return 1
  L_TYPE=''; L_KEY_ID=''; L_VALID_FROM=''; L_VALID_TO=''; L_CA=''
  L_PRINCIPALS=(); L_CRIT=(); L_EXT=()
  while IFS= read -r line; do
    if [[ $line =~ ^[[:space:]]*Type:[[:space:]]+(.*)$ ]]; then
      L_TYPE=${BASH_REMATCH[1]}; section=''
    elif [[ $line =~ ^[[:space:]]*Key\ ID:[[:space:]]+\"?(.*)\"?$ ]]; then
      L_KEY_ID=${BASH_REMATCH[1]}; section=''
    elif [[ $line =~ ^[[:space:]]*Valid:[[:space:]]+from[[:space:]]+([^[:space:]]+)[[:space:]]+to[[:space:]]+([^[:space:]]+) ]]; then
      L_VALID_FROM=${BASH_REMATCH[1]}; L_VALID_TO=${BASH_REMATCH[2]}; section=''
    elif [[ $line =~ ^[[:space:]]*Principals:[[:space:]]*$ ]]; then
      section=principals
    elif [[ $line =~ ^[[:space:]]*Critical\ Options:[[:space:]]*$ ]]; then
      section=critical
    elif [[ $line =~ ^[[:space:]]*Extensions:[[:space:]]*$ ]]; then
      section=extensions
    elif [[ $line =~ ^[[:space:]]*Signing\ CA:[[:space:]]+(.*)$ ]]; then
      L_CA=${BASH_REMATCH[1]}; section=''
    elif [[ -n $section && $line =~ ^[[:space:]]{2,}(.+)$ ]]; then
      item=$(trim "${BASH_REMATCH[1]}")
      case $section in
        principals) L_PRINCIPALS+=("$item") ;;
        critical)   L_CRIT+=("$item") ;;
        extensions) L_EXT+=("$item") ;;
      esac
    elif [[ -n $line && ! $line =~ ^[[:space:]] ]]; then
      section=''
    fi
  done <<< "$out"
  return 0
}

# ---------------------------------------------------------------------------
# CIDR handling (IPv4 and IPv6), pure bash
# ---------------------------------------------------------------------------
CIDR_FAMILY=''; CIDR_ADDR=''; CIDR_PREFIX=0

ipv4_to_hex() {
  local ip=$1 a b c d
  [[ $ip =~ ^([0-9]{1,3})\.([0-9]{1,3})\.([0-9]{1,3})\.([0-9]{1,3})$ ]] || return 1
  a=$((10#${BASH_REMATCH[1]})); b=$((10#${BASH_REMATCH[2]}))
  c=$((10#${BASH_REMATCH[3]})); d=$((10#${BASH_REMATCH[4]}))
  (( a <= 255 && b <= 255 && c <= 255 && d <= 255 )) || return 1
  printf '%02x%02x%02x%02x' "$a" "$b" "$c" "$d"
}

ipv6_to_hex() {
  local ip=$1
  ip=${ip%%\%*}                            # remove zone id
  [[ $ip == *:* ]] || return 1
  if [[ $ip == *.* ]]; then
    local head=${ip%:*} v4=${ip##*:} v4hex g1 g2
    v4hex=$(ipv4_to_hex "$v4") || return 1
    g1=${v4hex:0:4}; g2=${v4hex:4:4}
    ip="${head}:${g1}:${g2}"
  fi
  local left right
  local -a lg=() rg=() full=()
  local fill=0
  if [[ $ip == *::* ]]; then
    left=${ip%%::*}; right=${ip##*::}
    [[ $right != *::* ]] || return 1
    fill=1
  else
    left=$ip; right=''
  fi
  if [[ -n $left ]]; then IFS=':' read -r -a lg <<< "$left"; fi
  if [[ -n $right ]]; then IFS=':' read -r -a rg <<< "$right"; fi
  (( fill )) || (( ${#lg[@]} + ${#rg[@]} == 8 )) || return 1
  if (( fill )); then
    local total=$(( ${#lg[@]} + ${#rg[@]} ))
    (( total < 8 )) || return 1
    local zeros=$((8 - total)) i g
    for g in "${lg[@]}"; do full+=("$g"); done
    for ((i=0; i<zeros; i++)); do full+=("0"); done
    for g in "${rg[@]}"; do full+=("$g"); done
    lg=("${full[@]}")
  fi
  local out='' n
  for n in "${lg[@]}"; do
    [[ $n =~ ^[0-9A-Fa-f]{1,4}$ ]] || return 1
    printf -v n '%04x' $((16#$n))
    out+=$n
  done
  ((${#out} == 32)) || return 1
  printf '%s' "$out"
}

parse_cidr() {
  local cidr prefix addr
  cidr=$(trim "$1")
  [[ -n $cidr ]] || return 1
  if [[ $cidr == */* ]]; then
    addr=${cidr%/*}; prefix=${cidr##*/}
  else
    addr=$cidr
    if [[ $cidr == *:* ]]; then prefix=128; else prefix=32; fi
  fi
  [[ $prefix =~ ^[0-9]{1,3}$ ]] || return 1
  prefix=$((10#$prefix))
  if [[ $addr == *:* ]]; then
    (( prefix <= 128 )) || return 1
    addr=$(ipv6_to_hex "$addr") || return 1
    CIDR_FAMILY=6
  else
    (( prefix <= 32 )) || return 1
    addr=$(ipv4_to_hex "$addr") || return 1
    CIDR_FAMILY=4
  fi
  CIDR_ADDR=$addr; CIDR_PREFIX=$prefix
  return 0
}

hex_prefix_equal() {
  local a=$1 b=$2 bits=$3 full=$(( $3 / 4 )) rem=$(( $3 % 4 ))
  local ca cb na nb shift
  if (( full > 0 )); then
    [[ ${a:0:full} == "${b:0:full}" ]] || return 1
  fi
  if (( rem > 0 )); then
    ca=${a:full:1}; cb=${b:full:1}
    [[ -n $ca && -n $cb ]] || return 1
    na=$((16#$ca)); nb=$((16#$cb))
    shift=$((4 - rem))
    (( (na >> shift) == (nb >> shift) )) || return 1
  fi
  return 0
}

cidr_contains() {
  local cf=$1 ca=$2 cp=$3 af=$4 aa=$5 ap=$6
  [[ $cf == "$af" ]] || return 1
  (( cp >= ap )) || return 1
  hex_prefix_equal "$ca" "$aa" "$ap"
}

POLICY_DIR=''
POLICY_CERT_TYPE=user
POLICY_ALLOW_WILDCARD=true
POLICY_VALID_AFTER=''
POLICY_VALID_BEFORE=''
POLICY_CA_LIMIT=''
POLICY_CA_CERT_PATH=''
declare -a POLICY_CA_TYPES=() POLICY_DENIED_PRINCIPALS=() POLICY_ALLOWED_PRINCIPALS=()
declare -a POLICY_CRITICAL=() POLICY_FORCE_PREFIXES=() POLICY_EXTENSIONS=()
declare -a POLICY_REQUIRED_EXT=() POLICY_DENIED_EXT=()
declare -a POLICY_SRC_FAMILY=() POLICY_SRC_ADDR=() POLICY_SRC_PREFIX=()

load_policy() {
  local raw line key val
  [[ -r $POLICY_FILE ]] || die "policy file not readable: $POLICY_FILE"
  POLICY_DIR=$(cd "$(dirname "$POLICY_FILE")" && pwd)
  while IFS= read -r raw || [[ -n $raw ]]; do
    line=${raw%$'\r'}
    [[ $line =~ ^[[:space:]]*# ]] && continue
    [[ $line == *=* ]] || continue
    key=$(trim "${line%%=*}")
    val=$(trim "${line#*=}")
    [[ -n $key ]] && POLICY["$key"]=$val
  done < "$POLICY_FILE"

  POLICY_CERT_TYPE=$(pol_get cert_type); POLICY_CERT_TYPE=${POLICY_CERT_TYPE:-user}
  POLICY_ALLOW_WILDCARD=$(pol_get allow_wildcard_principals); POLICY_ALLOW_WILDCARD=${POLICY_ALLOW_WILDCARD:-true}

  split_csv "$(pol_get ca_key_types)"; POLICY_CA_TYPES=("${SPLIT[@]}")
  split_csv "$(pol_get denied_principals)"; POLICY_DENIED_PRINCIPALS=("${SPLIT[@]}")
  split_csv "$(pol_get allowed_principals)"; POLICY_ALLOWED_PRINCIPALS=("${SPLIT[@]}")
  split_csv "$(pol_get critical_options)"; POLICY_CRITICAL=("${SPLIT[@]}")
  split_csv "$(pol_get force_command_prefixes)"; POLICY_FORCE_PREFIXES=("${SPLIT[@]}")
  split_csv "$(pol_get extensions)"; POLICY_EXTENSIONS=("${SPLIT[@]}")
  split_csv "$(pol_get required_extensions)"; POLICY_REQUIRED_EXT=("${SPLIT[@]}")
  split_csv "$(pol_get denied_extensions)"; POLICY_DENIED_EXT=("${SPLIT[@]}")

  local t
  t=$(pol_get valid_after)
  if [[ -n $t ]]; then POLICY_VALID_AFTER=$(parse_time_epoch "$t") || die "invalid valid_after: $t"; fi
  t=$(pol_get valid_before)
  if [[ -n $t ]]; then POLICY_VALID_BEFORE=$(parse_time_epoch "$t") || die "invalid valid_before: $t"; fi

  # Optional CA certificate: compare against its valid_before (valid-after limit).
  t=$(pol_get ca_cert)
  if [[ -n $t ]]; then
    [[ $t == /* ]] || t="$POLICY_DIR/$t"
    POLICY_CA_CERT_PATH=$t
    [[ -r $POLICY_CA_CERT_PATH ]] || die "CA certificate not readable: $POLICY_CA_CERT_PATH"
    load_cert_blob "$POLICY_CA_CERT_PATH" || die "cannot parse CA certificate: $POLICY_CA_CERT_PATH"
    POLICY_CA_LIMIT=$C_VALID_BEFORE
  fi
  t=$(pol_get ca_valid_before)
  if [[ -n $t ]]; then
    t=$(parse_time_epoch "$t") || die "invalid ca_valid_before: $t"
    if [[ -z $POLICY_CA_LIMIT ]] || (( t < POLICY_CA_LIMIT )); then POLICY_CA_LIMIT=$t; fi
  fi

  local r
  split_csv "$(pol_get source_addresses)"
  for r in "${SPLIT[@]}"; do
    parse_cidr "$r" || die "invalid source_addresses entry: $r"
    POLICY_SRC_FAMILY+=("$CIDR_FAMILY")
    POLICY_SRC_ADDR+=("$CIDR_ADDR")
    POLICY_SRC_PREFIX+=("$CIDR_PREFIX")
  done
}

# Rule checks
# ---------------------------------------------------------------------------
validate_source_address() {
  local value=$1
  local -a ranges=()
  [[ -n $value ]] || { emit source_address_malformed high "source-address critical option is empty" value "$value"; return; }
  IFS=',' read -r -a ranges <<< "$value"
  local r cf ca cp i ok
  for r in "${ranges[@]}"; do
    r=$(trim "$r")
    if [[ -z $r ]] || ! parse_cidr "$r"; then
      emit source_address_malformed high "source-address is not a valid CIDR" value "$r"
      continue
    fi
    cf=$CIDR_FAMILY; ca=$CIDR_ADDR; cp=$CIDR_PREFIX
    if (( ${#POLICY_SRC_FAMILY[@]} )); then
      ok=0
      for ((i=0; i<${#POLICY_SRC_FAMILY[@]}; i++)); do
        if cidr_contains "$cf" "$ca" "$cp" "${POLICY_SRC_FAMILY[i]}" "${POLICY_SRC_ADDR[i]}" "${POLICY_SRC_PREFIX[i]}"; then
          ok=1; break
        fi
      done
      if (( ! ok )); then
        emit source_address_out_of_scope high "certificate source-address is outside policy allowed ranges" value "$r"
      fi
    fi
  done
}

check_principals() {
  local p d a matched
  for p in "${C_PRINCIPALS[@]}"; do
    for d in "${POLICY_DENIED_PRINCIPALS[@]}"; do
      # Either a literal cert principal matches a denied glob, or a glob
      # cert principal covers a denied literal name.
      if [[ $p == $d || $d == $p ]]; then
        if [[ $p == *[\*\?\[]* ]]; then
          emit principal_scope_escalation high "wildcard certificate principal covers an explicitly denied principal" principal "$p" denied "$d"
        else
          emit principal_denied high "certificate principal is explicitly denied" principal "$p" denied "$d"
        fi
        break
      fi
    done
    if [[ $POLICY_ALLOW_WILDCARD != true && $p == *[\*\?\[]* ]]; then
      emit principal_wildcard_forbidden high "wildcard principals are not permitted by policy" principal "$p"
    fi
    if (( ${#POLICY_ALLOWED_PRINCIPALS[@]} )); then
      matched=0
      for a in "${POLICY_ALLOWED_PRINCIPALS[@]}"; do
        if [[ $p == $a ]]; then matched=1; break; fi
      done
      (( matched )) || emit principal_not_allowed high "certificate principal is not allowed by policy" principal "$p"
    fi
  done
}

check_critical_options() {
  local i name value malformed prefix ok
  local -A seen=()
  for i in "${!C_CRIT_NAMES[@]}"; do
    name=${C_CRIT_NAMES[i]}
    value=${C_CRIT_VALUES[i]}
    malformed=${C_CRIT_MALFORMED[i]}
    if [[ -n ${seen[$name]+x} ]]; then
      emit critical_option_duplicate high "critical option appears more than once; sshd rejects duplicated critical options" option "$name" count "duplicate"
    fi
    seen[$name]=1

    case $name in
      force-command|source-address) ;;
      *)
        emit critical_option_unknown high "unknown critical option; sshd rejects certificates with unknown critical options" option "$name"
        continue
        ;;
    esac
    if (( ${#POLICY_CRITICAL[@]} )) && ! in_list "$name" "${POLICY_CRITICAL[@]}"; then
      emit critical_option_not_allowed high "critical option is not enabled by policy" option "$name"
    fi
    if (( malformed )); then
      emit critical_option_malformed high "critical option data is not in the form sshd expects" option "$name"
      continue
    fi

    case $name in
      force-command)
        if [[ -z $value ]]; then
          emit force_command_malformed high "force-command is empty" option "$name"
        elif (( ${#POLICY_FORCE_PREFIXES[@]} )); then
          ok=0
          for prefix in "${POLICY_FORCE_PREFIXES[@]}"; do
            if [[ $value == "$prefix"* ]]; then ok=1; break; fi
          done
          (( ok )) || emit force_command_not_allowed high "force-command is outside policy allowed prefixes" command "$value"
        fi
        ;;
      source-address)
        validate_source_address "$value"
        ;;
    esac
  done
}

check_extensions() {
  local i name data_len
  local -A seen=()
  for i in "${!C_EXT_NAMES[@]}"; do
    name=${C_EXT_NAMES[i]}
    data_len=${C_EXT_DATA_LEN[i]}
    if [[ -n ${seen[$name]+x} ]]; then
      emit extension_duplicate medium "extension appears more than once" extension "$name"
    fi
    seen[$name]=1
    if in_list "$name" "${POLICY_DENIED_EXT[@]}"; then
      emit extension_denied high "extension is explicitly denied by policy" extension "$name"
    fi
    if has_policy extensions && ! in_list "$name" "${POLICY_EXTENSIONS[@]}"; then
      emit extension_not_allowed high "extension is not enabled by policy" extension "$name"
    fi
    if (( data_len > 0 )); then
      emit extension_nonempty_data low "extension has unexpected data; OpenSSH extensions normally carry empty data" extension "$name" length "$data_len"
    fi
  done
  local req
  for req in "${POLICY_REQUIRED_EXT[@]}"; do
    if [[ -z ${seen[$req]+x} ]]; then
      emit extension_required_missing high "required extension is missing" extension "$req"
    fi
  done
}

cross_check_view() {
  local i
  if (( ${#L_PRINCIPALS[@]} != ${#C_PRINCIPALS[@]} )); then
    emit certificate_view_mismatch high "raw principal list disagrees with ssh-keygen -L" field "principals" raw_count "${#C_PRINCIPALS[@]}" sshkeygen_count "${#L_PRINCIPALS[@]}"
    return
  fi
  for i in "${!C_PRINCIPALS[@]}"; do
    if [[ ${L_PRINCIPALS[i]} != "${C_PRINCIPALS[i]}" ]]; then
      emit certificate_view_mismatch high "raw principal list disagrees with ssh-keygen -L" field "principals" raw "${C_PRINCIPALS[i]}" sshkeygen "${L_PRINCIPALS[i]}"
      break
    fi
  done

  if (( ${#L_CRIT[@]} != ${#C_CRIT_NAMES[@]} )); then
    emit certificate_view_mismatch high "raw critical-option list disagrees with ssh-keygen -L" field "critical_options" raw_count "${#C_CRIT_NAMES[@]}" sshkeygen_count "${#L_CRIT[@]}"
  else
    for i in "${!C_CRIT_NAMES[@]}"; do
      if [[ ${L_CRIT[i]%% *} != "${C_CRIT_NAMES[i]}" ]]; then
        emit certificate_view_mismatch high "raw critical-option list disagrees with ssh-keygen -L" field "critical_options" raw "${C_CRIT_NAMES[i]}" sshkeygen "${L_CRIT[i]}"
        break
      fi
    done
  fi

  if (( ${#L_EXT[@]} != ${#C_EXT_NAMES[@]} )); then
    emit certificate_view_mismatch high "raw extension list disagrees with ssh-keygen -L" field "extensions" raw_count "${#C_EXT_NAMES[@]}" sshkeygen_count "${#L_EXT[@]}"
  else
    for i in "${!C_EXT_NAMES[@]}"; do
      if [[ ${L_EXT[i]%% *} != "${C_EXT_NAMES[i]}" ]]; then
        emit certificate_view_mismatch high "raw extension list disagrees with ssh-keygen -L" field "extensions" raw "${C_EXT_NAMES[i]}" sshkeygen "${L_EXT[i]}"
        break
      fi
    done
  fi
}

audit_cert() {
  CURRENT_CERT=$1
  if [[ ! -r $CURRENT_CERT ]]; then
    emit certificate_unreadable high "certificate file is not readable"
    return
  fi
  if ! load_cert_blob "$CURRENT_CERT"; then
    emit certificate_unreadable high "certificate does not contain a decodable OpenSSH certificate blob"
    return
  fi

  local l_ok=1
  if ! parse_sshkeygen_L "$CURRENT_CERT"; then
    l_ok=0
    emit certificate_sshkeygen_decode_failed high "ssh-keygen -L could not decode the certificate"
  fi

  local raw_type_name
  case $C_TYPE in
    1) raw_type_name=user ;;
    2) raw_type_name=host ;;
    *) raw_type_name="unknown($C_TYPE)" ;;
  esac

  # Cross-check the two views.  Differences are a finding because an auditor
  # must never silently disagree with the tool sshd operators use.
  if (( l_ok )) && [[ -n $L_TYPE ]]; then
    local l_type_name=unknown
    [[ $L_TYPE == *" user certificate"* ]] && l_type_name=user
    [[ $L_TYPE == *" host certificate"* ]] && l_type_name=host
    if [[ $l_type_name != "$raw_type_name" ]]; then
      emit certificate_view_mismatch high "raw certificate type disagrees with ssh-keygen -L" raw "$raw_type_name" sshkeygen "$l_type_name"
    fi
    cross_check_view
  fi

  # Certificate type / host certificate for user authentication.
  if [[ $POLICY_CERT_TYPE == user && $C_TYPE -eq 2 ]]; then
    emit host_certificate_for_user_auth high "host-type certificate cannot authenticate a user" cert_type "$raw_type_name"
  elif [[ $C_TYPE -ne 1 && $C_TYPE -ne 2 ]]; then
    emit certificate_type_unknown high "certificate type field is neither user (1) nor host (2)" cert_type "$raw_type_name"
  elif [[ $POLICY_CERT_TYPE != "$raw_type_name" ]]; then
    emit certificate_type_mismatch high "certificate type does not match policy" expected "$POLICY_CERT_TYPE" actual "$raw_type_name"
  fi

  # CA key type.
  if (( ${#POLICY_CA_TYPES[@]} )) && ! in_list "$C_CA_TYPE" "${POLICY_CA_TYPES[@]}"; then
    emit ca_key_type_not_allowed high "signing CA key type is not allowed by policy" ca_key_type "$C_CA_TYPE"
  fi

  # Validity window and CA upper bound.
  if (( C_VALID_AFTER >= C_VALID_BEFORE )); then
    emit validity_inverted high "certificate valid_after is not before valid_before" valid_after "$C_VALID_AFTER" valid_before "$C_VALID_BEFORE"
  fi
  if [[ -n $POLICY_VALID_AFTER ]] && (( C_VALID_AFTER < POLICY_VALID_AFTER )); then
    emit validity_outside_policy high "certificate starts before policy validity window" bound "valid_after" cert_value "$C_VALID_AFTER" policy_value "$POLICY_VALID_AFTER"
  fi
  if [[ -n $POLICY_VALID_BEFORE ]] && (( C_VALID_BEFORE > POLICY_VALID_BEFORE )); then
    emit validity_outside_policy high "certificate ends after policy validity window" bound "valid_before" cert_value "$C_VALID_BEFORE" policy_value "$POLICY_VALID_BEFORE"
  fi
  if [[ -n $POLICY_CA_LIMIT ]] && (( C_VALID_BEFORE > POLICY_CA_LIMIT )); then
    emit validity_exceeds_ca high "certificate is valid beyond its signing CA valid-after" cert_valid_before "$C_VALID_BEFORE" ca_valid_before "$POLICY_CA_LIMIT"
  fi

  check_principals
  check_critical_options
  check_extensions
}

pubblob() {
  local f=$1 out=$2 b
  b=$(awk '{print $2}' "$f")
  [[ -n $b ]] || die "missing base64 field in $f"
  printf '%s' "$b" | base64 -d > "$out"
}
pubblob "$WORK/ca.pub" "$CA_BLOB"
pubblob "$WORK/subject.pub" "$SUBJECT_BLOB"

# ---------------------------------------------------------------------------
# SSH wire-format helpers
# ---------------------------------------------------------------------------
u32to() {
  local f=$1 v=$2
  printf "\\$(printf '%03o' $(((v >> 24) & 255)))\\$(printf '%03o' $(((v >> 16) & 255)))\\$(printf '%03o' $(((v >> 8) & 255)))\\$(printf '%03o' $((v & 255)))" >> "$f"
}
u64to() {
  local f=$1 v=$2
  u32to "$f" $(((v >> 32) & 0xffffffff))
  u32to "$f" $((v & 0xffffffff))
}
strto() {
  local f=$1 s=$2
  u32to "$f" "${#s}"
  printf '%s' "$s" >> "$f"
}
strfileto() {
  local f=$1 g=$2 n
  n=$(stat -c %s "$g")
  u32to "$f" "$n"
  cat "$g" >> "$f"
}
get_u32_file() {
  local f=$1 a b c d
  read -r a b c d < <(od -An -v -tu1 -N4 "$f")
  printf '%s' $(( (a << 24) | (b << 16) | (c << 8) | d ))
}
pubkey_body() {
  local src=$1 out=$2 tl off n
  tl=$(get_u32_file "$src")
  off=$((4 + tl + 4))
  n=$(( $(stat -c %s "$src") - off ))
  (( n > 0 )) || die "cannot extract public key body from $src"
  dd if="$src" of="$out" bs=1 skip="$off" count="$n" status=none
}
epoch() { date -u -d "$1" +%s; }

# Append a critical option whose data is sshbuf-string(value), as OpenSSH does
# for force-command and source-address.
crit_known_to() {
  local f=$1 name=$2 value=$3 t
  t=$(mktemp "$WORK/nested.XXXXXX")
  strto "$t" "$value"
  strto "$f" "$name"
  strfileto "$f" "$t"
  rm -f "$t"
}
# Append an arbitrary critical option with raw data (used for unknown options).
crit_raw_to() {
  local f=$1 name=$2 data=$3
  strto "$f" "$name"
  strto "$f" "$data"
}

# Globals consumed by build_cert.
declare -a F_PRINCIPALS=() F_CRITS=() F_EXTS=()
reset_fixture() { F_PRINCIPALS=(); F_CRITS=(); F_EXTS=(); }
known_crit() { F_CRITS+=("known"$'\t'"$1"$'\t'"$2"); }
raw_crit()   { F_CRITS+=("raw"$'\t'"$1"$'\t'"$2"); }
ext()        { F_EXTS+=("$1"$'\t'"${2:-}"); }

# build_cert OUTFILE TYPE VALID_AFTER VALID_BEFORE KEY_ID CA_BLOB SUBJECT_BLOB
build_cert() {
  local outfile=$1 type=$2 va=$3 vb=$4 keyid=$5 ca_blob=$6 subj_blob=$7
  local out="$WORK/cert.$$.blob" tmp="$WORK/list.$$" sig="$WORK/sig.$$"
  : > "$out"; : > "$tmp"

  strto "$out" '<email>'
  strto "$out" '0123456789abcdef0123456789abcdef'   # nonce: exactly 32 bytes
  pubkey_body "$subj_blob" "$WORK/pk.$$"
  strfileto "$out" "$WORK/pk.$$"
  u64to "$out" 42
  u32to "$out" "$type"
  strto "$out" "$keyid"

  # principals: string containing a sequence of strings
  : > "$tmp"
  local p
  for p in "${F_PRINCIPALS[@]}"; do strto "$tmp" "$p"; done
  strfileto "$out" "$tmp"

  u64to "$out" "$va"
  u64to "$out" "$vb"

  # critical options: string containing (name,data) tuples
  : > "$tmp"
  local spec kind name value
  for spec in "${F_CRITS[@]}"; do
    IFS=$'\t' read -r kind name value <<< "$spec"
    case $kind in
      known) crit_known_to "$tmp" "$name" "$value" ;;
      raw)   crit_raw_to "$tmp" "$name" "$value" ;;
      *) die "bad critical option spec: $spec" ;;
    esac
  done
  strfileto "$out" "$tmp"

  # extensions: string containing (name,data) tuples; data is normally empty
  : > "$tmp"
  for spec in "${F_EXTS[@]}"; do
    IFS=$'\t' read -r name value <<< "$spec"
    strto "$tmp" "$name"
    strto "$tmp" "$value"
  done
  strfileto "$out" "$tmp"

  strto "$out" ''                                   # reserved
  strfileto "$out" "$ca_blob"                       # signing CA public key

  # SSH signature blob: string algorithm + string signature bytes
  openssl dgst -sha512 -sign "$CA_KEY" -out "$WORK/rawsig.$$" "$out"
  : > "$sig"
  strto "$sig" 'rsa-sha2-512'
  strfileto "$sig" "$WORK/rawsig.$$"
  strfileto "$out" "$sig"

  printf '<email> %s fixture\n' "$(base64 -w0 "$out")" > "$outfile"
  ssh-keygen -L -f "$outfile" >/dev/null 2>&1 || die "fixture does not decode: $outfile"
  rm -f "$out" "$tmp" "$sig" "$WORK/rawsig.$$" "$WORK/pk.$$"
}

#!/usr/bin/env bash
# End-to-end verification for cert-audit.sh and the fixture set.
set -u
export LC_ALL=C TZ=UTC
cd "$(dirname "$0")"
fail=0
checks=0

pass() { printf 'ok   %s\n' "$*"; ((checks++)); }
bad()  { printf 'FAIL %s\n' "$*" >&2; fail=1; ((checks++)); }

# Every fixture must be a structurally valid certificate that ssh-keygen -L
# decodes (signatures are real; the hand-built blobs are signed by an RSA CA).
for f in fixtures/*.pub; do
  if ssh-keygen -L -f "$f" >/dev/null 2>&1; then
    pass "decode ${f##*/}"
  else
    bad "decode ${f##*/}"
  fi
done

run_clean() {
  local fixture=$1 out rc
  out=$(./cert-audit.sh -p policy.conf "fixtures/$fixture.pub" 2>&1); rc=$?
  if (( rc == 0 )) && [[ -z $out ]]; then
    pass "clean $fixture"
  else
    bad "clean $fixture (rc=$rc out=$out)"
  fi
}

run_code() {
  local fixture=$1 code=$2 out rc
  out=$(./cert-audit.sh -p policy.conf "fixtures/$fixture.pub" 2>&1); rc=$?
  if (( rc != 1 )); then
    bad "$fixture expected rc=1 got rc=$rc out=$out"
  elif ! grep -Fq "\"code\":\"$code\"" <<< "$out"; then
    bad "$fixture missing code $code out=$out"
  else
    pass "$fixture -> $code"
  fi
}

run_clean valid-ipv6

# A mixed invocation must fail as a whole while still reporting only the bad cert.
out=$(./cert-audit.sh -p policy.conf fixtures/valid.pub fixtures/wildcard-denied.pub 2>&1); rc=$?
if (( rc == 1 )) && [[ $(grep -Fc '"code":"principal_scope_escalation"' <<< "$out") -eq 1 ]]; then
  pass "mixed invocation"
else
  bad "mixed invocation (rc=$rc out=$out)"
fi

# Policy error must be a hard non-zero error (not silently accepted).
out=$(./cert-audit.sh -p /no/such/policy fixtures/valid.pub 2>&1); rc=$?
if (( rc == 2 )); then pass "missing policy exits 2"; else bad "missing policy exits $rc"; fi

if (( fail )); then
  printf '\n%d checks, FAILURES\n' "$checks" >&2
  exit 1
fi
printf '\n%d checks, all passed\n' "$checks"
