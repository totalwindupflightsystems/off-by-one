import type { MemoryPressureLevel, MemoryThresholds } from './types.js';
import { DEFAULT_THRESHOLDS } from './constants.js';

export function calculatePressureLevel(usagePercent: number, thresholds: MemoryThresholds = DEFAULT_THRESHOLDS): MemoryPressureLevel {
  if (!Number.isFinite(usagePercent) || usagePercent < 0) return 'low';
  if (usagePercent >= thresholds.critical) return 'critical';
  if (usagePercent >= thresholds.warn) return 'high';
  if (usagePercent >= thresholds.warn / 2) return 'moderate';
  return 'low';
}

export function detectMaxHeapSize(): number {
  const heapTotal = process.memoryUsage().heapTotal;
  return heapTotal > 0 ? heapTotal : Math.floor(os.totalmem() * DEFAULT_MAX_HEAP_PERCENT);
}

core.ts — class + factory; stateful privates stay here:

import type { EventBus } from '../../services/event-bus.js';
import type { MemoryPressureLevel, MemoryStats, MemoryThresholds } from './types.js';
import { DEFAULT_CHECK_INTERVAL_MS, DEFAULT_THRESHOLDS } from './constants.js';
import { calculatePressureLevel, detectMaxHeapSize } from './pressure-utils.js';
import { emitMemoryEvent } from './events.js';
import { createThresholdCacheStrategy } from './strategies.js';

export class MemoryManager {
  private readonly thresholds: MemoryThresholds;
  private readonly eventBus: EventBus;
  private level: MemoryPressureLevel = 'low';
  private maxHeapSize = detectMaxHeapSize();
  private timer?: NodeJS.Timeout;
  private readonly cacheStrategy = createThresholdCacheStrategy();

  constructor(eventBus: EventBus, options: { thresholds?: MemoryThresholds; checkIntervalMs?: number } = {}) {
    this.eventBus = eventBus;
    this.thresholds = options.thresholds ?? DEFAULT_THRESHOLDS;
  }

  start(): void {
    if (this.timer) return;
    this.timer = setInterval(() => this.check(), DEFAULT_CHECK_INTERVAL_MS);
  }
  stop(): void {
    if (this.timer) clearInterval(this.timer);
    this.timer = undefined;
  }
  check(): MemoryStats {
    const stats = this.sample();
    const next = calculatePressureLevel(stats.usagePercent, this.thresholds);
    if (next !== this.level) {
      this.level = next;
      emitMemoryEvent(this.eventBus, 'pressure-changed', { level: next, usagePercent: stats.usagePercent });
    }
    return stats;
  }
  private sample(): MemoryStats {
    const { heapUsed, heapTotal, rss } = process.memoryUsage();
    return { heapUsed, heapTotal, rss, usagePercent: (heapUsed / this.maxHeapSize) * 100 };
  }
}

export function createMemoryManager(eventBus: EventBus, options?: ConstructorParameters<typeof MemoryManager>[1]): MemoryManager {
  return new MemoryManager(eventBus, options);
}
