Answer

Bug (USABILITY-001): main() hardcoded the config path, so committed alternate configs (e.g. config.e2e.json) were unusable, and the load path was untestable.

Fix (stdlib only): 1. flag.String("config", defaultConfigPath, ...) — --config/-config with the original path as default, so no-flag behavior is byte-for-byte unchanged. 2. Extracted run(configPath string) error out of main for direct unit testing. 3. All config-load failures are wrapped with a stable CONFIG_LOAD_ERROR marker; main prints to stderr and exits 1.

main.go

package main

import (
    "flag"
    "fmt"
    "os"
)

// defaultConfigPath is the original hardcoded config path. Keeping it as the
// flag default means deployments that never passed --config behave identically,
// while committed alternate configs become usable.
const defaultConfigPath = "config.json"

func main() {
    configPath := flag.String("config", defaultConfigPath,
        "path to the JSON config file (default: "+defaultConfigPath+")")
    flag.Parse()

    if err := run(*configPath); err != nil {
        fmt.Fprintln(os.Stderr, err)
        os.Exit(1)
    }
}

config.go (marker + validation)

const ErrConfigLoad = "CONFIG_LOAD_ERROR"

func loadConfig(path string) (Config, error) {
    b, err := os.ReadFile(path)
    if err != nil {
        return Config{}, fmt.Errorf("%s: %w", ErrConfigLoad, err)
    }
    cfg := defaultConfig() // {Port:8080, Host:"127.0.0.1", HealthPath:"/health"}
    if err := json.Unmarshal(b, &cfg); err != nil {
        return Config{}, fmt.Errorf("%s: %w", ErrConfigLoad, err)
    }
    if cfg.Port < 1 || cfg.Port > 65535 {
        return Config{}, fmt.Errorf("%s: invalid port %d in %s", ErrConfigLoad, cfg.Port, path)
    }
    // ...empty Host/HealthPath fall back to defaults...
    return cfg, nil
}

server.go (extracted run)

// run loads config at configPath and serves HTTP until SIGINT/SIGTERM.
// Extracted from main so tests can drive it directly (e.g. error path).
func run(configPath string) error {
    cfg, err := loadConfig(configPath)
    if err != nil {
        return err
    }
    srv := &http.Server{Addr: fmt.Sprintf("%s:%d", cfg.Host, cfg.Port), Handler: newHandler(cfg)}
    errCh := make(chan error, 1)
    go func() { errCh <- srv.ListenAndServe() }()
    log.Printf("listening on %s (config: %s)", srv.Addr, configPath)

    stop := make(chan os.Signal, 1)
    signal.Notify(stop, os.Interrupt, syscall.SIGTERM)
    select {
    case err := <-errCh:
        if err == http.ErrServerClosed {
            return nil
        }
        return fmt.Errorf("server error: %w", err)
    case <-stop:
        ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
        defer cancel()
        return srv.Shutdown(ctx)
    }
}
