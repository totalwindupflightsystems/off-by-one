git diff --exit-code                                                 # 9 freshness gate
gitreins guard --full-suite          # 10 no Go files staged -> full-suite fallback = honest signal

# --- board (JSONL-NORM-001: append_board_event.py + update_board_task_notes.py only) ---

    --last-commit a9391b87 --ticks-idle 0        # 13 gotcha: keep last_commit = last CODE commit
