Offline DNS Zone, Delegation, CNAME/DNAME and Glue Audit

Root-cause analysis

The failure mode is not a single broken command; it is treating an unordered collection of zone files as ordered text.

A correct audit must:

  1. Parse zone-file state correctly — $ORIGIN, relative names, inherited owners, parentheses, comments, TTL/class fields, and canonical lowercase FQDNs.
  2. Track the source zone of every record — an address record outside the file’s SOA apex is still associated with that file. Losing this makes out-of-bailiwick glue invisible or causes false positives.
  3. Build the delegation/zone-cut graph — every non-apex NS record in a file is a cut from that file’s SOA apex to the delegated owner.
  4. Treat CNAME/DNAME as a reference graph — loops need a visited set; the 16-reference limit needs a hop counter; DNAME needs longest-ancestor suffix rewriting.
  5. Measure DNS names in wire octets — sum(1 + label_length) + 1 root <= 255, with escaped octets counted correctly.
  6. Make output deterministic — canonical names, de-duplicated findings, shortest path retained, and a final sort by finding type and key.

Exact fix

Save the following as dns-audit.sh, then run it with the zone files in any order. With no arguments it reads zone data from stdin.

#!/usr/bin/env bash
# dns-audit.sh - offline DNS zone/delegation/CNAME audit
# Usage: ./dns-audit.sh [zone-file ...]   (reads stdin when no files are given)
# Emits one sorted JSON object per finding on stdout.
set -euo pipefail
export LC_ALL=C

if (($# == 0)); then
  set -- -

function trim(s) { sub(/^[ \t]+/, "", s); sub(/[ \t]+$/, "", s); return s }

function is_escaped(s, pos,   i, n, c) {
  n = 0
  for (i = pos - 1; i >= 1 && substr(s, i, 1) == "\\"; i--) n++
  return (n % 2) == 1
}

function strip_comment(s,   i, L, c, q, out) {
  L = length(s); q = 0; out = ""
  for (i = 1; i <= L; i++) {
    c = substr(s, i, 1)
    if (q) {
      if (c == "\\" && i < L) { out = out c substr(s, i + 1, 1); i++; continue }
      if (c == "\"") q = 0
      out = out c
    } else {
      if (c == "\"") { q = 1; out = out c; continue }
      if (c == ";") break
      out = out c
    }
  }
  return out
}

function paren_delta(s,   i, L, c, q, esc, d) {
  L = length(s); q = 0; d = 0
  for (i = 1; i <= L; i++) {
    c = substr(s, i, 1)
    if (q) {
      if (c == "\\") { i++; continue }
      if (c == "\"") q = 0
    } else {
      if (c == "\"") q = 1
      else if (c == "(") d++
      else if (c == ")") d--
    }
  }
  return d
}

# Tokenize a logical resource-record line.  Quotes are retained only for
# determining token boundaries; the surrounding quotes are removed.
function tokenize(s, A,   i, L, c, tok, q, n) {
  L = length(s); tok = ""; q = 0; n = 0
  for (i = 1; i <= L; i++) {
    c = substr(s, i, 1)
    if (q) {
      if (c == "\\" && i < L) { tok = tok c substr(s, i + 1, 1); i++; continue }
      if (c == "\"") { q = 0; continue }
      tok = tok c
    } else {
      if (c == "\"") q = 1
      else if (c == " " || c == "\t") {
        if (tok != "") { A[++n] = tok; tok = "" }
      } else tok = tok c
    }
  }
  if (tok != "") A[++n] = tok
  return n
}

function unescaped_final_dot(s,   L) {
  L = length(s)
  return (L > 0 && substr(s, L, 1) == "." && !is_escaped(s, L))
}

function canonical_name(tok, org,   s) {
  if (tok == "@") return org
  if (tok == ".") return ""
  if (tok == "") return ""
  if (unescaped_final_dot(tok)) s = substr(tok, 1, length(tok) - 1)
  else if (org != "") s = tok "." org
  else s = tok
  return tolower(s)
}

function display_name(n) { return (n == "") ? "." : n "." }

function subdomain(n, p) {
  if (p == "") return 1
  if (n == p) return 1
  return (length(n) > length(p) && substr(n, length(n) - length(p)) == "." p)
}

function is_ttl(t) {
  return t ~ /^[0-9]+([smhdwSMHDW][0-9]+)*[smhdwSMHDW]?$/
}
function is_class(t) {
  t = toupper(t)
  return (t == "IN" || t == "CH" || t == "HS" || t == "CS" || t == "NONE" || t == "ANY")
}

function split_labels(s, A,   i, L, c, cur, n) {
  A[0] = 0; n = 0; cur = ""
  L = length(s)
  if (s == "") { A[0] = 0; return 0 }
  for (i = 1; i <= L; i++) {
    c = substr(s, i, 1)
    if (c == "\\" && i < L) { cur = cur c substr(s, i + 1, 1); i++ }
    else if (c == ".") { A[++n] = cur; cur = "" }
    else cur = cur c
  }
  A[++n] = cur
  A[0] = n
  return n
}

function label_octets(lab,   i, L, c, n) {
  L = length(lab); n = 0
  for (i = 1; i <= L; i++) {
    c = substr(lab, i, 1)
    if (c == "\\") {
      if (i + 3 <= L && substr(lab, i + 1, 3) ~ /^[0-9][0-9][0-9]$/) i += 3
      else if (i < L) i += 1
      n++
    } else n++
  }
  return n
}

function wire_len(name,   labs, n, i, total) {
  if (name == "") return 1
  n = split_labels(name, labs)
  total = 1                    # terminal root label
  for (i = 1; i <= n; i++) total += 1 + label_octets(labs[i])
  return total
}

function json_escape(s,   out, i, L, c) {
  out = ""; L = length(s)
  for (i = 1; i <= L; i++) {
    c = substr(s, i, 1)
    if (c == "\\") out = out "\\\\"
    else if (c == "\"") out = out "\\\""
    else if (c == "\n") out = out "\\n"
    else if (c == "\r") out = out "\\r"
    else if (c == "\t") out = out "\\t"
    else if (c < " ") { out = out sprintf("\\u%04x", ord(c)) }
    else out = out c
  }
  return out
}

# mawk has no ord(); this portable helper is enough for JSON escaping.
function ord(c,   i) {
  for (i = 1; i < 256; i++) if (sprintf("%c", i) == c) return i
  return 63
}

function join_path(A, n, sep,   i, s) {
  s = ""
  for (i = 1; i <= n; i++) s = s (i > 1 ? sep : "") A[i]
  return s
}

function path_json(A, n,   i, s) {
  s = "["
  for (i = 1; i <= n; i++) s = s (i > 1 ? "," : "") "\"" json_escape(display_name(A[i])) "\""
  return s "]"
}

function add_finding(type, key, A, n, detail,   idx, ps) {
  idx = type SUBSEP key
  ps = join_path(A, n, "\034")
  if (!(idx in fpath) || n < flen[idx] || (n == flen[idx] && ps < fpath[idx])) {
    ftype[idx] = type
    fkey[idx] = key
    fpath[idx] = ps
    flen[idx] = n
    fdetail[idx] = detail
    fjsonpath[idx] = path_json(A, n)
  }
}

function add_path_unique(A, n, v,   i) {
  for (i = 1; i <= n; i++) if (A[i] == v) return n
  A[++n] = v
  return n
}

# Use the SOA apex of this file, falling back to the most specific global apex.
function record_zone(f, name,   z, best, bestlen, k) {
  best = ""; bestlen = -1
  for (k = 1; k <= nfile_soa[f]; k++) {
    z = file_soa[f, k]
    if (subdomain(name, z) && length(z) > bestlen) { best = z; bestlen = length(z) }
  }
  if (bestlen >= 0) return best
  # Out-of-zone glue can legally appear in a zone file (the very thing this
  # audit checks).  Assign it to the file SOA apex rather than to the global
  # root.  Multiple zones per file are handled best-effort by the suffix match.
  if (nfile_soa[f] >= 1) return file_soa[f, 1]
  best = ""; bestlen = -1
  for (k = 1; k <= nzones; k++) {
    z = zones[k]
    if (subdomain(name, z) && length(z) > bestlen) { best = z; bestlen = length(z) }
  }
  return best
}

function closest_dname(name,   i, d, best, bestlen) {
  best = ""; bestlen = -1
  for (i = 1; i <= ndnames; i++) {
    d = dname_owner_list[i]
    if (name != d && subdomain(name, d) && length(d) > bestlen) {
      best = d; bestlen = length(d)
    }
  }
  return best
}

function dname_synth(q, d, t,   prefix) {
  prefix = substr(q, 1, length(q) - length(d) - 1)
  if (t == "") return prefix
  if (prefix == "") return t
  return prefix "." t
}

function canonical_cycle(A, starti, n, C,   m, i, best, bestpos) {
  # A[starti..n] are the distinct cycle vertices, in edge order.
  m = n - starti + 1
  best = A[starti]; bestpos = 1
  for (i = starti + 1; i <= n; i++) {
    if (A[i] < best) { best = A[i]; bestpos = i - starti + 1 }
  }
  for (i = 1; i <= m; i++) C[i] = A[starti + ((bestpos - 1 + i - 1) % m)]
  C[m + 1] = C[1]
  return m + 1
}

function resolve_from(start, do_limit,   cur, nxt, d, syn, i, s, hops, plen, sidx, clen, C, seen_local, p, cyc_key) {
  split("", seen_local)
  plen = 1; p[1] = start; seen_local[start] = 1; hops = 0

  while (1) {
    cur = p[plen]
    if (cur in cname) {
      nxt = cname[cur]
      hops++
      if (nxt in seen_local) {
        sidx = seen_local[nxt]
        clen = canonical_cycle(p, sidx, plen, C)
        cyc_key = join_path(C, clen - 1, "|")
        add_finding("cname-loop", cyc_key, C, clen, "CNAME/DNAME reference cycle")
        break
      }
      p[++plen] = nxt; seen_local[nxt] = plen
    } else {
      d = closest_dname(cur)
      if (d == "") break
      syn = dname_synth(cur, d, dname[d])
      hops++
      if (wire_len(syn) > 255) {
        # Path records the query, DNAME owner, and synthesized target.
        s = 1; p2[1] = cur
        p2[++s] = d
        p2[++s] = syn
        add_finding("dname-synthesis-too-long", display_name(d) "|" display_name(cur) "|" display_name(syn), p2, s,
                    "synthesized name is " wire_len(syn) " octets (limit 255), DNAME owner " display_name(d))
      }
      if (syn == cur || (syn in seen_local)) {
        if (syn in seen_local) {
          sidx = seen_local[syn]
          clen = canonical_cycle(p, sidx, plen, C)
          cyc_key = join_path(C, clen - 1, "|")
          add_finding("cname-loop", cyc_key, C, clen, "CNAME/DNAME reference cycle")
        }
        break
      }
      p[++plen] = syn; seen_local[syn] = plen
    }

    if (do_limit && hops > 16) {
      add_finding("cname-query-limit", start, p, plen,
                  "chain requires " hops " references; resolver limit is 16")
      break
    }
  }
}

BEGIN {
  nrec = 0; origin = ""; last_owner = ""; in_paren = 0; logical = ""
  nzones = 0; ndnames = 0; curfile = ""
}

{
  raw = $0
  # Starting a new input file resets zone-file state.  FNR==1 also handles
  # the same path being supplied more than once.
  if (FNR == 1) {
    curfile = FILENAME; origin = ""; last_owner = ""; in_paren = 0; logical = ""
  }

  line = strip_comment(raw)
  if (in_paren) {
    logical = logical " " line
    in_paren += paren_delta(line)
    if (in_paren <= 0) { in_paren = 0; parse_logical(logical); logical = "" }
    next
  }
  if (trim(line) == "") next
  in_paren = paren_delta(line)
  if (in_paren > 0) { logical = line; next }
  in_paren = 0
  parse_logical(line)
}

function parse_logical(line,   A, n, i, omitted, owner, typ, data, t, j, val) {
  n = tokenize(line, A)
  if (n == 0) return
  omitted = (line ~ /^[ \t]/)
  if (substr(A[1], 1, 1) == "$") {
    t = toupper(A[1])
    if (t == "$ORIGIN" && n >= 2) origin = canonical_name(A[2], origin)
    # $TTL/$INCLUDE/$GENERATE are intentionally ignored; pass included files
    # explicitly on the command line for deterministic, network-free input.
    return
  }

  if (omitted) {
    owner = last_owner
    i = 1
    if (owner == "") return
  } else {
    owner = canonical_name(A[1], origin)
    last_owner = owner
    i = 2
  }

  while (i <= n && (is_ttl(A[i]) || is_class(A[i]))) i++
  if (i > n) return
  typ = toupper(A[i])
  i++

  data = ""
  for (j = i; j <= n; j++) data = data (j > i ? " " : "") A[j]

  nrec++
  rec_owner[nrec] = owner
  rec_type[nrec] = typ
  rec_data[nrec] = data
  rec_file[nrec] = FILENAME

  if (typ == "SOA") {
    if (!(owner in soa_seen)) { soa_seen[owner] = 1; zones[++nzones] = owner }
    if (!((FILENAME, owner) in file_soa_seen)) {
      file_soa_seen[FILENAME, owner] = 1
      file_soa[FILENAME, ++nfile_soa[FILENAME]] = owner
    }
  }

  # Remember every name that can be an offline query/reference endpoint.
  all_names[owner] = 1
  if (typ == "CNAME" && i <= n) {
    val = canonical_name(A[i], origin)
    rec_target[nrec] = val
    all_names[val] = 1
    if (!(owner in cname) || val < cname[owner]) cname[owner] = val
    if (!(val in cname_incoming)) cname_incoming[val] = 1
  } else if (typ == "DNAME" && i <= n) {
    val = canonical_name(A[i], origin)
    rec_target[nrec] = val
    all_names[val] = 1
    if (!(owner in dname) || val < dname[owner]) dname[owner] = val
  } else if ((typ == "NS" || typ == "A" || typ == "AAAA") && i <= n) {
    val = canonical_name(A[i], origin)
    rec_target[nrec] = val
    all_names[val] = 1
  }
}

function sorted_names(   k, i, j, tmp) {
  # Simple insertion sort; input sizes are small and this is deterministic.
  for (k in all_names) name_list[++nname_list] = k
  for (i = 2; i <= nname_list; i++) {
    tmp = name_list[i]; j = i - 1
    while (j >= 1 && name_list[j] > tmp) { name_list[j + 1] = name_list[j]; j-- }
    name_list[j + 1] = tmp
  }
}

END {
  # 1. Assign each record to the zone represented by its source file.
  for (r = 1; r <= nrec; r++) rec_zone[r] = record_zone(rec_file[r], rec_owner[r])

  # 2. Build the DNAME list deterministically before any traversal.
  for (r = 1; r <= nrec; r++) if (rec_type[r] == "DNAME" && (rec_owner[r] in dname))
    dname_owner_list[++ndnames] = rec_owner[r]
  # de-duplicate DNAME owners
  nd2 = 0
  for (i = 1; i <= ndnames; i++) if (!(dname_owner_list[i] in dname_owner_seen)) {
    dname_owner_seen[dname_owner_list[i]] = 1
    dname_owner_list2[++nd2] = dname_owner_list[i]
  }
  ndnames = nd2
  for (i = 1; i <= ndnames; i++) dname_owner_list[i] = dname_owner_list2[i]

  sorted_names()

  # 3. Delegations and glue bailiwick checks.  An NS below a file SOA apex
  #    is a zone cut in that file, even if the same owner has an SOA in a
  #    different (child) file.
  for (r = 1; r <= nrec; r++) {
    if (rec_type[r] != "NS") continue
    parent = rec_zone[r]
    child = rec_owner[r]
    if (child == parent) continue                 # apex NS, not a cut
    nsname = rec_target[r]
    for (g = 1; g <= nrec; g++) {
      if (rec_type[g] != "A" && rec_type[g] != "AAAA") continue
      if (rec_owner[g] != nsname) continue
      if (rec_zone[g] != parent) continue        # not glue in the parent file
      if (!subdomain(nsname, parent)) {
        p[1] = parent; p[2] = child
        n = 2
        n = add_path_unique(p, n, nsname)
        n = add_path_unique(p, n, rec_owner[g])
        key = display_name(parent) "|" display_name(child) "|" display_name(nsname) "|" display_name(rec_owner[g])
        add_finding("glue-out-of-bailiwick", key, p, n,
                    "glue " display_name(rec_owner[g]) " is outside delegating zone " display_name(parent))
      }
    }
  }

  # 4. Exhaustive DNAME-synthesis length check, with closure over synthesized
  #    names.  Stop growing a branch once it exceeds 255 octets (otherwise a
  #    self-referential DNAME could generate an infinite sequence).
  for (round = 0; round < 64; round++) {
    added = 0
    for (ni = 1; ni <= nname_list; ni++) {
      q = name_list[ni]
      for (di = 1; di <= ndnames; di++) {
        d = dname_owner_list[di]
        if (q == d || !subdomain(q, d)) continue
        syn = dname_synth(q, d, dname[d])
        if (wire_len(syn) > 255 && !((d SUBSEP q SUBSEP syn) in dname_bad)) {
          dname_bad[d SUBSEP q SUBSEP syn] = 1
          p[1] = q; p[2] = d; p[3] = syn
          add_finding("dname-synthesis-too-long", display_name(d) "|" display_name(q) "|" display_name(syn), p, 3,
                      "synthesized name is " wire_len(syn) " octets (limit 255), DNAME owner " display_name(d))
        }
        if (!(syn in all_names) && wire_len(syn) <= 255) {
          all_names[syn] = 1; name_list[++nname_list] = syn; added = 1
        }
      }
    }
    if (!added) break
  }

  # 5. CNAME chains.  Resolve from maximal CNAME roots for the 16-query
  #    finding; resolve from every CNAME owner as well so cycles with no root
  #    (for example a->b->a) are still found.
  for (k in cname) {
    is_root = !(k in cname_incoming)
    resolve_from(k, is_root)
  }
  # The exhaustive scan above may have missed a DNAME encountered only through
  # a non-root CNAME; resolve all owners once more for loop/DNAME findings.
  for (k in cname) resolve_from(k, 0)

  # 6. Emit JSON lines; jq applies the final deterministic (type, key) order.
  for (idx in ftype) {
    json = "{\"type\":\"" json_escape(ftype[idx]) "\",\"key\":\"" json_escape(fkey[idx]) "\",\"path\":" fjsonpath[idx] ","
    json = json "\"detail\":\"" json_escape(fdetail[idx]) "\"}"
    print json
  }
}
' "$@" | jq -cs 'sort_by(.type, .key)[]'

Output schema

Each line is one sorted JSON finding:

{
  "type": "cname-loop | cname-query-limit | dname-synthesis-too-long | glue-out-of-bailiwick",
  "key": "stable identity for de-duplication",
  "path": ["shortest", "reference", "path"],
  "detail": "human-readable explanation"
}

The CNAME/DNAME hop limit is enforced as more than 16 references. A loop path is canonicalized to start at its lexicographically smallest vertex. The shortest path wins, with lexicographic tie-breaking.

tmp=$(mktemp -d)

L63=$(printf 'a%.0s' {1..63})
L61=$(printf 'b%.0s' {1..61})
DNATARGET="$L63.$L63.$L63.$L61.example."

{
  printf '$ORIGIN example.com.\n'
  printf '@ 3600 IN SOA ns.example.com. hostmaster.example.com. 1 2 3 4 5\n'
  printf '@ IN NS ns.example.com.\n'
  printf 'ns IN A <ip-address>\n'

  # Valid in-bailiwick delegation
  printf 'child IN NS ns.child.example.com.\n'
  printf 'ns.child IN A <ip-address>\n'

  # Out-of-bailiwick glue stored in the delegating parent zone
  printf 'evil IN NS ns.other.net.\n'
  printf 'ns.other.net. IN A <ip-address>\n'

  # CNAME loop
  printf 'loop1 IN CNAME loop2\n'
  printf 'loop2 IN CNAME loop1\n'

  # 17 CNAME references: exceeds the 16-reference limit
  for i in $(seq 0 16); do
    printf 'c%d IN CNAME c%d\n' "$i" "$((i + 1))"
  done
  printf 'c17 IN A <ip-address>\n'

  # DNAME target is 255 octets; adding the "x" label exceeds 255
  printf 'd IN DNAME %s\n' "$DNATARGET"
  printf 'x.d IN CNAME c0.example.com.\n'
} > "$tmp/example.zone"

Notes:

  • $INCLUDE is intentionally not expanded by the parser; pass all included files explicitly as arguments so input order stays deterministic and no filesystem traversal is implicit.
  • Multiple SOA zones per file are handled best-effort, but one zone per file is the reliable layout.
  • Glue is considered only when the address record is sourced from the same parent zone file as the delegation, preventing authoritative address data in another zone from being misclassified as glue.
