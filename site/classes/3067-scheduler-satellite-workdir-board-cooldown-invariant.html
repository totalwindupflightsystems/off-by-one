Fixing the scheduler satellite workdir / board / cooldown invariant

Problem class: scheduler-satellite-workdir-board-cooldown-invariant Status of live acceptance: OPEN — completed bounded read-only audit + guarded repair plan. No live repair is authorized without a project-scoped durable policy transaction.

1. Invariant and symptom

For every enabled scheduler lane:

# Invariant Source of truth
1 namespace == FAMILY_NAMESPACE[family] trusted checker
2 cadence_seconds == FAMILY_CADENCE[family] trusted checker
3 resolved board == realpath(primary board) operator config
4 <board>/tasks.jsonl exists and is valid JSONL board on disk
5 API detail.project == lanes.project DB, not the API envelope

Symptom. Satellite lanes ran with cooldown 3600 regardless of family (only docs is canonically 3600; core = 300, ci = 900), some satellite boards did not resolve to the primary board, and the canonical checker still reported "clean".

2. Root-cause analysis

  1. Fleet-wide policy writer — proposed a single literal 3600 for every lane instead of each family's canonical cadence; a configuration-wide stomp, not a scoped repair.
  2. Narrow canonical checker — it walks only files with four suffixes (.jsonl, .yaml, .yml, .json) and accepts an empty board directory, so it cannot see per-lane cadence/namespace drift, missing boards, or non-primary boards.
  3. Envelope-vs-detail bug — the prior audit compared the API envelope (metadata.project) instead of authoritative detail.project.
  4. No fail-closed behavior — malformed checker JSON, zero-match censuses, and inconsistent exit/result combinations were treated as success.

3. The fix

3.1 scripts/trusted_checker.py — single source of canonical literals

#!/usr/bin/env python3
"""Trusted canonical lane checker (operator-owned).

Single source of truth for canonical family namespace/cadence literals.
`audit-lanes.py` imports these maps instead of re-declaring constants.

Known limitations the audit must compensate for: it validates only files with
one of SUFFIXES, and an empty board directory produces zero findings.
"""
from __future__ import annotations

import argparse
import json
import os
import sys

FAMILY_NAMESPACE: dict[str, str] = {
    "core": "factory.core",
    "ci": "factory.ci",
    "docs": "factory.docs",
}
FAMILY_CADENCE: dict[str, int] = {
    "core": 300,
    "ci": 900,
    "docs": 3600,
}
SUFFIXES = (".jsonl", ".yaml", ".yml", ".json")


def canonical_namespace(family: str) -> str | None:
    return FAMILY_NAMESPACE.get(family)


def canonical_cadence(family: str) -> int | None:
    return FAMILY_CADENCE.get(family)


def _walk_suffixes(board: str):
    for root, _dirs, files in os.walk(board):
        for name in sorted(files):
            suffix = os.path.splitext(name)[1]
            if suffix in SUFFIXES:
                yield os.path.join(root, name), suffix


def audit(db: str, toml: str, board: str, skills: str) -> dict:
    results = []
    for path, suffix in _walk_suffixes(board):
        ok = os.path.getsize(path) >= 0
        results.append(
            {"path": path, "suffix": suffix, "ok": ok, "code": "ok" if ok else "bad"}
        )
    return {"results": results}


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="canonical lane checker")
    parser.add_argument("--db", required=True)
    parser.add_argument("--toml", required=True)
    parser.add_argument("--board", required=True)
    parser.add_argument("--skills", required=True)
    args = parser.parse_args(argv)
    payload = audit(args.db, args.toml, args.board, args.skills)
    json.dump(payload, sys.stdout)
    sys.stdout.write("\n")
    return 0 if all(r["ok"] for r in payload["results"]) else 1


if __name__ == "__main__":
    raise SystemExit(main())

3.2 scripts/audit-lanes.py — read-only, safe-field, fail-closed audit

#!/usr/bin/env python3
"""Bounded, read-only audit of the scheduler lane board/cooldown invariant.

Read-only: opens SQLite with mode=ro and never writes DB/TOML/boards/skills.
No dispatch, provisioning or remote execution.

Exit codes: 0 clean, 1 BLOCK findings, 2 untrusted input/shape.
"""
from __future__ import annotations

import argparse
import importlib.util
import json
import os
import sqlite3
import subprocess
import sys
from pathlib import Path
from typing import Any

SCHEMA = "lane-audit/v1"

# Only these keys may ever appear in projected output. Prompts, gateway fields
# and arbitrary checker diagnostics are deliberately absent.
SAFE_LANE_FIELDS = (
    "lane_id", "lane_name", "family", "enabled", "scope", "severity", "code",
    "message", "board", "cadence_seconds", "expected_cadence_seconds",
    "namespace", "expected_namespace", "project", "workdir", "skills",
)
SEVERITY_BLOCK = "BLOCK"
REQUIRED_CHECKER_RESULT_KEYS = {"path", "suffix", "ok", "code"}


class HardError(Exception):
    """Audit could not be trusted -> exit 2."""


def _require_file(path: str, label: str) -> str:
    p = Path(path)
    if not p.is_file():
        raise HardError(f"{label} is not a readable file: {path}")
    if not os.access(p, os.R_OK):
        raise HardError(f"{label} is not readable: {path}")
    return str(p)


def _require_dir(path: str, label: str) -> str:
    p = Path(path)
    if not p.is_dir():
        raise HardError(f"{label} is not a directory: {path}")
    return str(p)


def load_canonical(checker_path: str) -> tuple[dict[str, str], dict[str, int]]:
    """Canonical maps come from the trusted checker, never local literals."""
    _require_file(checker_path, "checker")
    spec = importlib.util.spec_from_file_location("_trusted_checker", checker_path)
    if spec is None or spec.loader is None:
        raise HardError(f"cannot import checker: {checker_path}")
    module = importlib.util.module_from_spec(spec)
    try:
        spec.loader.exec_module(module)
    except Exception as exc:  # noqa: BLE001
        raise HardError(f"checker import failed: {exc}") from exc
    ns = getattr(module, "FAMILY_NAMESPACE", None)
    cad = getattr(module, "FAMILY_CADENCE", None)
    if not isinstance(ns, dict) or not all(
        isinstance(k, str) and isinstance(v, str) for k, v in ns.items()
    ):
        raise HardError("checker FAMILY_NAMESPACE is missing or malformed")
    if not isinstance(cad, dict) or not all(
        isinstance(k, str) and isinstance(v, int) and not isinstance(v, bool)
        for k, v in cad.items()
    ):
        raise HardError("checker FAMILY_CADENCE is missing or malformed")
    if set(ns) != set(cad):
        raise HardError("checker namespace/cadence families disagree")
    return ns, cad


def read_lanes(db_path: str, table: str, prefix: str) -> list[dict[str, Any]]:
    if not table.replace("_", "").isalnum():
        raise HardError(f"invalid lane table name: {table!r}")
    try:
        con = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
    except sqlite3.Error as exc:
        raise HardError(f"cannot open DB read-only: {exc}") from exc
    con.row_factory = sqlite3.Row
    try:
        # No `WHERE enabled = 1`: disabled lanes stay in the census so drift is
        # visible; only enforcement is skipped for them.
        sql = (
            f"SELECT id, name, family, project, workdir, board, skills, "
            f"enabled, cadence_seconds, namespace FROM {table} "
            f"WHERE name LIKE ? ORDER BY id"
        )
        try:
            rows = [dict(r) for r in con.execute(sql, (f"{prefix}%",))]
        except sqlite3.Error as exc:
            raise HardError(f"lane query failed: {exc}") from exc
    finally:
        con.close()
    if not rows:
        raise HardError("zero-match lane census (refusing to report success)")
    return rows


def run_checker(
    checker_path: str, db: str, toml: str, board: str, skills: str
) -> list[dict[str, Any]]:
    proc = subprocess.run(
        [sys.executable, checker_path, "--db", db, "--toml", toml,
         "--board", board, "--skills", skills],
        capture_output=True, text=True, timeout=120,
    )
    try:
        payload = json.loads(proc.stdout)
    except json.JSONDecodeError as exc:
        raise HardError(f"checker emitted non-JSON output: {exc}") from exc
    # Exact-shape rejection: only {"results": [{path,suffix,ok,code}, ...]}.
    if not isinstance(payload, dict) or set(payload) != {"results"}:
        raise HardError(f"malformed checker JSON envelope: {sorted(payload)[:8]}")
    results = payload["results"]
    if not isinstance(results, list):
        raise HardError("malformed checker JSON: results is not a list")
    for item in results:
        if not isinstance(item, dict) or not REQUIRED_CHECKER_RESULT_KEYS.issubset(item):
            raise HardError(f"malformed checker result row: {item!r}")
        if not isinstance(item["ok"], bool) or not isinstance(item["path"], str):
            raise HardError(f"malformed checker result types: {item!r}")
    all_ok = all(item["ok"] for item in results)
    if all_ok and proc.returncode != 0:
        raise HardError(f"checker exit {proc.returncode} but all results ok")
    if not all_ok and proc.returncode == 0:
        raise HardError("checker exit 0 but results contain failures")
    findings = [
        {"scope": "global", "severity": SEVERITY_BLOCK, "code": "checker.failure",
         "message": f"{item['code']}: {item['path']}"}
        for item in results if not item["ok"]
    ]
    # Empty board dir => checker sees zero files; surface the coverage gap.
    if not results:
        findings.append({
            "scope": "primary-board", "severity": SEVERITY_BLOCK,
            "code": "checker.coverage_gap",
            "message": "checker produced zero results (empty board dir?)",
        })
    return findings


def _resolve_board(lane: dict[str, Any]) -> str:
    board = str(lane["board"])
    if not os.path.isabs(board):
        board = os.path.join(str(lane["workdir"]), board)
    return os.path.realpath(board)


def _invalid_jsonl_line(path: str) -> int | None:
    try:
        with open(path, "r", encoding="utf-8") as fh:
            for lineno, raw in enumerate(fh, start=1):
                text = raw.strip()
                if not text:
                    continue
                try:
                    obj = json.loads(text)
                except json.JSONDecodeError:
                    return lineno
                if not isinstance(obj, dict):
                    return lineno
    except OSError:
        return 1
    return None


def check_lanes(lanes, primary_board, namespace, cadence) -> list[dict[str, Any]]:
    findings: list[dict[str, Any]] = []
    resolved_primary = os.path.realpath(primary_board)
    if not os.path.isdir(resolved_primary):
        findings.append({
            "scope": "primary-board", "severity": SEVERITY_BLOCK,
            "code": "primary.missing",
            "message": f"primary board missing: {resolved_primary}",
        })
    for lane in lanes:
        enabled = bool(lane["enabled"])
        base = {
            "lane_id": lane["id"], "lane_name": lane["name"],
            "family": lane["family"], "enabled": enabled,
            "board": str(lane["board"]), "workdir": str(lane["workdir"]),
            "skills": str(lane["skills"]), "namespace": str(lane["namespace"]),
            "cadence_seconds": lane["cadence_seconds"], "project": lane["project"],
        }
        if not enabled:
            continue  # census only, not enforced
        family = str(lane["family"])
        exp_ns, exp_cad = namespace.get(family), cadence.get(family)
        if exp_ns is None or exp_cad is None:
            findings.append({**base, "scope": "per-lane", "severity": SEVERITY_BLOCK,
                "code": "family.unknown",
                "message": f"family {family!r} absent from trusted checker",
                "expected_namespace": None, "expected_cadence_seconds": None})
            continue
        if str(lane["namespace"]) != exp_ns:
            findings.append({**base, "scope": "per-lane", "severity": SEVERITY_BLOCK,
                "code": "namespace.drift",
                "message": f"namespace {lane['namespace']!r} != {exp_ns!r}",
                "expected_namespace": exp_ns, "expected_cadence_seconds": exp_cad})
        if lane["cadence_seconds"] != exp_cad:
            findings.append({**base, "scope": "per-lane", "severity": SEVERITY_BLOCK,
                "code": "cadence.drift",
                "message": f"cadence {lane['cadence_seconds']} != canonical {exp_cad}",
                "expected_namespace": exp_ns, "expected_cadence_seconds": exp_cad})
        lane_board = _resolve_board(lane)
        if lane_board != resolved_primary:
            findings.append({**base, "scope": "per-lane", "severity": SEVERITY_BLOCK,
                "code": "board.not_primary",
                "message": f"resolved board {lane_board} != {resolved_primary}",
                "expected_namespace": exp_ns, "expected_cadence_seconds": exp_cad})
        tasks = os.path.join(lane_board, "tasks.jsonl")
        if not os.path.isfile(tasks):
            findings.append({**base, "scope": "per-lane", "severity": SEVERITY_BLOCK,
                "code": "board.tasks_missing",
                "message": f"tasks.jsonl missing under {lane_board}",
                "expected_namespace": exp_ns, "expected_cadence_seconds": exp_cad})
        else:
            bad = _invalid_jsonl_line(tasks)
            if bad is not None:
                findings.append({**base, "scope": "per-lane", "severity": SEVERITY_BLOCK,
                    "code": "board.tasks_invalid",
                    "message": f"tasks.jsonl invalid at line {bad}",
                    "expected_namespace": exp_ns, "expected_cadence_seconds": exp_cad})
    return findings


def check_api(api_path: str | None, lanes) -> list[dict[str, Any]]:
    if api_path is None:
        return []
    _require_file(api_path, "api-dump")
    try:
        payload = json.loads(Path(api_path).read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        raise HardError(f"api-dump is not JSON: {exc}") from exc
    items = payload.get("items") if isinstance(payload, dict) else None
    if not isinstance(items, list):
        raise HardError("api-dump must be {'items': [...]}")
    by_id = {str(lane["id"]): lane for lane in lanes}
    findings: list[dict[str, Any]] = []
    for item in items:
        if not isinstance(item, dict):
            raise HardError("api-dump item is not an object")
        detail = item.get("detail")
        if not isinstance(detail, dict):
            continue
        lane_id = detail.get("lane_id")
        if lane_id is None:
            continue
        lane = by_id.get(str(lane_id))
        if lane is None:
            continue
        # Exact field comparison; envelope (metadata.project) is ignored.
        if "project" not in detail:
            findings.append({"scope": "per-lane", "severity": SEVERITY_BLOCK,
                "code": "api.detail_missing",
                "message": f"detail.project absent for lane {lane_id}",
                "lane_id": lane["id"], "lane_name": lane["name"],
                "project": lane["project"]})
        elif str(detail["project"]) != str(lane["project"]):
            findings.append({"scope": "per-lane", "severity": SEVERITY_BLOCK,
                "code": "api.project_mismatch",
                "message": f"detail.project {detail['project']!r} != DB {lane['project']!r}",
                "lane_id": lane["id"], "lane_name": lane["name"],
                "project": lane["project"]})
    return findings


def build_root_map(db, toml, board, skills):
    home = os.path.expanduser("~")
    pairs = [
        (os.path.realpath(board), "$BOARD"),
        (os.path.realpath(os.path.dirname(db)), "$DB_ROOT"),
        (os.path.realpath(os.path.dirname(toml)), "$CONFIG_ROOT"),
        (os.path.realpath(skills), "$SKILLS"),
        (os.path.realpath(home), "$HOME"),
    ]
    return sorted(pairs, key=lambda p: len(p[0]), reverse=True)


def normalize(value: str, roots) -> str:
    out = str(value)
    for root, token in roots:
        if root and out.startswith(root):
            return token + out[len(root):]
    return out


def project(finding, roots):
    return {
        key: normalize(finding[key], roots) if isinstance(finding.get(key), str)
        else finding[key]
        for key in SAFE_LANE_FIELDS if key in finding
    }


def audit(args: argparse.Namespace) -> tuple[int, dict[str, Any]]:
    db = _require_file(args.db, "db")
    toml = _require_file(args.toml, "toml")
    board = _require_dir(args.board, "board")
    skills = _require_dir(args.skills, "skills")
    namespace, cadence = load_canonical(args.checker)
    lanes = read_lanes(db, args.lanes_table, args.prefix)
    findings = []
    findings.extend(run_checker(args.checker, db, toml, board, skills))
    findings.extend(check_lanes(lanes, board, namespace, cadence))
    findings.extend(check_api(args.api_dump, lanes))
    roots = build_root_map(db, toml, board, skills)
    safe_findings = [project(f, roots) for f in findings]
    enabled = sum(1 for lane in lanes if lane["enabled"])
    block = any(f.get("severity") == SEVERITY_BLOCK for f in safe_findings)
    result = {
        "schema": SCHEMA, "ok": not block,
        "census": {"lanes": len(lanes), "enabled": enabled,
                   "disabled": len(lanes) - enabled},
        "findings": safe_findings,
    }
    return (1 if block else 0), result


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--db", required=True)
    parser.add_argument("--toml", required=True)
    parser.add_argument("--board", required=True)
    parser.add_argument("--skills", required=True)
    parser.add_argument("--checker", required=True)
    parser.add_argument("--lanes-table", default="lanes")
    parser.add_argument("--prefix", default="")
    parser.add_argument("--api-dump", default=None)
    args = parser.parse_args(argv)
    try:
        code, result = audit(args)
    except HardError as exc:
        json.dump({"schema": SCHEMA, "ok": False, "error": str(exc)},
                  sys.stdout, sort_keys=True)
        sys.stdout.write("\n")
        return 2
    json.dump(result, sys.stdout, indent=2, sort_keys=True)
    sys.stdout.write("\n")
    return code


if __name__ == "__main__":
    raise SystemExit(main())

3.3 scripts/scoped-lane-repair.py — reject fleet-wide writers, require durable txn

#!/usr/bin/env python3
"""Scoped cadence repair planner / guarded applier.

Reads canonical cadences from the trusted checker, refuses non-scoped plans,
refuses a non-canonical 3600 target (the previous fleet-wide writer), protects
the primary board/skills, is dry-run by default, and requires a durable
project-scoped policy transaction before any write.
"""
from __future__ import annotations

import argparse
import importlib.util
import json
import re
import sqlite3
import sys

TXN_RE = re.compile(r"^policytxn:[A-Za-z0-9._/-]+$")
FLEET_WIDE_SENTINEL = 3600
PROTECTED_FIELDS = ("board", "skills")


class Refused(Exception):
    pass


def load_canonical(checker_path: str) -> dict[str, int]:
    spec = importlib.util.spec_from_file_location("_trusted_checker", checker_path)
    if spec is None or spec.loader is None:
        raise Refused(f"cannot import checker {checker_path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    cad = getattr(module, "FAMILY_CADENCE", None)
    if not isinstance(cad, dict):
        raise Refused("checker FAMILY_CADENCE malformed")
    return cad


def read_lanes(db_path: str, lane_ids: list[int]) -> list[dict]:
    con = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
    con.row_factory = sqlite3.Row
    try:
        placeholders = ",".join("?" for _ in lane_ids)
        rows = [dict(r) for r in con.execute(
            f"SELECT id,name,family,board,skills,cadence_seconds FROM lanes "
            f"WHERE id IN ({placeholders}) ORDER BY id", lane_ids)]
    finally:
        con.close()
    if len(rows) != len(set(lane_ids)):
        raise Refused(f"unknown lane id(s): {sorted(set(lane_ids) - {r['id'] for r in rows})}")
    return rows


def plan_repair(checker_path, db_path, lane_ids, apply, policy_txn, set_cadence=None) -> dict:
    if not lane_ids:
        raise Refused("no --lanes scope: fleet-wide repair refused")
    if len(set(lane_ids)) != len(lane_ids):
        raise Refused("duplicate lane ids in scope")
    cadence = load_canonical(checker_path)
    lanes = read_lanes(db_path, lane_ids)

    # A caller-supplied target is valid only if canonical for every family.
    if set_cadence is not None:
        if set_cadence == FLEET_WIDE_SENTINEL:
            offending = sorted({str(l["family"]) for l in lanes
                                if cadence.get(str(l["family"])) != FLEET_WIDE_SENTINEL})
            if offending:
                raise Refused("fleet-wide 3600 write is not canonical for families: "
                              + ", ".join(offending))
        for lane in lanes:
            family = str(lane["family"])
            if cadence.get(family) != set_cadence:
                raise Refused(f"target cadence {set_cadence} is not canonical for "
                              f"family {family!r} ({cadence.get(family)})")

    changes = []
    for lane in lanes:
        family = str(lane["family"])
        expected = cadence.get(family)
        if expected is None:
            raise Refused(f"family {family!r} not in trusted checker")
        target = set_cadence if set_cadence is not None else expected
        if lane["cadence_seconds"] != target:
            changes.append({"lane_id": lane["id"], "lane_name": lane["name"],
                            "family": family, "field": "cadence_seconds",
                            "from": lane["cadence_seconds"], "to": target})

    plan = {"scope": {"lane_ids": sorted(set(lane_ids))},
            "protected_fields": list(PROTECTED_FIELDS),
            "changes": changes, "apply": bool(apply)}
    if apply:
        if not policy_txn or not TXN_RE.match(policy_txn):
            raise Refused("refusing live write: --policy-txn must be a scoped durable "
                          "handle of the form policytxn:&lt;project&gt;/<id>")
        plan["policy_txn"] = policy_txn
        plan["applied"] = False
        plan["note"] = "durable transaction API not available in this checkout"
    return plan


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--db", required=True)
    parser.add_argument("--checker", required=True)
    parser.add_argument("--lanes", required=True, help="comma-separated lane ids")
    parser.add_argument("--apply", action="store_true")
    parser.add_argument("--policy-txn", default=None)
    parser.add_argument("--set-cadence", type=int, default=None,
                        help="simulate/reject a fleet-wide writer target")
    args = parser.parse_args(argv)
    try:
        lane_ids = [int(x) for x in args.lanes.split(",") if x.strip()]
    except ValueError:
        print(json.dumps({"ok": False, "error": "invalid --lanes"}))
        return 2
    try:
        plan = plan_repair(args.checker, args.db, lane_ids, args.apply,
                           args.policy_txn, args.set_cadence)
    except Refused as exc:
        print(json.dumps({"ok": False, "refused": str(exc)}, sort_keys=True))
        return 3
    print(json.dumps({"ok": True, **plan}, indent=2, sort_keys=True))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
