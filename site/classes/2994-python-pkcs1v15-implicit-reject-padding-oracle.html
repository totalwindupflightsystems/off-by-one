RSA PKCS#1 v1.5 Decryption with Implicit Rejection

Root-cause analysis

A normal PKCS#1 v1.5 decryptor leaks validity through at least one of:

  • raising an exception or returning an error for bad padding,
  • short-circuiting when the separator is found,
  • using the separator position or PS length as a slice index, loop bound, or output length,
  • handling different malformed forms through visibly different code paths.

Any of these gives an attacker a Bleichenbacher-style validity/timing oracle.

Implicit rejection removes the error signal by returning deterministic pseudorandom plaintext for every invalid padding. To make the output length independent of the padding, the caller must supply the protocol’s fixed expected plaintext length L. Therefore a valid block must satisfy:

EM = 00 02 || PS || 00 || M
len(M) = L
separator index = k - L - 1

This lets the implementation extract the valid message from the public index k - L, without ever indexing with the secret separator position.

The implementation below:

  1. Performs the RSA operation for every fixed-length ciphertext.
  2. Scans all possible separator positions with a fixed loop bound.
  3. Uses arithmetic masks instead of branches on secret-derived values.
  4. Returns an HMAC-SHA256-derived fallback of exactly L bytes for all invalid cases.
  5. Mask-selects between the real message and fallback without a branch.

Exact fix

Save this as pkcs1v15_ir.py:

"""RSAES-PKCS1-v1_5 decryption with implicit rejection.

Only the Python standard library is used. The padding checker has a fixed
control-flow graph and performs a fixed number of byte operations for a
given RSA modulus size. It never branches on, or indexes with, the separator
position or the PS length.
"""

from __future__ import annotations

import hashlib
import hmac


def _eq(a: int, b: int) -> int:
    """Return 1 iff a == b, using arithmetic rather than a branch."""
    x = a ^ b
    return 1 & ((x - 1) >> 63)


def _ge_10(x: int) -> int:
    """Return 1 iff x >= 10."""
    return 1 & ((9 - x) >> 63)


def _derive_rejection_key(n: int, d: int) -> bytes:
    """Derive a private, per-key HMAC key from the RSA private key."""
    nb = n.to_bytes((n.bit_length() + 7) // 8, "big")
    db = d.to_bytes((d.bit_length() + 7) // 8, "big")
    return hashlib.sha256(
        b"RSAES-PKCS1-v1_5 implicit-rejection key\x00" + nb + b"\x00" + db
    ).digest()


def _expand(prf_key: bytes, ciphertext: bytes, length: int) -> bytes:
    """HMAC-SHA256 counter-mode expansion to ``length`` bytes."""
    out = bytearray()
    counter = 0
    prefix = (
        b"RSAES-PKCS1-v1_5 implicit-rejection plaintext\x00"
        + len(ciphertext).to_bytes(8, "big")
        + ciphertext
    )
    while len(out) < length:
        out += hmac.new(
            prf_key,
            prefix + counter.to_bytes(8, "big"),
            hashlib.sha256,
        ).digest()
        counter += 1
    return bytes(out[:length])


def decrypt_implicit_reject(
    ciphertext: bytes,
    n: int,
    d: int,
    expected_len: int | None = None,
    *,
    rejection_key: bytes | None = None,
) -> bytes:
    """Decrypt RSAES-PKCS1-v1_5 with implicit rejection.

    expected_len is the fixed plaintext length expected by the protocol.
    If omitted, it defaults to k-11, the maximum PKCS#1 v1.5 message length.

    Returns exactly expected_len bytes. Valid ciphertexts return the original
    message byte-for-byte. Invalid ciphertexts return a deterministic,
    HMAC-SHA256-derived fallback. No padding exception is raised.
    """
    k = (n.bit_length() + 7) // 8
    if expected_len is None:
        expected_len = k - 11
    if not (0 <= expected_len <= k - 11):
        raise ValueError("expected_len must be in [0, k-11]")

    if rejection_key is None:
        rejection_key = _derive_rejection_key(n, d)

    ct = bytes(ciphertext)
    fallback = _expand(rejection_key, ct, expected_len)

    # Wrong length is a public error; still return the rejection plaintext
    # rather than creating an error oracle.
    if len(ct) != k:
        return fallback

    c = int.from_bytes(ct, "big")
    c_is_in_range = int(c < n)

    # Reduce even for c >= n so the RSA work is unchanged.
    m = pow(c % n, d, n)
    em = m.to_bytes(k, "big")

    # Fixed loop over every possible separator position. sep is selected
    # arithmetically; it is never used as an index or loop bound.
    found = 0
    sep = 0
    for i in range(2, k):
        is_zero = _eq(em[i], 0)
        take = is_zero & (1 ^ found)
        sep = take * i + (1 - take) * sep
        found |= is_zero

    expected_sep = k - expected_len - 1
    valid = (
        _eq(em[0], 0)
        & _eq(em[1], 2)
        & found
        & _ge_10(sep)
        & _eq(sep, expected_sep)
        & c_is_in_range
    )

    # Branchless select between the real message and the fallback.
    # For a valid block, the message always starts at public index k-L.
    mask = (-valid) & 0xFF
    inverse = 0xFF ^ mask
    start = k - expected_len
    return bytes(
        ((em[start + i] & mask) | (fallback[i] & inverse))
        for i in range(expected_len)
    )


decrypt = decrypt_implicit_reject


def eme_pkcs1v15_encode(
    message: bytes, k: int, ps: bytes | None = None
) -> bytes:
    """Deterministic reference EME-PKCS1-v1_5 encoder."""
    ps_len = k - len(message) - 3
    if ps_len < 8:
        raise ValueError("message too long for PKCS#1 v1.5")
    if ps is None:
        ps = bytes((i % 255) + 1 for i in range(ps_len))
    if len(ps) != ps_len or 0 in ps:
        raise ValueError("invalid PS")
    return b"\x00\x02" + ps + b"\x00" + message


def rsa_encrypt_em(em: bytes, e: int, n: int) -> bytes:
    k = (n.bit_length() + 7) // 8
    c = pow(int.from_bytes(em, "big"), e, n)
    return c.to_bytes(k, "big")

Important: pass the protocol’s exact message length as expected_len. The default k - 11 only suits maximum-length messages. Derive rejection_key once and reuse it to avoid repeated key derivation. This fixes the padding-oracle surface; production RSA exponentiation should use a hardened constant-time RSA backend.

The following script checks:

  • byte-identical valid decryption,
  • deterministic fallback for zero-length PS, short PS, wrong separator, and missing separator,
  • timing indistinguishability across valid and four malformed classes,
  • 4096 adaptive chosen-ciphertext queries against the implicit-reject oracle.

    "zero_length_ps": b"\x00\x02\x00" + r(K - 3),
    "short_ps": b"\x00\x02\x01\x02\x03\x00" + r(K - 6),
    "wrong_separator": p.eme_pkcs1v15_encode(
        bytes((i * 29 + 7) & 255 for i in range(L + 1)), K
    ),
    "missing_separator": b"\x00\x02" + bytes(x or 1 for x in r(K - 2)),
}[kind]

    em = p.eme_pkcs1v15_encode(msg, K)

    "short_ps",
    "wrong_separator",
    "missing_separator",

# 3. Interleaved timing samples.

    "valid",
    "zero_length_ps",
    "short_ps",
    "wrong_separator",
    "missing_separator",

x = {k: [] for k in kinds}
for _ in range(500):
    for k in kinds:
        t = time.perf_counter_ns()
        p.decrypt(cts[k], N, D, L, rejection_key=RK)
        x[k].append(time.perf_counter_ns() - t)

def welch(a, b):
    se = math.sqrt(
        statistics.variance(a) / len(a)
        + statistics.variance(b) / len(b)
    )
    z = abs(statistics.mean(a) - statistics.mean(b)) / se
    return math.erfc(z / math.sqrt(2))

print("timing means (ns):", {
    k: round(statistics.mean(v), 1) for k, v in x.items()
})
print("minimum pairwise Welch p:", min(
    welch(x[a], x[b])
    for i, a in enumerate(kinds)
    for b in kinds[i + 1:]
))

# 4. 4096 adaptive queries: half valid with random messages, half malformed.
# The next malformed class is selected using the previous oracle output.
rng = random.Random(0xB16B00B5)
prev = b"\x00" * L
valid, invalid = [], []

for i in range(4096):
    if i & 1:
        msg = bytes(a ^ b for a, b in zip(rng.randbytes(L), prev))
        c = ct(p.eme_pkcs1v15_encode(msg, K))
        label = 1
    else:
        kind = (
            "zero_length_ps",
            "short_ps",
            "wrong_separator",
            "missing_separator",
        )[prev[0] & 3]

        if kind == "zero_length_ps":
            em = b"\x00\x02\x00" + rng.randbytes(K - 3)
        elif kind == "short_ps":
            em = (
                b"\x00\x02"
                + bytes(a or 1 for a in rng.randbytes(3))
                + b"\x00"
                + rng.randbytes(K - 6)
            )
        elif kind == "wrong_separator":
            em = p.eme_pkcs1v15_encode(rng.randbytes(L + 1), K)
        else:
            em = b"\x00\x02" + bytes(
                a or 1 for a in rng.randbytes(K - 2)
            )

        c = ct(em)
        label = 0

    out = p.decrypt(c, N, D, L, rejection_key=RK)
    assert len(out) == L
    (valid if label else invalid).append(out)
    prev = out

# A legacy validity oracle is a perfect classifier by definition.
# The implicit output is tested with a simple first-byte predictor.
acc_v = sum(b[0] & 1 for b in valid) / len(valid)
acc_i = sum(b[0] & 1 for b in invalid) / len(invalid)
balanced = 0.5 * (acc_v + 1 - acc_i)
se = math.sqrt(0.25 / len(valid) + 0.25 / len(invalid))

ca, cb = [0] * 16, [0] * 16
for b in valid:
    ca[b[0] >> 4] += 1
for b in invalid:
    cb[b[0] >> 4] += 1

chi2 = sum(
    (a - len(valid) * (a + b) / 4096) ** 2
    / (len(valid) * (a + b) / 4096)
    + (b - len(invalid) * (a + b) / 4096) ** 2
    / (len(invalid) * (a + b) / 4096)
    for a, b in zip(ca, cb)
    if a + b
)

df = 15
z = ((chi2 / df) ** (1 / 3) - (1 - 2 / (9 * df))) / math.sqrt(
    2 / (9 * df)
)
pval = 0.5 * math.erfc(z / math.sqrt(2))

print(
    f"4096 adaptive queries: legacy=1.000, "
    f"implicit predictor={balanced:.3f} +/- {1.96 * se:.3f}, "
    f"chi2={chi2:.2f} p={pval:.3f}"
)
