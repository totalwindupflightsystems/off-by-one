Relocating from __future__ import annotations After the Module Docstring

Symptom

After a mechanical AST pass injected from __future__ import annotations into ~100 scripts, ~45 unrelated pytest tests failed with two signatures:

  1. assert 'overload' in '' — a docs test read mod.__doc__ and got an empty/None value.
  2. AttributeError: 'NoneType' object has no attribute 'splitlines' — a CLI script did __doc__.splitlines() for help text and crashed.

The string literal is physically still present; only its role changed.

Root cause

In CPython, the module docstring is defined positionally: it is recognized only when the string literal is the first statement in the module body. ast.get_docstring mirrors:

body = tree.body
if body and isinstance(body[0], ast.Expr) and isinstance(body[0].value, ast.Constant) \
        and isinstance(body[0].value.value, str):
    return body[0].value.value   # this is __doc__
return None

from __future__ import ... is itself a statement, and PEP 236/PEP 563 require future imports near the top — but the module docstring must still precede them. In files whose first physical line is a shebang (optionally followed by blank/# comment lines), the docstring is not line 1. An injector using a line heuristic ("insert after the leading comments/blank lines") lands before the docstring:

#!/usr/bin/env python3
from __future__ import annotations   # <-- inserted here (wrong)
"""Real module docstring."""

Parse tree:

body[0] = ImportFrom(__future__, annotations)
body[1] = Expr(Constant("Real module docstring."))   # now a no-op expression

Because the string is no longer body[0], it is not a docstring and mod.__doc__ is None. The literal is still there, which is why the failure looks so confusing. Fixers/injectors must anchor on AST node positions (lineno/end_lineno), never shebang/comment/blank-line heuristics.

Exact fix

Detect a leading run of __future__ imports followed by a bare string Expr, remove those physical lines, and re-insert them immediately after the docstring's end_lineno. Works from AST positions; preserves shebang/encoding lines; handles multi-line docstrings, multiple future imports, CRLF; idempotent.

2. The fixer — scripts/tools/fix_future_import_docstring.py

#!/usr/bin/env python3
"""Relocate a misplaced ``from __future__ import annotations`` after the module docstring."""
from __future__ import annotations

import argparse
import ast
import io
import pathlib
import sys
import tokenize
from typing import Iterable

FUTURE = "__future__"


def _decode(raw: bytes) -> tuple[str, str]:
    """Return (text, encoding), honoring PEP 263 coding cookies."""
    try:
        encoding, _ = tokenize.detect_encoding(io.BytesIO(raw).readline)
    except SyntaxError:
        encoding = "utf-8"
    return raw.decode(encoding), encoding


def _is_future_import(node: ast.AST) -> bool:
    return isinstance(node, ast.ImportFrom) and node.module == FUTURE


def _is_string_expr(node: ast.AST) -> bool:
    return (
        isinstance(node, ast.Expr)
        and isinstance(node.value, ast.Constant)
        and isinstance(node.value.value, str)
    )


def find_misplaced(tree: ast.Module) -> tuple[list[ast.ImportFrom], ast.Expr] | None:
    """Return (future_imports, docstring_expr) if the docstring is displaced."""
    body = tree.body
    for index, node in enumerate(body):
        if _is_string_expr(node):
            if index == 0:
                return None                       # already correct
            imports = body[:index]
            if all(_is_future_import(n) for n in imports):
                return list(imports), node
            return None
        if not _is_future_import(node):
            return None                           # first real statement, nothing to do
    return None


def fix_source(text: str) -> str | None:
    """Return patched source, or None if no relocation is needed."""
    tree = ast.parse(text)
    found = find_misplaced(tree)
    if found is None:
        return None
    imports, doc = found

    lines = text.splitlines(keepends=True)

    # Full physical span of the contiguous future-import block (incl. inner comments).
    start = imports[0].lineno - 1
    end = imports[-1].end_lineno          # exclusive
    import_block = lines[start:end]

    del lines[start:end]                  # remove the block; docstring shifts up
    removed = end - start

    doc_end = doc.end_lineno - removed    # 0-indexed insert point after the docstring
    lines[doc_end:doc_end] = import_block
    return "".join(lines)


def verify_source(text: str) -> None:
    """Assert the recovered invariant: body[0] is the string, body[1] the future import."""
    body = ast.parse(text).body
    assert body, "empty module"
    assert _is_string_expr(body[0]), "body[0] is not the docstring Expr"
    assert _is_future_import(body[1]), "body[1] is not a __future__ import"
    assert any(a.name == "annotations" for a in body[1].names)
    assert ast.get_docstring(ast.parse(text)) is not None, "mod.__doc__ is still None"


def process(path: pathlib.Path, check_only: bool = False) -> bool:
    raw = path.read_bytes()
    text, encoding = _decode(raw)
    try:
        patched = fix_source(text)
    except SyntaxError as exc:
        print(f"SKIP (unparseable) {path}: {exc}", file=sys.stderr)
        return False
    if patched is None:
        return False
    verify_source(patched)
    if check_only:
        print(f"WOULD FIX {path}")
        return True
    with open(path, "w", encoding=encoding, newline="") as handle:  # newline="" preserves CRLF
        handle.write(patched)
    print(f"FIXED {path}")
    return True


def iter_python_files(paths: Iterable[pathlib.Path]) -> Iterable[pathlib.Path]:
    for target in paths:
        if target.is_dir():
            yield from sorted(target.rglob("*.py"))
        elif target.suffix == ".py":
            yield target


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("paths", nargs="+", type=pathlib.Path)
    parser.add_argument("--check", action="store_true", help="report without writing")
    args = parser.parse_args(argv)

    changed = sum(process(p, args.check) for p in iter_python_files(args.paths))
    print(f"{'would fix' if args.check else 'fixed'} {changed} file(s)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

#!/usr/bin/env python3
"""Fail if any file has a __future__ import before its module docstring."""
from __future__ import annotations

import ast
import pathlib
import sys


def is_string_expr(node):
    return (isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant)
            and isinstance(node.value.value, str))


def is_future(node):
    return isinstance(node, ast.ImportFrom) and node.module == "__future__"


def main(root: str = ".") -> int:
    failures = []
    for path in sorted(pathlib.Path(root).rglob("*.py")):
        try:
            body = ast.parse(path.read_bytes()).body
        except SyntaxError as exc:
            failures.append(f"{path}: SyntaxError: {exc}")
            continue
        if len(body) >= 2 and is_future(body[0]) and is_string_expr(body[1]):
            failures.append(f"{path}: __future__ import precedes the module docstring")
    if failures:
        print("\n".join(failures), file=sys.stderr)
        print(f"FAIL: {len(failures)} file(s)", file=sys.stderr)
        return 1
    print("OK: no displaced module docstrings")
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1] if len(sys.argv) > 1 else "."))

Prevention

Anchor the injector on AST positions:

  1. If body[0] is a string Expr, insert after body[0].end_lineno, then past any existing __future__ imports.
  2. Otherwise, insert at body[0].lineno - 1 (before the first statement), keeping the shebang/encoding cookie above the import.
def insert_future_import(source: str) -> str:
    tree = ast.parse(source)
    body = tree.body
    if any(isinstance(n, ast.ImportFrom) and n.module == "__future__"
           and any(a.name == "annotations" for a in n.names) for n in body):
        return source                                     # already present

    i = 0
    if body and is_string_expr(body[0]):
        i = 1
    while i < len(body) and isinstance(body[i], ast.ImportFrom) and body[i].module == "__future__":
        i += 1

    if i > 0:
        insert_at = body[i - 1].end_lineno                # after last frozen statement
    else:
        insert_at = body[0].lineno - 1 if body else 0     # before first statement

    lines = source.splitlines(keepends=True)
    lines.insert(insert_at, "from __future__ import annotations\n")
    return "".join(lines)

Never insert relative to shebang/comments/blank lines: comments carry no AST node, so only tree.body[*].lineno / .end_lineno reliably describe statement boundaries — and the module docstring is a statement whose position is semantically load-bearing.
