DOC-1 — Falsifying the README "session status" claims at runtime

Repo: get-h3/sdk-typescript · Commit: 9c1f209 · Task: DOC-1 Artifact: /tmp/doc1-runtime-probe.mts · Result: 8/8 PASS

1. What was claimed

Commit 9c1f209 added to the GET /v1/sessions/:id section of README.md: first /v1/process decision end ⇒ session starts completed, else active; a later /v1/process or /v1/result end ⇒ non-cancelled session becomes completed; /v1/cancel marks cancelled; DELETE removes it; a cancelled session cannot later complete; expired is in the enum but never produced.

2. Root-cause analysis

These are runtime transitions of an in-process Hono router, so the cheap gates can't decide them:

Gate Why it can't falsify
grep Tokens appear in enum, comments, and README — tokens ≠ branch reachability.
tsc --noEmit status is a union type; every assignment type-checks regardless of precedence.
Existing Vitest Asserts completed in exactly one place (harness.test.ts:751, /v1/result path). No test for first-process-end, later-process-end, cancel, or the negative terminal case. expired exists only in protocol.ts.

The state machine is in src/harness.ts: :129 initializer, :123 later-process flip, :206–211 result flip (guarded !== "cancelled"), :118 early-return that makes cancel terminal, :245/:256 cancel branches, :307 sessions.delete, and protocol.ts:337 enum.

Verified pattern: import the SDK by absolute path (probe stays in /tmp); let onProcess/onResult switch on a module-level mode; drive create/GET/result/GET/cancel/GET/DELETE/GET via app.request(); print PASS/FAIL + actual per claim and map each to its source line.

Sibling pitfall (reproduced): a /tmp/*.ts probe dies under tsx with Top-level await is currently not supported with the "cjs" output format because /tmp has no package.json "type": "module". Name it .mts.

import { createH3Router, SessionStatusSchema } from "/tmp/h3sdk/src/index.ts";
import type { Harness } from "/tmp/h3sdk/src/index.ts";

type Mode = "text" | "end";
let mode: Mode = "text";

function decision() {
  if (mode === "end") {
    return { decision: "end" as const, decision_id: crypto.randomUUID(),
      end: { reason: "task_complete" as const, summary: "done" } };
  }
  return { decision: "text" as const, decision_id: crypto.randomUUID(),
    text: { content: "hello", finished: true } };
}

function makeHarness(opts: { withCancel: boolean }): Harness {
  return {
    async onProcess() { return decision(); },
    async onResult() { return decision(); },
    ...(opts.withCancel ? { async onCancel() { return true; } } : {}),
    health() {
      return { status: "ok", version: "1.0.0", transport: "rest",
        protocol_version: "1.0", capabilities: ["text", "end"] };
    },
  };
}

const appWithCancel = createH3Router(makeHarness({ withCancel: true }));
const appDefault   = createH3Router(makeHarness({ withCancel: false }));

const baseProcess = (session_id: string) => ({
  session_id,
  message: { role: "user", content: "hi", timestamp: new Date().toISOString() },
  identity: { platform: "probe", chat_id: "probe", user_name: "probe", user_id: "probe" },
  context: {
    history: [], tools: [], models: [],
    config: { max_iterations: 10, timeout_seconds: 300 },
    session_state: { turn_count: 0, total_tool_calls: 0, total_llm_calls: 0,
      cost_so_far: 0, started_at: new Date().toISOString() },
  },
});

const post = (app: ReturnType<typeof createH3Router>, path: string, body: unknown) =>
  app.request(path, { method: "POST",
    headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });

const getSession = async (app: ReturnType<typeof createH3Router>, id: string) => {
  const res = await app.request(`/v1/sessions/${id}`);
  const body = (await res.json()) as { status?: string };
  return { status: res.status, body };
};

const TOTAL = 8;
let n = 0, pass = 0;
function check(label: string, actual: unknown, expected: unknown, proves: string) {
  n += 1;
  const ok = actual === expected;
  if (ok) pass += 1;
  console.log(`${ok ? "PASS" : "FAIL"} [${n}/${TOTAL}] ${label} — ` +
    `expected=${JSON.stringify(expected)} actual=${JSON.stringify(actual)}  ⟵ proves ${proves}`);
}

check("status enum is exactly {active,completed,expired,cancelled}",
  JSON.stringify(SessionStatusSchema.options),
  JSON.stringify(["active", "completed", "expired", "cancelled"]),
  "src/protocol.ts:337");

{ const id = "sess-first-end"; mode = "end";
  await post(appWithCancel, "/v1/process", baseProcess(id));
  const { body } = await getSession(appWithCancel, id);
  check("first /v1/process decision=end creates session completed",
    body.status, "completed", "src/harness.ts:129"); }

{ const id = "sess-first-text"; mode = "text";
  await post(appWithCancel, "/v1/process", baseProcess(id));
  const { body } = await getSession(appWithCancel, id);
  check("first /v1/process decision!=end creates session active",
    body.status, "active", "src/harness.ts:129"); }

{ const id = "sess-later-end"; mode = "text";
  await post(appWithCancel, "/v1/process", baseProcess(id));
  mode = "end"; await post(appWithCancel, "/v1/process", baseProcess(id));
  const { body } = await getSession(appWithCancel, id);
  check("later /v1/process decision=end transitions active->completed",
    body.status, "completed", "src/harness.ts:123"); }

{ const id = "sess-result-end"; mode = "text";
  const proc = await post(appWithCancel, "/v1/process", baseProcess(id));
  const pbody = (await proc.json()) as { decision_id: string };
  mode = "end";
  await post(appWithCancel, "/v1/result", { session_id: id, decision_id: pbody.decision_id,
    result: { type: "text_sent", success: true } });
  const { body } = await getSession(appWithCancel, id);
  check("/v1/result decision=end transitions active->completed",
    body.status, "completed", "src/harness.ts:206-211"); }

{ const idA = "sess-cancel-hook", idB = "sess-cancel-default"; mode = "text";
  await post(appWithCancel, "/v1/process", baseProcess(idA));
  await post(appDefault, "/v1/process", baseProcess(idB));
  await post(appWithCancel, "/v1/cancel", { session_id: idA, reason: "user_interrupt" });
  await post(appDefault, "/v1/cancel", { session_id: idB, reason: "user_interrupt" });
  const a = (await getSession(appWithCancel, idA)).body.status;
  const b = (await getSession(appDefault, idB)).body.status;
  check("POST /v1/cancel marks cancelled (onCancel hook AND default branch)",
    `${a}/${b}`, "cancelled/cancelled", "src/harness.ts:245,256"); }

{ const id = "sess-cancel-terminal"; mode = "text";
  await post(appWithCancel, "/v1/process", baseProcess(id));
  await post(appWithCancel, "/v1/cancel", { session_id: id, reason: "user_interrupt" });
  mode = "end";
  const proc = await post(appWithCancel, "/v1/process", baseProcess(id));
  const pbody = (await proc.json()) as { decision_id: string };
  await post(appWithCancel, "/v1/result", { session_id: id, decision_id: pbody.decision_id,
    result: { type: "text_sent", success: true } });
  const { body } = await getSession(appWithCancel, id);
  check("cancelled session does NOT flip to completed on a later end decision",
    body.status, "cancelled", "src/harness.ts:118,206"); }

{ const id = "sess-delete"; mode = "text";
  await post(appWithCancel, "/v1/process", baseProcess(id));
  const del = await appWithCancel.request(`/v1/sessions/${id}`, { method: "DELETE" });
  const delBody = (await del.json()) as { terminated?: boolean };
  const after = await appWithCancel.request(`/v1/sessions/${id}`);
  const afterBody = (await after.json()) as { error?: { code?: string } };
  check("DELETE returns 200 {terminated:true} then GET returns 404",
    `${del.status}/${delBody.terminated}/${after.status}/${afterBody.error?.code}`,
    "200/true/404/SESSION_NOT_FOUND", "src/harness.ts:307"); }

console.log(`\n${pass}/${TOTAL} documented claims verified against runtime behavior.`);
if (pass !== TOTAL) process.exitCode = 1;
