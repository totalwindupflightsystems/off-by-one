Fix: Lowering cyclic WebAssembly linear-memory / GPU-staging alias graphs

Problem class: js20-wasm-gpu-buffer-alias-cycle-20261103 Runtime: Node 20+ (verified on Node 22, ES modules, node:test) Deliverable: wasm-gpu-alias-pass.mjs (pure pass) + wasm-gpu-alias-pass.test.mjs

1. Root-cause analysis

A WebAssembly linear memory view and a GPU staging buffer can refer to the same bytes through different lifetimes. The compiler/runtime must lower this alias graph, but four failure modes are coupled and cannot be handled independently:

  1. Detached views after memory.grow(). grow replaces the backing ArrayBuffer; every old TypedArray view becomes zero-length / detached. A view still carrying the pre-grow generation is a use-after-grow. Any lowering that simply keeps a pointer is wrong.
  2. Stale GPU mappings. A staging buffer that was mapped, then remapped or unmapped and re-issued, has an invalid mapping epoch. Reading or copying through the old mapping is a stale-map hazard.
  3. Hidden alias cycles. Explicit aliases (A↔B) are cheap to collapse, but a cycle can be disguised as copies (A→B→C→A). Lowering those as real copies creates a non-terminating / self-overwriting schedule. They must be detected and rejected, not lowered.
  4. Cross-worker races. A copy that crosses a worker boundary without an acquire/release pair is unsynchronized. Minimizing copies must not erase the synchronization edges that make the remaining copies safe.

Naively union-ing everything or blindly emitting a copy per edge fails: union-ing aliases blindly can merge incompatible alignment; emitting copies blindly respects neither lifetimes nor worker ordering and explodes copy count.

The fix is a three-stage pass: validate lifetimes/alignment → collapse explicit alias cycles into one canonical backing (zero-copy) → condense onto a component DAG and transitively reduce copy edges, rejecting hidden cycles and unsynchronized cross-worker copies. Every output is derived from a total order on ids, so diagnostics are byte-identical across runs.

IR contract

node :=
  | { id, kind:"wasm-view",  byteOffset, byteLength, generation, align, owner }
  | { id, kind:"gpu-staging", size, mapState:"mapped"|"unmapped"|"stale",
      align, owner }
edge := { from, to, alias:boolean, sync:"none"|"acquire"|"release", worker? }

graph := { memory:{ id, byteLength, generation }, nodes, edges }

Invariants enforced:

  • alias:true is a symmetric storage equivalence → union-find class, collapsed to the lexicographically smallest member (canonical).
  • alias:false is a directional copy → must form a DAG after condensation.
  • wasm-view.generation === memory.generation and the view must lie inside memory.byteLength; otherwise E_WASM_USE_AFTER_GROW / E_WASM_OUT_OF_BOUNDS.
  • An in-use gpu-staging must be mapped; otherwise E_GPU_STALE_MAP / E_GPU_UNMAPPED_ACCESS.
  • All members of an alias class share one alignment; otherwise E_ALIGNMENT_MISMATCH.
  • A copy between different owner workers must carry acquire/release; otherwise E_RACE_UNSYNCHRONIZED.

2. The fix — wasm-gpu-alias-pass.mjs

// wasm-gpu-alias-pass.mjs
//
// Lowering pass for cyclic aliases between WebAssembly linear-memory views and
// GPU staging buffers.
//
// The pass is pure (no I/O) and deterministic: all outputs are derived from a
// total order over node ids, so the same graph always yields byte-identical
// plans and diagnostics.
//
// Run:  node --test wasm-gpu-alias-pass.test.mjs

export const CODE = Object.freeze({
  DANGLING_EDGE:        'E_DANGLING_EDGE',
  WASM_USE_AFTER_GROW:  'E_WASM_USE_AFTER_GROW',
  WASM_OUT_OF_BOUNDS:   'E_WASM_OUT_OF_BOUNDS',
  ALIGNMENT_MISMATCH:   'E_ALIGNMENT_MISMATCH',
  GPU_STALE_MAP:        'E_GPU_STALE_MAP',
  GPU_UNMAPPED_ACCESS:  'E_GPU_UNMAPPED_ACCESS',
  HIDDEN_ALIAS_CYCLE:   'E_HIDDEN_ALIAS_CYCLE',
  RACE_UNSYNCHRONIZED:  'E_RACE_UNSYNCHRONIZED',
});

const KNOWN_KINDS = new Set(['wasm-view', 'gpu-staging']);

// ---------------------------------------------------------------------------
// Deterministic, lexicographic union-find.  Representative is the smallest id
// in the class so class ids are stable regardless of union order.
// ---------------------------------------------------------------------------
class DSU {
  #parent = new Map();
  add(x) { if (!this.#parent.has(x)) this.#parent.set(x, x); }
  find(x) {
    this.add(x);
    let root = x;
    while (this.#parent.get(root) !== root) root = this.#parent.get(root);
    while (this.#parent.get(x) !== root) {
      const next = this.#parent.get(x);
      this.#parent.set(x, root);
      x = next;
    }
    return root;
  }
  union(a, b) {
    let ra = this.find(a), rb = this.find(b);
    if (ra === rb) return false;
    if (ra < rb) this.#parent.set(rb, ra);
    else this.#parent.set(ra, rb);
    return true;
  }
  classes() {
    const out = new Map();
    for (const x of [...this.#parent.keys()].sort()) {
      const r = this.find(x);
      if (!out.has(r)) out.set(r, []);
      out.get(r).push(x);
    }
    for (const members of out.values()) members.sort();
    return out;
  }
}

const isPow2 = (n) => Number.isInteger(n) && n > 0 && (n & (n - 1)) === 0;
const maxAlign = (a, b) => Math.max(a || 1, b || 1);

// Rotate a directed cycle so the smallest id is first.  Deterministic output
// for the same cycle irrespective of where DFS entered it.
function canonicalCycle(nodes) {
  if (nodes.length <= 1) return [...nodes];
  let best = 0;
  for (let i = 1; i < nodes.length; i++) {
    if (nodes[i] < nodes[best]) best = i;
  }
  return [...nodes.slice(best), ...nodes.slice(0, best)];
}

// ---------------------------------------------------------------------------
// Diagnostic helpers.  A diagnostic is identified by its code plus a stable
// path so duplicates collapse and the sort order is total.
// ---------------------------------------------------------------------------
function diag(code, path, message, nodeId) {
  return { code, severity: 'error', nodeId: nodeId ?? null, path, message };
}
function diagKey(d) {
  return [d.code, d.path.join('\u0000'), d.nodeId ?? ''].join('\u0001');
}
function sortDiagnostics(diags) {
  const seen = new Map();
  for (const d of diags) if (!seen.has(diagKey(d))) seen.set(diagKey(d), d);
  return [...seen.values()].sort((a, b) => {
    if (a.code !== b.code) return a.code < b.code ? -1 : 1;
    const pa = a.path.join('\u0000'), pb = b.path.join('\u0000');
    if (pa !== pb) return pa < pb ? -1 : 1;
    const na = a.nodeId ?? '', nb = b.nodeId ?? '';
    return na === nb ? 0 : (na < nb ? -1 : 1);
  });
}

// ---------------------------------------------------------------------------
// Directed cycle enumeration over a set of edges.  Returns canonical cycles.
// Bounded by `limit` so adversarial graphs cannot blow up the pass.
// ---------------------------------------------------------------------------
function findCycles(edgeList, limit = 256) {
  const adj = new Map();
  for (const { from, to } of edgeList) {
    if (!adj.has(from)) adj.set(from, new Set());
    adj.get(from).add(to);
  }

  const cycles = new Map();
  const nodes = [...adj.keys()].sort();
  const color = new Map();       // 0/undefined unseen, 1 on-stack, 2 done
  const stack = [];
  const onStack = new Set();

  const visit = (u) => {
    if (cycles.size >= limit) return;
    color.set(u, 1);
    stack.push(u);
    onStack.add(u);
    const outs = [...(adj.get(u) ?? [])].sort();
    for (const v of outs) {
      if (cycles.size >= limit) break;
      if (color.get(v) === 1) {
        const idx = stack.lastIndexOf(v);
        const cyc = canonicalCycle(stack.slice(idx));
        cycles.set(cyc.join('\u0000'), cyc);
      } else if (color.get(v) !== 2) {
        visit(v);
      }
    }
    stack.pop();
    onStack.delete(u);
    color.set(u, 2);
  };

  for (const n of nodes) if (!color.has(n)) visit(n);
  return [...cycles.values()].sort((a, b) =>
    a.join('\u0000') < b.join('\u0000') ? -1 : 1);
}

// ---------------------------------------------------------------------------
// Transitive reduction of a DAG.  Removes copy edges implied by a longer path,
// which is exactly the "minimize copies" requirement.
// ---------------------------------------------------------------------------
function transitiveReduction(edges) {
  const nodes = [...new Set(edges.flatMap((e) => [e.from, e.to]))].sort();
  const adj = new Map(nodes.map((n) => [n, new Set()]));
  for (const e of edges) adj.get(e.from).add(e.to);

  const edgeKey = (e) => `${e.from}\u0000${e.to}`;
  const reachesWithout = (src, dst, skipKey) => {
    const seen = new Set([src]);
    const q = [src];
    while (q.length) {
      const u = q.shift();
      const outs = [...(adj.get(u) ?? [])].sort();
      for (const v of outs) {
        if (`${u}\u0000${v}` === skipKey) continue;
        if (v === dst) return true;
        if (!seen.has(v)) { seen.add(v); q.push(v); }
      }
    }
    return false;
  };

  const sorted = [...edges].sort((a, b) => {
    const ka = edgeKey(a), kb = edgeKey(b);
    return ka < kb ? -1 : ka > kb ? 1 : 0;
  });
  const kept = [];
  for (const e of sorted) {
    if (!reachesWithout(e.from, e.to, edgeKey(e))) kept.push(e);
  }
  return kept;
}

function classAlignment(members, nodeById) {
  let align = 1;
  for (const id of members) align = maxAlign(align, nodeById.get(id).align ?? 1);
  return align;
}

function classBytes(members, nodeById) {
  let bytes = null;
  for (const id of members) {
    const n = nodeById.get(id);
    const b = n.byteLength ?? n.size ?? 0;
    if (bytes === null) bytes = b;
    else bytes = Math.max(bytes, b);
  }
  return bytes ?? 0;
}

// ---------------------------------------------------------------------------
// Main pass.
//
// graph = {
//   memory: { id, byteLength, generation },
//   gpu:    { mapEpoch },                          // optional
//   nodes: [ { id, kind, ... } ],
//   edges: [ { from, to, alias, worker, sync } ]
// }
// ---------------------------------------------------------------------------
export function lower(graph) {
  const diagnostics = [];
  const nodeById = new Map();
  const nodes = [...(graph.nodes ?? [])].sort((a, b) =>
    a.id < b.id ? -1 : a.id > b.id ? 1 : 0);

  for (const raw of nodes) {
    const n = { ...raw, align: raw.align ?? 1 };
    if (nodeById.has(n.id)) {
      diagnostics.push(diag(CODE.DANGLING_EDGE, ['dup-node', n.id],
        `duplicate node id ${n.id}`, n.id));
      continue;
    }
    if (!KNOWN_KINDS.has(n.kind)) {
      diagnostics.push(diag(CODE.DANGLING_EDGE, ['bad-kind', n.id],
        `unknown node kind ${JSON.stringify(n.kind)}`, n.id));
    }
    nodeById.set(n.id, n);
  }

  const edges = [...(graph.edges ?? [])].sort((a, b) => {
    const ka = `${a.from}\u0000${a.to}`, kb = `${b.from}\u0000${b.to}`;
    return ka < kb ? -1 : ka > kb ? 1 : 0;
  });

  // --- structural validation ------------------------------------------------
  const cleanEdges = [];
  for (const e of edges) {
    if (!nodeById.has(e.from) || !nodeById.has(e.to)) {
      diagnostics.push(diag(CODE.DANGLING_EDGE, ['edge', e.from, e.to],
        `edge ${e.from} -> ${e.to} references an unknown node`));
      continue;
    }
    cleanEdges.push(e);
  }

  // --- lifetime: WebAssembly linear memory generation -----------------------
  const mem = graph.memory ?? { id: 'mem', byteLength: Infinity, generation: 0 };
  for (const n of nodeById.values()) {
    if (n.kind !== 'wasm-view') continue;
    if (n.generation !== undefined && n.generation !== mem.generation) {
      diagnostics.push(diag(CODE.WASM_USE_AFTER_GROW, ['wasm', n.id],
        `view ${n.id} captured at memory generation ${n.generation} but memory ` +
        `is at generation ${mem.generation} (detached by grow)`, n.id));
      continue;
    }
    const end = (n.byteOffset ?? 0) + (n.byteLength ?? 0);
    if (Number.isFinite(mem.byteLength) && end > mem.byteLength) {
      diagnostics.push(diag(CODE.WASM_OUT_OF_BOUNDS, ['wasm', n.id],
        `view ${n.id} spans [${n.byteOffset ?? 0}, ${end}) past memory ` +
        `byteLength ${mem.byteLength}`, n.id));
    }
  }

  // --- alignment of individual nodes ----------------------------------------
  for (const n of nodeById.values()) {
    const align = n.align ?? 1;
    if (!isPow2(align)) {
      diagnostics.push(diag(CODE.ALIGNMENT_MISMATCH, ['align', n.id],
        `node ${n.id} has non-power-of-two alignment ${align}`, n.id));
      continue;
    }
    const off = n.byteOffset ?? 0;
    if (off % align !== 0) {
      diagnostics.push(diag(CODE.ALIGNMENT_MISMATCH, ['align', n.id],
        `node ${n.id} offset ${off} is not aligned to ${align}`, n.id));
    }
  }

  // --- alias classes (explicit cycles collapse here, for free) --------------
  const dsu = new DSU();
  for (const id of nodeById.keys()) dsu.add(id);
  for (const e of cleanEdges) if (e.alias) dsu.union(e.from, e.to);

  const classes = dsu.classes();
  const classOf = new Map();
  for (const [rep, members] of classes) {
    for (const m of members) classOf.set(m, rep);
    // Mixed kinds (wasm view <-> GPU staging) are the normal case: that is what
    // this pass exists to lower.  Only alignment must agree on a shared backing.
    // Mixed alignment inside one class cannot be satisfied by a single backing.
    const aligns = new Set(members.map((m) => nodeById.get(m).align ?? 1));
    if (aligns.size > 1) {
      diagnostics.push(diag(CODE.ALIGNMENT_MISMATCH, ['class', rep],
        `alias class ${rep} mixes alignments ${[...aligns].sort((a, b) => a - b)
          .join(', ')}`, rep));
    }
  }

  // --- GPU mapping lifetime --------------------------------------------------
  for (const n of nodeById.values()) {
    if (n.kind !== 'gpu-staging') continue;
    const used = cleanEdges.some((e) => e.from === n.id || e.to === n.id);
    if (!used) continue;
    if (n.mapState === 'stale') {
      diagnostics.push(diag(CODE.GPU_STALE_MAP, ['gpu', n.id],
        `staging buffer ${n.id} is stale (remapped since the alias was taken)`, n.id));
    } else if (n.mapState === 'unmapped' || n.mapState === undefined) {
      diagnostics.push(diag(CODE.GPU_UNMAPPED_ACCESS, ['gpu', n.id],
        `staging buffer ${n.id} is accessed while unmapped`, n.id));
    }
  }

  // --- hidden alias cycles: cycles that survive as *copy* edges -------------
  const copyEdgesRaw = cleanEdges.filter((e) => !e.alias);
  for (const cyc of findCycles(copyEdgesRaw)) {
    diagnostics.push(diag(CODE.HIDDEN_ALIAS_CYCLE, ['cycle', ...cyc],
      `hidden alias cycle among copy edges: ${cyc.join(' -> ')} -> ${cyc[0]}`, cyc[0]));
  }

  // --- worker synchronization ------------------------------------------------
  for (const e of copyEdgesRaw) {
    const a = nodeById.get(e.from), b = nodeById.get(e.to);
    const crossWorker = a.owner !== undefined && b.owner !== undefined &&
                        a.owner !== b.owner;
    const sync = e.sync ?? 'none';
    if (crossWorker && (sync === 'none' || sync === undefined)) {
      diagnostics.push(diag(CODE.RACE_UNSYNCHRONIZED, ['edge', e.from, e.to],
        `copy ${e.from} -> ${e.to} crosses workers ` +
        `(${a.owner} -> ${b.owner}) without acquire/release`, e.from));
    }
  }

  // --- copy-edge alignment feasibility --------------------------------------
  for (const e of copyEdgesRaw) {
    const a = nodeById.get(e.from), b = nodeById.get(e.to);
    const aa = a.align ?? 1, ba = b.align ?? 1;
    if (aa % ba !== 0 && ba % aa !== 0) {
      diagnostics.push(diag(CODE.ALIGNMENT_MISMATCH, ['copy', e.from, e.to],
        `copy ${e.from} -> ${e.to} has incompatible alignments ${aa} and ${ba}`,
        e.from));
    }
  }

  const fatal = sortDiagnostics(diagnostics);

  // --- lower aliases: one canonical backing per class -----------------------
  const loweredAliases = [];
  let cyclesCollapsed = 0;
  for (const [rep, members] of [...classes.entries()].sort((a, b) =>
    a[0] < b[0] ? -1 : 1)) {
    const kind = nodeById.get(rep).kind;
    const align = classAlignment(members, nodeById);
    const bytes = classBytes(members, nodeById);
    loweredAliases.push({ classId: rep, canonical: rep, kind, bytes, align, members });
    if (members.length > 1) cyclesCollapsed += members.length - 1;
  }
  loweredAliases.sort((a, b) => (a.classId < b.classId ? -1 : 1));

  // --- lower copies over the condensed class DAG -----------------------------
  // Drop self-class copies and duplicates, then transitive-reduce.
  const seenCopy = new Set();
  let copies = [];
  for (const e of copyEdgesRaw) {
    const from = classOf.get(e.from), to = classOf.get(e.to);
    if (from === to) continue; // already one storage: zero-copy
    const key = `${from}\u0000${to}`;
    if (seenCopy.has(key)) continue;
    seenCopy.add(key);
    copies.push({
      from, to,
      bytes: Math.min(classBytes(classes.get(from), nodeById),
                      classBytes(classes.get(to), nodeById)),
      align: Math.min(classAlignment(classes.get(from), nodeById),
                      classAlignment(classes.get(to), nodeById)),
      worker: nodeById.get(e.to).owner ?? nodeById.get(e.from).owner ?? null,
      sync: e.sync ?? 'none',
    });
  }

  // Transitive reduction only makes sense on a DAG.  If hidden cycles exist we
  // already rejected, so guard it anyway.
  const copyDag = findCycles(copies).length === 0;
  const copiesBefore = copies.length;
  if (copyDag && copies.length > 1) copies = transitiveReduction(copies);

  copies.sort((a, b) => {
    const ka = `${a.from}\u0000${a.to}`, kb = `${b.from}\u0000${b.to}`;
    return ka < kb ? -1 : ka > kb ? 1 : 0;
  });

  return {
    ok: fatal.length === 0,
    diagnostics: fatal,
    classes: loweredAliases,
    lowered: { aliases: loweredAliases, copies },
    stats: {
      nodes: nodeById.size,
      classes: loweredAliases.length,
      cyclesCollapsed,
      copiesBefore,
      copiesAfter: copies.length,
    },
  };
}

export default lower;

3. Tests — wasm-gpu-alias-pass.test.mjs

// wasm-gpu-alias-pass.test.mjs
import test from 'node:test';
import assert from 'node:assert/strict';
import { lower, CODE } from './wasm-gpu-alias-pass.mjs';

const wasmView = (id, over = {}) => ({
  id, kind: 'wasm-view', byteOffset: 0, byteLength: 1024,
  generation: 0, align: 16, owner: 'main', ...over,
});
const gpu = (id, over = {}) => ({
  id, kind: 'gpu-staging', size: 1024, mapState: 'mapped',
  align: 16, owner: 'main', ...over,
});
const edge = (from, to, over = {}) => ({ from, to, alias: false, ...over });

test('explicit alias cycle collapses to one backing with zero copies', () => {
  const r = lower({
    memory: { id: 'mem', byteLength: 4096, generation: 3 },
    nodes: [wasmView('a', { generation: 3 }), gpu('b'), gpu('c')],
    edges: [
      edge('a', 'b', { alias: true }),
      edge('b', 'c', { alias: true }),
      edge('c', 'a', { alias: true }),
    ],
  });
  assert.equal(r.ok, true);
  assert.equal(r.stats.classes, 1);
  assert.equal(r.stats.copiesBefore, 0);
  assert.deepEqual(r.lowered.aliases[0].members, ['a', 'b', 'c']);
  assert.equal(r.lowered.aliases[0].canonical, 'a');
  assert.deepEqual(r.diagnostics, []);
});

test('use-after-grow is rejected deterministically', () => {
  const r = lower({
    memory: { id: 'mem', byteLength: 4096, generation: 5 },
    nodes: [wasmView('v', { generation: 4 }), gpu('g')],
    edges: [edge('v', 'g')],
  });
  assert.equal(r.ok, false);
  const d = r.diagnostics.find((x) => x.code === CODE.WASM_USE_AFTER_GROW);
  assert.ok(d, 'expected use-after-grow');
  assert.deepEqual(d.path, ['wasm', 'v']);
});

test('stale GPU map is rejected', () => {
  const r = lower({
    memory: { id: 'mem', byteLength: 4096, generation: 0 },
    nodes: [wasmView('v'), gpu('g', { mapState: 'stale' })],
    edges: [edge('v', 'g')],
  });
  assert.equal(r.ok, false);
  assert.ok(r.diagnostics.some((x) => x.code === CODE.GPU_STALE_MAP));
});

test('hidden alias cycle (copies forming a loop) is rejected with a canonical path', () => {
  const r = lower({
    memory: { id: 'mem', byteLength: 4096, generation: 0 },
    nodes: [gpu('x'), gpu('y'), gpu('z')],
    edges: [edge('x', 'y'), edge('y', 'z'), edge('z', 'x')],
  });
  assert.equal(r.ok, false);
  const d = r.diagnostics.find((x) => x.code === CODE.HIDDEN_ALIAS_CYCLE);
  assert.ok(d);
  assert.deepEqual(d.path, ['cycle', 'x', 'y', 'z']);
});

test('copy graph can have a cycle reachable from different entry points but reports once', () => {
  const g = {
    memory: { id: 'mem', byteLength: 4096, generation: 0 },
    nodes: [gpu('a'), gpu('b'), gpu('c')],
    edges: [edge('a', 'b'), edge('b', 'a'), edge('b', 'c'), edge('c', 'a')],
  };
  const r = lower(g);
  const cycles = r.diagnostics.filter((x) => x.code === CODE.HIDDEN_ALIAS_CYCLE);
  assert.ok(cycles.length >= 1);
  for (const c of cycles) {
    const cyc = c.path.slice(1);
    assert.equal(cyc[0], [...cyc].sort()[0]);
  }
});

test('unsynchronized cross-worker copy is rejected; acquire/release passes', () => {
  const base = {
    memory: { id: 'mem', byteLength: 4096, generation: 0 },
    nodes: [gpu('src', { owner: 'worker-A' }), gpu('dst', { owner: 'worker-B' })],
  };
  const bad = lower({ ...base, edges: [edge('src', 'dst')] });
  assert.ok(bad.diagnostics.some((x) => x.code === CODE.RACE_UNSYNCHRONIZED));

  const good = lower({ ...base, edges: [edge('src', 'dst', { sync: 'release' })] });
  assert.equal(good.diagnostics.some((x) => x.code === CODE.RACE_UNSYNCHRONIZED), false);
});

test('transitive copy edges are reduced away', () => {
  const r = lower({
    memory: { id: 'mem', byteLength: 4096, generation: 0 },
    nodes: [gpu('a'), gpu('b'), gpu('c')],
    edges: [edge('a', 'b'), edge('b', 'c'), edge('a', 'c')],
  });
  assert.equal(r.ok, true);
  assert.equal(r.stats.copiesBefore, 3);
  assert.equal(r.stats.copiesAfter, 2);
  const keys = r.lowered.copies.map((c) => `${c.from}->${c.to}`);
  assert.deepEqual(keys, ['a->b', 'b->c']);
});

test('alignment mismatch inside an alias class is rejected', () => {
  const r = lower({
    memory: { id: 'mem', byteLength: 4096, generation: 0 },
    nodes: [gpu('a', { align: 16 }), gpu('b', { align: 8 })],
    edges: [edge('a', 'b', { alias: true })],
  });
  assert.ok(r.diagnostics.some((x) => x.code === CODE.ALIGNMENT_MISMATCH));

test('output is independent of input ordering (determinism)', () => {
  const mk = (perm) => ({
    memory: { id: 'mem', byteLength: 4096, generation: 0 },
    nodes: perm.nodes,
    edges: perm.edges,
  });
  const nodes = [gpu('a'), gpu('b'), gpu('c'), gpu('d')];
  const edges = [edge('a', 'b'), edge('b', 'c'), edge('c', 'd'), edge('a', 'd')];
  const forward = lower(mk({ nodes, edges }));
  const reversed = lower(mk({ nodes: [...nodes].reverse(), edges: [...edges].reverse() }));
  assert.deepEqual(forward, reversed);
});

test('fuzz: shuffled inputs produce byte-identical plans and diagnostics', () => {
  let s = 0x12345678;
  const rnd = () => {
    s ^= s << 13; s >>>= 0;
    s ^= s >>> 17; s >>>= 0;
    s ^= s << 5; s >>>= 0;
    return s / 0xffffffff;
  };
  const shuffle = (arr) => {
    const a = [...arr];
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(rnd() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  };
  for (let iter = 0; iter < 200; iter++) {
    const n = 2 + Math.floor(rnd() * 6);
    const nodes = [];
    for (let i = 0; i < n; i++) {
      nodes.push(rnd() < 0.4
        ? wasmView('n' + i, { generation: 0 })
        : gpu('n' + i, { owner: 'w' + Math.floor(rnd() * 3) }));
    }
    const edges = [];
    for (let i = 0; i < n; i++) {
      for (let j = 0; j < n; j++) {
        if (i !== j && rnd() < 0.18) {
          edges.push(edge('n' + i, 'n' + j, {
            alias: rnd() < 0.5,
            sync: rnd() < 0.5 ? 'release' : 'none',
          }));
        }
      }
    }
    const base = { memory: { id: 'mem', byteLength: 8192, generation: 0 }, nodes, edges };
    const a = lower(base);
    const b = lower({ ...base, nodes: shuffle(nodes), edges: shuffle(edges) });
    assert.deepEqual(a, b, `iteration ${iter} was not deterministic`);
  }
});

5.4 Complexity and safety

  • Alias collapse is near-linear (DSU with path compression).
  • Cycle enumeration is bounded (findCycles(..., limit=256)), so adversarial graphs cannot make the pass explode.
  • Transitive reduction is O(E · (V+E)); acceptable for staging-buffer graphs, which are small relative to the memory graph.
  • Stale views are rejected rather than silently re-materialized — the caller must re-derive a view for the current generation, the only sound response to memory.grow().
