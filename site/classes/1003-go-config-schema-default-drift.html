Two changes: (1) kill the schema/loader default drift, (2) the GET /api/v1/config companion endpoint.

1. Align schema literal + doc comment with loader const

internal/config/loader.go — the loader const (single source of truth):

package config

// defaultProjectCooldown is the per-project cool-down in seconds applied by the
// loader when cooldown_s is unset. Operators rely on this as the documented
// behavior, so it must match the value published in the JSON Schema and docs.
const defaultProjectCooldown = 7200 // 2h

internal/config/schema.go — the --schema output literal, previously 900, now driven from the const so it can never drift again:

package config

import "fmt"

// Schema returns the JSON Schema for the daemon configuration.
// The cooldown_s default is interpolated from defaultProjectCooldown so the
// schema and the loader cannot silently diverge (see go-config-schema-default-drift).
func Schema() string {
    return fmt.Sprintf(`{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "title": "schedulerd configuration",
  "type": "object",
  "properties": {
    "cooldown_s": {
      "type": "integer",
      "minimum": 0,
      "default": %d,
      "description": "Seconds to wait between scheduled runs of the same project (default %d, i.e. 2h)."
    }
  }
}`, defaultProjectCooldown, defaultProjectCooldown)
}

internal/config/schema_test.go — schema-output test: generate the actual schema and assert the published default equals the loader const (regression guard for exactly this bug):

package config

import (
    "encoding/json"
    "testing"
)

func TestSchemaCooldownDefaultMatchesLoader(t *testing.T) {
    var schema struct {
        Properties struct {
            CooldownS struct {
                Default int `json:"default"`
            } `json:"cooldown_s"`
        } `json:"properties"`
    }
    if err := json.Unmarshal([]byte(Schema()), &schema); err != nil {
        t.Fatalf("schema is not valid JSON: %v", err)
    }
    if got := schema.Properties.CooldownS.Default; got != defaultProjectCooldown {
        t.Fatalf("schema cooldown_s default = %d, want %d (loader const): schema and loader have drifted",
            got, defaultProjectCooldown)
    }
}

2. Companion: GET /api/v1/config (resolved three-layer config)

internal/api/resolved.go — snapshot storage:

package api

import (
    "sync"

    "schedulerd/internal/config"
)

var (
    resolvedMu sync.RWMutex
    resolved   *config.Config
)

// SetResolvedConfig snapshots the fully-resolved three-layer configuration
// (defaults < file < flags). Call it once in main.go after flag resolution.
func SetResolvedConfig(cfg *config.Config) {
    resolvedMu.Lock()
    defer resolvedMu.Unlock()
    c := cfg.Clone()
    resolved = &c
}

// ResolvedConfig returns a copy of the snapshot; ok=false if not yet set.
func ResolvedConfig() (config.Config, bool) {
    resolvedMu.RLock()
    defer resolvedMu.RUnlock()
    if resolved == nil {
        return config.Config{}, false
    }
    return resolved.Clone(), true
}

internal/api/handler.go — handler with 405 on non-GET and first-4-chars gateway-key masking:

func (s *Server) handleConfig(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodGet {
        w.Header().Set("Allow", http.MethodGet)
        http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
        return
    }
    cfg, ok := ResolvedConfig()
    if !ok {
        http.Error(w, "resolved config not ready", http.StatusServiceUnavailable)
        return
    }
    w.Header().Set("Content-Type", "application/json")
    if err := json.NewEncoder(w).Encode(maskSecrets(cfg)); err != nil {
        s.logger.Error("encode /api/v1/config", "err", err)
    }
}

// maskSecrets redacts sensitive fields. gateway_key keeps only its first 4
// characters visible so operators can identify the key without leaking it.
func maskSecrets(cfg config.Config) map[string]any {
    out := map[string]any{
        "cooldown_s":   cfg.CooldownS,
        "project_dir":  cfg.ProjectDir,
        "log_level":    cfg.LogLevel,
    }
    if k := cfg.GatewayKey; k != "" {
        visible := k
        if len(k) > 4 {
            visible = k[:4]
        }
        out["gateway_key"] = visible + strings.Repeat("*", len(k)-len(visible))
    }
    return out
}

cmd/schedulerd/main.go — populate the snapshot from the resolved flag vars after the three-layer merge:

func main() {
    // flags -> resolved over defaults < config file
    cooldownS := flag.Int("cooldown-s", 0, "override cooldown_s (seconds); 0 = use file/default")
    cfgFile := flag.String("config", "", "path to config file")
    flag.Parse()

    cfg, err := config.Load(*cfgFile)         // defaults < file
    if err != nil { log.Fatal(err) }
    if *cooldownS > 0 {                       // flags win
        cfg.CooldownS = *cooldownS
    }

    api.SetResolvedConfig(&cfg)               // snapshot for GET /api/v1/config
    srv := api.New(cfg)
    log.Fatal(srv.ListenAndServe(":8080"))
}
