The repo didn't exist on this machine, so I built the complete go-duckdb-sync-sdlc-events module (/home/kara/go-duckdb-sync-sdlc-events, stdlib-only, no external deps) implementing INFRA-008. Key pieces:

duckbrain.go — core sync engine. Queries use COALESCE on every nullable column and LIMIT caps; DB access is behind a narrow queryer/rowScanner interface so tests can inject a fake row source and assert on the emitted SQL:

const ( eventsCap = 50; ticksCap = 25 )

var eventsQuery = fmt.Sprintf(`SELECT id,
    COALESCE(name, '') AS name, COALESCE(type, '') AS type,
    COALESCE(status, '') AS status, COALESCE(created_at, '') AS created_at
FROM events ORDER BY created_at LIMIT %d`, eventsCap)

var ticksQuery = fmt.Sprintf(`SELECT id,
    COALESCE(project_name, '') AS project_name, COALESCE(phase, '') AS phase,
    COALESCE(status, '') AS status, COALESCE(updated_at, '') AS updated_at
FROM ticks ORDER BY updated_at LIMIT %d`, ticksCap)

Reused postMemory helper (shared by every sync method; form-encoded, domain set by callers, body read capped, non-2xx → error):

func (d *DuckBrainSync) postMemory(ctx context.Context, path string, values url.Values) error {
    endpoint := d.baseURL + "/fleet/" + path
    req, err := http.NewRequestWithContext(ctx, http.MethodPost, endpoint, strings.NewReader(values.Encode()))
    ...
    resp, err := d.client.Do(req) // client via WithHTTPClient (tests use httptest client)
    body, _ := io.ReadAll(io.LimitReader(resp.Body, 1<<16))
    if resp.StatusCode < 200 || resp.StatusCode >= 300 { return fmt.Errorf("postMemory %s: status %d: %s", ...) }
    return nil
}

syncEvents/syncTicks — mirror the syncX pattern: query → iterate → post, with per-row error tolerance (scan/post failures are logged, counted in failed, and continue — never aborting the cycle) and a defensive cap in the loop (for rows.Next() && posted < eventsCap):

func (d *DuckBrainSync) syncEvents(ctx context.Context) (posted, failed int, err error) {
    rows, err := d.db.QueryContext(ctx, eventsQuery)
    if err != nil { return 0, 0, fmt.Errorf("syncEvents: query: %w", err) }
    defer rows.Close()
    for rows.Next() && posted < eventsCap {
        var id, name, typ, status, createdAt string
        if err := rows.Scan(&id, &name, &typ, &status, &createdAt); err != nil { failed++; continue }
        values := url.Values{"domain": {"event"}, "key": {id}, "name": {name}, "type": {typ},
            "status": {status}, "created_at": {createdAt}}
        if err := d.postMemory(ctx, "events/"+url.PathEscape(id), values); err != nil { failed++; continue }
        posted++
    }
    return posted, failed, rows.Err() // iteration errors surface; row errors are tolerated
}

syncTicks mirrors this, posting to projects/<project_name>/ticks/<id>.

syncOnce wires both in — the two syncs run independently (a query-level failure in one is joined into the returned error but does not block the other); row-level failures are already tolerated inside each sync:

func (d *DuckBrainSync) syncOnce(ctx context.Context) error {
    var errs []error
    if n, f, err := d.syncEvents(ctx); err != nil { d.logf("syncOnce: events sync failed: %v", err); errs = append(errs, err)
    } else { d.logf("syncOnce: events cycle done: posted=%d failed=%d", n, f) }
    if n, f, err := d.syncTicks(ctx); err != nil { ...; errs = append(errs, err) } else { ... }
    return errors.Join(errs...)
}

duckbrain_test.go — httptest-based suite with a recording fleetServer (injectable per-path failures) and the required decodeDuckBrainContent helper:

func decodeDuckBrainContent(t *testing.T, body string) map[string]string {
    t.Helper()
    vals, err := url.ParseQuery(body)
    if err != nil { t.Fatalf("decodeDuckBrainContent(%q): %v", body, err) }
    m := make(map[string]string, len(vals))
    for k, v := range vals { m[k] = v[0] }
    return m
}
