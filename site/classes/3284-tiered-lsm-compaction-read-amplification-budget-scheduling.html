Tiered-LSM Compaction under a Read-Amplification Budget — Exact Minimum-I/O Scheduling

Problem class: tiered-lsm-compaction-read-amplification-budget-scheduling Environment: go1.26 (verified with go1.26.8, linux/amd64)

1. Diagnosis / root-cause analysis

Given per-level run counts Count[i], aggregate sizes Bytes[i], fanout limits MaxRuns[i], and a read-amp budget, we must find the minimum-byte compaction plan (or prove infeasibility). The obvious approaches fail for concrete reasons:

  1. Read amp is global; cost is local and coupled. RA is one scalar across the tree, but compacting level i changes the byte volume level i+1 must rewrite. "Cheapest level first" is not optimal.
  2. Write amplification is super-additive in block length. In a cascade [i..j], level-k bytes are read+rewritten once per level crossed: cost(i,j) = 2·Σ_{k=i..j}(j−k+1)·Bytes[k]. Two shallow compactions can be far cheaper — or far costlier — than one deep merge. This coupling is the crux.
  3. Fanout forces cascades. If a block's output would exceed MaxRuns[j+1], the block must extend upward; a purely local fix yields an infeasible plan.
  4. "Already under budget ⇒ do nothing" is a bug. Fanout is a hard invariant; a level can exceed MaxRuns while RA ≤ budget. (The first implementation had this exact bug; the randomized replay test caught it: plan violates fanout at level 0: 6 > 5 (plan []).)
  5. The bottom level is terminal (no level below to write to). If it alone violates budget/fanout, the instance is infeasible.

Winning structure: any plan decomposes uniquely into maximal consecutive compacted blocks. Block [i..j] produces one run on j+1 and removes Σ Count − 1 runs. Maximal blocks are non-adjacent intervals, so this is weighted interval selection plus a global run-budget dimension — solvable exactly by a DAG shortest-path DP.

2. Formal model

  • Read amplification (tiered): RA = Σ_i runs_i.
  • Block [i..j], 0 ≤ i ≤ j ≤ L−2:
  • cost(i,j) = 2·Σ_{k=i..j}(j−k+1)·Bytes[k]
  • reward(i,j) = (Σ_{k=i..j} Count[k]) − 1
  • Feasibility: non-compacted p has ≤ MaxRuns[p] runs; a destination j+1 needs Count[j+1]+1 ≤ MaxRuns[j+1]; bottom level never compacted; final RA ≤ budget.

3. Exact algorithm

With E = max(0, RA_before − budget), define dp[pos][r] = min cost after deciding levels 0..pos−1 with min(r,E) runs removed. Transitions: leave pos untouched (if fanout holds); or start block [pos..j] → dp[j+2][min(E, r+reward)]. Answer dp[L][E]; INF ⇒ infeasible. Complexity O(L²·E) time, O(L·E) memory. Execute plans top-down so every destination is empty when written (no transient fanout violation, natural pipelining of overlapping compactions).

4. Exact fix — solver.go

// Package main implements an exact solver for the tiered LSM compaction
// read-amplification budget scheduling problem.  See SOLUTION.md for the model.
package main

type Level struct {
    Count int   `json:"count"`
    Bytes int64 `json:"bytes"`
}

type Instance struct {
    Levels        []Level `json:"levels"`
    ReadAmpBudget int     `json:"read_amp_budget"`
    MaxRuns       []int   `json:"max_runs"`
}

type Compaction struct {
    From  int   `json:"from"`
    To    int   `json:"to"`
    Bytes int64 `json:"bytes"`
    Cost  int64 `json:"cost"`
}

type Plan struct {
    Feasible      bool         `json:"feasible"`
    Reason        string       `json:"reason,omitempty"`
    TotalCost     int64        `json:"total_cost"`
    ReadAmpBefore int          `json:"read_amp_before"`
    ReadAmpAfter  int          `json:"read_amp_after"`
    Compactions   []Compaction `json:"compactions"`
}

const infCost int64 = 1 << 62
const unboundedRuns = 1 << 30

type decision struct {
    prevPos int
    prevR   int
    action  int // 0 origin, 1 skip, 2 block ending at j
    j       int
}

func Solve(inst Instance) Plan {
    l := len(inst.Levels)

    maxRuns := make([]int, l)
    for i := 0; i < l; i++ {
        if i < len(inst.MaxRuns) && inst.MaxRuns[i] > 0 {
            maxRuns[i] = inst.MaxRuns[i]
        } else {
            maxRuns[i] = unboundedRuns
        }
    }

    totalCount := 0
    for _, lv := range inst.Levels {
        totalCount += lv.Count
    }
    raBefore := totalCount
    plan := Plan{ReadAmpBefore: raBefore, ReadAmpAfter: raBefore, Compactions: []Compaction{}}

    // A fanout violation must be repaired even if RA is already in budget.
    if raBefore <= inst.ReadAmpBudget {
        ok := true
        for i := 0; i < l; i++ {
            if inst.Levels[i].Count > maxRuns[i] {
                ok = false
                break
            }
        }
        if ok {
            plan.Feasible = true
            plan.ReadAmpAfter = raBefore
            return plan
        }
    }
    need := raBefore - inst.ReadAmpBudget
    if need < 0 {
        need = 0
    }
    if l == 0 {
        plan.Feasible = false
        plan.Reason = "no levels present"
        return plan
    }

    dp := make([][]int64, l+1)
    par := make([][]decision, l+1)
    for i := range dp {
        dp[i] = make([]int64, need+1)
        par[i] = make([]decision, need+1)
        for r := range dp[i] {
            dp[i][r] = infCost
            par[i][r] = decision{prevPos: -1}
        }
    }
    dp[0][0] = 0

    relax := func(npos, nr int, cost int64, d decision) {
        if npos < 0 || npos > l || nr < 0 || nr > need {
            return
        }
        if cost < dp[npos][nr] {
            dp[npos][nr] = cost
            par[npos][nr] = d
        }
    }

    for pos := 0; pos <= l; pos++ {
        for r := 0; r <= need; r++ {
            cur := dp[pos][r]
            if cur == infCost || pos == l {
                continue
            }
            if inst.Levels[pos].Count <= maxRuns[pos] {
                relax(pos+1, r, cur, decision{prevPos: pos, prevR: r, action: 1, j: -1})
            }
            for j := pos; j <= l-2; j++ {
                var sumCount int
                for k := pos; k <= j; k++ {
                    sumCount += inst.Levels[k].Count
                }
                if sumCount == 0 {
                    continue
                }
                if inst.Levels[j+1].Count+1 > maxRuns[j+1] {
                    continue
                }
                nr := r + sumCount - 1
                if nr > need {
                    nr = need
                }
                relax(j+2, nr, cur+intervalCost(inst.Levels, pos, j),
                    decision{prevPos: pos, prevR: r, action: 2, j: j})
            }
        }
    }

    if dp[l][need] == infCost {
        plan.Feasible = false
        plan.Reason = "read-amplification budget cannot be met even after compacting every compactable level (fanout and bottom-level limits apply)"
        return plan
    }

    plan.Feasible = true
    plan.TotalCost = dp[l][need]
    pos, r := l, need
    var blocks []Compaction
    for pos > 0 {
        d := par[pos][r]
        if d.prevPos < 0 {
            break
        }
        if d.action == 2 {
            blocks = append(blocks, Compaction{
                From: d.prevPos, To: d.j,
                Bytes: sumBytes(inst.Levels, d.prevPos, d.j),
                Cost:  intervalCost(inst.Levels, d.prevPos, d.j),
            })
        }
        pos, r = d.prevPos, d.prevR
    }
    for i, k := 0, len(blocks)-1; i < k; i, k = i+1, k-1 {
        blocks[i], blocks[k] = blocks[k], blocks[i]
    }
    plan.Compactions = blocks
    runs, _ := Simulate(inst, blocks)
    raAfter := 0
    for _, c := range runs {
        raAfter += c
    }
    plan.ReadAmpAfter = raAfter
    return plan
}

func intervalCost(levels []Level, i, j int) int64 {
    var cost, pending int64
    for k := i; k <= j; k++ {
        pending += levels[k].Bytes
        cost += 2 * pending
    }
    return cost
}

func sumBytes(levels []Level, i, j int) int64 {
    var s int64
    for k := i; k <= j; k++ {
        s += levels[k].Bytes
    }
    return s
}

// Simulate replays a plan and is the test-side oracle.
func Simulate(inst Instance, blocks []Compaction) ([]int, int64) {
    l := len(inst.Levels)
    runs := make([]int, l)
    for i := range runs {
        runs[i] = inst.Levels[i].Count
    }
    var io int64
    for _, b := range blocks {
        var pending int64
        for k := b.From; k <= b.To; k++ {
            pending += inst.Levels[k].Bytes
            io += 2 * pending
        }
        runs[b.From] = 0
        for k := b.From + 1; k <= b.To; k++ {
            runs[k] = 0
        }
        runs[b.To+1]++
    }
    return runs, io
}

package main

import (
    "encoding/json"
    "fmt"
    "os"
)

func main() {
    var in *os.File
    if len(os.Args) > 1 {
        f, err := os.Open(os.Args[1])
        if err != nil {
            fail(err)
        }
        defer f.Close()
        in = f
    } else {
        in = os.Stdin
    }
    var inst Instance
    dec := json.NewDecoder(in)
    dec.DisallowUnknownFields()
    if err := dec.Decode(&inst); err != nil {
        fail(fmt.Errorf("parse instance: %w", err))
    }
    if len(inst.Levels) == 0 {
        fail(fmt.Errorf("instance has no levels"))
    }
    enc := json.NewEncoder(os.Stdout)
    enc.SetIndent("", "  ")
    if err := enc.Encode(Solve(inst)); err != nil {
        fail(err)
    }
}

func fail(err error) {
    fmt.Fprintln(os.Stderr, "error:", err)
    os.Exit(1)
}

6. Adapting the model

  • Leveled read amp: use "one occupied level per merged interior level" for the reward; same DP.
  • Size-ratio fanout (S_{i+1} ≤ T·S_i): precompute per-(pos,j) feasibility; same skeleton.
  • Reading the destination level too: add +2·Σ_{t=i+1}^{j+1}Bytes[t] to intervalCost; only that function changes.
  • Per-run non-uniform sizes: pick smallest runs first; the aggregate-Bytes level model is exact for level-granular compactions.
