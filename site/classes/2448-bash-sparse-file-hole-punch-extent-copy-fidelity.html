Hole-for-hole sparse copy in bash with SEEK_DATA/SEEK_HOLE

Files created: ~/sparse-copy (tested, executable) and ~/SOLUTION.md (full write-up). The code block below was diffed byte-for-byte against the tested script.

Root-cause analysis

Sparseness is a property of a file's extent map, not of its bytes. It can only be read with lseek(2) SEEK_DATA/SEEK_HOLE:

  • lseek(fd, off, SEEK_DATA) → start of the next data extent at/after off (ENXIO when the tail is a hole).
  • lseek(fd, off, SEEK_HOLE) → start of the next hole; the virtual hole at EOF is always reported.

Any copier that inspects bytes or copies [0,size) linearly destroys the mapping:

command src st_blocks dst st_blocks failure
naive read/write loop 2056 131072 holes materialized as zeros
cp --sparse=always 2056 8 allocated all-zero data extent collapsed into a hole
dd conv=sparse 2056 8 zero-detection ≠ hole detection
this copier 2056 2056 preserved

Bash cannot call these lseek modes, so the extent walk lives in an embedded python3 helper (a C helper with the same loop also works). The concurrency requirement means a single pass is not trusted: it must never leave a partial tail, retry on source change, and exit nonzero with a mapping report when divergence remains.

The fix

Save as sparse-copy, chmod +x sparse-copy.

#!/usr/bin/env bash
# sparse-copy -- hole-for-hole sparse file copier using SEEK_DATA/SEEK_HOLE.
#
# Usage: sparse-copy SRC DST
#
# Exit status:
#   0  destination mapping/logical size/allocated blocks match the source
#   1  mapping diverged (per-extent fidelity report is printed)
#   2  usage / setup error
#
# Bash cannot call lseek(2) with SEEK_DATA/SEEK_HOLE itself, so the byte-level
# extent walk is delegated to an embedded Python 3 helper.  Bash owns argument
# handling, process plumbing and the final exit status; Python owns the
# SEEK_DATA/SEEK_HOLE extent walk, the sparse pwrite copy and the report.

set -u -o pipefail

die() { printf 'sparse-copy: %s\n' "$*" >&2; exit 2; }

[[ $# -eq 2 ]] || die "usage: sparse-copy SRC DST"
src=$1
dst=$2

[[ -e $src ]] || die "source does not exist: $src"
[[ -f $src ]] || die "source is not a regular file: $src"
[[ -d $dst ]] && die "destination is a directory: $dst"

command -v python3 >/dev/null 2>&1 || die "python3 is required for SEEK_DATA/SEEK_HOLE"

exec python3 - "$src" "$dst" <<'PY'
import errno
import os
import sys

SRC, DST = sys.argv[1], sys.argv[2]
CHUNK = 1 << 20       # read/write unit for a single data extent
MAX_ATTEMPTS = 8      # bounded retries while a concurrent writer is active

ENXIO = errno.ENXIO


def file_extents(fd, size):
    """Return [(kind, start, end), ...] covering [0, size) using SEEK_DATA/HOLE."""
    out = []
    off = 0
    while off < size:
        try:
            data = os.lseek(fd, off, os.SEEK_DATA)
        except OSError as e:
            if e.errno == ENXIO:          # no more data: remainder is a hole
                if off < size:
                    out.append(("hole", off, size))
                break
            raise
        if data > off:
            out.append(("hole", off, data))
        if data >= size:
            break
        try:
            hole = os.lseek(fd, data, os.SEEK_HOLE)
        except OSError:
            hole = size
        # A filesystem without hole support reports hole == data.
        if hole <= data:
            hole = size
        hole = min(hole, size)
        out.append(("data", data, hole))
        off = hole
    return out


def extents_of_path(path):
    fd = os.open(path, os.O_RDONLY)
    try:
        st = os.fstat(fd)
        return file_extents(fd, st.st_size), st
    finally:
        os.close(fd)


def copy_once(sfd, dfd):
    """One pass.  Returns (status, copied_bytes, torn_at_or_None)."""
    st0 = os.fstat(sfd)
    size0 = st0.st_size
    os.ftruncate(dfd, size0)              # establish logical size; holes stay holes
    copied = 0
    off = 0
    torn_at = None
    while off < size0:
        try:
            data = os.lseek(sfd, off, os.SEEK_DATA)
        except OSError as e:
            if e.errno == ENXIO:
                break
            raise
        if data >= size0:
            break
        try:
            hole = os.lseek(sfd, data, os.SEEK_HOLE)
        except OSError:
            hole = size0
        if hole <= data:
            hole = size0
        hole = min(hole, size0)

        pos = data
        while pos < hole:
            want = min(CHUNK, hole - pos)
            buf = os.pread(sfd, want, pos)
            if len(buf) < want:
                # Source shrank under us mid-extent: never keep a partial tail.
                os.ftruncate(dfd, data)
                torn_at = data
                return ("torn", copied, torn_at)
            n = os.pwrite(dfd, buf, pos)
            if n <= 0:
                raise OSError(errno.EIO, "short write at offset %d" % pos)
            copied += n
            pos += n
        off = hole

    os.fsync(dfd)
    st1 = os.fstat(sfd)
    stable = (
        st1.st_size == st0.st_size
        and st1.st_mtime_ns == st0.st_mtime_ns
        and st1.st_ctime_ns == st0.st_ctime_ns
        and st1.st_blocks == st0.st_blocks
    )
    return ("stable" if stable else "changed", copied, None)


def segment_union(src_ex, dst_ex, size):
    """Slice [0,size) at every boundary from both mappings; label each slice."""
    bounds = {0, size}
    for _, a, b in src_ex + dst_ex:
        bounds.add(a)
        bounds.add(b)
    bounds = sorted(b for b in bounds if 0 <= b <= size)
    return list(zip(bounds, bounds[1:]))


def kind_at(ext_list, pos):
    for kind, a, b in ext_list:
        if a <= pos < b:
            return kind
    return "hole"


def report(src, dst, copied):
    src_ex, sst = extents_of_path(src)
    dst_ex, dst_st = extents_of_path(dst)

    size = sst.st_size
    src_data = sum(b - a for k, a, b in src_ex if k == "data")
    dst_data = sum(b - a for k, a, b in dst_ex if k == "data")
    src_holes = sum(1 for k, _, _ in src_ex if k == "hole")
    dst_holes = sum(1 for k, _, _ in dst_ex if k == "hole")

    same_size = sst.st_size == dst_st.st_size
    same_blocks = sst.st_blocks == dst_st.st_blocks

    rows = []
    mismatch = (not same_size) or (not same_blocks) or (src_ex != dst_ex)
    for a, b in segment_union(src_ex, dst_ex, size):
        ks = kind_at(src_ex, a)
        kd = kind_at(dst_ex, a)
        ok = ks == kd
        mismatch |= not ok
        rows.append((a, b, ks, kd, (b - a) if ks == "data" else 0, ok))

    out = sys.stdout
    out.write("=== sparse-copy fidelity report ===\n")
    out.write("source: %s\n" % src)
    out.write("dest:   %s\n" % dst)
    out.write("logical size:     src=%d dst=%d %s\n"
              % (sst.st_size, dst_st.st_size, "OK" if same_size else "DIVERGED"))
    out.write("allocated blocks: src=%d dst=%d %s (512B units)\n"
              % (sst.st_blocks, dst_st.st_blocks, "OK" if same_blocks else "DIVERGED"))
    out.write("data bytes:       copied=%d src=%d dst=%d\n" % (copied, src_data, dst_data))
    out.write("holes preserved:  src=%d dst=%d\n" % (src_holes, dst_holes))
    out.write("extents (union of source/destination mappings):\n")
    out.write("  %-4s %-28s %-6s %-6s %-10s %s\n"
              % ("#", "range", "src", "dst", "bytes", "status"))
    for i, (a, b, ks, kd, nbytes, ok) in enumerate(rows, 1):
        out.write("  %-4d [%-10d,%-10d) %-6s %-6s %-10d %s\n"
                  % (i, a, b, ks, kd, nbytes, "ok" if ok else "MISMATCH"))
    out.write("RESULT: %s\n" % ("MATCH" if not mismatch else "DIVERGED"))
    return 0 if not mismatch else 1


def main():
    sfd = os.open(SRC, os.O_RDONLY)
    sst = os.fstat(sfd)
    mode = sst.st_mode & 0o7777
    try:
        dst_st = os.stat(DST)
        if (dst_st.st_dev, dst_st.st_ino) == (sst.st_dev, sst.st_ino):
            sys.stderr.write("sparse-copy: refusing to copy a file onto itself\n")
            return 2
    except FileNotFoundError:
        pass
    dfd = os.open(DST, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, mode)
    try:
        last = None
        for attempt in range(1, MAX_ATTEMPTS + 1):
            status, copied, torn = copy_once(sfd, dfd)
            last = (status, copied, torn)
            if status == "stable" or attempt == MAX_ATTEMPTS:
                break
            # Source moved under us: rewind destination and take a fresh snapshot.
            os.ftruncate(dfd, 0)
        status, copied, torn = last
        if status != "stable":
            sys.stderr.write(
                "sparse-copy: source changed during copy "
                "(attempts=%d, last=%s%s)\n"
                % (MAX_ATTEMPTS, status,
                   ", torn extent @%d" % torn if torn is not None else ""))
    finally:
        os.close(sfd)
        os.close(dfd)

    # Preserve mode on an existing destination too.
    try:
        os.chmod(DST, mode)
    except OSError:
        pass

    rc = report(SRC, DST, copied)
    return rc


if __name__ == "__main__":
    sys.exit(main())

Key points

  • ftruncate(dst, size0) before writing extends the file without allocating, so untouched regions stay holes.
  • pwrite at the original offset for each [data, hole) range — no seek bookkeeping can drift.
  • Allocated zero data survives because extents come from SEEK_DATA, never from inspecting bytes.
  • No torn tail: a short pread immediately does ftruncate(dst, data).
  • Bounded retry (8×) re-snapshots on source change; the destination is only zeroed between retries, so the last coherent pass is kept.
  • Exits 1 with a per-extent report whenever size, extent kinds, or st_blocks diverge.
