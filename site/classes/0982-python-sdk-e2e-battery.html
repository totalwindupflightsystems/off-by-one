Three contract bugs were fixed during bring-up: (1) body: Any = None was parsed by FastAPI as a query param, never consuming the JSON body → declare Body(None); (2) the middleware wrote the request-id into the response only, so it couldn't be echoed in the payload → inject it into the incoming scope headers as well; (3) Starlette decodes %2F in routed paths → the canonical echo returns the decoded path and the battery asserts urllib.parse.unquote(path).

from fastapi import Body, FastAPI, Request
from uvicorn import Config, Server

class EchoHarness:
    def echo(self, request: Request, body: Any = Body(None)) -> dict[str, Any]:
        return {"service": "EchoHarness", "method": request.method,
                "path": request.url.path,
                "query": {k: v for k, v in sorted(request.query_params.items())},
                "headers": self._echo_headers(request), "body": body, "ts": time.time()}
    def health(self, request): ...   # status/uptime
    def version(self, request): ...  # service/sdk/version

def create_router(harness):
    router = APIRouter()
    for method in ("GET", "POST", "PUT", "DELETE"):
        router.add_api_route("/echo", harness.echo, methods=[method], response_model=dict[str, Any])
        router.add_api_route("/echo/{path:path}", harness.echo, methods=[method], response_model=dict[str, Any])
    router.add_api_route("/health", harness.health, methods=["GET"], ...)
    router.add_api_route("/version", harness.version, methods=["GET"], ...)
    return router

class EchoMiddleware:  # canonical add_middleware: request-id + timing
    async def __call__(self, scope, receive, send):
        ... # compute/echo x-echo-request-id into scope headers; set
            # x-echo-request-id + x-echo-ms response headers

def build_app():
    app = FastAPI(title="h3py EchoHarness", version=EchoHarness.VERSION)
    app.add_middleware(EchoMiddleware)
    app.include_router(create_router(EchoHarness()))   # SDK class preferred if importable
    return app

if __name__ == "__main__":
    Server(Config(build_app(), host=args.host, port=args.port, log_level="warning")).run()
