jsonl-board-event-id-inversion-recurrence — Diagnosis & Verified Fix

TL;DR

A closeout row was appended to events.jsonl by hand with a literal id: 327 while the board head was 353. The canonical appender stamps id = MAX(int id) + 1, so the correct value was 354. The backwards walk (353 -> 327) made the repo guard tests RED, but the canary only emitted a generic event_seq=gap with no ids/direction, so it did not distinguish an inversion from a legitimate forward gap.

Fix is two-part: 1. Repair the live board with a byte-preserving single-token renumber (327 -> 354), leaving every other byte untouched. 2. Harden detection so the probe prints offending id pairs and classifies decrease => inversion vs jump => forward, and the wrapper verdict follows that classification.

1. Root-cause analysis

Symptom. Guard suite RED with the pair (353, 327); canary verdict only event_seq=gap.

Why the row exists. The board file has more than one writer. The canonical appender computes the next id from the current maximum (MAX(int id) + 1), but a closeout/backfill path appended a hand-built row with a hard-coded id taken from stale context (327) after the head had already advanced to 353.

Evidence chain (as executed):

  1. pytest on the repo showed two failures naming the pair:
  2. test_no_id_inversion → id inversion(s): [(353, 327)]
  3. test_head_stamp_is_next_id → id pair (353,327), expected id 354
  4. git log -S '"id":327' cast the id token to the exact commit that added the row. The row shape was {"id":327,"ts":null,"tick_id":null,"kind":null} — ts/tick_id/kind are null, i.e. not canonical appender output (a hand-built row).
  5. Canonical appender rule ⇒ MAX(353) + 1 = 354, confirming the expected stamp.

Why detection missed it. The board probe collapsed every anomaly into a single residual label event_seq=gap. It discarded the neighboring ids and their direction, so an inversion (data corruption) looked identical to a forward jump (legitimate missing legacy ids). The wrapper had no way to escalate.

The probe now emits exactly:

event_seq=gap (inversion: 353->327; forward: 322->333, 341->343)

2b. repair_board_ids.py — byte-preserving single-token renumber

Reads raw bytes, splits physical lines, replaces only the bytes of the target id value, then re-asserts the invariant set before writing.

#!/usr/bin/env python3
"""Byte-preserving single-token renumber of an inverted board row."""
from __future__ import annotations

import argparse
import json
import re
import sys
from pathlib import Path

# Documented legacy forward gaps for this board; adjust per repo.
LEGACY_FORWARD_GAPS: set[tuple[int, int]] = {(322, 333), (341, 343)}


def parse_rows(raw: bytes) -> list[dict]:
    return [json.loads(l.decode("utf-8"))
            for l in raw.split(b"\n") if l.strip()]


def walk(ids: list[int]) -> tuple[list[tuple[int, int]], list[tuple[int, int]]]:
    """Return (inversions, forward_gaps) over adjacent ids in file order."""
    inversions, forward = [], []
    for a, b in zip(ids, ids[1:]):
        if b < a:
            inversions.append((a, b))
        elif b > a + 1:
            forward.append((a, b))
    return inversions, forward


def renumber(path: Path, old: int, new: int) -> dict:
    raw = path.read_bytes()
    nl = b"\r\n" if b"\r\n" in raw else b"\n"
    lines = raw.split(nl)
    had_final_newline = raw.endswith(nl)
    physical = lines[:-1] if had_final_newline and lines and lines[-1] == b"" else lines

    # 1. exactly one row carries the stale id
    id_re = re.compile(rb'"id"\s*:\s*(-?\d+)')
    candidates = []
    for idx, ln in enumerate(physical):
        if not ln.strip():
            continue
        m = id_re.search(ln)
        if m and int(m.group(1)) == old:
            candidates.append((idx, m))
    if len(candidates) != 1:
        raise SystemExit(f"expected exactly one row with id={old}, found {len(candidates)}")

    idx, m = candidates[0]
    # Replace ONLY the bytes of the id value; everything else stays byte-for-byte.
    new_lines = list(physical)
    new_lines[idx] = (
        physical[idx][: m.start(1)] + str(new).encode() + physical[idx][m.end(1):]
    )

    changed = sum(1 for a, b in zip(physical, new_lines) if a != b)

    out_raw = nl.join(new_lines) + (nl if had_final_newline else b"")
    path.write_bytes(out_raw)

    # ---------------- invariant assertions ----------------
    reparsed = parse_rows(out_raw)
    ids = [r["id"] for r in reparsed]
    inversions, forward = walk(ids)
    dupes = sorted({i for i in ids if ids.count(i) > 1})

    assert changed == 1, f"expected 1 changed line, got {changed}"
    assert len(reparsed) == len(physical), "row count changed"
    assert not inversions, f"inversions remain: {inversions}"
    assert not dupes, f"duplicate ids: {dupes}"
    assert set(forward) == LEGACY_FORWARD_GAPS, (
        f"forward gaps {forward} != documented {sorted(LEGACY_FORWARD_GAPS)}"
    )

    return {"old": old, "new": new, "rows": len(reparsed),
            "changed_lines": changed, "inversions": inversions,
            "forward_gaps": forward, "duplicates": dupes}


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("board", type=Path)
    ap.add_argument("--old", type=int, required=True)
    ap.add_argument("--new", type=int, required=True)
    a = ap.parse_args()
    print(json.dumps(renumber(a.board, a.old, a.new), indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())

2c. probe_board_seq.py — id-pair classification + verdict

#!/usr/bin/env python3
"""Board sequence probe: print offending id pairs and classify them."""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

LEGACY_FORWARD_GAPS: set[tuple[int, int]] = {(322, 333), (341, 343)}


def load_ids(path: Path) -> list[int]:
    return [json.loads(l)["id"]
            for l in path.read_text(encoding="utf-8").splitlines() if l.strip()]


def classify(ids: list[int]):
    inversions: list[tuple[int, int]] = []
    forward: list[tuple[int, int]] = []
    for a, b in zip(ids, ids[1:]):
        if b < a:
            inversions.append((a, b))      # went backwards  -> corruption
        elif b > a + 1:
            forward.append((a, b))         # jumped forward  -> gap
    return inversions, forward


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("board", type=Path)
    a = ap.parse_args()
    ids = load_ids(a.board)
    inversions, forward = classify(ids)

    parts = []
    if inversions:
        parts.append("inversion: " + ", ".join(f"{x}->{y}" for x, y in inversions))
    if forward:
        parts.append("forward: " + ", ".join(f"{x}->{y}" for x, y in forward))

    label = "gap" if (inversions or forward) else "ok"
    print(f"event_seq={label}" + (f" ({'; '.join(parts)})" if parts else ""))

    unknown_forward = [p for p in forward if p not in LEGACY_FORWARD_GAPS]
    if inversions:
        print("verdict=RED (id inversion)")            # inversion is never OK
        return 1
    if unknown_forward:
        print(f"verdict=RED (unexpected forward gap: {unknown_forward})")
        return 1
    if forward:
        print(f"verdict=GREEN (documented legacy forward gap: {forward})")
        return 0
    print("verdict=GREEN (contiguous)")
    return 0


if __name__ == "__main__":
    sys.exit(main())

2d. tests/test_board_guard.py — guards that name the pair

import json
from pathlib import Path

from probe_board_seq import LEGACY_FORWARD_GAPS, classify

BOARD = Path(__file__).resolve().parents[1] / "events.jsonl"


def load_ids() -> list[int]:
    return [json.loads(l)["id"] for l in BOARD.read_text().splitlines() if l.strip()]


def test_no_id_inversion() -> None:
    inversions, _ = classify(load_ids())
    assert not inversions, f"id inversion(s): {inversions}"


def test_no_duplicate_ids() -> None:
    ids = load_ids()
    assert len(ids) == len(set(ids))


def test_forward_gaps_are_documented_legacy_set() -> None:
    _, forward = classify(load_ids())
    assert set(forward) == LEGACY_FORWARD_GAPS


def test_head_stamp_is_next_id() -> None:
    ids = load_ids()
    head, last = max(ids[:-1]), ids[-1]
    assert last == head + 1, (
        f"closeout row was not stamped by the canonical appender: "
        f"id pair ({head},{last}), expected id {head + 1}"
    )


def test_board_is_valid_jsonl() -> None:
    for line in BOARD.read_text().splitlines():
        if line.strip():
            json.loads(line)
