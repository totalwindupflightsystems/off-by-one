The correct working pattern for a docs-drift-sweep-foreman-direct run, which is what this problem class encodes, looks like:

# 1) Foreman-direct sweep: verify every candidate finding with a live grep BEFORE any edit
guard() {  # fail the sweep if a live check does not match reality
  local desc="$1" cmd="$2" want="$3"
  local got; got=$(eval "$cmd")
  if [ "$got" != "$want" ]; then
    echo "DRIFT: $desc (expected '$want', got '$got')" >&2; return 1
  fi
  echo "OK: $desc"; return 0
}

guard "schema path refs resolve" \
  'grep -rn "docs/schema" . --include="*.md" | wc -l' "0"
guard "corpus count matches file list" \
  "$(find docs -name '*.md' | wc -l) vs corpus.md count" "0"
guard "env vars documented for each referenced var" \
  'grep -roh "\${[A-Z_]*}" docs | sort -u | wc -l' "0"
guard "structure tree lists every doc" \
  'diff <(grep -o "docs/[a-z-]*\.md" STRUCTURE.md | sort -u) <(find docs -name "*.md" | sort) | wc -l' "0"

# 2) Only after a guard passes for a REAL finding: batch same-file edits into ONE commit
