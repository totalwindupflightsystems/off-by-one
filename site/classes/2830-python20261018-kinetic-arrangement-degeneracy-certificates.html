Exact kinetic arrangement degeneracy certificates

Root-cause analysis

A naive kinetic arrangement solver fails for four separate reasons:

  1. Epsilon-based event ordering. Floating-point or t ± eps comparisons split events that occur at the same rational time into an arbitrary order. A triple point becomes three independent pair crossings, a triangle can appear to collapse and reappear, and before/after topology becomes order-dependent.

  2. One-event-at-a-time processing. Simultaneous degeneracies must be treated as a batch. Advancing the arrangement after only the first of several equal-time events makes the remaining queued events stale and can apply invalid topology updates.

  3. Incomplete event generation. Pairwise events alone are insufficient. A bounded face can collapse at a triple concurrence where no endpoint incidence or pair-intersection birth/death occurs. Collinear overlaps also require splitting at projected endpoints and merging duplicate geometric edges.

  4. No independently checkable evidence. Returning only a time does not prove which event occurred or why the invariant failed. The certificate must include exact event witnesses plus exact before/at/after topology records that can be recomputed from the input.

The fix below uses exact Fraction arithmetic, batches equal times, invalidates stale queue epochs, revalidates every witness exactly, and emits a compact verifiable certificate.

Exact fix

Event model

For moving endpoints p(t) = p0 + v*t, the candidate event times are rational roots of:

  • endpoint equality: two linear equations;
  • endpoint-on-supporting-line: a quadratic orientation polynomial;
  • triple-line concurrence: the determinant of three moving line-coefficient polynomials;
  • collinear overlap boundaries are covered by endpoint equality and endpoint-on-line incidences.

Every candidate is still revalidated exactly at its timestamp. Candidates that do not represent an actual incidence are discarded as stale.

Topology

At a fixed rational time:

  • split every segment at every endpoint incidence and every pairwise intersection or overlap boundary;
  • merge duplicate collinear geometric edges;
  • compute vertices V, edges E, connected components C;
  • use Euler’s formula F = E - V + C + 1;
  • optionally build half-edge face orbits for same_face invariants.

Certificate

Before/after snapshots are taken at exact rational midpoints between adjacent candidate times, never at t ± eps. The certificate contains:

  • exact failure time and event witnesses;
  • before, at, and after topology records;
  • Euler identity, vertex/edge/component/face counts, and canonical graph hashes;
  • invariant values before/at/after;
  • stale-event counters.

#!/usr/bin/env python3
"""
Exact event-driven kinetic arrangement solver.

Domain
------
Moving points are affine in time:

    p(t) = (px + vx*t, py + vy*t)

with rational px, py, vx, vy.  A moving segment has two moving endpoints.
The solver works with exact Fraction arithmetic and processes all events at
the same rational time as one batch.  It is intended for instances in which
the critical event times are rational; rational roots are extracted exactly.

The main entry points are:

    solve(segments, invariant, t_start, t_end) -> dict
    verify_certificate(cert, segments, invariant) -> (bool, list[str])

The returned dictionary is a compact certificate: exact event witnesses,
before/at/after topology counters and canonical graph fingerprints.
"""

def Q(x: Any) -> Fraction:
    """Convert input to an exact rational without going through binary float."""
    if isinstance(x, Fraction):
        return x
    if isinstance(x, int):
        return Fraction(x)
    return Fraction(str(x))


def qstr(x: Fraction) -> str:
    return f"{x.numerator}/{x.denominator}"


@dataclass(frozen=True, order=True)
class Vec2:
    x: Fraction
    y: Fraction

    def __add__(self, o: "Vec2") -> "Vec2":
        return Vec2(self.x + o.x, self.y + o.y)

    def __sub__(self, o: "Vec2") -> "Vec2":
        return Vec2(self.x - o.x, self.y - o.y)

    def __mul__(self, s: Fraction) -> "Vec2":
        return Vec2(self.x * s, self.y * s)

    __rmul__ = __mul__

    def dot(self, o: "Vec2") -> Fraction:
        return self.x * o.x + self.y * o.y

    def cross(self, o: "Vec2") -> Fraction:
        return self.x * o.y - self.y * o.x

    def is_zero(self) -> bool:
        return self.x == 0 and self.y == 0

    def key(self) -> tuple[Fraction, Fraction]:
        return (self.x, self.y)


@dataclass(frozen=True)
class MovingPoint:
    px: Fraction
    py: Fraction
    vx: Fraction
    vy: Fraction

    def at(self, t: Fraction) -> Vec2:
        return Vec2(self.px + self.vx * t, self.py + self.vy * t)

    def xpoly(self) -> list[Fraction]:
        return ptrim([self.px, self.vx])

    def ypoly(self) -> list[Fraction]:
        return ptrim([self.py, self.vy])


@dataclass(frozen=True)
class MovingSegment:
    a: MovingPoint
    b: MovingPoint
    label: str = ""

    def at(self, t: Fraction) -> "FixedSegment":
        return FixedSegment(self.a.at(t), self.b.at(t), self.label)


@dataclass(frozen=True)
class FixedSegment:
    a: Vec2
    b: Vec2
    label: str = ""

    @property
    def d(self) -> Vec2:
        return self.b - self.a

    def degenerate(self) -> bool:
        return self.a == self.b


def make_moving_point(px, py, vx, vy) -> MovingPoint:
    return MovingPoint(Q(px), Q(py), Q(vx), Q(vy))


def make_segment(a, b, label: str = "") -> MovingSegment:
    """a/b may be MovingPoint, 4-tuples, or 2-tuples (static point)."""
    def mp(v):
        if isinstance(v, MovingPoint):
            return v
        if len(v) == 4:
            return make_moving_point(*v)
        if len(v) == 2:
            return make_moving_point(v[0], v[1], 0, 0)
        raise ValueError("point must be MovingPoint, [x,y], or [x,y,vx,vy]")
    return MovingSegment(mp(a), mp(b), label)


def vec_dict(p: Vec2) -> dict[str, str]:
    return {"x": qstr(p.x), "y": qstr(p.y)}


def on_segment(p: Vec2, a: Vec2, b: Vec2) -> bool:
    d = b - a
    if d.is_zero():
        return p == a
    return d.cross(p - a) == 0 and 0 <= (p - a).dot(d) <= d.dot(d)


def segment_param(p: Vec2, a: Vec2, b: Vec2) -> Optional[Fraction]:
    d = b - a
    if d.is_zero():
        return None
    if d.cross(p - a) != 0:
        return None
    den = d.dot(d)
    num = (p - a).dot(d)
    if 0 <= num <= den:
        return num / den
    return None


def line_intersection(a: Vec2, b: Vec2, c: Vec2, d: Vec2):
    """Return (point, s, u) for a+s(b-a)=c+u(d-c), or None if parallel."""
    d1, d2 = b - a, d - c
    den = d1.cross(d2)
    if den == 0:
        return None
    diff = c - a
    s = diff.cross(d2) / den
    u = diff.cross(d1) / den
    return a + d1 * s, s, u


# ---------------------------------------------------------------------------
# Exact univariate polynomials (coefficients ascending)
# ---------------------------------------------------------------------------

def ptrim(p: Iterable[Fraction]) -> list[Fraction]:
    q = [Q(c) for c in p]
    while q and q[-1] == 0:
        q.pop()
    return q


def padd(a: list[Fraction], b: list[Fraction]) -> list[Fraction]:
    n = max(len(a), len(b))
    out = [Fraction(0)] * n
    for i in range(n):
        out[i] = (a[i] if i < len(a) else 0) + (b[i] if i < len(b) else 0)
    return ptrim(out)


def psub(a: list[Fraction], b: list[Fraction]) -> list[Fraction]:
    n = max(len(a), len(b))
    out = [Fraction(0)] * n
    for i in range(n):
        out[i] = (a[i] if i < len(a) else 0) - (b[i] if i < len(b) else 0)
    return ptrim(out)


def pmul(a: list[Fraction], b: list[Fraction]) -> list[Fraction]:
    if not a or not b:
        return []
    out = [Fraction(0)] * (len(a) + len(b) - 1)
    for i, ai in enumerate(a):
        if ai:
            for j, bj in enumerate(b):
                if bj:
                    out[i + j] += ai * bj
    return ptrim(out)


def peval(p: list[Fraction], x: Fraction) -> Fraction:
    r = Fraction(0)
    for c in reversed(p):
        r = r * x + c
    return r


def _integer_coeffs(p: list[Fraction]) -> list[int]:
    lcm = 1
    for c in p:
        lcm = lcm * c.denominator // math.gcd(lcm, c.denominator)
    vals = [int(c * lcm) for c in p]
    g = 0
    for v in vals:
        g = math.gcd(g, abs(v))
    if g > 1:
        vals = [v // g for v in vals]
    if vals and vals[-1] < 0:
        vals = [-v for v in vals]
    return vals


def divisors(n: int) -> list[int]:
    n = abs(n)
    if n == 0:
        return []
    small, large = [], []
    i = 1
    while i * i <= n:
        if n % i == 0:
            small.append(i)
            if i * i != n:
                large.append(n // i)
        i += 1
    return small + large[::-1]


def sqrt_fraction(q: Fraction) -> Optional[Fraction]:
    if q < 0:
        return None
    n, d = q.numerator, q.denominator
    rn, rd = math.isqrt(n), math.isqrt(d)
    if rn * rn == n and rd * rd == d:
        return Fraction(rn, rd)
    return None


def rational_roots(poly: list[Fraction]) -> list[Fraction]:
    """
    All rational roots of a polynomial with rational coefficients.

    Rational-root theorem plus exact quadratic formula.  The input contract
    of the solver is that every relevant critical time is rational; any
    irrational root is deliberately not approximated.
    """
    p = ptrim(poly)
    roots: set[Fraction] = set()
    if not p:
        return []

    # Factor out x^k.
    k = 0
    while k < len(p) and p[k] == 0:
        roots.add(Fraction(0))
        k += 1
    p = p[k:]
    if len(p) <= 1:
        return sorted(roots)

    if len(p) == 2:
        roots.add(-p[0] / p[1])
        return sorted(roots)

    if len(p) == 3:
        a, b, c = p[2], p[1], p[0]
        disc = b * b - 4 * a * c
        s = sqrt_fraction(disc)
        if s is not None:
            roots.add((-b + s) / (2 * a))
            roots.add((-b - s) / (2 * a))
        return sorted(roots)

    ints = _integer_coeffs(p)
    a0, an = ints[0], ints[-1]
    if a0 == 0:
        # Already stripped all zero constant terms.
        return sorted(roots)

    candidates: set[Fraction] = set()
    for pdiv in divisors(a0):
        for qdiv in divisors(an):
            g = math.gcd(pdiv, qdiv)
            pp, qq = pdiv // g, qdiv // g
            candidates.add(Fraction(pp, qq))
            candidates.add(Fraction(-pp, qq))

    for cand in candidates:
        if peval(p, cand) == 0:
            roots.add(cand)

    return sorted(roots)

def orientation_poly(seg: MovingSegment, p: MovingPoint) -> list[Fraction]:
    """cross(b(t)-a(t), p(t)-a(t)) as a polynomial in t."""
    ax, ay = seg.a.xpoly(), seg.a.ypoly()
    bx, by = seg.b.xpoly(), seg.b.ypoly()
    px, py = p.xpoly(), p.ypoly()
    dx, dy = psub(bx, ax), psub(by, ay)
    qx, qy = psub(px, ax), psub(py, ay)
    return psub(pmul(dx, qy), pmul(dy, qx))


def line_coeffs_poly(seg: MovingSegment):
    """L(t)=(Lx,Ly,Lz) with L.(x,y,1)=0 for the supporting line."""
    ax, ay = seg.a.xpoly(), seg.a.ypoly()
    bx, by = seg.b.xpoly(), seg.b.ypoly()
    lx = psub(ay, by)
    ly = psub(bx, ax)
    lz = psub(pmul(ax, by), pmul(ay, bx))
    return [lx, ly, lz]


def det3_poly(l1, l2, l3) -> list[Fraction]:
    return padd(
        psub(
            pmul(l1[0], psub(pmul(l2[1], l3[2]), pmul(l2[2], l3[1]))),
            pmul(l1[1], psub(pmul(l2[0], l3[2]), pmul(l2[2], l3[0]))),
        ),
        pmul(l1[2], psub(pmul(l2[0], l3[1]), pmul(l2[1], l3[0]))),
    )


def common_linear_root(px: list[Fraction], py: list[Fraction]) -> Optional[Fraction]:
    """t with px(t)=py(t)=0, or None (no isolated common root)."""
    for p in (ptrim(px), ptrim(py)):
        if not p:
            continue
        if len(p) == 1:
            if p[0] != 0:
                return None
        else:
            t = -p[0] / p[1]
            if peval(px, t) == 0 and peval(py, t) == 0:
                return t
            return None
    return None  # identically zero: equality holds on an interval

@dataclass
class Orbit:
    idx: int
    comp: int
    area: Fraction
    walk: list[tuple[tuple[Fraction, Fraction], tuple[Fraction, Fraction]]]
    polygon: list[Vec2]
    rep: Vec2


@dataclass
class TopologySnapshot:
    t: Fraction
    vertices: set[tuple[Fraction, Fraction]]
    edges: dict[tuple[tuple[Fraction, Fraction], tuple[Fraction, Fraction]], set[int]]
    adjacency: dict[tuple[Fraction, Fraction], set[tuple[Fraction, Fraction]]]
    components: int
    faces: int
    bounded_faces: int
    orbits: list[Orbit]
    face_roots: dict[int, int]
    unbounded_root: int
    orbit_face_count: int
    signature: str
    empty: bool = False

    def locate_face(self, p: Vec2) -> int:
        """Face containing p. p must not lie on the arrangement."""
        # First check the boundary.
        for (u, v) in self.edges:
            if on_segment(p, Vec2(*u), Vec2(*v)):
                raise ValueError("query point lies on the arrangement")
        positive = [o for o in self.orbits if o.area > 0]
        best = None
        for o in positive:
            if point_in_polygon_strict(p, o.polygon):
                if best is None or o.area < best.area:
                    best = o
        if best is None:
            return self.unbounded_root
        return self.face_roots[best.idx]

    def same_face(self, p: Vec2, q: Vec2) -> bool:
        return self.locate_face(p) == self.locate_face(q)


def point_in_polygon_strict(p: Vec2, poly: list[Vec2]) -> bool:
    """Exact even-odd point-in-polygon. Returns False on the boundary."""
    if len(poly) < 3:
        return False
    inside = False
    n = len(poly)
    for i in range(n):
        a, b = poly[i], poly[(i + 1) % n]
        if on_segment(p, a, b):
            return False
        if (a.y > p.y) != (b.y > p.y):
            xint = a.x + (p.y - a.y) * (b.x - a.x) / (b.y - a.y)
            if p.x < xint:
                inside = not inside
    return inside


def _angle_cmp_at(v, a, b) -> int:
    ax, ay = a[0] - v[0], a[1] - v[1]
    bx, by = b[0] - v[0], b[1] - v[1]
    ha = 0 if (ay > 0 or (ay == 0 and ax >= 0)) else 1
    hb = 0 if (by > 0 or (by == 0 and bx >= 0)) else 1
    if ha != hb:
        return -1 if ha < hb else 1
    c = ax * by - ay * bx
    if c != 0:
        return -1 if c > 0 else 1
    da, db = ax * ax + ay * ay, bx * bx + by * by
    if da != db:
        return -1 if da < db else 1
    return 0


def _add_point_to_segment(split: set[Fraction], seg: FixedSegment, p: Vec2) -> None:
    s = segment_param(p, seg.a, seg.b)
    if s is not None:
        split.add(s)


def build_topology(moving_segments: list[MovingSegment], t: Fraction) -> TopologySnapshot:
    """
    Build the exact planar graph of the segment union at time t and its faces.

    Every segment is split at every endpoint incidence and every pairwise
    intersection/overlap boundary.  Overlapping collinear pieces are merged
    into one geometric edge.
    """
    fixed = [s.at(t) for s in moving_segments]
    n = len(fixed)
    split: list[set[Fraction]] = [set() for _ in range(n)]

    for i, s in enumerate(fixed):
        if not s.degenerate():
            split[i].update((Fraction(0), Fraction(1)))

    for i in range(n):
        for j in range(i + 1, n):
            A, B = fixed[i], fixed[j]
            dA, dB = A.d, B.d
            da, db = dA.is_zero(), dB.is_zero()
            if da and db:
                continue
            if da:
                _add_point_to_segment(split[j], B, A.a)
                continue
            if db:
                _add_point_to_segment(split[i], A, B.a)
                continue

            den = dA.cross(dB)
            diff = B.a - A.a
            if den != 0:
                s = diff.cross(dB) / den
                u = diff.cross(dA) / den
                if 0 <= s <= 1 and 0 <= u <= 1:
                    split[i].add(s)
                    split[j].add(u)
            else:
                # Parallel.  Only a collinear overlap can create incidences.
                if diff.cross(dA) == 0:
                    for q in (B.a, B.b):
                        _add_point_to_segment(split[i], A, q)
                    for q in (A.a, A.b):
                        _add_point_to_segment(split[j], B, q)

    vertices: set[tuple[Fraction, Fraction]] = set()
    edge_labels: dict[tuple[tuple[Fraction, Fraction], tuple[Fraction, Fraction]], set[int]] = {}

    for i, s in enumerate(fixed):
        if s.degenerate():
            vertices.add(s.a.key())
            continue
        params = sorted(split[i])
        for k in range(len(params) - 1):
            u, v = params[k], params[k + 1]
            if u == v:
                continue
            p = s.a + s.d * u
            q = s.a + s.d * v
            if p == q:
                continue
            pk, qk = p.key(), q.key()
            key = (pk, qk) if pk < qk else (qk, pk)
            vertices.add(pk)
            vertices.add(qk)
            edge_labels.setdefault(key, set()).add(i)

    adjacency: dict[tuple[Fraction, Fraction], set[tuple[Fraction, Fraction]]] = {
        v: set() for v in vertices
    }
    for (u, v) in edge_labels:
        adjacency[u].add(v)
        adjacency[v].add(u)

    # Connected components (isolated vertices included).
    comp: dict[tuple[Fraction, Fraction], int] = {}
    cid = 0
    for start in vertices:
        if start in comp:
            continue
        stack = [start]
        comp[start] = cid
        while stack:
            u = stack.pop()
            for v in adjacency[u]:
                if v not in comp:
                    comp[v] = cid
                    stack.append(v)
        cid += 1
    C = cid

    V, E = len(vertices), len(edge_labels)
    F_euler = E - V + C + 1

    # Sorted outgoing half-edges at every vertex.
    neighbors_sorted: dict[tuple[Fraction, Fraction], list[tuple[Fraction, Fraction]]] = {}
    pos: dict[tuple[Fraction, Fraction], dict[tuple[Fraction, Fraction], int]] = {}
    for v in vertices:
        ns = sorted(adjacency[v], key=cmp_to_key(lambda a, b, v=v: _angle_cmp_at(v, a, b)))
        neighbors_sorted[v] = ns
        pos[v] = {w: k for k, w in enumerate(ns)}

    visited: set[tuple[tuple[Fraction, Fraction], tuple[Fraction, Fraction]]] = set()
    orbits: list[Orbit] = []
    for u in vertices:
        for v in neighbors_sorted[u]:
            if (u, v) in visited:
                continue
            walk = []
            cur = (u, v)
            while cur not in visited:
                visited.add(cur)
                walk.append(cur)
                a, b = cur
                idx = pos[b][a]
                nb = neighbors_sorted[b]
                nxt = nb[(idx - 1) % len(nb)]  # immediately clockwise
                cur = (b, nxt)
            if cur != (u, v):
                raise RuntimeError("non-planar half-edge traversal")
            area = sum(a[0] * b[1] - a[1] * b[0] for a, b in walk) / 2
            poly = [Vec2(*a) for a, _ in walk]
            orbits.append(
                Orbit(
                    idx=len(orbits),
                    comp=comp[u],
                    area=area,
                    walk=walk,
                    polygon=poly,
                    rep=Vec2(*u),
                )
            )

    # Merge boundary orbits belonging to the same actual face.
    parent = list(range(len(orbits) + 1))
    UNB = len(orbits)

    def find(x: int) -> int:
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    positive = [o for o in orbits if o.area > 0]
    for o in orbits:
        if o.area > 0:
            continue
        # This is the exterior boundary of its component.  It belongs to the
        # smallest positive orbit of another component that strictly contains
        # its representative vertex; otherwise it is the unbounded face.
        best = None
        for p in positive:
            if p.comp == o.comp:
                continue
            if point_in_polygon_strict(o.rep, p.polygon):
                if best is None or p.area < best.area:
                    best = p
        union(o.idx, best.idx if best is not None else UNB)

    face_roots = {o.idx: find(o.idx) for o in orbits}
    orbit_face_count = len({find(o.idx) for o in orbits} | {find(UNB)})

    data = {
        "vertices": sorted([list(v) for v in vertices]),
        "edges": sorted(
            (list(u), list(v), sorted(edge_labels[(u, v)]))
            for (u, v) in edge_labels
        ),
    }
    sig = hashlib.sha256(repr(data).encode("utf-8")).hexdigest()

    return TopologySnapshot(
        t=t,
        vertices=vertices,
        edges=edge_labels,
        adjacency=adjacency,
        components=C,
        faces=F_euler,
        bounded_faces=F_euler - 1,
        orbits=orbits,
        face_roots=face_roots,
        unbounded_root=find(UNB),
        orbit_face_count=orbit_face_count,
        signature=sig,
        empty=(V == 0),
    )

class Invariant:
    name = "invariant"

    def baseline(self, snap: TopologySnapshot) -> Any:
        raise NotImplementedError

    def check(self, snap: TopologySnapshot) -> bool:
        raise NotImplementedError

    def observed(self, snap: TopologySnapshot) -> Any:
        raise NotImplementedError

    def expected_value(self) -> Any:
        raise NotImplementedError


class FaceCountInvariant(Invariant):
    """The number of faces of the segment union must remain 'expected'."""

    name = "face_count"

    def __init__(self, expected: Optional[int] = None):
        self.expected = expected

    def baseline(self, snap: TopologySnapshot) -> int:
        if self.expected is None:
            self.expected = snap.faces
        return self.expected

    def check(self, snap: TopologySnapshot) -> bool:
        return snap.faces == self.expected

    def observed(self, snap: TopologySnapshot) -> int:
        return snap.faces

    def expected_value(self) -> int:
        return self.expected


class ComponentCountInvariant(Invariant):
    """The number of connected components of the segment union must stay fixed."""

    name = "component_count"

    def __init__(self, expected: Optional[int] = None):
        self.expected = expected

    def baseline(self, snap: TopologySnapshot) -> int:
        if self.expected is None:
            self.expected = snap.components
        return self.expected

    def check(self, snap: TopologySnapshot) -> bool:
        return snap.components == self.expected

    def observed(self, snap: TopologySnapshot) -> int:
        return snap.components

    def expected_value(self) -> int:
        return self.expected

@dataclass(frozen=True)
class Witness:
    kind: str
    idxs: tuple[int, ...]


def _add_candidate(by_time, t: Optional[Fraction], kind: str, idxs: tuple[int, ...],
                   t0: Fraction, t1: Optional[Fraction]) -> None:
    if t is None or t < t0 or (t1 is not None and t > t1):
        return
    by_time[t][(kind, idxs)] = Witness(kind, idxs)


def endpoint_of(segments: list[MovingSegment], eid: int) -> MovingPoint:
    i, side = divmod(eid, 2)
    return segments[i].a if side == 0 else segments[i].b


def generate_candidates(segments: list[MovingSegment], t0: Fraction,
                        t1: Optional[Fraction]) -> dict[Fraction, dict[tuple, Witness]]:
    """
    Generate an over-approximation of all topology-changing times.

    Candidates:
      * every pair of endpoints coincide,
      * every endpoint lies on every other segment's supporting line,
      * every triple of supporting lines is concurrent.

    The first two families cover pair intersection births/deaths and all
    collinear overlap boundaries.  The third covers triple (and higher)
    concurrency, which is where a face can collapse without any pairwise
    incidence changing.  Non-applicable roots are retained as queued
    candidates and are exactly rejected later (stale events).
    """
    by_time: dict[Fraction, dict[tuple, Witness]] = defaultdict(dict)
    n = len(segments)
    endpoints = []
    for i, s in enumerate(segments):
        endpoints.append((2 * i, s.a))
        endpoints.append((2 * i + 1, s.b))

    for x in range(len(endpoints)):
        eid1, p1 = endpoints[x]
        for y in range(x + 1, len(endpoints)):
            eid2, p2 = endpoints[y]
            dx = psub(p1.xpoly(), p2.xpoly())
            dy = psub(p1.ypoly(), p2.ypoly())
            t = common_linear_root(dx, dy)
            _add_candidate(by_time, t, "endpoint_equal", (eid1, eid2), t0, t1)

    for i, s in enumerate(segments):
        for eid, p in endpoints:
            if eid // 2 == i:
                continue
            roots = rational_roots(orientation_poly(s, p))
            for t in roots:
                _add_candidate(by_time, t, "point_on_line", (i, eid), t0, t1)

    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                det = det3_poly(
                    line_coeffs_poly(segments[i]),
                    line_coeffs_poly(segments[j]),
                    line_coeffs_poly(segments[k]),
                )
                for t in rational_roots(det):
                    _add_candidate(by_time, t, "triple_concurrent", (i, j, k), t0, t1)

    return by_time


def validate_witness(w: Witness, segments: list[MovingSegment], t: Fraction) -> bool:
    """Exact revalidation of a queued event at its own timestamp."""
    if w.kind == "endpoint_equal":
        e1, e2 = w.idxs
        return endpoint_of(segments, e1).at(t) == endpoint_of(segments, e2).at(t)

    if w.kind == "point_on_line":
        i, eid = w.idxs
        s = segments[i].at(t)
        p = endpoint_of(segments, eid).at(t)
        if s.degenerate():
            return False
        return s.d.cross(p - s.a) == 0 and on_segment(p, s.a, s.b)

    if w.kind == "triple_concurrent":
        i, j, k = w.idxs
        A, B, C = segments[i].at(t), segments[j].at(t), segments[k].at(t)
        if A.degenerate() or B.degenerate() or C.degenerate():
            return False
        r1 = line_intersection(A.a, A.b, B.a, B.b)
        r2 = line_intersection(A.a, A.b, C.a, C.b)
        if r1 is None or r2 is None:
            return False
        p, _, _ = r1
        if p != r2[0]:
            return False
        return on_segment(p, A.a, A.b) and on_segment(p, B.a, B.b) and on_segment(p, C.a, C.b)

    return False


def witness_dict(w: Witness, segments: list[MovingSegment], t: Fraction) -> dict:
    out = {"kind": w.kind, "indices": list(w.idxs)}
    if w.kind == "endpoint_equal":
        e1, e2 = w.idxs
        p1 = endpoint_of(segments, e1).at(t)
        out.update({
            "endpoints": [e1, e2],
            "labels": [segments[e1 // 2].label, segments[e2 // 2].label],
            "point": vec_dict(p1),
        })
    elif w.kind == "point_on_line":
        i, eid = w.idxs
        p = endpoint_of(segments, eid).at(t)
        s = segments[i].at(t)
        out.update({
            "segment": segments[i].label,
            "endpoint": eid,
            "endpoint_label": segments[eid // 2].label,
            "point": vec_dict(p),
            "segment_param": qstr(segment_param(p, s.a, s.b)),
        })
    elif w.kind == "triple_concurrent":
        i, j, k = w.idxs
        A, B = segments[i].at(t), segments[j].at(t)
        r = line_intersection(A.a, A.b, B.a, B.b)
        out.update({
            "segments": [segments[i].label, segments[j].label, segments[k].label],
            "point": vec_dict(r[0]) if r else None,
            "pair_params": (
                {"on_first": qstr(r[1]), "on_second": qstr(r[2])} if r else None
            ),
        })
    return out


# ---------------------------------------------------------------------------
# Exact priority queue with epoch-based stale invalidation
# ---------------------------------------------------------------------------

class BatchQueue:
    """
    Priority queue keyed by exact Fraction time.

    After a real event batch the epoch is advanced and all future batches are
    re-inserted with the new epoch.  Any older copy of a future batch is
    discarded as stale when popped.  This mirrors kinetic queues that
    invalidate events computed from an old topology, while keeping ordering
    purely exact.
    """

    def __init__(self, batches: list[tuple[Fraction, list[Witness]]]):
        self.batches = batches
        self.heap: list[tuple] = []
        self.epoch = 0
        self.stale_discarded = 0
        self.seq = 0
        for idx, (t, ws) in enumerate(batches):
            self._push(idx, t, ws)
        self.next_reschedule = 1

    def _push(self, idx: int, t: Fraction, ws: list[Witness]) -> None:
        heapq.heappush(self.heap, (t, self.seq, idx, self.epoch, ws))
        self.seq += 1

    def pop_valid(self):
        while self.heap:
            t, _, idx, ep, ws = heapq.heappop(self.heap)
            if ep != self.epoch:
                self.stale_discarded += 1
                continue
            return t, idx, ws
        return None

    def advance_and_reschedule(self, start_idx: int) -> None:
        self.epoch += 1
        for idx in range(start_idx, len(self.batches)):
            t, ws = self.batches[idx]
            self._push(idx, t, ws)

def _midpoint(a: Fraction, b: Fraction) -> Fraction:
    return (a + b) / 2


def _snapshot_record(snap: TopologySnapshot, inv: Invariant) -> dict:
    return {
        "time": qstr(snap.t),
        "vertices": len(snap.vertices),
        "edges": len(snap.edges),
        "components": snap.components,
        "faces": snap.faces,
        "bounded_faces": snap.bounded_faces,
        "euler": f"{len(snap.vertices)}-{len(snap.edges)}+{snap.faces}=={1 + snap.components}",
        "orbit_face_count": snap.orbit_face_count,
        "signature": snap.signature,
        "invariant": {
            "name": inv.name,
            "expected": _json_value(inv.expected_value()),
            "observed": _json_value(inv.observed(snap)),
            "ok": bool(inv.check(snap)),
        },
    }


def _json_value(v):
    if isinstance(v, Fraction):
        return qstr(v)
    if isinstance(v, Vec2):
        return vec_dict(v)
    return v


def _input_digest(segments: list[MovingSegment]) -> str:
    data = []
    for s in segments:
        data.append([
            s.label,
            [qstr(s.a.px), qstr(s.a.py), qstr(s.a.vx), qstr(s.a.vy)],
            [qstr(s.b.px), qstr(s.b.py), qstr(s.b.vx), qstr(s.b.vy)],
        ])
    return hashlib.sha256(repr(data).encode("utf-8")).hexdigest()


def solve(segments: list[MovingSegment],
          invariant: Optional[Invariant] = None,
          t_start: Fraction = Fraction(0),
          t_end: Optional[Fraction] = None,
          strict_sweep_check: bool = False) -> dict:
    """
    Sweep until the first failure of the invariant, or the horizon.

    A failure exactly at event time t is reported at t.  If the event makes
    the invariant fail immediately afterwards, it is also reported at t;
    the certificate contains a rational sample point strictly after t.
    """
    if not segments:
        raise ValueError("at least one segment is required")
    if invariant is None:
        invariant = FaceCountInvariant()
    t_start = Q(t_start)
    explicit_end = t_end is not None
    t_end_q = Q(t_end) if explicit_end else None

    by_time = generate_candidates(segments, t_start, t_end_q)
    if t_end_q is None:
        t_end_q = max(by_time.keys(), default=t_start) + 1
    t_end = t_end_q
    if t_end < t_start:
        raise ValueError("t_end must be >= t_start")

    times = sorted(by_time)
    batches = [(t, [by_time[t][k] for k in sorted(by_time[t], key=repr)]) for t in times]

    start_snap = build_topology(segments, t_start)
    invariant.baseline(start_snap)

    # The invariant may already fail at the initial time.
    if not invariant.check(start_snap):
        next_t = times[0] if times and times[0] > t_start else t_start
        after_t = _midpoint(t_start, next_t) if next_t > t_start else t_start
        after_snap = build_topology(segments, after_t)
        return _failure_certificate(
            segments, invariant, t_start, [Witness("initial_state", ())],
            [], 0, 0, start_snap, start_snap, after_snap, t_start,
            failing="initial",
        )

    queue = BatchQueue(batches)
    prev_time = t_start
    last_snap = start_snap
    stale_witness_total = 0
    actual_batches = 0

    while True:
        item = queue.pop_valid()
        if item is None:
            break
        t, idx, witnesses = item
        if t < t_start:
            continue

        at_snap = build_topology(segments, t)
        actual = [w for w in witnesses if validate_witness(w, segments, t)]
        stale_witness_total += len(witnesses) - len(actual)

        if strict_sweep_check:
            # Check that a topology change across a candidate time is never
            # missed by the witness set.
            nt = times[idx + 1] if idx + 1 < len(times) else t_end
            sample_after = _midpoint(t, nt) if nt > t else t
            after_snap_probe = build_topology(segments, sample_after)
            if at_snap.signature != after_snap_probe.signature and not actual:
                raise RuntimeError(
                    f"missed event at {qstr(t)}: topology changed but no witness validated"
                )

        if not actual:
            # Over-approximated/stale candidate; no topology-changing event.
            prev_time = t
            continue

        actual_batches += 1
        # Strictly-before sample: between the previous queued time and t.
        before_t = _midpoint(prev_time, t) if t > prev_time else t
        before_snap = build_topology(segments, before_t) if before_t != t else at_snap

        # Strictly-after sample: before the next queued time (or horizon end).
        next_t = times[idx + 1] if idx + 1 < len(times) else t_end
        after_t = _midpoint(t, next_t) if next_t > t else t
        after_snap = build_topology(segments, after_t) if after_t != t else at_snap

        # First failure at the exact event time.
        if not invariant.check(at_snap):
            return _failure_certificate(
                segments, invariant, t, actual, witnesses,
                stale_witness_total, queue.stale_discarded,
                before_snap, at_snap, after_snap, t, failing="at",
                actual_batches=actual_batches,
            )

        # Or immediately after the event.
        if after_t > t and not invariant.check(after_snap):
            return _failure_certificate(
                segments, invariant, t, actual, witnesses,
                stale_witness_total, queue.stale_discarded,
                before_snap, at_snap, after_snap, t, failing="after",
                actual_batches=actual_batches,
            )

        prev_time = t
        last_snap = after_snap
        queue.advance_and_reschedule(idx + 1)

    return {
        "status": "ok",
        "input_digest": _input_digest(segments),
        "invariant": invariant.name,
        "expected": _json_value(invariant.expected_value()),
        "time_range": [qstr(t_start), qstr(t_end)],
        "last": _snapshot_record(last_snap, invariant),
        "events_processed": actual_batches,
        "stale_witnesses_discarded": stale_witness_total,
        "queue_stale_discarded": queue.stale_discarded,
        "message": "invariant held through the horizon",
    }


def _failure_certificate(segments, invariant, t, actual, all_witnesses,
                         stale_witness_total, queue_stale, before_snap,
                         at_snap, after_snap, failure_time, failing: str,
                         actual_batches: int = 0) -> dict:
    return {
        "status": "failed",
        "input_digest": _input_digest(segments),
        "invariant": invariant.name,
        "expected": _json_value(invariant.expected_value()),
        "time": qstr(failure_time),
        "failing": failing,
        "event": {
            "actual_witnesses": [witness_dict(w, segments, t) for w in actual],
            "batch_witness_count": len(all_witnesses),
            "stale_witnesses_discarded": stale_witness_total,
            "queue_stale_discarded": queue_stale,
            "events_processed": actual_batches,
        },
        "before": _snapshot_record(before_snap, invariant),
        "at": _snapshot_record(at_snap, invariant),
        "after": _snapshot_record(after_snap, invariant),
        "policy": {
            "arithmetic": "exact rational (Fraction)",
            "ordering": "exact Fraction keys, never epsilon",
            "simultaneity": "all events at the same rational time form one batch",
            "stale_events": "epoch invalidation plus exact witness revalidation",
        },
    }

def verify_certificate(cert: dict, segments: list[MovingSegment],
                       invariant: Optional[Invariant] = None) -> tuple[bool, list[str]]:
    """
    Recompute every topology record from the geometry and revalidate every
    event witness.  Returns (ok, problems).
    """
    problems: list[str] = []

    def bad(msg: str) -> None:
        problems.append(msg)

    if cert.get("input_digest") != _input_digest(segments):
        bad("input digest mismatch")

    if invariant is None:
        # Reconstruct a suitable invariant from the certificate when possible.
        typ = cert.get("invariant")
        exp = cert.get("expected")
        if typ == "face_count" and exp is not None:
            invariant = FaceCountInvariant(int(exp))
        elif typ == "component_count" and exp is not None:
            invariant = ComponentCountInvariant(int(exp))
        elif typ == "bounded_face_count" and exp is not None:
            invariant = BoundedFaceCountInvariant(int(exp))
        else:
            bad("no invariant supplied and certificate invariant is not reconstructible")
            return False, problems

    if cert.get("status") == "failed":
        t = Q(cert["time"])
        for name in ("before", "at", "after"):
            rec = cert.get(name)
            if rec is None:
                bad(f"missing {name} snapshot")
                continue
            ts = Q(rec["time"])
            snap = build_topology(segments, ts)
            if len(snap.vertices) != rec["vertices"]:
                bad(f"{name}: vertex count mismatch")
            if len(snap.edges) != rec["edges"]:
                bad(f"{name}: edge count mismatch")
            if snap.components != rec["components"]:
                bad(f"{name}: component count mismatch")
            if snap.faces != rec["faces"]:
                bad(f"{name}: face count mismatch")
            if snap.bounded_faces != rec["bounded_faces"]:
                bad(f"{name}: bounded-face count mismatch")
            if snap.signature != rec["signature"]:
                bad(f"{name}: graph signature mismatch")
            if snap.orbit_face_count != rec["orbit_face_count"]:
                bad(f"{name}: orbit-derived face count mismatch")
            if len(snap.vertices) - len(snap.edges) + snap.faces != 1 + snap.components:
                bad(f"{name}: Euler identity fails")
            ok = invariant.check(snap)
            if bool(rec["invariant"]["ok"]) != bool(ok):
                bad(f"{name}: stored invariant result does not recompute")

        # Validate the event witnesses exactly at the failure time.
        if "event" not in cert:
            bad("missing event block")
            actual = []
        else:
            actual = cert["event"].get("actual_witnesses", [])
            if not actual:
                bad("event has no actual witness")
        for wd in actual:
            kind = wd.get("kind")
            idxs = tuple(wd.get("indices", ()))
            if kind == "initial_state":
                continue
            if kind not in ("endpoint_equal", "point_on_line", "triple_concurrent"):
                bad(f"unknown witness kind {kind!r}")
                continue
            if not validate_witness(Witness(kind, idxs), segments, t):
                bad(f"witness {kind}{idxs} does not validate at {qstr(t)}")

        before = cert.get("before")
        at = cert.get("at")
        after = cert.get("after")
        if cert.get("failing") != "initial" and before and before["invariant"]["ok"] is not True:
            bad("invariant was already false before the event")
        if at and after:
            if at["invariant"]["ok"] is True and after["invariant"]["ok"] is True:
                bad("certificate reports failure but neither at nor after snapshot fails")
            if Q(before["time"]) > Q(at["time"]):
                bad("before sample is after event time")
            if Q(after["time"]) < Q(at["time"]):
                bad("after sample is before event time")

        # Check expected value agrees with what the invariant actually enforces.
        if cert.get("expected") is not None:
            exp = cert["expected"]
            if isinstance(exp, str):
                try:
                    exp_cmp = Q(exp)
                except Exception:
                    exp_cmp = exp
                if isinstance(invariant.expected_value(), Fraction):
                    exp_cmp = Q(exp_cmp)
                if exp_cmp != invariant.expected_value():
                    bad("certificate expected value mismatch")

    elif cert.get("status") == "ok":
        rec = cert.get("last")
        if rec is None:
            bad("missing last snapshot")
        else:
            snap = build_topology(segments, Q(rec["time"]))
            if snap.signature != rec["signature"]:
                bad("last graph signature mismatch")
            if snap.faces != rec["faces"]:
                bad("last face count mismatch")
            if not invariant.check(snap):
                bad("success certificate but invariant fails at last snapshot")
    else:
        bad("unknown certificate status")

    return (len(problems) == 0), problems

def segments_from_json(data: list[dict]) -> list[MovingSegment]:
    out = []
    for item in data:
        out.append(make_segment(item["a"], item["b"], item.get("label", "")))
    return out


def invariant_from_json(obj: Optional[dict]) -> Invariant:
    if not obj:
        return FaceCountInvariant()
    typ = obj.get("type", "face_count")
    exp = obj.get("expected")
    if typ == "face_count":
        return FaceCountInvariant(None if exp is None else int(exp))
    if typ == "component_count":
        return ComponentCountInvariant(None if exp is None else int(exp))
    if typ == "bounded_face_count":
        return BoundedFaceCountInvariant(None if exp is None else int(exp))
    if typ == "same_face":
        p = Vec2(Q(obj["p"][0]), Q(obj["p"][1]))
        q = Vec2(Q(obj["q"][0]), Q(obj["q"][1]))
        return SameFaceInvariant(p, q, None if exp is None else bool(exp))
    raise ValueError(f"unknown invariant type {typ!r}")


def main(argv: list[str]) -> int:
    if len(argv) < 2:
        print("usage: kinetic_arrangement.py input.json [--verify certificate.json]",
              file=sys.stderr)
        return 2
    with open(argv[1], "r", encoding="utf-8") as f:
        spec = json.load(f)
    segments = segments_from_json(spec["segments"])
    inv = invariant_from_json(spec.get("invariant"))
    t_start = Q(spec.get("t_start", 0))
    t_end = None if spec.get("t_end") is None else Q(spec["t_end"])
    cert = solve(segments, inv, t_start, t_end,
                 strict_sweep_check=bool(spec.get("strict_sweep_check", False)))

    if "--verify" in argv:
        vpath = argv[argv.index("--verify") + 1]
        with open(vpath, "r", encoding="utf-8") as f:
            vcert = json.load(f)
        ok, problems = verify_certificate(vcert, segments, inv)
        cert["verification"] = {"ok": ok, "problems": problems}

    print(json.dumps(cert, indent=2, sort_keys=True))
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv))
