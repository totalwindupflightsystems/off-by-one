Key-Committing AEAD Envelope over AES-GCM

Problem class: go-aead-key-commitment-partition-oracle-multikey Environment: Go 1.26 (go1.26.8 linux/amd64), standard library only (crypto/aes, crypto/cipher, crypto/subtle).

1. Root-cause analysis

For a 96-bit nonce IV, let J0 = IV‖0³¹‖1 and H = E_K(0¹²⁸). AES-GCM's tag is

T = E_K(J0) XOR GHASH_H(A, C)
GHASH_H(X) = X_1·H^m XOR X_2·H^(m-1) XOR … XOR X_m·H

GHASH is linear in the AAD/ciphertext blocks, and the key enters only through H and E_K(J0). Given two keys K1, K2, the same nonce/ciphertext, and one free 16-byte AAD block A, both tags verify iff

GHASH_H1 − GHASH_H2 = E_K1(J0) XOR E_K2(J0) =: D

which is a single linear equation in GF(2¹²⁸):

A = ( D XOR C1·(H1³ XOR H2³) XOR C2·(H1² XOR H2²) XOR L·(H1 XOR H2) )
    · (H1⁴ XOR H2⁴)⁻¹

where L = [len(A)]₆₄‖[len(C)]₆₄. The same 2-block ciphertext then authenticates under two distinct keys and decrypts to two different plaintexts — an invisible salamander. A decryptor trying a key ring sees |accepted set| ≥ 2: a partition oracle. GCM's 2⁻¹²⁸ forgery bound does not give key commitment.

2. The fix

Add a domain-separated commitment tag CMAC(K_com, header‖ciphertext) and check it in constant time before GCM. Derive independent subkeys from the master key via AES-as-PRP: kEnc=E_K(1)‖E_K(2), kMac=E_K(3)‖E_K(4), kCom=E_K(5)‖E_K(6). Use a length-prefixed synthetic GCM nonce for nonce-misuse safety.

Envelope: version(1) | nonce(12) | adLen(4) | ctLen(4) | synthNonce(12) | ad | ct | gcmTag(16) | commitTag(16).

Proof sketch. Acc(E) ⊆ {K : CMAC(kCom(K), M) = commitTag} because GCM is only tried after the commitment matches. CMAC is a PRF under distinct AES-derived subkeys, so two distinct keys collide on the commitment with probability ≤ 2⁻¹²⁸ + negl; union-bounding over an n-key ring gives |Acc| ≤ 1. The honest key passes both checks, giving |Acc| = 1. Forging a new envelope needs a 128-bit GCM and 128-bit commitment forgery ≤ 2⁻¹²⁸. Nonce reuse across different messages changes synthNonce w.h.p. (2⁻⁹⁶), so GCM counters are not reused.

gf128.go — GF(2¹²⁸), GHASH, inversion

package commitaead

import "encoding/binary"

type gf128 [16]byte

var gfOne = gf128{0x80}

func gfXOR(a, b gf128) gf128 {
    var z gf128
    for i := range z {
        z[i] = a[i] ^ b[i]
    }
    return z
}

func gfMul(a, b gf128) gf128 {
    var z gf128
    v := a
    for i := 0; i < 128; i++ {
        if (b[i/8]>>(7-uint(i%8)))&1 != 0 {
            z = gfXOR(z, v)
        }
        lsb := v[15] & 1
        for j := 15; j > 0; j-- {
            v[j] = v[j]>>1 | v[j-1]<<7
        }
        v[0] >>= 1
        if lsb != 0 {
            v[0] ^= 0xe1
        }
    }
    return z
}

func gfSquare(a gf128) gf128 { return gfMul(a, a) }

// gfInv returns a^(2^128-2) = a^-1.
func gfInv(a gf128) gf128 {
    x := a
    for i := 0; i < 126; i++ {
        x = gfSquare(x)
        x = gfMul(x, a)
    }
    return gfSquare(x)
}

func pad16(b []byte) []byte {
    if len(b)%16 == 0 {
        return b
    }
    p := make([]byte, (len(b)/16+1)*16)
    copy(p, b)
    return p
}

func lengthBlock(aad, ct []byte) [16]byte {
    var l [16]byte
    binary.BigEndian.PutUint64(l[:8], uint64(len(aad))*8)
    binary.BigEndian.PutUint64(l[8:], uint64(len(ct))*8)
    return l
}

func ghashBlocks(h gf128, blocks ...[]byte) gf128 {
    var y gf128
    for _, b := range blocks {
        for len(b) > 0 {
            var blk gf128
            copy(blk[:], b[:16])
            y = gfMul(gfXOR(y, blk), h)
            b = b[16:]
        }
    }
    return y
}

func ghashGCM(h gf128, aad, ct []byte) gf128 {
    lb := lengthBlock(aad, ct)
    return ghashBlocks(h, pad16(aad), pad16(ct), lb[:])
}

cmac.go — AES-CMAC (NIST SP 800-38B)

package commitaead

import "crypto/cipher"

func cmac(block cipher.Block, msg []byte) [16]byte {
    var k1 [16]byte
    block.Encrypt(k1[:], k1[:])
    msb := shiftLeft(&k1)
    k1[15] ^= msb * 0x87
    k2 := k1
    msb = shiftLeft(&k2)
    k2[15] ^= msb * 0x87

    var x [16]byte
    if len(msg) == 0 {
        x = k2
        x[0] ^= 0x80
        block.Encrypt(x[:], x[:])
        return x
    }
    m := msg
    for len(m) >= 16 {
        for i := 0; i < 16; i++ {
            x[i] ^= m[i]
        }
        if len(m) == 16 {
            for i := 0; i < 16; i++ {
                x[i] ^= k1[i]
            }
        }
        block.Encrypt(x[:], x[:])
        m = m[16:]
    }
    if len(m) > 0 {
        for i := range m {
            x[i] ^= m[i]
        }
        for i := 0; i < 16; i++ {
            x[i] ^= k2[i]
        }
        x[len(m)] ^= 0x80
        block.Encrypt(x[:], x[:])
    }
    return x
}

func shiftLeft(x *[16]byte) byte {
    var msb byte
    for i := len(x) - 1; i >= 0; i-- {
        msb, x[i] = x[i]>>7, x[i]<<1|msb
    }
    return msb
}

commitaead.go — envelope API (full file)

package commitaead

import (
    "bytes"
    "crypto/aes"
    "crypto/cipher"
    "crypto/subtle"
    "encoding/binary"
    "errors"
)

const (
    KeySize    = 32
    NonceSize  = 12
    TagSize    = 16
    CommitSize = 16
    synthNonceSize = 12
    Version    = 0x02

    headerFixed = 1 + NonceSize + 4 + 4 + synthNonceSize
    overhead    = headerFixed + TagSize + CommitSize
)

var (
    ErrKeySize   = errors.New("commitaead: key must be 32 bytes (AES-256)")
    ErrNonceSize = errors.New("commitaead: nonce must be 12 bytes")
    ErrFormat    = errors.New("commitaead: malformed envelope")
    ErrNoMatch   = errors.New("commitaead: no key in the ring authenticates the envelope")
    ErrAmbiguous = errors.New("commitaead: envelope authenticates under more than one key")
)

func deriveKey(key []byte) (kEnc, kMac, kCom []byte, err error) {
    if len(key) != KeySize {
        return nil, nil, nil, ErrKeySize
    }
    block, err := aes.NewCipher(key)
    if err != nil {
        return nil, nil, nil, err
    }
    derive := func(domain byte) []byte {
        var in, out [16]byte
        in[15] = domain
        block.Encrypt(out[:], in[:])
        return out[:]
    }
    kEnc = append(append([]byte{}, derive(1)...), derive(2)...)
    kMac = append(append([]byte{}, derive(3)...), derive(4)...)
    kCom = append(append([]byte{}, derive(5)...), derive(6)...)
    return kEnc, kMac, kCom, nil
}

func EncryptCommitted(key, nonce, ad, plaintext []byte) ([]byte, error) {
    if len(key) != KeySize {
        return nil, ErrKeySize
    }
    if len(nonce) != NonceSize {
        return nil, ErrNonceSize
    }
    kEnc, kMac, kCom, err := deriveKey(key)
    if err != nil {
        return nil, err
    }
    if uint64(len(ad)) > 0xffffffff || uint64(len(plaintext)) > 0xffffffff {
        return nil, ErrFormat
    }
    macBlock, err := aes.NewCipher(kMac)
    if err != nil {
        return nil, err
    }
    // length-prefixed, injective SIV input
    synInput := make([]byte, 0, 8+len(nonce)+len(ad)+len(plaintext))
    var lp [4]byte
    binary.BigEndian.PutUint32(lp[:], uint32(len(ad)))
    synInput = append(synInput, lp[:]...)
    synInput = append(synInput, ad...)
    binary.BigEndian.PutUint32(lp[:], uint32(len(plaintext)))
    synInput = append(synInput, lp[:]...)
    synInput = append(synInput, plaintext...)
    synInput = append(synInput, nonce...)
    synFull := cmac(macBlock, synInput)
    var synthNonce [synthNonceSize]byte
    copy(synthNonce[:], synFull[:synthNonceSize])

    header := make([]byte, 0, headerFixed+len(ad))
    header = append(header, Version)
    header = append(header, nonce...)
    var lenBuf [4]byte
    binary.BigEndian.PutUint32(lenBuf[:], uint32(len(ad)))
    header = append(header, lenBuf[:]...)
    binary.BigEndian.PutUint32(lenBuf[:], uint32(len(plaintext)))
    header = append(header, lenBuf[:]...)
    header = append(header, synthNonce[:]...)
    header = append(header, ad...)

    encBlock, err := aes.NewCipher(kEnc)
    if err != nil {
        return nil, err
    }
    aead, err := cipher.NewGCM(encBlock)
    if err != nil {
        return nil, err
    }
    sealed := aead.Seal(nil, synthNonce[:], plaintext, ad)
    ct := sealed[:len(sealed)-TagSize]
    gcmTag := sealed[len(sealed)-TagSize:]

    comBlock, err := aes.NewCipher(kCom)
    if err != nil {
        return nil, err
    }
    commitInput := make([]byte, 0, len(header)+len(ct))
    commitInput = append(commitInput, header...)
    commitInput = append(commitInput, ct...)
    commit := cmac(comBlock, commitInput)

    env := make([]byte, 0, overhead+len(ad)+len(ct))
    env = append(env, header...)
    env = append(env, ct...)
    env = append(env, gcmTag...)
    env = append(env, commit[:]...)
    return env, nil
}

type envelopeView struct {
    nonce, synthNonce, ad, ct, gcmTag, commitTag, header []byte
}

func parseEnvelope(env []byte) (*envelopeView, error) {
    if len(env) < overhead {
        return nil, ErrFormat
    }
    if env[0] != Version {
        return nil, ErrFormat
    }
    adLen := uint64(binary.BigEndian.Uint32(env[13:17]))
    ctLen := uint64(binary.BigEndian.Uint32(env[17:21]))
    if adLen+ctLen+overhead != uint64(len(env)) {
        return nil, ErrFormat
    }
    adStart := headerFixed
    adEnd := adStart + int(adLen)
    ctEnd := adEnd + int(ctLen)
    return &envelopeView{
        nonce:      env[1:13],
        synthNonce: env[21:33],
        header:     env[:adEnd],
        ad:         env[adStart:adEnd],
        ct:         env[adEnd:ctEnd],
        gcmTag:     env[ctEnd : ctEnd+TagSize],
        commitTag:  env[ctEnd+TagSize : ctEnd+TagSize+CommitSize],
    }, nil
}

func DecryptCommitted(ring [][]byte, envelope []byte) (plaintext []byte, keyIndex int, err error) {
    v, err := parseEnvelope(envelope)
    if err != nil {
        return nil, -1, err
    }
    commitInput := make([]byte, 0, len(v.header)+len(v.ct))
    commitInput = append(commitInput, v.header...)
    commitInput = append(commitInput, v.ct...)

    // Deduplicate: a repeated identical key is one key, not an ambiguity.
    type entry struct {
        key []byte
        idx int
    }
    uniq := make([]entry, 0, len(ring))
    for i, key := range ring {
        seen := false
        for _, e := range uniq {
            if bytes.Equal(e.key, key) {
                seen = true
                break
            }
        }
        if !seen {
            uniq = append(uniq, entry{key, i})
        }
    }

    matches, keyIndex := 0, -1
    var result []byte
    for _, e := range uniq {
        if len(e.key) != KeySize {
            continue
        }
        kEnc, _, kCom, err := deriveKey(e.key)
        if err != nil {
            continue
        }
        comBlock, err := aes.NewCipher(kCom)
        if err != nil {
            continue
        }
        expected := cmac(comBlock, commitInput)
        if subtle.ConstantTimeCompare(expected[:], v.commitTag) != 1 {
            continue
        }
        encBlock, err := aes.NewCipher(kEnc)
        if err != nil {
            continue
        }
        aead, err := cipher.NewGCM(encBlock)
        if err != nil {
            continue
        }
        ctTag := make([]byte, 0, len(v.ct)+TagSize)
        ctTag = append(ctTag, v.ct...)
        ctTag = append(ctTag, v.gcmTag...)
        p, err := aead.Open(nil, v.synthNonce, ctTag, v.ad)
        if err != nil {
            continue
        }
        matches++
        keyIndex = e.idx
        result = p
    }
    switch {
    case matches == 1:
        return result, keyIndex, nil
    case matches == 0:
        return nil, -1, ErrNoMatch
    default:
        return nil, -1, ErrAmbiguous
    }
}

func AcceptedKeys(ring [][]byte, envelope []byte) []int {
    var out []int
    for i, key := range ring {
        if _, _, err := DecryptCommitted([][]byte{key}, envelope); err == nil {
            out = append(out, i)
        }
    }
    return out
}

salamander.go — the partition oracle

package commitaead

import (
    "crypto/aes"
    "crypto/cipher"
    "encoding/binary"
    "fmt"
)

type InvisibleSalamander struct {
    Nonce      []byte
    AAD        []byte
    Ciphertext []byte
    Tag        []byte
    Plaintext1 []byte
    Plaintext2 []byte
}

func NewInvisibleSalamander(k1, k2, nonce, aadPrefix, ct []byte) (*InvisibleSalamander, error) {
    if len(nonce) != 12 {
        return nil, fmt.Errorf("commitaead: salamander nonce must be 12 bytes")
    }
    if len(aadPrefix)%16 != 0 {
        return nil, fmt.Errorf("commitaead: aadPrefix must be a multiple of 16 bytes")
    }
    if len(ct)%16 != 0 || len(ct) == 0 {
        return nil, fmt.Errorf("commitaead: ciphertext must be a non-empty multiple of 16 bytes")
    }
    b1, err := aes.NewCipher(k1)
    if err != nil {
        return nil, err
    }
    b2, err := aes.NewCipher(k2)
    if err != nil {
        return nil, err
    }
    var zero [16]byte
    var H1, H2 gf128
    b1.Encrypt(H1[:], zero[:])
    b2.Encrypt(H2[:], zero[:])

    j0 := make([]byte, 16)
    copy(j0, nonce)
    j0[15] = 1
    var E1, E2 gf128
    b1.Encrypt(E1[:], j0)
    b2.Encrypt(E2[:], j0)
    delta := gfXOR(E1, E2)

    totalAADBlocks := len(aadPrefix)/16 + 1
    ctBlocks := len(ct) / 16
    m := totalAADBlocks + ctBlocks + 1

    h1p := make([]gf128, m+1)
    h2p := make([]gf128, m+1)
    h1p[0], h2p[0] = gfOne, gfOne
    for i := 1; i <= m; i++ {
        h1p[i] = gfMul(h1p[i-1], H1)
        h2p[i] = gfMul(h2p[i-1], H2)
    }
    coeff := func(p int) gf128 { return gfXOR(h1p[p], h2p[p]) }

    var sum gf128
    for a := 0; a < totalAADBlocks-1; a++ {
        var blk gf128
        copy(blk[:], aadPrefix[a*16:a*16+16])
        sum = gfXOR(sum, gfMul(blk, coeff(m-a)))
    }
    for c := 0; c < ctBlocks; c++ {
        var blk gf128
        copy(blk[:], ct[c*16:c*16+16])
        sum = gfXOR(sum, gfMul(blk, coeff(m-(totalAADBlocks+c))))
    }
    var lb [16]byte
    aadLen := totalAADBlocks * 16
    binary.BigEndian.PutUint64(lb[:8], uint64(aadLen)*8)
    binary.BigEndian.PutUint64(lb[8:], uint64(len(ct))*8)
    {
        var blk gf128
        copy(blk[:], lb[:])
        sum = gfXOR(sum, gfMul(blk, coeff(1)))
    }

    pStar := m - (totalAADBlocks - 1)
    X := gfMul(gfXOR(delta, sum), gfInv(coeff(pStar)))

    aad := make([]byte, aadLen)
    copy(aad, aadPrefix)
    copy(aad[(totalAADBlocks-1)*16:], X[:])

    s1 := ghashGCM(H1, aad, ct)
    tag := make([]byte, TagSize)
    for i := range tag {
        tag[i] = E1[i] ^ s1[i]
    }

    ctTag := append(append([]byte{}, ct...), tag...)
    g1, err := cipher.NewGCM(b1)
    if err != nil {
        return nil, err
    }
    p1, err := g1.Open(nil, nonce, append([]byte{}, ctTag...), aad)
    if err != nil {
        return nil, fmt.Errorf("commitaead: salamander did not verify under key 1: %w", err)
    }
    g2, err := cipher.NewGCM(b2)
    if err != nil {
        return nil, err
    }
    p2, err := g2.Open(nil, nonce, append([]byte{}, ctTag...), aad)
    if err != nil {
        return nil, fmt.Errorf("commitaead: salamander did not verify under key 2: %w", err)
    }
    return &InvisibleSalamander{
        Nonce: nonce, AAD: aad, Ciphertext: ct, Tag: tag,
        Plaintext1: p1, Plaintext2: p2,
    }, nil
}

// TestCommittedVariantBoundsAcceptedSet drives the accepted set to exactly 0 or 1
// for genuine GCM salamanders (which plain GCM accepts under two keys).
func TestCommittedVariantBoundsAcceptedSet(t *testing.T) {
    commitTagOff := overhead - CommitSize
    for iter := 0; iter < 100; iter++ {
        k1, k2 := randBytes(t, KeySize), randBytes(t, KeySize)
        ring := [][]byte{randBytes(t, KeySize), k1, randBytes(t, KeySize), k2}
        nonce, callerNonce, ct := randBytes(t, NonceSize), randBytes(t, NonceSize), randBytes(t, 32)
        // GCM runs under derived subkeys.
        kEnc1, _, _, _ := deriveKey(k1)
        kEnc2, _, _, _ := deriveKey(k2)
        sal, _ := NewInvisibleSalamander(kEnc1, kEnc2, nonce, nil, ct)

        // commitment under k1 -> only k1 accepted (GCM part also valid under k2)
        env1 := buildEnvelopeFor(t, k1, callerNonce, sal.Nonce, sal.AAD, sal.Ciphertext, sal.Tag)
        if got := AcceptedKeys(ring, env1); len(got) != 1 || got[0] != 1 {
            t.Fatalf("env1 accepted set = %v, want [1]", got)
        }
        // commitment under k2 -> only k2 accepted
        env2 := buildEnvelopeFor(t, k2, callerNonce, sal.Nonce, sal.AAD, sal.Ciphertext, sal.Tag)
        if got := AcceptedKeys(ring, env2); len(got) != 1 || got[0] != 3 {
            t.Fatalf("env2 accepted set = %v, want [3]", got)
        }
        // random commitment -> nobody accepted
        envRandom := append([]byte{}, env1...)
        copy(envRandom[commitTagOff:], randBytes(t, CommitSize))
        if got := AcceptedKeys(ring, envRandom); len(got) != 0 {
            t.Fatalf("random commitment accepted set = %v, want []", got)
        }
    }
}

Constant-time: DecryptCommitted uses crypto/subtle.ConstantTimeCompare for the commitment; GCM's Open uses subtle.ConstantTimeCompare internally (confirmed in crypto/internal/fips140/aes/gcm/). No byte-wise short-circuit.

Two subtle bugs caught and fixed during development: (1) the embedded salamander must collide on the derived kEnc subkeys, not the master keys; (2) naive CMAC(kMac, nonce‖ad‖pt) is non-injective ((ad="a",pt="bc") ≡ (ad="ab",pt="c")), so the SIV input is length-prefixed.
