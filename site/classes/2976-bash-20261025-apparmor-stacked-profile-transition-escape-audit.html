AppArmor Stacked-Profile Transition Escape Auditor

Root-cause analysis

The escape class is not a single parser bug. It comes from reviewing only isolated AppArmor rules instead of the reachable label graph:

  • ux / pux transitions reach the unconfined label.
  • px / cx transitions can reach another profile that has additional file, capability, network, or mount permissions.
  • change_profile, delegate, and wildcard/unsafe transitions can bypass the expected confinement.
  • Hats and change_profile/change_hat edges are separate reachable labels.
  • attach_disconnected and delegate_deleted broaden the effective security state; a transition into a label carrying them is an escape even when the textual rules look similar.
  • Includes, live securityfs policy, and variables destroy naive file:line attribution unless preprocessing preserves provenance.
  • A clean policy often has different rules in source and target. It must not be flagged unless the target adds permissions not covered by the source; exact rule-string comparison creates false positives.

The fix is a policy reconstructor plus permission lattice and transition graph, not a grep rule checker.

Exact fix

The verified implementation is in:

~/apparmor-audit/
├── apparmor_escape_audit.sh
├── run-tests.sh
├── fixtures/escapes/...
└── fixtures/clean/...

Run it directly:

cd ~/apparmor-audit
./apparmor_escape_audit.sh \
  --etc fixtures/escapes/etc/apparmor.d \
  --securityfs fixtures/escapes/securityfs/apparmor \
  --out out-escapes

Exit codes:

  • 0 = no real escape
  • 1 = one or more escape witnesses
  • 2 = usage or I/O error

The complete auditor is below.

#!/usr/bin/env bash
# AppArmor policy escape auditor.
# Reconstructs policy from source + live securityfs text policy, builds the
# exec/hat/change_profile/stack transition graph, computes effective permission
# sets, and reports witness transitions that reach a less restrictive label.
set -u
shopt -s extglob

ETC_DIR="/etc/apparmor.d"
SEC_DIR="/sys/kernel/security/apparmor"
OUT_DIR=""
JSON_FILE=""

QUIET=0

usage() {
  cat <<EOF
Usage: $0 [options]
  --etc DIR          AppArmor source tree (default: $ETC_DIR)
  --securityfs DIR   securityfs apparmor root (default: $SEC_DIR)
  --out DIR          directory for report.json/report.txt
  --json FILE        JSON report path (overrides --out)
  --summary FILE     human summary path (overrides --out)
  -I DIR             extra include search directory (repeatable)
  -q                 quiet (no human summary on stdout)
  -h, --help         show this help
  --version          show version

Exit status: 0 = no escape found, 1 = escape(s) found, 2 = usage/IO error.
EOF
}

die() { printf 'error: %s\n' "$*" >&2; exit 2; }

done

clean_field() { local s=${1//$'\t'/ }; s=${s//$'\n'/ }; s=${s//$'\r'/ }; s=${s//$US/ }; printf '%s' "$s"; }
trim() { local s=$1; s=${s#"${s%%[![:space:]]*}"}; s=${s%"${s##*[![:space:]]}"}; printf '%s' "$s"; }

subtract_modes() {
  local a=$1 b=$2 out="" c i
  for ((i=0; i<${#a}; i++)); do
    c=${a:i:1}
    [[ $b == *"$c"* ]] || out+=$c

join_by() { local d=$1; shift; local f=$1; shift; printf '%s' "$f"; printf '%s%s' "$d" "$@" 2>/dev/null; }

expand_vars() {
  local text=$1 k pass found
  for pass in 1 2 3 4 5 6; do
    found=0
    for k in "${!VAR[@]}"; do
      if [[ $text == *"@{${k}}"* ]]; then
        text=${text//"@{${k}}"/${VAR[$k]}}
        found=1
      fi
    done
    (( found )) || break
  done
  printf '%s' "$text"
}

resolve_include() {
  local inc=$1 base=$2 p d
  inc=${inc#<}; inc=${inc%>}; inc=${inc#\"}; inc=${inc%\"}
  inc=$(expand_vars "$inc")
  if [[ $inc == /* ]]; then
    [[ -f $inc ]] && { printf '%s' "$inc"; return 0; }
  else
    for d in "$base" "$ETC_DIR" /etc/apparmor.d "${INCLUDE_DIRS[@]}"; do
      [[ -n $d ]] || continue
      p="$d/$inc"
      [[ -f $p ]] && { printf '%s' "$p"; return 0; }
    done
  fi
  return 1
}

include_target() {
  local line=$1 base=$2 rest pat exists=0
  line=$(trim "$line")
  [[ $line == \#include* || $line == include* ]] || return 1
  if [[ $line == \#include* ]]; then rest=${line#\#include}; else rest=${line#include}; fi
  rest=$(trim "$rest")
  if [[ $rest == if[[:space:]]+exists* ]]; then
    exists=1
    rest=${rest#if}
    rest=${rest#exists}
    rest=$(trim "$rest")
  fi
  rest=${rest%,}; rest=$(trim "$rest")
  pat=$rest
  local resolved
  if ! resolved=$(resolve_include "$pat" "$base"); then
    (( exists )) && { printf ''; return 0; }
    printf 'missing-include:%s' "$pat"; return 0
  fi
  printf '%s' "$resolved"
  return 0
}

collect_file() {
  local file=$1 depth=${2:-0} line inc base
  (( depth > 40 )) && return
  [[ -f $file ]] || return
  base=$(dirname "$file")
  while IFS= read -r line || [[ -n $line ]]; do
    inc=$(include_target "$line" "$base" 2>/dev/null) || true
    if [[ -n $inc && $inc != missing-include:* ]]; then
      collect_file "$inc" $((depth+1))
      continue
    fi
    line=$(trim "$line")
    [[ $line == \#* || -z $line ]] && continue
    if [[ $line =~ ^abi[[:space:]] ]]; then continue; fi
    if [[ $line =~ ^@\{([A-Za-z_][A-Za-z0-9_]*)\}[[:space:]]*(\+?=|\?=|:=)[[:space:]]*(.*)$ ]]; then
      local name=${BASH_REMATCH[1]} op=${BASH_REMATCH[2]} val=${BASH_REMATCH[3]}
      val=${val%%#*}; val=$(trim "${val%,}")
      case "$op" in
        +=) VAR[$name]="${VAR[$name]:-}${VAR[$name]:+ }$val" ;;
        ?=) [[ -n ${VAR[$name]:-} ]] || VAR[$name]=$val ;;
        *)  VAR[$name]=$val ;;
      esac
    fi
  done < "$file"
}

emit_file() {
  local file=$1 depth=${2:-0} line inc base text lineno=0
  (( depth > 40 )) && return
  [[ -f $file ]] || return
  base=$(dirname "$file")
  while IFS= read -r line || [[ -n $line ]]; do
    ((lineno++))
    inc=$(include_target "$line" "$base" 2>/dev/null) || true
    if [[ -n $inc ]]; then
      if [[ $inc == missing-include:* ]]; then
        printf '%s%s%s%sinclude-error: %s\n' "$file" "$US" "$lineno" "$US" "${inc#missing-include:}" >> "$TMPDIR/warnings"
      else
        emit_file "$inc" $((depth+1))
      fi
      continue
    fi
    text=$(trim "$line")
    [[ -z $text || $text == \#* ]] && continue
    [[ $text =~ ^abi[[:space:]] ]] && continue
    [[ $text =~ ^@\{[A-Za-z_][A-Za-z0-9_]*\}[[:space:]]*(\+?=|\?=|:=) ]] && continue
    text=${text%%#*}
    text=$(trim "$text")
    [[ -z $text ]] && continue
    text=$(expand_vars "$text")
    printf '%s%s%s%s%s\n' "$file" "$US" "$lineno" "$US" "$text"
  done < "$file"
}

normalize_label_name() {
  local name=$1 parent=$2
  name=$(trim "$name")
  name=${name#\"}; name=${name%\"}
  name=${name%,}
  if [[ $name == \&* ]]; then name=${name#\&}; fi
  if [[ -n $parent ]]; then
    if [[ $name == "$parent//"* ]]; then
      :
    elif [[ $name == //* ]]; then
      name="$parent$name"
    else
      name="$parent//$name"
    fi
  fi
  printf '%s' "$name"
}

add_label() {
  local name=$1 parent=$2 flags=$3 attach=$4 origin=$5 type=$6
  [[ -n $name ]] || return
  if [[ -z ${LABEL_SEEN[$name]:-} ]]; then
    LABEL_SEEN[$name]=1
    LABEL_ORDER+=("$name")
  fi
  if [[ -z ${LABEL_ORIGIN[$name]:-} ]]; then LABEL_ORIGIN[$name]=$origin; fi
  [[ -n $parent ]] && PARENT[$name]=$parent
  [[ -n $type && -z ${LABEL_TYPE[$name]:-} ]] && LABEL_TYPE[$name]=$type
  if [[ -n $flags ]]; then
    if [[ -z ${FLAGS[$name]:-} ]]; then FLAGS[$name]=$flags; else FLAGS[$name]="${FLAGS[$name]},$flags"; fi
  fi
  [[ -n $attach && -z ${ATTACH[$name]:-} ]] && ATTACH[$name]=$attach
  if [[ $flags == *unconfined* ]]; then UNCONFINED[$name]=1; fi
}

parse_profile_header() {
  local raw=$1 parent=$2 origin=$3 text name attach="" flags="" type="profile" rest
  text=$(trim "$raw")
  text=${text%\{}
  text=$(trim "$text")
  if [[ $text == profile[[:space:]]* ]]; then
    rest=${text#profile}; rest=$(trim "$rest")
  elif [[ $text == hat[[:space:]]* ]]; then
    rest=${text#hat}; rest=$(trim "$rest"); type="hat"
  elif [[ $text == ^* ]]; then
    rest=${text#^}; rest=$(trim "$rest"); type="hat"
  else
    rest=$text
  fi
  if [[ $rest == \"* ]]; then
    name=${rest#\"}; name=${name%%\"*}; rest=${rest#*\"}; rest=$(trim "$rest")
  else
    name=${rest%%[[:space:]]*}
    if [[ $rest == "$name"* ]]; then rest=$(trim "${rest#"$name"}"); fi
  fi
  if [[ $rest =~ flags=\(([^\)]*)\) ]]; then
    flags=${BASH_REMATCH[1]}
    rest=${rest/ flags=\(${BASH_REMATCH[1]}\)/}
    rest=${rest/flags=\(${BASH_REMATCH[1]}\)/}
  elif [[ $rest =~ (^|[[:space:]])flags=([^[:space:]]+) ]]; then
    flags=${BASH_REMATCH[2]}
    rest=${rest/${BASH_REMATCH[1]}flags=${BASH_REMATCH[2]}/}
  fi
  rest=$(trim "$rest")
  if [[ $rest == /* ]]; then
    attach=${rest%%[[:space:]]*}
    attach=${attach#\"}; attach=${attach%\"}
  elif [[ $name == /* ]]; then
    attach=$name
  fi
  name=$(normalize_label_name "$name" "$parent")
  flags=$(printf '%s' "$flags" | tr ',' ' ' | xargs 2>/dev/null | tr ' ' ',')
  add_label "$name" "$parent" "$flags" "$attach" "$origin" "$type"
  PARSED_LABEL="$name"
}

is_perm_token() { [[ $1 =~ ^[rwakmlxipucPCUIL]+$ ]]; }

parse_file_rule() {
  local action=$1 line=$2 file=$3 lineno=$4 raw=$5
  local left=$line right="" part tokens t perm="" path="" exec="" base="" target stack=0
  local -a targets=()
  if [[ $line == *'->'* ]]; then
    left=${line%% -> *}; right=${line#* -> }
  elif [[ $line == *'->'* ]]; then
    left=${line%%->*}; right=${line#*->}
  fi
  left=$(trim "${left%,}"); right=$(trim "${right%,}")
  if [[ -n $right ]]; then
    right=${right#\{}; right=${right%\}}
    IFS=',' read -ra targets <<< "$right"
    if ((${#targets[@]} == 1)); then
      IFS=' ' read -ra targets <<< "${targets[0]}"
    fi
  fi
  ((${#targets[@]})) || targets=("")
  read -r -a tokens <<< "$left"
  local -a path_candidates=()
  for t in "${tokens[@]}"; do
    t=${t#\"}; t=${t%\"}
    if is_perm_token "$t" && [[ $t == *[rwakmlx]* ]]; then
      perm=$t
    elif [[ $t != *=* && $t != file && $t != owner && $t != audit && $t != deny && $t != allow ]]; then
      path_candidates+=("$t")
    fi
  done
  if [[ -z $perm ]]; then return 1; fi
  path=${path_candidates[0]:-}
  [[ -n $path ]] || return 1
  path=${path#\"}; path=${path%\"}
  if [[ $perm == *[xX]* ]]; then
    local i=${perm%%[xX]*}
    local after=${perm#*[xX]}
    local l=${#i} r=${#after}
    while (( l > 0 )); do
      local c=${i:l-1:1}
      [[ $c == [iIpPcCuU] ]] || break
      ((l--))
    done
    while (( r > 0 )); do
      local c=${after:0:1}
      [[ $c == [iIpPcCuU] ]] || break
      ((r--)); after=${after:1}
    done
    exec="${i:l}x${after}"
    base="${i:0:l}${after}"
  else
    base=$perm
  fi
  base=$(canon_modes "$base")
  [[ -n $exec ]] && base=$(canon_modes "${base}x")
  R_LABEL+=("$CURRENT_LABEL"); R_KIND+=("file"); R_ACTION+=("$action")
  R_PATH+=("$path"); R_MODES+=("$base"); R_TARGET+=("$right"); R_FILE+=("$file"); R_LINE+=("$lineno"); R_RAW+=("$raw")
  if [[ $action == allow ]]; then
    local key="$CURRENT_LABEL$US$path"
    ALLOW[$key]="${ALLOW[$key]:-}$base"
    [[ -n $exec ]] || return 0
    local tmpl tstack
    for tmpl in "${targets[@]}"; do
      tmpl=$(trim "$tmpl")
      tstack=0
      if [[ $tmpl == \&* ]]; then tstack=1; tmpl=${tmpl#\&}; fi
      tmpl=$(normalize_label_name "$tmpl" "")
      if [[ -z $tmpl ]]; then
        case "$exec" in
          *[pP]*|*[cC]*) tmpl="__attachment__:$path" ;;
        esac
      fi
      local ch
      for ((i=0; i<${#exec}; i++)); do
        ch=${exec:i:1}
        case "$ch" in
          u|U)
            E_FROM+=("$CURRENT_LABEL"); E_TO+=("unconfined"); E_KIND+=("exec-$exec")
            E_STACKED+=(0); E_FLAGS+=(""); E_FILE+=("$file"); E_LINE+=("$lineno"); E_RAW+=("$raw")
            ;;
          p|P)
            E_FROM+=("$CURRENT_LABEL"); E_TO+=("${tmpl:-__unresolved_px__}"); E_KIND+=("exec-$exec")
            E_STACKED+=("$tstack"); E_FLAGS+=(""); E_FILE+=("$file"); E_LINE+=("$lineno"); E_RAW+=("$raw")
            ;;
          c|C)
            E_FROM+=("$CURRENT_LABEL"); E_TO+=("${tmpl:-__unresolved_cx__}"); E_KIND+=("exec-$exec")
            E_STACKED+=("$tstack"); E_FLAGS+=(""); E_FILE+=("$file"); E_LINE+=("$lineno"); E_RAW+=("$raw")
            ;;
        esac
      done
    done
  fi
  return 0
}

add_transition() {

  E_FROM+=("$from"); E_TO+=("$to"); E_KIND+=("$kind"); E_STACKED+=("$stacked")

parse_change_rule() {
  local keyword=$1 line=$2 file=$3 lineno=$4 raw=$5
  local rest left="" right="" has_arrow=0 flags="" unsafe=0 subset=0 t
  local -a targets=() words=() cleaned=()
  rest=${line#*"$keyword"}; rest=$(trim "$rest"); rest=${rest%,}
  if [[ $rest == *'->'* ]]; then
    left=${rest%%->*}; right=${rest#*->}; has_arrow=1
  else
    left=$rest
  fi
  left=$(trim "$left"); right=$(trim "${right%,}"); right=${right#\{}; right=${right%\}}
  read -r -a words <<< "$left"
  for t in "${words[@]}"; do
    case "$t" in
      unsafe) unsafe=1 ;;
      subset) subset=1 ;;
      safe) ;;
      *) cleaned+=("$t") ;;
    esac
  done
  left=$(trim "${cleaned[*]:-}")
  [[ $unsafe == 1 ]] && flags="${flags:+$flags,}unsafe"
  [[ $subset == 1 ]] && flags="${flags:+$flags,}subset"
  R_LABEL+=("$CURRENT_LABEL"); R_KIND+=("$keyword"); R_ACTION+=("allow")
  R_PATH+=("$left"); R_MODES+=(""); R_TARGET+=("$right"); R_FILE+=("$file"); R_LINE+=("$lineno"); R_RAW+=("$raw")
  if ((has_arrow)); then
    IFS=',' read -ra targets <<< "$right"
    if ((${#targets[@]} == 1)); then IFS=' ' read -ra targets <<< "${targets[0]}"; fi
    for t in "${targets[@]}"; do
      t=$(trim "$t"); [[ -n $t ]] || continue
      add_transition "$CURRENT_LABEL" "$t" "$keyword" "$flags" "$file" "$lineno" "$raw"
    done
  else
    if [[ $keyword == change_hat ]]; then
      HAT_RULE[$CURRENT_LABEL]="$file:$lineno:$raw"
    elif [[ $subset == 1 ]]; then
      add_transition "$CURRENT_LABEL" "(subset)" "$keyword" "$flags" "$file" "$lineno" "$raw"
    elif [[ $unsafe == 1 ]]; then
      add_transition "$CURRENT_LABEL" "(wildcard)" "$keyword" "$flags" "$file" "$lineno" "$raw"
    fi
  fi
}

add_simple_permission() {
  local kind=$1 action=$2 rest=$3 file=$4 lineno=$5 raw=$6 items item
  rest=$(trim "$rest"); rest=${rest%,}
  R_LABEL+=("$CURRENT_LABEL"); R_KIND+=("$kind"); R_ACTION+=("$action")
  R_PATH+=(""); R_MODES+=("$rest"); R_TARGET+=(""); R_FILE+=("$file"); R_LINE+=("$lineno"); R_RAW+=("$raw")
  [[ $action == allow ]] || return 0
  case "$kind" in
    capability)
      [[ -z $rest ]] && rest='*'
      rest=${rest//,/ }
      for item in $rest; do CAPS["$CURRENT_LABEL$US$item"]=1; done
      ;;
    network)
      [[ -z $rest ]] && rest='*'
      rest=$(printf '%s' "$rest" | tr -s '[:space:]' ' ')
      NETS["$CURRENT_LABEL$US$rest"]=1
      ;;
    mount|umount|remount|pivot_root)
      [[ -z $rest ]] && rest='*'
      MOUNTS["$CURRENT_LABEL$US$kind $rest"]=1
      ;;
  esac
}

parse_rule_line() {
  local raw=$1 file=$2 lineno=$3 line action=allow keyword="" rest t
  line=$(trim "$raw"); line=${line%,}
  while [[ $line =~ ^(audit|quiet|allow|deny|owner|prompt)[[:space:]]+(.*)$ ]]; do
    t=${BASH_REMATCH[1]}; line=${BASH_REMATCH[2]}
    [[ $t == deny ]] && action=deny
    [[ $t == allow ]] && action=allow
  done
  if [[ $line =~ ^priority=[^[:space:]]+[[:space:]]+(.*)$ ]]; then line=${BASH_REMATCH[1]}; fi
  while [[ $line =~ ^(audit|quiet|allow|deny|owner|prompt)[[:space:]]+(.*)$ ]]; do
    t=${BASH_REMATCH[1]}; line=${BASH_REMATCH[2]}
    [[ $t == deny ]] && action=deny
    [[ $t == allow ]] && action=allow
  done
  line=$(trim "$line")
  [[ -z $line ]] && return 0
  keyword=${line%%[[:space:]]*}
  rest=$(trim "${line#"$keyword"}")
  case "$keyword" in
    file)   parse_file_rule "$action" "$rest" "$file" "$lineno" "$raw" || true ;;
    capability|network|mount|umount|remount|pivot_root)
            add_simple_permission "$keyword" "$action" "$rest" "$file" "$lineno" "$raw" ;;
    change_profile|delegate|change_hat)
            parse_change_rule "$keyword" "$line" "$file" "$lineno" "$raw" ;;
    all)
            R_LABEL+=("$CURRENT_LABEL"); R_KIND+=("all"); R_ACTION+=("$action"); R_PATH+=(""); R_MODES+=("")
            R_TARGET+=(""); R_FILE+=("$file"); R_LINE+=("$lineno"); R_RAW+=("$raw") ;;
    link)
            R_LABEL+=("$CURRENT_LABEL"); R_KIND+=("link"); R_ACTION+=("$action"); R_PATH+=(""); R_MODES+=("$rest")
            R_TARGET+=(""); R_FILE+=("$file"); R_LINE+=("$lineno"); R_RAW+=("$raw") ;;
    signal|ptrace|unix|dbus|mqueue|rlimit|userns|io_uring|userns_create|prompt)
            R_LABEL+=("$CURRENT_LABEL"); R_KIND+=("$keyword"); R_ACTION+=("$action"); R_PATH+=(""); R_MODES+=("$rest")
            R_TARGET+=(""); R_FILE+=("$file"); R_LINE+=("$lineno"); R_RAW+=("$raw") ;;
    *)
      if is_perm_token "$keyword" || [[ $keyword == /* || $keyword == @\{* || $keyword == \"* ]]; then
        parse_file_rule "$action" "$line" "$file" "$lineno" "$raw" || true
      fi
      ;;
  esac
}

scan_live() {
  local pf="$SEC_DIR/profiles" d name f
  if [[ -f $pf ]]; then
    while IFS= read -r line || [[ -n $line ]]; do
      line=$(trim "$line"); [[ -n $line ]] || continue
      name=${line%% (*}; name=$(trim "$name")
      [[ -n $name ]] && LIVE_NAMES+=("$name")
    done < "$pf"
  fi
  [[ -d "$SEC_DIR/policy/profiles" ]] || return 0
  for d in "$SEC_DIR"/policy/profiles/*/; do
    [[ -d $d ]] || continue
    d=${d%/}
    name=""
    [[ -f $d/name ]] && name=$(trim "$(<"$d/name")")
    [[ -z $name ]] && name=$(basename "$d")
    [[ -n $name ]] && LIVE_NAMES+=("$name")
    for f in "$d/raw_data" "$d/.load" "$d/load" "$d/profile"; do
      [[ -f $f ]] || continue
      if LC_ALL=C grep -Iq . "$f" 2>/dev/null; then
        LIVE_TEXT_FILES+=("$f")
      fi
    done
  done
}

parse_stream() {
  local file lineno text line name parent origin
  while IFS="$US" read -r file lineno text; do
    [[ -n $text ]] || continue
    line=$(trim "$text")
    [[ -z $line ]] && continue
    if [[ $line =~ ^\}[[:space:]]*([,;])?[[:space:]]*$ ]]; then
      if ((${#STACK[@]})); then unset 'STACK[${#STACK[@]}-1]'; fi
      if ((${#STACK[@]})); then CURRENT_LABEL=${STACK[${#STACK[@]}-1]}; else CURRENT_LABEL=""; fi
      continue
    fi
    if [[ $line == *'{' && ( $line == profile[[:space:]]* || $line == hat[[:space:]]* || $line == \^* || $line == /* || $line == \"* ) ]]; then
      if [[ $line == *',' ]]; then
        :
      else
        parent=$CURRENT_LABEL
        origin="$file:$lineno"
        PARSED_LABEL=""
        parse_profile_header "$line" "$parent" "$origin"
        name=$PARSED_LABEL
        if [[ -n $name ]]; then
          STACK+=("$name")
          CURRENT_LABEL=$name
          continue
        fi
      fi
    fi
    [[ -n $CURRENT_LABEL ]] || continue
    parse_rule_line "$line" "$file" "$lineno"
  done < "$1"
}

finalize_permissions() {
  local i label key path modes
  for i in "${!R_LABEL[@]}"; do
    label=${R_LABEL[$i]}; kind=${R_KIND[$i]}; action=${R_ACTION[$i]}
    if [[ $kind == file && $action == deny ]]; then
      key="$label$US${R_PATH[$i]}"
      DENY[$key]="${DENY[$key]:-}${R_MODES[$i]}"
    fi
  done
  for key in "${!ALLOW[@]}"; do
    label=${key%%"$US"*}; path=${key#*"$US"}
    modes=${ALLOW[$key]}
    if [[ -n ${DENY[$key]:-} ]]; then modes=$(subtract_modes "$modes" "${DENY[$key]}"); fi
    modes=$(canon_modes "$modes")
    [[ -n $modes ]] && FILE_PERMS[$key]=$modes
  done
  for label in "${LABEL_ORDER[@]}"; do
    local fl=${FLAGS[$label]:-}
    [[ $fl == *attach_disconnected* ]] && META["$label${US}flag:attach_disconnected"]=1
    [[ $fl == *delegate_deleted* ]] && META["$label${US}flag:delegate_deleted"]=1
    [[ $fl == *unconfined* ]] && UNCONFINED[$label]=1
  done
}

path_matches() {
  local target=$1 pattern=$2 p
  [[ $target == "$pattern" ]] && return 0
  p=${pattern//\*\*/\*}
  if [[ $p == *\{*\}* ]]; then
    local pre rest post alt
    pre=${p%%\{*}; rest=${p#*\{}; post=${rest#*\}}; alt=${rest%%\}*}
    p="$pre@(${alt//,/|})$post"
  fi
  shopt -s extglob
  [[ $target == $p ]]
}

source_file_modes() {
  local label=$1 target=$2 k sp sm out=""
  local pre="$label$US"
  for k in "${!FILE_PERMS[@]}"; do
    [[ $k == "$pre"* ]] || continue
    sp=${k#"$pre"}
    if path_matches "$target" "$sp" || [[ $target == "$sp" ]]; then
      sm=${FILE_PERMS[$k]}
      out+=$sm
    fi
  done
  printf '%s' "$out"
}

net_covers() {
  local src=$1 tgt=$2 sf tt st
  [[ $src == '*' ]] && return 0
  [[ $src == "$tgt" ]] && return 0
  sf=${src%% *}; tt=${tgt%% *}
  st=""
  [[ $src == *" "* ]] && st=${src#* }
  [[ $tt == "$sf" ]] || return 1
  [[ -z $st || $st == '*' || $st == "${tgt#* }" ]] && return 0
  return 1
}

mount_covers() {
  local src=$1 tgt=$2
  [[ $src == "$tgt" ]] && return 0
  [[ $src == 'mount *' || $src == 'mount' ]] && [[ $tgt == mount* ]] && return 0
  return 1
}

compute_added() {
  local from=$1 to=$2 added=() k pre item tpath tmodes srcmodes newmodes
  if [[ $to == unconfined || -n ${UNCONFINED[$to]:-} ]]; then
    printf 'ALL'
    return
  fi
  if [[ -z ${LABEL_SEEN[$to]:-} ]]; then
    printf 'UNRESOLVED'
    return
  fi
  pre="$to$US"
  for k in "${!FILE_PERMS[@]}"; do
    [[ $k == "$pre"* ]] || continue
    tpath=${k#"$pre"}; tmodes=${FILE_PERMS[$k]}
    srcmodes=$(source_file_modes "$from" "$tpath")
    newmodes=$(subtract_modes "$tmodes" "$srcmodes")
    [[ -n $newmodes ]] && added+=("file:$tpath:$newmodes")
  done
  pre="$to$US"
  for k in "${!CAPS[@]}"; do
    [[ $k == "$pre"* ]] || continue
    item=${k#"$pre"}
    if [[ $item == '*' ]]; then
      [[ -n ${CAPS["$from${US}*"]:-} ]] || added+=("capability:*")
    else
      [[ -n ${CAPS["$from${US}$item"]:-} || -n ${CAPS["$from${US}*"]:-} ]] || added+=("capability:$item")
    fi
  done
  pre="$to$US"
  for k in "${!NETS[@]}"; do
    [[ $k == "$pre"* ]] || continue
    item=${k#"$pre"}; local covered=0 sk
    for sk in "${!NETS[@]}"; do
      [[ $sk == "$from$US"* ]] || continue
      if net_covers "${sk#"$from$US"}" "$item"; then covered=1; break; fi
    done
    ((covered)) || added+=("network:$item")
  done
  pre="$to$US"
  for k in "${!MOUNTS[@]}"; do
    [[ $k == "$pre"* ]] || continue
    item=${k#"$pre"}; local covered=0 sk
    for sk in "${!MOUNTS[@]}"; do
      [[ $sk == "$from$US"* ]] || continue
      if mount_covers "${sk#"$from$US"}" "$item"; then covered=1; break; fi
    done
    ((covered)) || added+=("mount:$item")
  done
  pre="$to$US"
  for k in "${!META[@]}"; do
    [[ $k == "$pre"* ]] || continue
    item=${k#"$pre"}
    [[ -n ${META["$from${US}$item"]:-} ]] || added+=("$item")
  done
  local IFS=,
  printf '%s' "${added[*]}"
}

evaluate_edges() {
  local i from to kind flags added reason sev escaped stacked t ef el er p h
  local n=${#E_FROM[@]}
  for ((i=0; i<n; i++)); do
    to=${E_TO[$i]}
    [[ $to == *[\*\?\[\{]* ]] || continue
    [[ $to == __* ]] && continue
    for t in "${LABEL_ORDER[@]}"; do
      if path_matches "$t" "$to"; then
        E_FROM+=("${E_FROM[$i]}"); E_TO+=("$t"); E_KIND+=("${E_KIND[$i]}")
        E_STACKED+=("${E_STACKED[$i]}"); E_FLAGS+=("${E_FLAGS[$i]}")
        E_FILE+=("${E_FILE[$i]}"); E_LINE+=("${E_LINE[$i]}"); E_RAW+=("${E_RAW[$i]}")
      fi
    done
  done
  for p in "${!HAT_RULE[@]}"; do
    for h in "${LABEL_ORDER[@]}"; do
      [[ ${PARENT[$h]:-} == "$p" && ${LABEL_TYPE[$h]:-} == hat ]] || continue
      IFS=':' read -r ef el er <<< "${HAT_RULE[$p]}"
      add_transition "$p" "$h" "change_hat" "" "$ef" "$el" "$er"
    done
  done
  for i in "${!E_FROM[@]}"; do
    from=${E_FROM[$i]}; to=${E_TO[$i]}; kind=${E_KIND[$i]}; flags=${E_FLAGS[$i]}; stacked=${E_STACKED[$i]}
    added=""; reason=""; sev="info"; escaped=0
    if [[ $stacked == 1 ]]; then
      reason="stacked profile transition intersects confinement; it cannot add permissions"
    elif [[ $to == unconfined ]]; then
      escaped=1; added="ALL"; sev="critical"; reason="unconfined exec target"
    elif [[ -n ${UNCONFINED[$to]:-} ]]; then
      escaped=1; added="ALL"; sev="critical"; reason="target profile is marked unconfined"
    elif [[ $to == __unresolved* ]]; then
      if [[ $flags == *unsafe* || $kind == delegate ]]; then
        escaped=1; added="UNRESOLVED"; sev="high"; reason="unsafe/delegate transition to unresolved target"
      else
        reason="target not resolved from policy; not proven less restrictive"
      fi
    elif [[ -z ${LABEL_SEEN[$to]:-} ]]; then
      if [[ $to == '(subset)' ]]; then
        reason="explicit subset transition (safe by definition)"
      elif [[ $to == '(wildcard)' ]]; then
        escaped=1; added="ALL"; sev="critical"; reason="unsafe/wildcard change_profile with no bounded target"
      elif [[ $flags == *unsafe* || $kind == delegate ]]; then
        escaped=1; added="UNRESOLVED"; sev="high"; reason="unsafe/delegate transition to unknown target"
      else
        reason="target not present in reconstructed policy"
      fi
    else
      added=$(compute_added "$from" "$to")
      if [[ -n $added ]]; then
        escaped=1; sev="high"; reason="target grants permissions not present in source label"
        [[ $flags == *unsafe* ]] && reason+="; change_profile uses unsafe exec mode"
        [[ $kind == delegate ]] && reason+="; delegate transition"
      elif [[ $flags == *unsafe* ]]; then
        escaped=1; sev="medium"; reason="change_profile unsafe flag disables secure-exec environment (permissions are subset)"
      elif [[ $flags == *subset* ]]; then
        reason="target is permission-subset (explicit subset transition)"
      else
        reason="target is permission-subset of source"
      fi
    fi
    E_ESCAPE[$i]=$escaped; E_ADDED[$i]=$added; E_REASON[$i]=$reason; E_SEVERITY[$i]=$sev
  done
}
