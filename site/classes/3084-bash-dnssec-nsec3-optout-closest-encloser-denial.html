DNSSEC NSEC3 Opt-Out Denial-of-Existence Validator (bash + dig + openssl)

Root-cause analysis

NSEC3 Opt-Out denial is a minefield because the resolver's AD bit is not a validation result you can delegate to, and Opt-Out deliberately removes insecure delegations from the NSEC3 chain. The specific failure modes a validator must defend against:

  1. Trusting AD. A forged/partial response can carry AD=1. The validator must recompute every NSEC3 RRSIG and the coverage relation itself.
  2. Confusing match with cover. Coverage is a circular interval check over the canonical base32hex form of the hash (owner < H < next, wrapping when next <= owner). An exact match is not coverage, and owner==next is invalid.
  3. Treating Opt-Out NODATA as NXDOMAIN. For a DS query at an insecure delegation in an Opt-Out zone there is no matching NSEC3; the proof is: NSEC3 matching the closest encloser (the parent), plus an NSEC3 that covers the next-closer with the Opt-Out flag set. That is a NOERROR NODATA, not an NXDOMAIN. A true NXDOMAIN additionally requires a third NSEC3 covering *.closest-encloser.
  4. Broken RRSIG input. The signed data is RRSIG_RDATA(no signature) || canonical-RRset, where the NSEC3 RDATA must be reconstructed in wire form — salt, 1-byte-length-prefixed next hash, and the window/bitmap type encoding — using the original TTL from the RRSIG, owner names lower-cased, RRs sorted by canonical octet order.
  5. Not checking signed parameters. Iteration count, salt, and hash algorithm are inside the signed NSEC3 RDATA; all NSEC3 records in a proof must agree (and match any expectation).
  6. Opaque failure. Every rejection path needs a distinct, machine-readable reason and nonzero exit.

The script below implements all of the above with only bash builtins, dig, and openssl. It supports ECDSA P-256/P-384 (alg 13/14, the common case for TLDs such as .com), Ed25519 (15), and RSA (8/10), with a DS trust anchor for the zone apex.

Exact fix

Save as nsec3_optout_validate.sh and chmod +x.

#!/usr/bin/env bash
#=====================================================================
# nsec3_optout_validate.sh
#
# DNSSEC denial-of-existence validator for NSEC3 Opt-Out zones.
# Uses only bash builtins, dig and openssl.
#
# Validates:
#   * RRSIG over every NSEC3 record (cryptographic verification)
#   * NSEC3 hash algorithm / iteration count / salt consistency
#   * canonical (base32hex) owner-name ordering and coverage
#   * NXDOMAIN closest-encloser proof (match CE, cover next-closer,
#     cover wildcard)
#   * NODATA for an insecure delegation via Opt-Out coverage of the
#     next-closer name (distinct from true NXDOMAIN)
#
# It never trusts the resolver AD bit.  A response whose NSEC3 records
# do not actually cover the queried hash is rejected even if AD is set.
#
# Exit / reason codes:
#   0  ACCEPT_NXDOMAIN | ACCEPT_NXDOMAIN_OPTOUT | ACCEPT_NODATA |
#      ACCEPT_NODATA_OPTOUT_DELEGATION
#   1  REJECT_USAGE
#   2  REJECT_QUERY
#   3  REJECT_ANCHOR
#   4  REJECT_RRSIG_INVALID
#   5  REJECT_PARAM_MISMATCH
#   6  REJECT_NSEC3_NOT_COVERING
#   7  REJECT_CLOSEST_ENCLOSER
#   8  REJECT_NODATA_PROOF
#   9  REJECT_NSEC3_ORDERING
#  10  REJECT_UNSUPPORTED_ALG
#  11  REJECT_NO_NSEC3
#  12  REJECT_RCODE
#=====================================================================
set -uo pipefail

#---------------------------------------------------------------------
# small pure-bash helpers
#---------------------------------------------------------------------
B32H='0123456789ABCDEFGHIJKLMNOPQRSTUV'
B64CHARS='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'

_b64val() { local c=$1 i; for ((i=0;i<64;i++)); do [[ ${B64CHARS:i:1} == "$c" ]] && { REPLY=$i; return 0; }; done; REPLY=0; }
b64_to_hex() {
  local s=$1 out="" i c v acc=0 nbits=0
  s=${s//[[:space:]]/}
  for ((i=0;i<${#s};i++)); do
    c=${s:i:1}; [[ $c == '=' ]] && break
    _b64val "$c"; v=$REPLY
    acc=$(( (acc<<6)|v )); nbits=$((nbits+6))
    while (( nbits>=8 )); do nbits=$((nbits-8)); printf -v out '%s%02x' "$out" $(( (acc>>nbits)&0xff )); done
  done
  printf '%s' "$out"
}
_b32val() { local c=${1^^} i; for ((i=0;i<32;i++)); do [[ ${B32H:i:1} == "$c" ]] && { REPLY=$i; return 0; }; done; REPLY=0; }
b32_to_hex() {
  local s=${1^^} out="" i c v acc=0 nbits=0
  s=${s//=/}
  for ((i=0;i<${#s};i++)); do
    c=${s:i:1}; _b32val "$c"; v=$REPLY
    acc=$(( (acc<<5)|v )); nbits=$((nbits+5))
    while (( nbits>=8 )); do nbits=$((nbits-8)); printf -v out '%s%02x' "$out" $(( (acc>>nbits)&0xff )); done
  done
  printf '%s' "$out"
}
hex_to_b32() {
  local h=$1 i out="" acc=0 nbits=0 v
  for ((i=0;i<${#h};i+=2)); do
    v=$((16#${h:i:2})); acc=$(( (acc<<8)|v )); nbits=$((nbits+8))
    while (( nbits>=5 )); do nbits=$((nbits-5)); printf -v out '%s%s' "$out" "${B32H:$(( (acc>>nbits)&31 )):1}"; done
  done
  (( nbits>0 )) && printf -v out '%s%s' "$out" "${B32H:$(( (acc<<(5-nbits))&31 )):1}"
  printf '%s' "$out"
}
hex2bin() { local h=$1 i; for ((i=0;i<${#h};i+=2)); do printf "\\x${h:i:2}"; done; }
sha1_hex()   { local o; o=$(openssl dgst -sha1   -r 2>/dev/null); o=${o%% *}; printf '%s' "$o"; }
sha256_hex() { local o; o=$(openssl dgst -sha256 -r 2>/dev/null); o=${o%% *}; printf '%s' "$o"; }
sha384_hex() { local o; o=$(openssl dgst -sha384 -r 2>/dev/null); o=${o%% *}; printf '%s' "$o"; }

# canonical lower-case wire-format name, returned as hex
name_to_hex() {
  local n=${1,,} out="" lab j
  n=${n%.}
  if [[ -n $n ]]; then
    IFS='.' read -ra _labs <<< "$n"
    for lab in "${_labs[@]}"; do
      printf -v out '%s%02x' "$out" ${#lab}
      for ((j=0;j<${#lab};j++)); do printf -v out '%s%02x' "$out" "'${lab:j:1}"; done
    done
  fi
  printf '%s00' "$out"
}

#---------------------------------------------------------------------
# DER helpers (DNSKEY -> SubjectPublicKeyInfo)
#---------------------------------------------------------------------
der_len() { local n=$1; if ((n<128)); then printf '%02x' "$n"; elif ((n<256)); then printf '81%02x' "$n"; elif ((n<65536)); then printf '82%04x' "$n"; else printf '83%06x' "$n"; fi; }
der_tlv() { local tag=$1 v=$2; printf '%s%s%s' "$tag" "$(der_len $(( ${#v}/2 )))" "$v"; }
der_int() {
  local v=${1,,}
  while [[ ${#v} -gt 2 && ${v:0:2} == "00" ]]; do v=${v:2}; done
  [[ -z $v ]] && v=00
  (( 16#${v:0:2} >= 128 )) && v="00$v"
  der_tlv 02 "$v"
}
dnskey_spki() { # <alg> <pubkey_hex>
  local alg=$1 pubhex=$2
  case $alg in
    13) printf '3059301306072a8648ce3d020106082a8648ce3d03010703420004%s' "$pubhex" ;;
    14) printf '3076301006072a8648ce3d020106052b8104002203620004%s' "$pubhex" ;;
    15) printf '302a300506032b6570032100%s' "$pubhex" ;;
    8|10)
      local pos=1 elen=$((16#${pubhex:0:2}))
      (( elen==0 )) && { elen=$((16#${pubhex:2:4})); pos=2; }
      local e=${pubhex:$((pos*2)):$((elen*2))}
      local n=${pubhex:$(((pos+elen)*2))}
      local rsa; rsa=$(der_tlv 30 "$(der_int "$n")$(der_int "$e")")
      local algid='300d06092a864886f70d0101010500'
      local bitstr; bitstr=$(der_tlv 03 "00$rsa")
      printf '30%s%s%s' "$(der_len $(( (${#algid}+${#bitstr})/2 )))" "$algid" "$bitstr"
      ;;
    *) return 1 ;;
  esac
}
ecdsa_raw_to_der() {
  local raw=$1 half=$2 r s
  r=${raw:0:$((half*2))}; s=${raw:$((half*2))}
  printf '%s' "$(der_tlv 30 "$(der_int "$r")$(der_int "$s")")"
}

#---------------------------------------------------------------------
# time, rrtype helpers
#---------------------------------------------------------------------
ymdhms_to_epoch() {
  local s=$1
  local Y=$((10#${s:0:4})) M=$((10#${s:4:2})) D=$((10#${s:6:2}))
  local h=$((10#${s:8:2})) m=$((10#${s:10:2})) sec=$((10#${s:12:2}))
  local y=$Y
  (( M<=2 )) && y=$((y-1))
  local era=$(( (y>=0 ? y : y-399) / 400 ))
  local yoe=$(( y - era*400 ))
  local mp=$(( M + (M>2 ? -3 : 9) ))
  local doy=$(( (153*mp+2)/5 + D - 1 ))
  local doe=$(( yoe*365 + yoe/4 - yoe/100 + doy ))
  local days=$(( era*146097 + doe - 719468 ))
  printf '%s' $(( days*86400 + h*3600 + m*60 + sec ))
}
type_num() {
  case ${1^^} in
    A) echo 1;; NS) echo 2;; CNAME) echo 5;; SOA) echo 6;; PTR) echo 12;;
    HINFO) echo 13;; MX) echo 15;; TXT) echo 16;; RP) echo 17;; AFSDB) echo 18;;
    SIG) echo 24;; KEY) echo 25;; AAAA) echo 28;; LOC) echo 29;; SRV) echo 33;;
    NAPTR) echo 35;; KX) echo 36;; CERT) echo 37;; DNAME) echo 39;; DS) echo 43;;
    SSHFP) echo 44;; IPSECKEY) echo 45;; RRSIG) echo 46;; NSEC) echo 47;;
    DNSKEY) echo 48;; DHCID) echo 49;; NSEC3) echo 50;; NSEC3PARAM) echo 51;;
    TLSA) echo 52;; SMIMEA) echo 53;; HIP) echo 55;; CDS) echo 59;; CDNSKEY) echo 60;;
    OPENPGPKEY) echo 61;; CSYNC) echo 62;; ZONEMD) echo 63;; SVCB) echo 64;;
    HTTPS) echo 65;; SPF) echo 99;; EUI48) echo 108;; EUI64) echo 109;;
    TKEY) echo 249;; TSIG) echo 250;; IXFR) echo 251;; AXFR) echo 252;;
    MAILB) echo 253;; MAILA) echo 254;; ANY) echo 255;; URI) echo 256;; CAA) echo 257;;
    *) echo 0;;
  esac
}
type_bitmap_hex() {
  local types="$*" out="" w off tn t b maxb
  local -A bits win
  for t in $types; do
    tn=$(type_num "$t"); (( tn==0 )) && continue
    w=$((tn/256)); off=$((tn%256)); win[$w]=1
    printf -v bits["$w:$((off/8))"] '%d' $(( ${bits["$w:$((off/8))"]:-0} | (1 << (7-(off%8))) ))
  done
  for ((w=0;w<=255;w++)); do
    [[ -z ${win[$w]:-} ]] && continue
    maxb=-1
    for ((b=0;b<32;b++)); do [[ -n ${bits["$w:$b"]:-} && ${bits["$w:$b"]} -ne 0 ]] && maxb=$b; done
    (( maxb<0 )) && continue
    printf -v out '%s%02x%02x' "$out" "$w" $((maxb+1))
    for ((b=0;b<=maxb;b++)); do printf -v out '%s%02x' "$out" "${bits["$w:$b"]:-0}"; done
  done
  printf '%s' "$out"
}
# NSEC3 hash of a name: base32hex(SHA1^iter(name||salt))
nsec3_hash() {
  local name=$1 salt=$2 it=$3 h i
  h=$(hex2bin "$(name_to_hex "$name")$salt" | sha1_hex)
  for ((i=0;i<it;i++)); do h=$(hex2bin "${h}${salt}" | sha1_hex); done
  hex_to_b32 "$h"
}
# canonical NSEC3 RDATA
nsec3_rdata_hex() {
  local alg=$1 flags=$2 iters=$3 salt=$4 next=$5; shift 5
  local out nh
  printf -v out '%02x%02x%04x' "$alg" "$flags" "$iters"
  if [[ $salt == '-' || -z $salt ]]; then out+='00'; else printf -v out '%s%02x%s' "$out" $(( ${#salt}/2 )) "$salt"; fi
  nh=$(b32_to_hex "$next")
  printf -v out '%s%02x%s' "$out" $(( ${#nh}/2 )) "$nh"
  printf '%s%s' "$out" "$(type_bitmap_hex $*)"
}
# DNSKEY key tag, RFC 4034 App. B
dnskey_keytag() {
  local flags=$1 proto=$2 alg=$3 b64=$4
  local rdata_hex; printf -v rdata_hex '%04x%02x%02x' "$flags" "$proto" "$alg"
  rdata_hex+=$(b64_to_hex "$b64")
  local i b ac=0
  for ((i=0;i<${#rdata_hex};i+=2)); do
    b=$((16#${rdata_hex:i:2}))
    (( i%4==0 )) && ac=$((ac+(b<<8))) || ac=$((ac+b))
  done
  ac=$(( ac + ((ac>>16)&0xffff) ))
  printf '%d' $(( ac & 0xffff ))
}
# lexicographic octet comparison of two hex strings (RFC 4034 6.3)
# returns 1 if a<b, 2 if a>b, 0 if equal
hex_cmp() {
  local a=$1 b=$2
  local la=${#a} lb=${#b}
  local n i ba bb
  n=$(( la<lb ? la : lb ))
  for ((i=0;i<n;i+=2)); do
    ba=$((16#${a:i:2})); bb=$((16#${b:i:2}))
    if (( ba<bb )); then return 1; elif (( ba>bb )); then return 2; fi
  done
  if (( la<lb )); then return 1; elif (( la>lb )); then return 2; fi
  return 0
}
# insertion sort of canonical RR hex strings
canon_sort() {
  local -a a=("$@"); local i j n=${#a[@]} x
  for ((i=1;i<n;i++)); do
    x=${a[i]}; j=$((i-1))
    while (( j>=0 )); do
      hex_cmp "${a[j]}" "$x"; local ord=$?
      (( ord==2 )) || break
      a[j+1]=${a[j]}; j=$((j-1))
    done
    a[j+1]=$x
  done
  printf '%s\n' "${a[@]}"
}
# canonical RR hex: owner type class ttl rdlen rdata
canonical_rr() { local owner=$1 type=$2 rdata=$3 ttl=$4; printf '%s%04x%04x%08x%04x%s' "$(name_to_hex "$owner")" "$type" 1 "$ttl" "$(( ${#rdata}/2 ))" "$rdata"; }

#---------------------------------------------------------------------
# record reader (joins dig multi-line presentation records)
#---------------------------------------------------------------------
read_records() { # <file> -> REC_LINES[]
  local file=$1 line prev="" cont
  REC_LINES=()
  while IFS= read -r line; do
    [[ -z $line || $line == ';'* ]] && continue
    if [[ $line == [[:space:]]* || $line == '('* ]]; then
      cont=${line#"${line%%[![:space:]]*}"}; cont=${cont%)}; cont=${cont#(}
      prev="$prev $cont"
    else
      [[ -n $prev ]] && REC_LINES+=("$prev")
      prev=$line
    fi
  done < "$file"
  [[ -n $prev ]] && REC_LINES+=("$prev")
}

#---------------------------------------------------------------------
# RRSIG verification.  Globals: DNSKEY_LINES
#   verify_rrsig <sigline> <owner> <rrtype> <rdata_hex>...
#---------------------------------------------------------------------
verify_rrsig() {
  local sigline=$1 owner=$2 rrtype=$3; shift 3
  local rdata_list=("$@")

  local o ttl cls typ rest
  read -r o ttl cls typ rest <<< "$sigline"
  local str=$rest covered alg labels origttl exp inc keytag signer sig
  read -r covered str <<< "$str"
  read -r alg str <<< "$str"
  read -r labels str <<< "$str"
  read -r origttl str <<< "$str"
  read -r exp str <<< "$str"
  read -r inc str <<< "$str"
  read -r keytag str <<< "$str"
  read -r signer sig <<< "$str"
  sig=${sig//[[:space:]]/}

  local rdata_hex
  printf -v rdata_hex '%04x%02x%02x%08x%08x%08x%04x' \
      "$rrtype" "$alg" "$labels" "$origttl" \
      "$(ymdhms_to_epoch "$exp")" "$(ymdhms_to_epoch "$inc")" "$keytag"
  rdata_hex+=$(name_to_hex "$signer")

  local now; printf -v now '%(%s)T' -1
  local eh; eh=$(ymdhms_to_epoch "$exp")
  (( eh < now )) && { REASON=REJECT_RRSIG_INVALID; DETAIL="RRSIG expired"; return 4; }

  # canonical RRset
  local canon=() r
  for r in "${rdata_list[@]}"; do canon+=("$(canonical_rr "$owner" "$rrtype" "$r" "$origttl")"); done
  local sorted_hex="$rdata_hex"
  while IFS= read -r l; do sorted_hex+="$l"; done < <(canon_sort "${canon[@]}")

  # locate DNSKEY
  local kl kflags kproto kalg kb64 kt hit=0
  for kl in "${DNSKEY_LINES[@]}"; do
    read -r _ _ _ _ kflags kproto kalg kb64 <<< "$kl"
    kt=$(dnskey_keytag "$kflags" "$kproto" "$kalg" "$kb64")
    [[ $kt == "$keytag" && $kalg == "$alg" ]] && { hit=1; break; }
  done
  (( hit )) || { REASON=REJECT_RRSIG_INVALID; DETAIL="no trusted DNSKEY tag=$keytag"; return 4; }

  local pubhex spki pem sighex ok=0
  pubhex=$(b64_to_hex "$kb64")
  spki=$(dnskey_spki "$alg" "$pubhex") || { REASON=REJECT_UNSUPPORTED_ALG; return 10; }
  pem=$(hex2bin "$spki" | openssl pkey -pubin -inform DER 2>/dev/null) || { REASON=REJECT_UNSUPPORTED_ALG; return 10; }
  sighex=$(b64_to_hex "$sig")

  case $alg in
    8)  openssl dgst -sha256 -verify <(printf '%s\n' "$pem") -signature <(hex2bin "$sighex") <(hex2bin "$sorted_hex") >/dev/null 2>&1 && ok=1 ;;
    10) openssl dgst -sha512 -verify <(printf '%s\n' "$pem") -signature <(hex2bin "$sighex") <(hex2bin "$sorted_hex") >/dev/null 2>&1 && ok=1 ;;
    13) openssl dgst -sha256 -verify <(printf '%s\n' "$pem") -signature <(hex2bin "$(ecdsa_raw_to_der "$sighex" 32)") <(hex2bin "$sorted_hex") >/dev/null 2>&1 && ok=1 ;;
    14) openssl dgst -sha384 -verify <(printf '%s\n' "$pem") -signature <(hex2bin "$(ecdsa_raw_to_der "$sighex" 48)") <(hex2bin "$sorted_hex") >/dev/null 2>&1 && ok=1 ;;
    15) openssl pkeyutl -verify -pubin -inkey <(printf '%s\n' "$pem") -rawin -in <(hex2bin "$sorted_hex") -sigfile <(hex2bin "$sighex") >/dev/null 2>&1 && ok=1 ;;
    *)  REASON=REJECT_UNSUPPORTED_ALG; return 10 ;;
  esac
  (( ok )) && return 0
  REASON=REJECT_RRSIG_INVALID; DETAIL="crypto verify failed alg=$alg tag=$keytag"
  return 4
}

# verify DNSKEY RRset against a DS anchor (4 trailing fields)
verify_dnskey_against_ds() {
  local dsline=$1 owner=$2
  local keytag alg dt digest
  # Accept either a full presentation line (owner TTL class DS ...) or bare rdata
  dsline=${dsline//$'\t'/ }
  local rdata=$dsline
  if [[ " $dsline " == *" DS "* ]]; then rdata=${dsline#* DS }; fi
  set -- $rdata
  if (( $# >= 4 )); then keytag=$1; alg=$2; dt=$3; shift 3; digest="$*"; else return 3; fi
  digest=${digest//[[:space:]]/}
  local kl kflags kproto kalg kb64 kt got input rdata_hex
  for kl in "${DNSKEY_LINES[@]}"; do
    read -r _ _ _ _ kflags kproto kalg kb64 <<< "$kl"
    kt=$(dnskey_keytag "$kflags" "$kproto" "$kalg" "$kb64")
    [[ $kt == "$keytag" && $kalg == "$alg" ]] || continue
    printf -v rdata_hex '%04x%02x%02x' "$kflags" "$kproto" "$kalg"
    rdata_hex+=$(b64_to_hex "$kb64")
    input=$(name_to_hex "$owner")$rdata_hex
    case $dt in
      1) got=$(hex2bin "$input" | sha1_hex);;
      2) got=$(hex2bin "$input" | sha256_hex);;
      4) got=$(hex2bin "$input" | sha384_hex);;
      *) return 2;;
    esac
    [[ ${got^^} == "${digest^^}" ]] && return 0
  done
  return 3
}

# does NSEC3 (owner,next) cover hash h?
covers() { local o=${1^^} n=${2^^} h=${3^^}; [[ $h == "$o" ]] && return 1;
  if [[ $o < $n ]]; then [[ $o < $h && $h < $n ]]; else [[ $h > $o || $h < $n ]]; fi; }

#=====================================================================
# Core validation over already-loaded records.
# Globals in:  AUTH_LINES[], DNSKEY_LINES[], RCODE, AD
# Returns 0 on accept; on reject sets REASON and returns exit code.
#=====================================================================
N3_OWNER=(); N3_FLAGS=(); N3_ITER=(); N3_SALT=(); N3_NEXT=(); N3_TYPES=(); N3_SIG=(); N3_ALG=()

index_nsec3() {
  N3_OWNER=(); N3_FLAGS=(); N3_ITER=(); N3_SALT=(); N3_NEXT=(); N3_TYPES=(); N3_SIG=(); N3_ALG=()
  local line own ttl cls typ rest alg fl it sa nx types l so st sc st2 srest scov sig
  for line in "${AUTH_LINES[@]}"; do
    read -r own ttl cls typ rest <<< "$line"
    if [[ ${typ^^} == NSEC3 ]]; then
      read -r alg fl it sa nx types <<< "$rest"
      # find RRSIG NSEC3 with same owner
      sig=""
      for l in "${AUTH_LINES[@]}"; do
        read -r so st sc st2 srest <<< "$l"
        [[ ${so,,} == "${own,,}" && ${st2^^} == RRSIG ]] || continue
        read -r scov srest <<< "$srest"
        [[ ${scov^^} == NSEC3 ]] && { sig=$l; break; }
      done
      N3_OWNER+=("${own^^}"); N3_FLAGS+=("$fl"); N3_ITER+=("$it")
      N3_SALT+=("$sa"); N3_NEXT+=("$nx"); N3_TYPES+=("$types"); N3_SIG+=("$sig"); N3_ALG+=("$alg")
    fi
  done
}

validate() {
  local zone=$1 qname=$2 qtype=$3 exp_iter=${4:-} exp_salt=${5:-}
  local szone=${zone,,}; szone=${szone%.}
  local sq=${qname,,}; sq=${sq%.}
  local qtnum; qtnum=$(type_num "$qtype")

  index_nsec3
  ((${#N3_OWNER[@]})) || { REASON=REJECT_NO_NSEC3; DETAIL="no NSEC3 in authority"; return 11; }

  # ---- 1. verify RRSIG over every NSEC3 and parameters -------------
  local i owner lab nxhex rdata canon rc
  local ref_iter="" ref_salt=""
  for ((i=0;i<${#N3_OWNER[@]};i++)); do
    owner=${N3_OWNER[$i]}
    lab=${owner%%.*}
    nxhex=$(b32_to_hex "${N3_NEXT[$i]}")

    # canonical owner-name ordering / validity
    [[ ${#lab} -eq 32 && ${lab} =~ ^[0-9A-V]{32}$ ]] || { REASON=REJECT_NSEC3_ORDERING; DETAIL="bad owner hash '$lab'"; return 9; }
    [[ ${#nxhex} -eq 40 ]] || { REASON=REJECT_NSEC3_ORDERING; DETAIL="next hash not 20 octets"; return 9; }
    [[ ${N3_NEXT[$i]} != "$lab" ]] || { REASON=REJECT_NSEC3_ORDERING; DETAIL="owner==next"; return 9; }

    local alg=${N3_ALG[$i]}
    [[ $alg == 1 ]] || { REASON=REJECT_UNSUPPORTED_ALG; DETAIL="NSEC3 hash alg $alg != 1"; return 10; }

    # parameters
    if [[ -n $exp_iter && ${N3_ITER[$i]} != "$exp_iter" ]]; then
      REASON=REJECT_PARAM_MISMATCH; DETAIL="iterations ${N3_ITER[$i]} != expected $exp_iter"; return 5; fi
    local cursalt=${N3_SALT[$i]}; [[ $cursalt == '-' ]] && cursalt=""
    local esalt=""; [[ $exp_salt == '-' ]] && esalt=""; esalt=${exp_salt:-}
    if [[ -n $exp_salt && ${cursalt^^} != ${esalt^^} ]]; then
      REASON=REJECT_PARAM_MISMATCH; DETAIL="salt '$cursalt' != expected '$esalt'"; return 5; fi
    if [[ -z $ref_iter ]]; then ref_iter=${N3_ITER[$i]}; ref_salt=$cursalt
    else
      [[ ${N3_ITER[$i]} == "$ref_iter" && ${cursalt^^} == ${ref_salt^^} ]] || {
        REASON=REJECT_PARAM_MISMATCH; DETAIL="NSEC3 parameters differ within response"; return 5; }
    fi

    # ---- RRSIG verification ----
    [[ -n ${N3_SIG[$i]} ]] || { REASON=REJECT_RRSIG_INVALID; DETAIL="missing RRSIG for $owner"; return 4; }
    rdata=$(nsec3_rdata_hex "$alg" "${N3_FLAGS[$i]}" "${N3_ITER[$i]}" "${N3_SALT[$i]}" "${N3_NEXT[$i]}" "${N3_TYPES[$i]}")
    verify_rrsig "${N3_SIG[$i]}" "$owner" 50 "$rdata"
    rc=$?
    (( rc == 0 )) || return $rc
  done

  # zone's effective salt/iterations for hashing
  local salt_hex=""; [[ $ref_salt == '-' ]] && salt_hex=""; salt_hex=${ref_salt:-}
  local iters=$ref_iter

  # ---- 2. build ancestor chain and find closest encloser -----------
  local -a chain=()
  local cur=$sq
  while :; do
    chain+=("$cur")
    [[ $cur == "$szone" ]] && break
    [[ $cur != *.* ]] && break
    cur=${cur#*.}
  done
  # ensure apex present
  [[ ${chain[$((${#chain[@]}-1))]} == "$szone" ]] || chain+=("$szone")

  # index owner hash -> index
  local -A owner_idx
  for ((i=0;i<${#N3_OWNER[@]};i++)); do owner_idx[${N3_OWNER[$i]%%.*}]=$i; done

  match_idx() { local h=${1^^}; printf '%s' "${owner_idx[$h]:-}"; }
  cover_idx() {
    local h=${1^^}
    for ((i=0;i<${#N3_OWNER[@]};i++)); do
      covers "${N3_OWNER[$i]%%.*}" "${N3_NEXT[$i]}" "$h" && { printf '%s' "$i"; return 0; }
    done
    printf '%s' ""
  }

  local CE="" ce_hash="" anc ah idx
  for anc in "${chain[@]}"; do
    ah=$(nsec3_hash "$anc" "$salt_hex" "$iters")
    idx=$(match_idx "$ah")
    if [[ -n $idx ]]; then CE=$anc; ce_hash=$ah; break; fi
  done
  [[ -n $CE ]] || { REASON=REJECT_CLOSEST_ENCLOSER; DETAIL="no closest encloser found"; return 7; }

  # next closer: ancestor just below CE on the path to qname
  local next_closer="" nc_hash=""
  if [[ $CE != "$sq" ]]; then
    local prev=""
    for anc in "${chain[@]}"; do
      if [[ $anc == "$CE" && -n $prev ]]; then next_closer=$prev; break; fi
      prev=$anc
    done
  fi

  local q_hash; q_hash=$(nsec3_hash "$sq" "$salt_hex" "$iters")
  local wild_hash; wild_hash=$(nsec3_hash "*.$CE" "$salt_hex" "$iters")

  # ---- 3. classify ------------------------------------------------
  if [[ ${RCODE^^} == NXDOMAIN ]]; then
    local cidx nidx widx
    cidx=$(match_idx "$ce_hash")
    [[ -n $cidx ]] || { REASON=REJECT_CLOSEST_ENCLOSER; DETAIL="no NSEC3 matches closest encloser $CE"; return 7; }
    [[ -n $next_closer ]] || { REASON=REJECT_CLOSEST_ENCLOSER; DETAIL="no next-closer"; return 7; }
    nc_hash=$(nsec3_hash "$next_closer" "$salt_hex" "$iters")
    nidx=$(cover_idx "$nc_hash")
    [[ -n $nidx ]] || { REASON=REJECT_NSEC3_NOT_COVERING; DETAIL="next-closer $next_closer hash $nc_hash not covered (AD=$AD)"; return 6; }
    widx=$(cover_idx "$wild_hash")
    [[ -n $widx ]] || { REASON=REJECT_NSEC3_NOT_COVERING; DETAIL="wildcard *.$CE hash $wild_hash not covered (AD=$AD)"; return 6; }
    if [[ ${N3_FLAGS[$nidx]} == 1 ]]; then
      RESULT=ACCEPT_NXDOMAIN_OPTOUT; DETAIL="CE=$CE next-closer $next_closer covered by opt-out NSEC3"
    else
      RESULT=ACCEPT_NXDOMAIN; DETAIL="CE=$CE next-closer covered, wildcard denied"
    fi
    return 0
  fi

  if [[ ${RCODE^^} == NOERROR ]]; then
    local midx; midx=$(match_idx "$q_hash")
    if [[ -n $midx ]]; then
      # existing name: NODATA if qtype absent from the matching NSEC3 bitmap
      local types=" ${N3_TYPES[$midx]} " t found=0
      for t in ${N3_TYPES[$midx]}; do [[ ${t^^} == "${qtype^^}" ]] && found=1; done
      if (( found )); then REASON=REJECT_NODATA_PROOF; DETAIL="matching NSEC3 still asserts $qtype"; return 8; fi
      if [[ ${N3_TYPES[$midx]^^} == *" CNAME "* && ${qtype^^} != CNAME ]]; then
        REASON=REJECT_NODATA_PROOF; DETAIL="CNAME present, alias NODATA not proven"; return 8; fi
      RESULT=ACCEPT_NODATA; DETAIL="matching NSEC3 for $sq omits $qtype"
      return 0
    fi
    # no matching NSEC3: Opt-Out insecure-delegation NODATA
    local cidx; cidx=$(match_idx "$ce_hash")
    [[ -n $cidx ]] || { REASON=REJECT_CLOSEST_ENCLOSER; DETAIL="no NSEC3 matches closest encloser $CE"; return 7; }
    [[ -n $next_closer ]] || { REASON=REJECT_CLOSEST_ENCLOSER; DETAIL="no next-closer"; return 7; }
    nc_hash=$(nsec3_hash "$next_closer" "$salt_hex" "$iters")
    local nidx; nidx=$(cover_idx "$nc_hash")
    [[ -n $nidx ]] || { REASON=REJECT_NSEC3_NOT_COVERING; DETAIL="next-closer $next_closer hash $nc_hash not covered (AD=$AD)"; return 6; }
    if [[ ${N3_FLAGS[$nidx]} != 1 ]]; then
      REASON=REJECT_NODATA_PROOF; DETAIL="next-closer covered but Opt-Out flag not set"; return 8; fi
    RESULT=ACCEPT_NODATA_OPTOUT_DELEGATION
    DETAIL="CE=$CE next-closer $next_closer covered by Opt-Out NSEC3 (insecure delegation)"
    return 0
  fi

  REASON=REJECT_RCODE; DETAIL="unexpected rcode $RCODE"; return 12
}

# find the first DS record in a (possibly multiline) dig output file
read_ds_from_file() {
  local f=$1 line prev="" own ttl cls typ rest
  while IFS= read -r line; do
    [[ -z $line || $line == ';'* ]] && continue
    if [[ $line == [[:space:]]* ]]; then prev="$prev ${line#"${line%%[![:space:]]*}"}"; continue; fi
    [[ -n $prev ]] && { read -r own ttl cls typ rest <<< "$prev"; [[ ${typ^^} == DS ]] && { printf '%s' "$prev"; return; }; }
    prev=$line
  done < "$f"
  [[ -n $prev ]] && { read -r own ttl cls typ rest <<< "$prev"; [[ ${typ^^} == DS ]] && printf '%s' "$prev"; }
}

do_query() { # <outfile> <dig args...>

  return 1
