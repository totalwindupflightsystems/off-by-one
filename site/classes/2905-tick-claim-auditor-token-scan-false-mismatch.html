Tick-Claim Audit Fix: Read the Staleness Clause, Not the Worst Token on the Line

Root Cause

scripts/check_tick_claim.py found the first line containing staleness, then scanned the entire line for the most severe severity token:

_SEVERITY_RE = re.compile(r"\b(CLOCK_SKEW|unknown|FAIL|WARN|ok)\b")

Tick commits are canary summaries containing pass-count clauses such as:

canary 15 PASS/1 WARN/0 FAIL exit 0 (... staleness ok @21883s)

The unrelated FAIL in 0 FAIL outranked the actual claim staleness ok. The auditor consequently reported:

claim_severity=FAIL
verdict_at_tick=ok
claim_audit=MISMATCH

while its own arithmetic correctly produced ok.

There were two related defects:

  1. A line mentioning staleness without any severity token was treated like a claim, causing false MISMATCH.
  2. board.jsonl was assumed to start with a single-line JSON header. Pretty-printed headers followed by JSONL records raised JSONDecodeError / exit 2.

The fix must extract severity from the staleness clause itself and only use the old line-wide scan as a fallback.

Exact Fix

Apply the following replacements in scripts/check_tick_claim.py.

1. Add clause-aware claim extraction

Replace the existing severity regex / extractor with:

import re

# Canonical severities, least to most severe.
_SEV_RANK = {
    "ok": 0,
    "warn": 1,
    "fail": 2,
    "clock_skew": 3,
    "unknown": 4,
}

# The severity belongs to the staleness clause, not to any other token.
_STALENESS_CLAUSE_RE = re.compile(
    r"\bstaleness\s*[\s=:]*\b(ok|warn|fail|clock_skew|unknown)\b",
    re.IGNORECASE,
)

# Legacy fallback, used ONLY when the staleness line has no explicit clause.
_SEVERITY_RE = re.compile(
    r"\b(clock_skew|unknown|fail|warn|ok)\b",
    re.IGNORECASE,
)


def _canon(token: str) -> str:
    return token.lower()


def extract_claim(message: str):
    """Return the claimed staleness severity, or None for NO_CLAIM."""
    line = next(
        (ln for ln in message.splitlines() if "staleness" in ln.lower()),
        None,
    )
    if line is None:
        return None

    # First and preferred path: severity attached to "staleness".
    clause = _STALENESS_CLAUSE_RE.search(line)
    if clause:
        return _canon(clause.group(1))

    # Fallback only for prose mentioning staleness without a clause.
    tokens = [_canon(t) for t in _SEVERITY_RE.findall(line)]
    if not tokens:
        return None  # a mention without a claim must never become MISMATCH
    return max(tokens, key=_SEV_RANK.__getitem__)

2. Parse the board header tolerantly

Replace the single-line json.loads(...) header read with:

import json


def parse_board_header(blob: str) -> dict:
    """Parse the first JSON object in board.jsonl, tolerating pretty printing."""
    text = blob.lstrip("\ufeff \t\r\n")
    if not text:
        raise ValueError("board.jsonl is empty")

    try:
        obj = json.loads(text)
    except json.JSONDecodeError:
        # Header is pretty-printed and/or followed by JSONL records.
        try:
            obj, _end = json.JSONDecoder().raw_decode(text)
        except json.JSONDecodeError as exc:
            raise ValueError("board.jsonl has no parseable JSON header") from exc

    if not isinstance(obj, dict):
        raise ValueError("board.jsonl header must be a JSON object")
    if "last_tick" not in obj:
        raise ValueError("board.jsonl header missing last_tick")
    return obj


def read_board_header(path: str) -> dict:
    with open(path, encoding="utf-8") as fh:
        return parse_board_header(fh.read())

json.loads() handles a normal single-object header. If the header is pretty-printed and followed by JSONL records, the resulting Extra data error is handled by raw_decode(), which reads exactly one JSON object.

3. Make NO_CLAIM non-failing

Ensure the audit path handles extract_claim(...) is None before comparing severities:

def audit(message: str, header: dict):
    claim = extract_claim(message)
    verdict = verdict_at_tick(header, extract_tick(message))

    if claim is None:
        result = "NO_CLAIM"
    elif verdict is None:
        result = "NO_VERDICT"
    elif claim == verdict:
        result = "MATCH"
    else:
        result = "MISMATCH"

    return claim, verdict, result

A NO_CLAIM result should exit 0, never 1 / MISMATCH.

import pytest
from check_tick_claim import (
    audit,
    extract_claim,
    parse_board_header,
    read_board_header,
    verdict_at_tick,
)

HEADER = {"last_tick": 21800, "cooldown_s": 600}

HONEST = (
    "canary 15 PASS/1 WARN/0 FAIL exit 0 "
    "(all checks passed; staleness ok @21883s)"
)


def test_clause_beats_unrelated_pass_count_fail():
    msg = (
        "canary 15 PASS/1 WARN/0 FAIL exit 0 "
        "(2 FAIL in old log, but staleness ok @21883s)"
    )
    assert extract_claim(msg) == "ok"


def test_true_fail_clause_is_not_hidden_by_other_ok():
    msg = "canary 15 PASS/0 FAIL exit 0 (staleness FAIL @21883s)"
    assert extract_claim(msg) == "fail"
    assert audit(msg, HEADER) == ("fail", "ok", "MISMATCH")


def test_fallback_when_line_mentions_staleness_but_has_no_clause():
    msg = "staleness n/a; canary 15 PASS/0 FAIL exit 1"
    assert extract_claim(msg) == "fail"


def test_mention_without_any_token_is_no_claim_not_mismatch():
    msg = "note: staleness was reviewed manually @21883s"
    assert extract_claim(msg) is None
    assert audit(msg, HEADER) == (None, "ok", "NO_CLAIM")


def test_honest_canary_is_match_and_ok():
    assert audit(HONEST, HEADER) == ("ok", "ok", "MATCH")


def test_pretty_printed_header_before_jsonl_records():
    blob = (
        '{\n'
        '  "last_tick": 21800,\n'
        '  "cooldown_s": 600\n'
        '}\n'
        '{"tick": 21883, "kind": "canary"}\n'
    )
    header = parse_board_header(blob)
    assert header["last_tick"] == 21800
    assert header["cooldown_s"] == 600
    assert verdict_at_tick(header, 21883) == "ok"


def test_pretty_printed_header_file(tmp_path):
    p = tmp_path / "board.jsonl"
    p.write_text(
        '{\n  "last_tick": 21800,\n  "cooldown_s": 600\n}\n'
        '{"tick": 21883}\n',
        encoding="utf-8",
    )
    header = read_board_header(str(p))
    assert (header["last_tick"], header["cooldown_s"]) == (21800, 600)


def test_corpus_sweep_no_false_mismatch():
    # 400 honest canaries with unrelated pass-count FAILs.
    for i in range(400):
        tick = 21883 + i
        header = {"last_tick": tick - 83, "cooldown_s": 600}
        msg = (
            f"canary 15 PASS/1 WARN/0 FAIL exit 0 "
            f"(last run (i-1) had 2 FAIL, staleness ok @{tick}s)"
        )
        claim, verdict, result = audit(msg, header)
        assert (claim, verdict, result) == ("ok", "ok", "MATCH"), (i, msg)
