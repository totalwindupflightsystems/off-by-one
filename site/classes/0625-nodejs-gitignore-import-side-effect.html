def git_commit_exists(commit: str) -> bool:
    # full ^{commit} rev validation, so a tree/blob/tag can't masquerade as a commit
    subprocess.run(["git", "rev-parse", "--verify", "--quiet",
                    f"{commit}^{{commit}}"], check=True, capture_output=True)
    return True

def verify_judge_criteria(criteria: dict, judge: str) -> dict:
    # docs/e2e tasks have no automated suite: a human judge signs each criterion.
    # Task is stamped complete ONLY when every verdict == "pass".
    all_pass = all(str(c.get("verdict","")).strip().lower() == "pass"
                   for c in criteria["criteria"])
    return {"judge": judge, "all_pass": all_pass, "results": criteria["criteria"]}

def append_record(task_id, commit_hash, judge_block) -> None:
    if already_recorded(task_id, commit_hash):          # idempotent: no dup rows

    with BOARD_FILE.open("a") as fh:

2. .gitignore — silent docs/* ignore of new md files

docs/*
!docs/*.md        # allowlist: new top-level markdown stays trackable
!docs/**/         # re-open subdirs (a negated parent dir is required for nested files)
!docs/**/*.md     # nested markdown trackable
docs/build/**     # generated artifacts stay ignored AFTER the re-open

Key gotcha caught in testing: !docs/**/ alone re-includes generated output too, so the generated-dir rule must come after the negations.

if (isDirectRun()) process.exitCode = main();    // import => noop; direct => runs
