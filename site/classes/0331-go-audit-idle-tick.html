// audit-idle-tick: runs every consensus tick; returns NoOp when idle.
type Audit struct {
    GitCrossRef  Gate // new commits since last tick
    Suite        Gate // full suite re-run result
    GitReins     Gate // 22/22 zero drift
    Hilo         Gate // 1195/188 stable
    SchedPin     Gate // fleet.toml CooldownS=7200 intact
    PendingQueue Gate // real pending == 0, no PUT issued
}

func RunTick(tick int, prevState State) Result {
    r := Audit{
        GitCrossRef:  Gate{Name: "no-new-commits",   OK: prevState.HeadHash == currentHead()},
        Suite:        Gate{Name: "suite-green",      OK: suiteGreen()}, // chronicle 142s, harness 148s, demo 32s, shim 4.1s
        GitReins:     Gate{Name: "drift-zero",       OK: gitReinsDrift() == 0}, // 22/22
        Hilo:         Gate{Name: "hilo-stable",      OK: hiloStable()},   // 1195/188
        SchedPin:     Gate{Name: "sched-pin",        OK: fleetPin("CooldownS") == 7200},
        PendingQueue: Gate{Name: "no-pending",       OK: pendingCount() == 0},
    }

    if all(r) {
        return Result{Action: NoOp, ConsecutiveIdle: prevState.ConsecutiveIdle + 1,
                      SchedPin: prevState.SchedPin, // do NOT PUT; pin stays 7200
                      Gates: r}
    }
    return Result{Action: Reconcile, Gates: failed(r)}
}

// Invariant: NoOp must never (a) emit a PUT, (b) create a task,
// (c) mutate the fleet.toml pin, or (d) claim a commit that does not exist.
