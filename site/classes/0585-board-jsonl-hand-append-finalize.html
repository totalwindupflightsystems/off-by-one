python3 - <<'PY'
import json
path = "tasks.jsonl"
rows = [json.loads(l) for l in open(path) if l.strip()]
for r in rows:
    if r["id"] == "JSONL-NORM-001":
        assert r["status"] == "pending", r                     # stale-pending invariant
        r["status"] = "done"
        r["foreman_note"] = "PASS; recovery=board-jsonl-hand-append-finalize; jsonl-only-patch"
open(path, "w").write("".join(json.dumps(r, sort_keys=True) + "\n" for r in rows))
PY
