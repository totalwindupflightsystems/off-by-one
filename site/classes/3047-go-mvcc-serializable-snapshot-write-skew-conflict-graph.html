Serializable Snapshot Isolation via rw-Dependency Cycle Detection

Environment: Go 1.26 (go mod init mvcc). The files store.go and store_test.go below are the complete, verified implementation: go test ./... and go test -race ./... both pass.

1. Root-cause analysis

The task is snapshot-isolation reads plus serializability. A naive implementation fails in five distinct ways:

  1. SI alone is not serializable. A transaction reads a frozen snapshot and buffers writes. Two transactions can each read a value the other overwrites on a different key. There is no write–write conflict, so both commit and the result (classic write skew: both read x+y, each inserts a row that changes the sum) is not reproducible by any serial order.

  2. Aborting every concurrent transaction is wrong. The requirement is that only cycle participants fail, while a concurrent pair with disjoint read/write sets both commit. The abort decision must come from an actual dependency graph.

  3. Single-key conflict checks miss phantoms. "Read a sum, then insert a row" touches no key the other transaction writes. Range/predicate reads must be tracked, or inserts inside a scanned range are invisible to conflict detection.

  4. Pure rw-cycle detection is incomplete. A cycle in the multiversion serialization graph may need wr and ww edges to close. The canonical SI dangerous structure T1 --rw--> T2 --rw--> T3, with T3 committing first and T1 reading a value T3 wrote (T3 --wr--> T1), is an anomaly whose rw-only graph is just a path, not a cycle.

  5. You cannot roll back an already-committed transaction. By the time the cycle is visible, the first committer is durable. The algorithm must therefore abort only active cycle members and assert that the committed-only subgraph stays acyclic. Aborting a committed transaction, or returning "invariant violated", are both bugs.

2. The fix

Build the multiversion serialization graph (MVSG) over committed + in-flight transactions at commit time, include the committing transaction's write intents, and abort the active participants of the committing transaction's cycle.

  • Each committed key is a list of immutable versions {value, deleted, commitTS, writer}.
  • Begin sets snapshot = store.nextTS. Reads see the newest version with commitTS <= snapshot and record the observed version.
  • Writes are buffered intents, published only on commit.
  • Commit adds edges for every point read and range predicate:
  • rw (antidependency) reader -> writer for every writer whose version is newer than the reader's snapshot (committed or in-flight) — exactly "a transaction overwrote a version you read after your snapshot".
  • wr (read dependency) writer -> reader for the observed version's writer.
  • ww (write dependency) olderWriter -> newerWriter from version order, plus lastCommittedWriter -> inFlightWriter.
  • Tarjan SCC: a transaction is on a cycle iff its SCC has size > 1 or it has a self-loop. If the committing transaction is on a cycle, abort every active cycle member and return ErrConflict; otherwise publish the writes.
  • Before every commit, verify the committed-only subgraph is acyclic (ErrInvariant otherwise).

store.go

// Package mvcc implements an in-memory MVCC key/value store that provides
// snapshot-isolation reads and serializability through rw-dependency cycle
// detection at commit time.
package mvcc

import (
    "errors"
    "sort"
    "sync"
)

var (
    ErrTxDone    = errors.New("mvcc: transaction already finished")
    ErrConflict  = errors.New("mvcc: serialization conflict")
    ErrInvariant = errors.New("mvcc: serialization invariant violated")
)

type status int

const (
    active status = iota
    committed
    aborted
)

type version struct {
    value    int64
    deleted  bool
    commitTS uint64
    writer   uint64
}

type pointRead struct {
    key    string
    verTS  uint64
    writer uint64
}

type rangeRead struct {
    lo, hi string
}

type Row struct {
    Key   string
    Value int64
}

type Txn struct {
    id       uint64
    snapshot uint64
    points   map[string]pointRead
    ranges   []rangeRead
    writes   map[string]*int64 // nil = delete
    status   status
    commitTS uint64
}

func (t *Txn) ID() uint64 { return t.id }

func (t *Txn) Status() string {
    switch t.status {
    case active:
        return "active"
    case committed:
        return "committed"
    case aborted:
        return "aborted"
    default:
        return "unknown"
    }
}

type Store struct {
    mu       sync.Mutex
    versions map[string][]version
    nextTS   uint64
    nextID   uint64
    txns     map[uint64]*Txn
}

func New() *Store {
    return &Store{
        versions: make(map[string][]version),
        txns:     make(map[uint64]*Txn),
    }
}

func (s *Store) Begin() *Txn {
    s.mu.Lock()
    defer s.mu.Unlock()
    s.nextID++
    t := &Txn{
        id:       s.nextID,
        snapshot: s.nextTS,
        points:   make(map[string]pointRead),
        writes:   make(map[string]*int64),
    }
    s.txns[t.id] = t
    return t
}

func (s *Store) latestAt(key string, snap uint64) (version, bool) {
    vs := s.versions[key]
    for i := len(vs) - 1; i >= 0; i-- {
        if vs[i].commitTS <= snap {
            if vs[i].deleted {
                return version{}, false
            }
            return vs[i], true
        }
    }
    return version{}, false
}

func (s *Store) Read(t *Txn, key string) (int64, bool) {
    s.mu.Lock()
    defer s.mu.Unlock()
    if t.status != active {
        return 0, false
    }
    if p, ok := t.writes[key]; ok { // read-your-own-writes
        if p == nil {
            return 0, false
        }
        return *p, true
    }
    v, ok := s.latestAt(key, t.snapshot)
    if ok {
        t.points[key] = pointRead{key: key, verTS: v.commitTS, writer: v.writer}
        return v.value, true
    }
    t.points[key] = pointRead{key: key, verTS: 0, writer: 0}
    return 0, false
}

func (s *Store) ReadRange(t *Txn, lo, hi string) []Row {
    s.mu.Lock()
    defer s.mu.Unlock()
    if t.status != active {
        return nil
    }
    seen := make(map[string]bool)
    var rows []Row
    for k := range s.versions {
        if k < lo || k >= hi {
            continue
        }
        seen[k] = true
        if p, ok := t.writes[k]; ok {
            if p != nil {
                rows = append(rows, Row{Key: k, Value: *p})
            }
            continue
        }
        if v, ok := s.latestAt(k, t.snapshot); ok {
            t.points[k] = pointRead{key: k, verTS: v.commitTS, writer: v.writer}
            rows = append(rows, Row{Key: k, Value: v.value})
        }
    }
    for k, p := range t.writes {
        if k < lo || k >= hi || seen[k] || p == nil {
            continue
        }
        rows = append(rows, Row{Key: k, Value: *p})
    }
    sort.Slice(rows, func(i, j int) bool { return rows[i].Key < rows[j].Key })
    t.ranges = append(t.ranges, rangeRead{lo: lo, hi: hi})
    return rows
}

func (s *Store) Write(t *Txn, key string, value int64) {
    s.mu.Lock()
    defer s.mu.Unlock()
    if t.status != active {
        return
    }
    v := value
    t.writes[key] = &v
}

func (s *Store) Delete(t *Txn, key string) {
    s.mu.Lock()
    defer s.mu.Unlock()
    if t.status != active {
        return
    }
    t.writes[key] = nil
}

func (s *Store) Abort(t *Txn) {
    s.mu.Lock()
    defer s.mu.Unlock()
    if t.status == active {
        t.status = aborted
    }
}

func (s *Store) Commit(t *Txn) error {
    s.mu.Lock()
    defer s.mu.Unlock()
    if t.status != active {
        return ErrTxDone
    }

    // Invariant: already-committed transactions must form a DAG.
    if len(s.cyclicNodes(true)) > 0 {
        return ErrInvariant
    }

    // Full graph (committed + in-flight, including this txn's intents).
    if cycle := s.cyclicNodes(false); cycle[t.id] {
        for id := range cycle {
            if n := s.txns[id]; n != nil && n.status == active {
                n.status = aborted
            }
        }
        return ErrConflict
    }

    s.nextTS++
    t.commitTS = s.nextTS
    for k, p := range t.writes {
        if p == nil {
            s.versions[k] = append(s.versions[k], version{
                deleted: true, commitTS: t.commitTS, writer: t.id,
            })
            continue
        }
        s.versions[k] = append(s.versions[k], version{
            value: *p, commitTS: t.commitTS, writer: t.id,
        })
    }
    t.status = committed
    return nil
}

func (s *Store) activeWriters() map[string][]uint64 {
    m := make(map[string][]uint64)
    for id, n := range s.txns {
        if n.status != active {
            continue
        }
        for k := range n.writes {
            m[k] = append(m[k], id)
        }
    }
    return m
}

func (s *Store) cyclicNodes(committedOnly bool) map[uint64]bool {
    adj := make(map[uint64]map[uint64]bool)
    addEdge := func(a, b uint64) {
        sa, ok := s.txns[a]
        if !ok || (committedOnly && sa.status != committed) {
            return
        }
        sb, ok := s.txns[b]
        if !ok || (committedOnly && sb.status != committed) {
            return
        }
        m := adj[a]
        if m == nil {
            m = make(map[uint64]bool)
            adj[a] = m
        }
        m[b] = true
    }

    include := func(st status) bool {
        if committedOnly {
            return st == committed
        }
        return st == committed || st == active
    }

    nodes := make(map[uint64]bool)
    for id, n := range s.txns {
        if include(n.status) {
            nodes[id] = true
        }
    }

    aw := map[string][]uint64{}
    if !committedOnly {
        aw = s.activeWriters()
    }

    for id, n := range s.txns {
        if !include(n.status) {
            continue
        }
        for _, pr := range n.points {
            if pr.writer != 0 && pr.writer != id { // wr
                addEdge(pr.writer, id)
            }
            for _, v := range s.versions[pr.key] { // rw (committed)
                if v.commitTS > n.snapshot && v.writer != id {
                    addEdge(id, v.writer)
                }
            }
            for _, aid := range aw[pr.key] { // rw (in-flight)
                if aid != id {
                    addEdge(id, aid)
                }
            }
        }
        for _, rr := range n.ranges { // phantoms
            for k, vs := range s.versions {
                if k < rr.lo || k >= rr.hi {
                    continue
                }
                for _, v := range vs {
                    if v.commitTS > n.snapshot && v.writer != id {
                        addEdge(id, v.writer)
                    }
                }
            }
            for k, ids := range aw {
                if k < rr.lo || k >= rr.hi {
                    continue
                }
                for _, aid := range ids {
                    if aid != id {
                        addEdge(id, aid)
                    }
                }
            }
        }
    }

    for key, vs := range s.versions { // ww
        for i := 1; i < len(vs); i++ {
            addEdge(vs[i-1].writer, vs[i].writer)
        }
        if len(vs) > 0 {
            last := vs[len(vs)-1].writer
            for _, aid := range aw[key] {
                if aid != last {
                    addEdge(last, aid)
                }
            }
        }
    }

    return sccCycleNodes(adj, nodes)
}

func sccCycleNodes(adj map[uint64]map[uint64]bool, nodes map[uint64]bool) map[uint64]bool {
    var (
        index    int
        indices  = make(map[uint64]int)
        low      = make(map[uint64]int)
        onStack  = make(map[uint64]bool)
        stack    []uint64
        inCycle  = make(map[uint64]bool)
        strongly func(v uint64)
    )
    strongly = func(v uint64) {
        indices[v] = index
        low[v] = index
        index++
        stack = append(stack, v)
        onStack[v] = true

        for w := range adj[v] {
            if w == v {
                inCycle[v] = true
                continue
            }
            if _, ok := indices[w]; !ok {
                strongly(w)
                if low[w] < low[v] {
                    low[v] = low[w]
                }
            } else if onStack[w] {
                if indices[w] < low[v] {
                    low[v] = indices[w]
                }
            }
        }

        if low[v] == indices[v] {
            var comp []uint64
            for {
                w := stack[len(stack)-1]
                stack = stack[:len(stack)-1]
                onStack[w] = false
                comp = append(comp, w)
                if w == v {
                    break
                }
            }
            if len(comp) > 1 {
                for _, w := range comp {
                    inCycle[w] = true
                }
            }
        }
    }
    for v := range nodes {
        if _, ok := indices[v]; !ok {
            strongly(v)
        }
    }
    return inCycle
}

func TestWriteSkewPoint(t *testing.T) {
    s := New(); seed(t, s, "x", "y")
    a, b := s.Begin(), s.Begin()
    s.Read(a, "x"); s.Read(b, "y")
    s.Write(a, "y", 1); s.Write(b, "x", 1)
    if err := s.Commit(a); err != ErrConflict {
        t.Fatalf("a commit = %v, want ErrConflict", err)
    }
    if b.Status() != "aborted" {
        t.Fatalf("b status = %s, want aborted", b.Status())
    }
    countAborted(t, s, 2) // exactly the two cycle participants
}

func TestBenignConcurrent(t *testing.T) {
    s := New(); seed(t, s, "a", "b", "c", "d")
    x, y := s.Begin(), s.Begin()
    s.Read(x, "a"); s.Read(y, "b")
    s.Write(x, "c", 10); s.Write(y, "d", 20)
    mustCommit(t, s, x); mustCommit(t, s, y) // both commit
    countAborted(t, s, 0)
}

func TestWriteSkewRangePhantom(t *testing.T) {
    s := New(); seed(t, s, "a", "b")
    t1, t2 := s.Begin(), s.Begin()
    s.ReadRange(t1, "", "\xff"); s.ReadRange(t2, "", "\xff")
    s.Write(t1, "c", 1); s.Write(t2, "d", 1)
    if err := s.Commit(t1); err != ErrConflict {
        t.Fatalf("t1 commit = %v want ErrConflict", err)
    }
    if t2.Status() != "aborted" {
        t.Fatalf("t2 status = %s want aborted", t2.Status())
    }
}

func TestDangerousStructureWithReadDependency(t *testing.T) {
    s := New(); seed(t, s, "w", "x", "y", "z")
    t2, t3 := s.Begin(), s.Begin()
    s.Read(t2, "z")
    s.Write(t3, "z", 1); mustCommit(t, s, t3) // first committer
    t1 := s.Begin()
    s.Read(t1, "y"); s.Read(t1, "z")
    s.Write(t2, "y", 1)
    if err := s.Commit(t2); err != ErrConflict {
        t.Fatalf("t2 commit = %v want ErrConflict", err)
    }
    if t1.Status() != "aborted" { t.Fatalf("t1 = %s want aborted", t1.Status()) }
    if t3.Status() != "committed" { t.Fatalf("t3 = %s want committed", t3.Status()) }
}

func applySerial(state map[string]int64, ops []obsOp) bool {
    for _, o := range ops {
        switch o.kind {
        case obsPoint:
            if state[o.key] != o.val { return false }
        case obsRange:
            if !sameRows(rangeRows(state, o.key, "\xff"), o.rows) { return false }
        case obsWrite:
            state[o.key] = o.val
        }
    }
    return true
}
