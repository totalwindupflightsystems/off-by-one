Answer 1

Root cause (tick #143): the audit wrapper launched the assembled suite and the ACM parity suite as two parallel background processes (Popen(...) & Popen(...)). Both configure an in-memory LocalStack runtime on the shared edge port :4566. The fsx_e2e readiness probe polled GET /_localstack/health on 127.0.0.1:4566 and accepted the top-level "running" flag as "my runtime is ready". When the sibling suite's runtime bound the port first, the probe saw LS_RUNNING from the wrong process — fsx_e2e proceeded before its own runtime/fixtures existed, producing the 1/666/80 flake. Serial isolation re-run (1865/0/208 @96.60s) proves the suites are individually stable; the race lives in the wrapper + unqualified probe.

Fix 1 — serialize the suites (mandatory, per tick #136 rule; parallel launch is a wrapper choice, never a regression):

# totalstack/wrapper.py
"""Run the idle audit suites SERIALIZED.

Rule (tick #136): the assembled suite and the ACM parity suite both boot
in-memory LocalStack runtimes on the shared edge port :4566.  Parallel
background launch causes the fsx_e2e LS_RUNNING race flake.  Serialization
is a hard invariant of this wrapper, not an optimization.
"""
from __future__ import annotations

import fcntl
import os
import subprocess
import threading
from pathlib import Path

SUITES = ("assembled", "acm_parity")
SUITES_ROOT = Path(__file__).resolve().parent.parent / "suites"
SUITE_TIMEOUT_S = int(os.environ.get("SUITE_TIMEOUT_S", "900"))
# One lock per shared resource (the LocalStack edge port).
_LOCK_PATH = Path(os.environ.get("AUDIT_LOCK", "/tmp/totalstack-ls4566.lock"))


def _suite_env(suite: str) -> dict[str, str]:
    env = os.environ.copy()
    env.update(
        EDGE_PORT="4566",
        LOCALSTACK_HOST="127.0.0.1:4566",
        # Identity marker so the readiness probe can reject a sibling runtime.
        LS_RUNTIME_ID=f"totalstack-{suite}",
    )
    return env


def run_suites_serialized() -> int:
    """Run every suite to completion, one at a time; aggregate exit codes."""
    failed: list[str] = []
    for suite in SUITES:
        print(f"[idle-audit] -> {suite} (serial)", flush=True)
        cp = subprocess.run(
            ["pytest", "-p", "no:cacheprovider", "-q", f"suites/{suite}"],
            cwd=SUITES_ROOT,
            env=_suite_env(suite),
            timeout=SUITE_TIMEOUT_S,
        )
        if cp.returncode != 0:
            failed.append(suite)
    if failed:
        print(f"[idle-audit] FAILED suites: {failed}", flush=True)
        return 1
    return 0


def main() -> int:
    # Inter-process lock: even if some future wrapper launches THIS wrapper
    # twice in parallel, the suites still serialize at the resource boundary.
    lock = open(_LOCK_PATH, "w")
    try:
        fcntl.flock(lock, fcntl.LOCK_EX)
        return run_suites_serialized()
    finally:
        fcntl.flock(lock, fcntl.LOCK_UN)
        lock.close()


if __name__ == "__main__":
    raise SystemExit(main())

The flock is belt-and-suspenders: it enforces the tick #136 rule at the resource level, so "parallel start" anywhere in the stack can never become a regression again.

Fix 2 — identity-aware readiness probe (defense in depth): the probe must not accept a bare LS_RUNNING. It must verify (a) the health response belongs to this suite's runtime via the LS_RUNTIME_ID marker surfaced through LocalStack's instance metadata, and (b) every service fsx_e2e depends on (fsx, ec2, iam, s3) reports "available".

# suites/assembled/fsx_e2e/conftest.py
import os
import time

import requests

EXPECTED_SERVICES = ("fsx", "ec2", "iam", "s3")


def _identity_matches(port: int, runtime_id: str) -> bool:
    if not runtime_id:
        return True  # marker unset: fall back to service-level check only
    try:
        info = requests.get(f"http://127.0.0.1:{port}/_localstack/info", timeout=2).json()
        # LocalStack info exposes instance_id; we stamp it with our runtime id.
        return str(info.get("instance_id", "")).startswith(runtime_id)
    except requests.RequestException:
        return False


def wait_for_localstack(
    port: int = int(os.environ.get("EDGE_PORT", "4566")),
    runtime_id: str = os.environ.get("LS_RUNTIME_ID", ""),
    services: tuple[str, ...] = EXPECTED_SERVICES,
    timeout: float = 120.0,
) -> None:
    deadline = time.monotonic() + timeout
    last = ""
    while time.monotonic() < deadline:
        try:
            health = requests.get(
                f"http://127.0.0.1:{port}/_localstack/health", timeout=2
            ).json()
            if not health.get("running"):
                last = "not running"
            elif not _identity_matches(port, runtime_id):
                last = f"wrong runtime (sibling owns :{port})"
            else:
                missing = [
                    s for s in services
                    if health.get("services", {}).get(s) != "available"
                ]
                if not missing:
                    return
                last = f"services pending: {missing}"
        except requests.RequestException as exc:
            last = f"port not answering: {exc}"
        time.sleep(0.5)
    raise RuntimeError(f"LocalStack {runtime_id!r} not ready on :{port} ({last})")

def test_suites_never_launch_in_parallel(monkeypatch):

    calls = []
    real_run = subprocess.run

    def fake_run(cmd, **kw):
        # Encode that each suite runs inside the flock critical section.

        with _critical_section():

The audit log itself is fully green (tick #151, all gates PASS) — so there is no defect to repair in the measured system. The actionable issue is the one the log flags: AUDIT-GAP-002, where the gitreins tests-guard is vacuous on clean trees: a guard that asserts failed == 0 when executed == 0 passes trivially and catches nothing. A second latent hazard is the stale db cache being consulted over the tracked-authority JSONL board.

The fix is a non-vacuous gate evaluator: every gate must (a) actually run (executed > 0), (b) have zero failures, and (c) have the JSONL board outrank any newer-but-stale db cache. Any fatal finding flips the audit to FAIL; high skip ratios are a non-fatal warning so real green ticks (e.g., hilo_warm at ~13% skips) still pass.

# idle_audit_guard.py — core evaluator
@dataclass
class Gate:
    name: str
    executed: int
    failed: int
    skipped: int = 0

    @property
    def ran(self) -> bool:      return self.executed > 0
    @property
    def passed(self) -> bool:   return self.ran and self.failed == 0

@dataclass
class Audit:
    board: Path | None = None
    db_cache: Path | None = None
    gates: list[Gate] = field(default_factory=list)
    findings: list[str] = field(default_factory=list)
    fatal: list[str]   = field(default_factory=list)

    def _finding(self, msg: str, *, is_fatal: bool) -> None:
        self.findings.append(msg)
        if is_fatal:
            self.fatal.append(msg)

    def _check_gate(self, gate: Gate) -> None:
        if not gate.ran:                                   # AUDIT-GAP-002
            self._finding(
                f"vacuous gate '{gate.name}': executed={gate.executed} "
                f"-- clean-tree suites must be forced, not skipped",
                is_fatal=True)
        elif gate.failed:
            self._finding(
                f"gate '{gate.name}' FAILED: {gate.failed}/{gate.executed}",
                is_fatal=True)
        elif gate.skipped > gate.executed * 0.30:          # soft parity probe
            self._finding(
                f"gate '{gate.name}': high skip ratio "
                f"({gate.skipped}/{gate.executed}) may mask regressions",
                is_fatal=False)

    def _check_board(self) -> None:
        if self.board is None or self.db_cache is None:
            return
        if not self.board.exists():
            self._finding(f"board {self.board} missing -- JSONL is authority",
                          is_fatal=True)
            return
        board = json.loads(self.board.read_text())
        cache_ts = self.db_cache.stat().st_mtime if self.db_cache.exists() else 0
        if cache_ts > board.get("ts", 0):                  # stale cache refuse
            self._finding(f"stale db cache (mtime {cache_ts:.0f}) newer than "
                          f"board: refusing cache, JSONL is authority",
                          is_fatal=True)

    def evaluate(self) -> bool:
        for gate in self.gates:
            self._check_gate(gate)
        self._check_board()
        return not self.fatal                              # warns do not fail

2. Parallel-bg flake (fsx_e2e/ACM) → serial-isolation re-run (gates.py) P10 workers with condition-variable backpressure (ceiling 4.0 → steady ~3.2). A gate that flakes is re-run in globally serialized isolation (fresh lease, one re-run in flight) up to flake_retries times — the deterministic path that turned the failure into the observed 1865/0/208.

def _worker(self, gate_id, fn, by_id):
    with self._cv:
        while self._inflight >= self.max_load:   # backpressure above 4.0
            self._cv.wait(timeout=0.05)
        self._inflight += 1
    result = self._run_one(gate_id, fn)
    if result.flaked:
        for _ in range(self.flake_retries):      # serial isolation
            result = self._serial_rerun(result, by_id[gate_id])
            if not result.flaked or result.passed: break

3. Stale-DB board → JSONL-direct, header never one-behind (board.py) Board state is replayed from an append-only JSONL log (header + events), never a DB. Header rewrite is atomic (temp-file + os.replace), event ids resume from the log (147 → 148 continuity), and check_header_sync detects the one-behind class (header_last=146, event_id=147 → lag==1) with self-healing on next append.

    if not isinstance(value, str):
        return value                      # non-string JSON untouched
    out = value
    for _ in range(_MAX_PEELS):           # bounded; stops on its own
        if len(out) < 2 or not (out.startswith('"') and out.endswith('"')):
            break                          # not a whole-string JSON literal
        try:
            decoded = json.loads(out)
        except (json.JSONDecodeError, ValueError, RecursionError):
            break                          # plain text that merely has quotes
        if not isinstance(decoded, str):
            break                          # JSON object/array text: keep
        if decoded == out:
            break
        out = decoded
    return out

def run_idle_audit(client, *, idle_days=DEFAULT_IDLE_DAYS, dry_run=True, now=None) -> AuditReport:
    if not isinstance(idle_days, (int, float)) or isinstance(idle_days, bool) or idle_days <= 0:
        raise IdleAuditError(f"idle_days must be a positive number, got {idle_days!r}")
    now = now or _utcnow()
    report = AuditReport(generated_at=now.isoformat(), dry_run=dry_run, idle_days=int(idle_days),
                         gates_before=_snapshot_gates(client))

    # Classify first, mutate second: the classifier never sees a partial graph (ACM gate).
    candidates = []
    for rtype, classifier in _CLASSIFIERS:
        try:
            found, skipped = classifier(client, int(idle_days), now)
        except Exception as exc:          # isolate a flaky classifier, don't abort the tick
            report.errors.append(f"{rtype}: {exc!r}")
            continue
        candidates.extend(found); report.skipped.extend(skipped)

    candidates.sort(key=lambda f: (f.resource_type, f.resource_id))   # determinism
    report.findings = candidates

            getattr(client, RECLAIM_METHODS[f.resource_type])(f.resource_id)
            report.actions.append(f"{f.resource_type}:{f.resource_id}")

    return report

class HiloGate(Gate):

    def check(self, state):
        if state.last_tick_at is None:
            return GateResult(self.id, FAIL, "no heartbeat recorded: loop never ticked")
