Solution: CF-005 rework — shell-dispatch-artifact-manifest-boundary

Repository: coding-hermes/code-factory (branch wt/CF-005-rework, based on wt/CF-005 @ 9a04a4a). All work is local/hermetic: no live bunker, Crier identity, gateway, credential, or inference proof.

Root-cause analysis

Three independent boundary defects, all of the same class — a checked-in script accepted a proxy for evidence instead of the measured thing:

  1. The shell-dispatch check was a substring match on a crash. tests/selftest.sh ran project-node.sh --check under PATH=/nonexistent and passed if the exit code was non-zero and the output contained the substring FAIL. A Bash crash (...: unbound variable) is non-zero and not a named checker row, so it satisfied both predicates. A green that cannot distinguish a real FAIL &lt;project&gt;_url row from a shell fault is decoration.
  2. The artifact boundary was a character blacklist. verify-node.sh did case "$ARTIFACT" in *[!A-Za-z0-9_./~:-]*) — a denylist, not an allowlist. ~/.ssh/id_rsa, .., //, and remote paths outside the manifest were all addressable, and the value was interpolated into bash -lc "sha256sum -- $ARTIFACT" (a shell string), so traversal and quoting were unchecked.
  3. Dispatch and manifest boundaries were advisory. node-create.sh validated nothing about --repo before bunker spawn/exec/cp; project-node.sh wrote the manifest with a bare open(path, "w") (non-atomic, no writer-status check) and would happily overwrite a manifest owned by a different node_id/project.

The rework replaces each proxy with a fail-closed, named measurement, then proves the check can go red with executable mutations.

1. scripts/artifact-path.py (new) — allowlist + descriptor-relative O_NOFOLLOW

#!/usr/bin/env python3
"""artifact-path.py — validate a node artifact path before any remote shell probe.

  lexical   `~/.node/node.json` or `~/.node/dispatch/<dispatch-id>/...` only.
  open      walk from HOME one component at a time with os.open(dir_fd=, O_NOFOLLOW),
            then require a regular file, st_nlink == 1, size <= --max-bytes.

Called with a direct argv (never interpolated into a shell).
Exit 0 accepted (relative path on stdout), 2 rejected (`REJECT <reason>` on stderr).
"""
from __future__ import annotations

import argparse
import os
import re
import stat
import sys

MANIFEST_REL = ".node/node.json"
DISPATCH_ROOT = ".node/dispatch"
DEFAULT_MAX_BYTES = 16 * 1024 * 1024

DISPATCH_ID = re.compile(r"^[A-Za-z0-9][A-Za-z0-9_-]{0,63}$")
COMPONENT = re.compile(r"^[A-Za-z0-9._-]+$")


def _reject(reason: str):
    sys.stderr.write("REJECT %s\n" % reason)
    return 2


def lexical(artifact: str):
    """Return (relative_path, None) or (None, reason)."""
    if not artifact:
        return None, "empty path"
    if artifact != artifact.strip():
        return None, "leading/trailing whitespace"
    if any(ord(c) < 0x20 or ord(c) == 0x7F for c in artifact):
        return None, "control character"
    if "\\" in artifact:
        return None, "backslash separator"
    if artifact.startswith("~/"):
        rel = artifact[2:]
    else:
        return None, "must be home-relative (~/...)"

    if not rel or rel.startswith("/") or rel.endswith("/") or "//" in rel:
        return None, "malformed separators"
    parts = rel.split("/")
    for part in parts:
        if part in ("", ".", ".."):
            return None, "dot component"
        if not COMPONENT.match(part):
            return None, "unsupported characters in %r" % part

    if rel == MANIFEST_REL:
        return rel, None

    if parts[0] == ".node" and len(parts) >= 4 and parts[1] == "dispatch":
        if not DISPATCH_ID.match(parts[2]):
            return None, "dispatch id outside grammar"
        return rel, None

    return None, "not the node manifest or a dispatch artifact"


def open_nofollow(home: str, rel: str, max_bytes: int):
    """Descriptor-relative traversal with O_NOFOLLOW. Returns (path, None)/(None, reason)."""
    parts = rel.split("/")
    home = os.path.abspath(home)
    try:
        fd = os.open(home, os.O_RDONLY | os.O_DIRECTORY)
    except OSError as exc:
        return None, "cannot open home: %s" % exc.strerror
    try:
        for index, part in enumerate(parts):
            last = index == len(parts) - 1
            flags = os.O_RDONLY | os.O_NOFOLLOW   # CF005-NOFOLLOW
            if not last:
                flags |= os.O_DIRECTORY
            try:
                nfd = os.open(part, flags, dir_fd=fd)
            except OSError as exc:
                if exc.errno == getattr(os, "ELOOP", 40):
                    return None, "symlink refused at %r" % part
                return None, "cannot open %r: %s" % (part, exc.strerror)
            finally:
                os.close(fd)
            fd = nfd
            if last:
                st = os.fstat(fd)
                if not stat.S_ISREG(st.st_mode):
                    return None, "not a regular file"
                if st.st_nlink != 1:
                    return None, "hard-linked file (nlink=%d)" % st.st_nlink
                if st.st_size > max_bytes:
                    return None, "size %d exceeds %d bytes" % (st.st_size, max_bytes)
        return os.path.join(home, rel), None
    finally:
        try:
            os.close(fd)
        except OSError:
            pass


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="validate a node artifact path")
    parser.add_argument("--artifact", required=True)
    parser.add_argument("--home", default=os.path.expanduser("~"))
    parser.add_argument("--max-bytes", type=int, default=DEFAULT_MAX_BYTES)
    parser.add_argument("--open", dest="do_open", action="store_true",
                        help="also walk the file with O_NOFOLLOW (local file mode)")
    parser.add_argument("--lexical", dest="do_lexical", action="store_true",
                        help="lexical scope only (default when --open is absent)")
    args = parser.parse_args(argv)

    rel, reason = lexical(args.artifact)
    if reason is not None:
        return _reject(reason)

    if args.do_open:
        path, reason = open_nofollow(args.home, rel, args.max_bytes)
        if reason is not None:
            return _reject(reason)
        print(path)
    else:
        print(rel)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

2. scripts/node-manifest.py (new) — identity check + atomic writer with explicit status

#!/usr/bin/env python3
"""node-manifest.py — identity check and atomic publish for the node manifest.

  identity-check  an --apply that targets an existing manifest must match its
                  node_id and project; mismatch is a named refusal.
  write           temp file in the same dir, flush + fsync(file), os.replace,
                  fsync(dir). Failure prints `WRITER-FAIL` and exits 4.
"""
from __future__ import annotations

import argparse
import json
import os
import sys
import tempfile

CREATED_BY = "project-node.sh v2"


def _identity_matches(path: str, node_id: str, project: str):
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, ValueError) as exc:
        print("cannot read manifest: %s" % exc, file=sys.stderr)
        return 3
    if data.get("node_id") != node_id:
        print("node_id %r != requested %r" % (data.get("node_id"), node_id), file=sys.stderr)
        return 1
    if data.get("project") != project:
        print("project %r != requested %r" % (data.get("project"), project), file=sys.stderr)
        return 1
    return 0


def _atomic_write(path: str, payload: dict) -> int:
    directory = os.path.dirname(os.path.abspath(path)) or "."
    tmp_path = None
    try:
        fd, tmp_path = tempfile.mkstemp(dir=directory, prefix=".node.json.", suffix=".tmp")
        with os.fdopen(fd, "w", encoding="utf-8") as handle:
            json.dump(payload, handle, indent=1)
            handle.write("\n")
            handle.flush()
            os.fsync(handle.fileno())          # CF005-WRITER-STATUS
        os.replace(tmp_path, path)             # atomic publish
        tmp_path = None
        dir_fd = os.open(directory, os.O_RDONLY | os.O_DIRECTORY)
        try:
            os.fsync(dir_fd)                   # CF005-WRITER-STATUS
        finally:
            os.close(dir_fd)
        return 0
    except OSError as exc:
        print("WRITER-FAIL %s" % exc, file=sys.stderr)
        return 4
    finally:
        if tmp_path is not None:
            try:
                os.unlink(tmp_path)
            except OSError:
                pass


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="node manifest identity and writer")
    sub = parser.add_subparsers(dest="command", required=True)

    ident = sub.add_parser("identity-check")
    ident.add_argument("manifest")
    ident.add_argument("--node-id", required=True)
    ident.add_argument("--project", required=True)

    write = sub.add_parser("write")
    write.add_argument("manifest")
    write.add_argument("--node-id", required=True)
    write.add_argument("--project", required=True)
    write.add_argument("--harness", required=True)
    write.add_argument("--version", default="unknown")
    write.add_argument("--port", type=int, default=10800)
    write.add_argument("--repo", default="")
    write.add_argument("--ref", default="main")
    write.add_argument("--bus-agent-id", default="")
    write.add_argument("--bus-base", default="")
    write.add_argument("--repo-path", default=os.path.expanduser("~/repo"))

    args = parser.parse_args(argv)

    if args.command == "identity-check":
        return _identity_matches(args.manifest, args.node_id, args.project)

    payload = {
        "node_id": args.node_id,
        "project": args.project,
        "repo": {"origin": args.repo, "ref": args.ref, "path": os.path.abspath(args.repo_path)},
        "harness": {"name": args.harness, "version": args.version, "port": args.port},
        "bus": {"agent_id": args.bus_agent_id or args.node_id, "base": args.bus_base},
        "created_by": CREATED_BY,
        "created_at": __import__("time").strftime("%Y-%m-%dT%H:%M:%SZ", __import__("time").gmtime()),
    }
    return _atomic_write(args.manifest, payload)


if __name__ == "__main__":
    raise SystemExit(main())

# The Git-origin grammar is deliberately narrow and documented in
# specs/NODE.md#git-origin-grammar. cf. tests/node-mutations.py.
repo_origin_ok() {
  case "$1" in
    ''|*' '*|*'..'*|*';'*|*'|'*|*'&'*|*'$'*|*'`'*|*'"'*|*"'"*|*'<'*|*'>'*|*'\\'*|*$'\n'*|*$'\t'*) return 1 ;;
  esac
  [[ "$1" =~ ^git@[A-Za-z0-9.-]+:[A-Za-z0-9._-]+(/[A-Za-z0-9._-]+)*\.git$ ]] && return 0
  [[ "$1" =~ ^https://[A-Za-z0-9.-]+(:[0-9]+)?/[A-Za-z0-9._-]+(/[A-Za-z0-9._-]+)*\.git$ ]] && return 0
  [[ "$1" =~ ^ssh://git@[A-Za-z0-9.-]+(:[0-9]+)?/[A-Za-z0-9._-]+(/[A-Za-z0-9._-]+)*\.git$ ]] && return 0
  return 1
}
