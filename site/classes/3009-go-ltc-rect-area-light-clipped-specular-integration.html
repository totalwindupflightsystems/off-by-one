LTC Rectangular Area-Light Integration (Go)

Problem class: go-ltc-rect-area-light-clipped-specular-integration · Go 1.26

1. What the routine computes

Under LTC (Heitz et al., SIGGRAPH 2016) the specular term is

L_o = L_e * ∫_A f(ω_i,ω_o) cosθ_i dω_i  ≈  L_e * ∫_A D_o(ω_i) dω_i

The 4-parameter matrix M is fitted so D_o matches the GGX lobe × cosine; the routine returns J = ∫_A D_o dω. Substituting ω = M u / |M u| (so dω = |det M|/|M u|³ du) gives

J = (1/π) ∫_{M⁻¹(A)} max(0, u·ẑ) du .

M⁻¹(A) is the convex polygon with vertices q_i = Minv·R·(p_i − P), where R = rows(T1,T2,N). Since u ∝ q, the horizon test u·ẑ ≥ 0 is just q_i.z ≥ 0. Stokes' theorem turns the projected solid angle into the exact edge sum, so the whole thing is

J = (1/2π) Σ_i α_i (n_i·ẑ),   α_i = ∠(q_i,q_{i+1}),  n_i = normalize(q_i × q_{i+1})

over the horizon-clipped transformed polygon.

2. Root-cause analysis

Reference values at grazing angles are missed unless all of these are right:

  1. Horizon clip must be in transformed space at z = 0 (Minv space). Clipping in world space is only correct when Minv = I; the true run-plane is row₃(Minv·R)·(x−P) = 0.
  2. Apply the (T1,T2,N) rotation before Minv. Invisible when V = N, large otherwise.
  3. Clip between transform and edge sum. Otherwise the radial projection wraps the sphere and arcs cancel incorrectly.
  4. Use atan2(|a×b|, a·b), not acos(dot) — acos loses precision near dot ≈ 1, exactly the stretched edge-on case.
  5. Correct normalisation: Σ α(n·ẑ)/(2π).
  6. V ∥ N tangent fallback to avoid 0/0.
  7. Interpolate da/(da−db) only when the edge crosses z=0, which keeps the denominator bounded.

3. The fix (ltc.go)

package ltc

import "math"

type Vec3 struct{ X, Y, Z float64 }

func (a Vec3) Add(b Vec3) Vec3          { return Vec3{a.X + b.X, a.Y + b.Y, a.Z + b.Z} }
func (a Vec3) Sub(b Vec3) Vec3          { return Vec3{a.X - b.X, a.Y - b.Y, a.Z - b.Z} }
func (a Vec3) Scale(s float64) Vec3     { return Vec3{a.X * s, a.Y * s, a.Z * s} }
func (a Vec3) Mul(b Vec3) Vec3          { return Vec3{a.X * b.X, a.Y * b.Y, a.Z * b.Z} }
func (a Vec3) Dot(b Vec3) float64       { return a.X*b.X + a.Y*b.Y + a.Z*b.Z }
func (a Vec3) Cross(b Vec3) Vec3 {
    return Vec3{a.Y*b.Z - a.Z*b.Y, a.Z*b.X - a.X*b.Z, a.X*b.Y - a.Y*b.X}
}
func (a Vec3) LengthSq() float64 { return a.Dot(a) }
func (a Vec3) Length() float64   { return math.Sqrt(a.Dot(a)) }
func (a Vec3) Normalize() Vec3 {
    l := a.Length()
    if l == 0 { return a }
    return a.Scale(1 / l)
}
func (a Vec3) Neg() Vec3 { return Vec3{-a.X, -a.Y, -a.Z} }

type Mat3 [3][3]float64

var Identity3 = Mat3{{1, 0, 0}, {0, 1, 0}, {0, 0, 1}}

func (m Mat3) Mul(n Mat3) Mat3 {
    var r Mat3
    for i := 0; i < 3; i++ {
        for j := 0; j < 3; j++ {
            r[i][j] = m[i][0]*n[0][j] + m[i][1]*n[1][j] + m[i][2]*n[2][j]
        }
    }
    return r
}
func (m Mat3) MulVec(v Vec3) Vec3 {
    return Vec3{
        m[0][0]*v.X + m[0][1]*v.Y + m[0][2]*v.Z,
        m[1][0]*v.X + m[1][1]*v.Y + m[1][2]*v.Z,
        m[2][0]*v.X + m[2][1]*v.Y + m[2][2]*v.Z,
    }
}
func (m Mat3) Transpose() Mat3 {
    return Mat3{{m[0][0], m[1][0], m[2][0]},
        {m[0][1], m[1][1], m[2][1]},
        {m[0][2], m[1][2], m[2][2]}}
}
func (m Mat3) Det() float64 {
    return m[0][0]*(m[1][1]*m[2][2]-m[1][2]*m[2][1]) -
        m[0][1]*(m[1][0]*m[2][2]-m[1][2]*m[2][0]) +
        m[0][2]*(m[1][0]*m[2][1]-m[1][1]*m[2][0])
}
func (m Mat3) Inverse() Mat3 {
    det := m.Det()
    if det == 0 { return Mat3{} }
    invDet := 1 / det
    c00 := m[1][1]*m[2][2] - m[1][2]*m[2][1]
    c01 := m[0][2]*m[2][1] - m[0][1]*m[2][2]
    c02 := m[0][1]*m[1][2] - m[0][2]*m[1][1]
    c10 := m[1][2]*m[2][0] - m[1][0]*m[2][2]
    c11 := m[0][0]*m[2][2] - m[0][2]*m[2][0]
    c12 := m[0][2]*m[1][0] - m[0][0]*m[1][2]
    c20 := m[1][0]*m[2][1] - m[1][1]*m[2][0]
    c21 := m[0][1]*m[2][0] - m[0][0]*m[2][1]
    c22 := m[0][0]*m[1][1] - m[0][1]*m[1][0]
    return Mat3{
        {c00 * invDet, c01 * invDet, c02 * invDet},
        {c10 * invDet, c11 * invDet, c12 * invDet},
        {c20 * invDet, c21 * invDet, c22 * invDet},
    }
}

// OrthoBasis builds the LTC frame (T1,T2,N); falls back when V is parallel to N.
func OrthoBasis(N, V Vec3) (T1, T2 Vec3) {
    T1 = V.Sub(N.Scale(V.Dot(N)))
    if T1.LengthSq() < 1e-24 {
        helper := Vec3{0, 0, 1}
        if math.Abs(N.Z) > 0.9 {
            helper = Vec3{1, 0, 0}
        }
        T1 = helper.Sub(N.Scale(helper.Dot(N)))
    }
    T1 = T1.Normalize()
    T2 = N.Cross(T1)
    return T1, T2
}

// ClipHorizon clips a convex polygon against z >= 0 (Sutherland–Hodgman).
func ClipHorizon(poly []Vec3) []Vec3 {
    if len(poly) == 0 { return nil }
    out := make([]Vec3, len(poly)+2)
    n := clipHorizonInto(out, poly)
    return out[:n]
}

// clipHorizonInto writes clipped vertices into dst (len(dst) >= len(src)+2); no allocation.
func clipHorizonInto(dst, src []Vec3) int {
    n := len(src)
    m := 0
    for i := 0; i < n; i++ {
        a := src[i]
        b := src[(i+1)%n]
        da := a.Z
        db := b.Z
        inA := da >= 0
        inB := db >= 0
        if inA {
            dst[m] = a
            m++
        }
        if inA != inB {
            t := da / (da - db)
            dst[m] = a.Add(b.Sub(a).Scale(t))
            m++
        }
    }
    return m
}

// IntegrateSphericalPolygon = (1/2π) Σ α_i (n_i·ẑ) over the upper-hemisphere polygon.
func IntegrateSphericalPolygon(poly []Vec3) float64 {
    n := len(poly)
    if n < 3 { return 0 }
    sum := 0.0
    for i := 0; i < n; i++ {
        a := poly[i].Normalize()
        b := poly[(i+1)%n].Normalize()
        cross := a.Cross(b)
        s := cross.Length()
        alpha := math.Atan2(s, a.Dot(b)) // stable angle in [0,π]
        if s > 0 {
            sum += alpha * (cross.Z / s)
        }
    }
    return sum / (2 * math.Pi)
}

// IntegrateRect returns J = ∫_A D_o(ω) dω. rect is [4]Vec3 in order.
func IntegrateRect(N, V, P Vec3, Minv Mat3, rect [4]Vec3) float64 {
    N = N.Normalize()
    V = V.Normalize()
    T1, T2 := OrthoBasis(N, V)

    R := Mat3{
        {T1.X, T1.Y, T1.Z},
        {T2.X, T2.Y, T2.Z},
        {N.X, N.Y, N.Z},
    }
    A := Minv.Mul(R) // compose rotation and inverse LTC matrix once

    var raw [4]Vec3
    for i, c := range rect {
        raw[i] = A.MulVec(c.Sub(P))
    }
    var clipped [6]Vec3 // a convex quad cut by one plane has at most 5 vertices
    n := clipHorizonInto(clipped[:], raw[:])
    return IntegrateSphericalPolygon(clipped[:n])
}

// IntegrateRectOneSided rejects emitters facing away from P in world space.
func IntegrateRectOneSided(N, V, P Vec3, Minv Mat3, rect [4]Vec3, twoSided bool) float64 {
    if !twoSided {
        lightN := rect[1].Sub(rect[0]).Cross(rect[3].Sub(rect[0]))
        if lightN.Dot(P.Sub(rect[0])) < 0 {
            return 0
        }
    }
    return IntegrateRect(N, V, P, Minv, rect)
}

LTCMatrix(a,b,c,d) builds the standard [[a,0,b],[0,1,0],[c,0,d]] fit matrix; LTCDensity and GGXSpecularCos are included in the repo for the verification harness.

4. Verification

The independent reference integrates the pointwise density over the rectangle's area,

J = ∫_A D_o(ω(x))·|cosθ_light|/|x−P|² dA,
D_o(ω) = max(0,(Minv R ω)·ẑ)/(|det M|·|Minv R ω|³),

with adaptive Gauss–Legendre, after splitting the rectangle along its horizon plane so the max(0,·) kink lies on a triangle edge. It shares no code with the edge-sum routine.
