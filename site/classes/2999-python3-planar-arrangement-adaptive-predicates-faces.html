Planar Arrangement of Integer Segments with Exact Adaptive Predicates

Reference solution for python3-planar-arrangement-adaptive-predicates-faces. Builds the full arrangement of integer line segments (split at every proper and improper intersection, merge coincident vertices), extracts bounded and unbounded faces from a DCEL half-edge structure, validates V - E + F, and separately reports segments that only touch at endpoints.

1. Root-cause analysis

  1. Inexact orientation corrupts topology. A float/atan2 orientation makes wrong left/right decisions for nearly collinear triples. One flipped sign drops an intersection or mis-sorts half-edges, so next/prev links form wrong cycles and faces self-intersect. Exactness must be guaranteed, not merely improved.
  2. Numeric intersection points are not canonical. The same geometric point computed by two float formulas becomes two vertices, disconnecting the graph and breaking Euler. Points must be exact (rationals) and merged by exact equality.
  3. Improper intersections are dropped. Collinear-overlap endpoints and endpoint/T contacts fail the classic d1*d2<0 and d3*d4<0 test, yet they must still be split points, otherwise overlapping sub-edges and duplicate edges survive.
  4. Duplicate/zero-length edges. Overlapping collinear inputs generate the same sub-edge repeatedly; without de-duplicating undirected edges the DCEL gets parallel half-edges and the angular sort has equal keys.
  5. Wrong next convention. Taking next(h) as the edge counter-clockwise from twin(h) yields clockwise bounded faces and breaks hole/left-side logic. It must be the edge immediately clockwise from twin(h).
  6. Face count on disconnected graphs and holes. A half-edge cycle is a boundary component, not a face. For C components the outer face has C cycles; a face with k holes has 1+k cycles. The correct rule: every bounded face has exactly one positive-area (CCW) cycle, so F = (#CCW cycles) + 1, with negative cycles attached as holes to the smallest containing CCW cycle or to the unbounded face. Euler generalizes to V - E + F = 1 + C.
  7. Endpoint-only touches misclassified. Segments sharing a single endpoint create a vertex without a topological crossing; they must be reported separately and must not add spurious split points.

from __future__ import annotations
from fractions import Fraction
from functools import cmp_to_key
from typing import Dict, List, Sequence, Set, Tuple

Point = Tuple[Fraction, Fraction]
Segment = Tuple[Point, Point]

_EPS = 2.0 ** -53
_CCW_ERRBOUND = (3.0 + 16.0 * _EPS) * _EPS
_INCIRCLE_ERRBOUND = (10.0 + 96.0 * _EPS) * _EPS

def orient2d(pa, pb, pc) -> int:
    """Sign of signed area of triangle (pa,pb,pc). Adaptive; exact fallback."""
    ax, ay = float(pa[0]), float(pa[1])
    bx, by = float(pb[0]), float(pb[1])
    cx, cy = float(pc[0]), float(pc[1])
    detleft = (ax - cx) * (by - cy)
    detright = (ay - cy) * (bx - cx)
    det = detleft - detright
    if detleft > 0.0:
        if detright <= 0.0:
            return 1 if det > 0.0 else (-1 if det < 0.0 else 0)
        detsum = detleft + detright
    elif detleft < 0.0:
        if detright >= 0.0:
            return 1 if det > 0.0 else (-1 if det < 0.0 else 0)
        detsum = -detleft - detright
    else:
        return 1 if det > 0.0 else (-1 if det < 0.0 else 0)
    errbound = _CCW_ERRBOUND * detsum
    if det > errbound:
        return 1
    if det < -errbound:
        return -1
    d = (pa[0]-pc[0])*(pb[1]-pc[1]) - (pa[1]-pc[1])*(pb[0]-pc[0])
    return 1 if d > 0 else (-1 if d < 0 else 0)

def incircle(pa, pb, pc, pd) -> int:
    """>0 inside, <0 outside, 0 cocircular (pa,pb,pc CCW). Adaptive; exact."""
    ax, ay = float(pa[0]), float(pa[1])
    bx, by = float(pb[0]), float(pb[1])
    cx, cy = float(pc[0]), float(pc[1])
    dx, dy = float(pd[0]), float(pd[1])
    adx, ady = ax-dx, ay-dy
    bdx, bdy = bx-dx, by-dy
    cdx, cdy = cx-dx, cy-dy
    bdxcdy, cdxbdy = bdx*cdy, cdx*bdy
    alift = adx*adx + ady*ady
    cdxady, adxcdy = cdx*ady, adx*cdy
    blift = bdx*bdx + bdy*bdy
    adxbdy, bdxady = adx*bdy, bdx*ady
    clift = cdx*cdx + cdy*cdy
    det = (alift*(bdxcdy-cdxbdy) + blift*(cdxady-adxcdy)
           + clift*(adxbdy-bdxady))
    permanent = ((abs(bdxcdy)+abs(cdxbdy))*alift
                 + (abs(cdxady)+abs(adxcdy))*blift
                 + (abs(adxbdy)+abs(bdxady))*clift)
    errbound = _INCIRCLE_ERRBOUND * permanent
    if det > errbound:
        return 1
    if det < -errbound:
        return -1
    return _incircle_exact(pa, pb, pc, pd)

def _incircle_exact(pa, pb, pc, pd) -> int:
    adx, ady = pa[0]-pd[0], pa[1]-pd[1]
    bdx, bdy = pb[0]-pd[0], pb[1]-pd[1]
    cdx, cdy = pc[0]-pd[0], pc[1]-pd[1]
    alift = adx*adx + ady*ady
    blift = bdx*bdx + bdy*bdy
    clift = cdx*cdx + cdy*cdy
    det = (alift*(bdx*cdy-cdx*bdy) + blift*(cdx*ady-adx*cdy)
           + clift*(adx*bdy-bdx*ady))
    return 1 if det > 0 else (-1 if det < 0 else 0)

def _on_segment(p, a, b) -> bool:
    if orient2d(a, b, p) != 0:
        return False
    return (min(a[0], b[0]) <= p[0] <= max(a[0], b[0])
            and min(a[1], b[1]) <= p[1] <= max(a[1], b[1]))

def _line_intersection(a, b, c, d):
    rx, ry = b[0]-a[0], b[1]-a[1]
    sx, sy = d[0]-c[0], d[1]-c[1]
    denom = rx*sy - ry*sx
    if denom == 0:
        return None
    qpx, qpy = c[0]-a[0], c[1]-a[1]
    t = (qpx*sy - qpy*sx) / denom
    u = (qpx*ry - qpy*rx) / denom
    if 0 <= t <= 1 and 0 <= u <= 1:
        return (a[0] + t*rx, a[1] + t*ry)
    return None

def segment_relation(a, b, c, d):
    """('none'|'proper'|'endpoint'|'t_junction'|'overlap'|'collinear_touch', points)."""
    o1 = orient2d(a, b, c); o2 = orient2d(a, b, d)
    o3 = orient2d(c, d, a); o4 = orient2d(c, d, b)
    if o1 == 0 and o2 == 0:
        pts = set()
        for p in (a, b):
            if _on_segment(p, c, d): pts.add(p)
        for p in (c, d):
            if _on_segment(p, a, b): pts.add(p)
        if len(pts) >= 2: return "overlap", pts
        if len(pts) == 1: return "collinear_touch", pts
        return "none", pts
    if o1 * o2 < 0 and o3 * o4 < 0:
        p = _line_intersection(a, b, c, d)
        return "proper", ({p} if p is not None else set())
    pts = set()
    if o3 == 0 and _on_segment(a, c, d): pts.add(a)
    if o4 == 0 and _on_segment(b, c, d): pts.add(b)
    if o1 == 0 and _on_segment(c, a, b): pts.add(c)
    if o2 == 0 and _on_segment(d, a, b): pts.add(d)
    if not pts: return "none", pts
    if len(pts) > 1: return "overlap", pts
    p = next(iter(pts))
    if p in {a, b} and p in {c, d}: return "endpoint", pts
    return "t_junction", pts

class HalfEdge:
    __slots__ = ("origin", "target", "twin", "next", "prev")
    def __init__(self, origin, target):
        self.origin = origin; self.target = target
        self.twin = self.next = self.prev = -1

def _angle_cmp(d1, d2) -> int:
    def half(d): return 0 if (d[1] > 0 or (d[1] == 0 and d[0] > 0)) else 1
    h1, h2 = half(d1), half(d2)
    if h1 != h2: return -1 if h1 < h2 else 1
    cross = d1[0]*d2[1] - d1[1]*d2[0]
    return -1 if cross > 0 else (1 if cross < 0 else 0)

def build_arrangement(segments):
    segs = [((Fraction(a[0]), Fraction(a[1])), (Fraction(b[0]), Fraction(b[1])))
            for a, b in segments]
    n = len(segs)
    candidates = [set() for _ in range(n)]
    for i in range(n):
        candidates[i].add(segs[i][0]); candidates[i].add(segs[i][1])
    relation_report = []
    for i in range(n):
        for j in range(i+1, n):
            a, b = segs[i]; c, d = segs[j]
            kind, pts = segment_relation(a, b, c, d)
            if kind == "none": continue
            for p in pts:
                candidates[i].add(p); candidates[j].add(p)
            relation_report.append({"i": i, "j": j, "kind": kind, "points": pts})
    point_to_id, points, edges = {}, [], set()
    def get_id(p):
        idx = point_to_id.get(p)
        if idx is None:
            idx = len(points); point_to_id[p] = idx; points.append(p)
        return idx
    for i, (a, b) in enumerate(segs):
        if a[0] != b[0]:
            def key(p, a=a, b=b): return (p[0]-a[0])/(b[0]-a[0])
        else:
            def key(p, a=a, b=b): return (p[1]-a[1])/(b[1]-a[1])
        ordered = sorted(candidates[i], key=key)
        uniq = []
        for p in ordered:
            if not uniq or p != uniq[-1]: uniq.append(p)
        for p, q in zip(uniq, uniq[1:]):
            if p == q: continue
            u, v = get_id(p), get_id(q)
            if u != v: edges.add((u, v) if u < v else (v, u))
    unique_edges = sorted(edges)
    ids_in_use = set()
    for u, v in unique_edges:
        ids_in_use.add(u); ids_in_use.add(v)
    remap, compact_points = {}, []
    for old in sorted(ids_in_use):
        remap[old] = len(compact_points); compact_points.append(points[old])
    points = compact_points
    unique_edges = [(remap[u], remap[v]) for u, v in unique_edges]
    half_edges, outgoing = [], {}
    for u, v in unique_edges:
        h1 = len(half_edges); half_edges.append(HalfEdge(u, v))
        h2 = len(half_edges); half_edges.append(HalfEdge(v, u))
        half_edges[h1].twin = h2; half_edges[h2].twin = h1
        outgoing.setdefault(u, []).append(h1)
        outgoing.setdefault(v, []).append(h2)
    for v, he_list in outgoing.items():
        vx, vy = points[v]
        def cmp_he(h1, h2, vx=vx, vy=vy):
            t1 = half_edges[h1].target; t2 = half_edges[h2].target
            return _angle_cmp((points[t1][0]-vx, points[t1][1]-vy),
                              (points[t2][0]-vx, points[t2][1]-vy))
        he_list.sort(key=cmp_to_key(cmp_he))
    pos_in_out = {}
    for v, he_list in outgoing.items():
        for k, h in enumerate(he_list): pos_in_out[h] = k
    for h, he in enumerate(half_edges):
        he_list = outgoing[he.target]
        k = pos_in_out[he.twin]
        nxt = he_list[(k - 1) % len(he_list)]
        he.next = nxt; half_edges[nxt].prev = h
    visited, cycles = [False]*len(half_edges), []
    for h in range(len(half_edges)):
        if visited[h]: continue
        cyc, cur = [], h
        while not visited[cur]:
            visited[cur] = True; cyc.append(cur); cur = half_edges[cur].next
        cycles.append(cyc)
    cycle_info = []
    for cyc in cycles:
        area2 = Fraction(0); poly = []
        for h in cyc:
            p = points[half_edges[h].origin]; q = points[half_edges[h].target]
            area2 += p[0]*q[1] - p[1]*q[0]; poly.append(p)
        cycle_info.append({"cycle": cyc, "area2": area2, "poly": poly})
    positive = [ci for ci in cycle_info if ci["area2"] > 0]
    negative = [ci for ci in cycle_info if ci["area2"] < 0]
    for ci in positive: ci["holes"] = []
    unbounded_holes = []
    for ci in negative:
        rep = ci["poly"][0]; best = best_area = None
        for pc in positive:
            if _point_in_polygon(rep, pc["poly"]) == 1:
                a = abs(pc["area2"])
                if best is None or a < best_area: best, best_area = pc, a
        if best is None: unbounded_holes.append(ci)
        else: best["holes"].append(ci)
    faces = []
    for pc in positive:
        faces.append({"outer": pc["poly"], "holes": [h["poly"] for h in pc["holes"]],
                      "area2": pc["area2"], "bounded": True})
    unbounded_poly = [h["poly"] for h in unbounded_holes]
    faces.append({"outer": unbounded_poly[0] if unbounded_poly else [],
                  "holes": unbounded_poly[1:], "area2": None, "bounded": False})
    parent = list(range(len(points)))
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]; x = parent[x]
        return x
    def union(x, y):
        rx, ry = find(x), find(y)
        if rx != ry: parent[rx] = ry
    for u, v in unique_edges: union(u, v)
    comps = len({find(i) for i in range(len(points))}) if points else 0
    V, E, F = len(points), len(unique_edges), len(positive) + 1
    euler, expected = V - E + F, 1 + comps
    def pick(k): return [r for r in relation_report if r["kind"] == k]
    return {"vertices": points, "edges": unique_edges, "half_edges": half_edges,
            "faces": faces, "num_faces": F, "num_bounded_faces": len(positive),
            "components": comps, "euler": euler, "euler_expected": expected,
            "euler_ok": euler == expected, "relation_report": relation_report,
            "proper_crossings": pick("proper"), "endpoint_touches": pick("endpoint"),
            "collinear_endpoint_touches": pick("collinear_touch"),
            "t_junctions": pick("t_junction"), "overlaps": pick("overlap")}

def _point_in_polygon(p, poly) -> int:
    """1 inside, -1 outside, 0 on boundary (exact winding number)."""
    wn = 0; n = len(poly)
    for i in range(n):
        a, b = poly[i], poly[(i+1) % n]
        if _on_segment(p, a, b): return 0
        if a[1] <= p[1]:
            if b[1] > p[1] and orient2d(a, b, p) > 0: wn += 1
        else:
            if b[1] <= p[1] and orient2d(a, b, p) < 0: wn -= 1
    return 1 if wn != 0 else -1
