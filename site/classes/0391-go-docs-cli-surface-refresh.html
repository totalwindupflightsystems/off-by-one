The memoryd CLI is cobra-based. Three surface defects were fixed: phantom commands documented in the README (mcp, import, bench), a wrong default port (8090) in the README, and a status command whose default port (9234) diverged from serve (8080).

1. Ground truth command inventory (not README prose). Commands were enumerated two ways because a bare Use:-string grep misses commands registered programmatically:

3. status default port aligned to 8080, keeping the MEMORYD_STATUS_PORT env override and adding explicit precedence (flag > env > default):

// cmd/status.go
const defaultStatusPort = 8080

func newStatusCmd() *cobra.Command {
    var port int
    cmd := &cobra.Command{
        Use:   "status",
        Short: "Show memoryd service health",
        RunE: func(cmd *cobra.Command, _ []string) error {
            if !cmd.Flags().Changed("port") {
                if v := os.Getenv("MEMORYD_STATUS_PORT"); v != "" {
                    p, err := strconv.Atoi(v)
                    if err != nil {
                        return fmt.Errorf("invalid MEMORYD_STATUS_PORT %q: %w", v, err)
                    }
                    port = p
                } else {
                    port = defaultStatusPort
                }
            }
            if port < 1 || port > 65535 {
                return fmt.Errorf("status port %d out of range 1-65535", port)
            }
            // probe http://127.0.0.1:%d/healthz
            return nil
        },
    }
    cmd.Flags().IntVar(&port, "port", defaultStatusPort,
        "status port (flag overrides MEMORYD_STATUS_PORT)")
    return cmd
}
