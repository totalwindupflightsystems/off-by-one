All gates are already green, so the correct "fix" is a guard that short-circuits before any diff, PUT, or release is attempted. The concrete artifact of an idle tick is that logic, e.g.:

// idle_tick.go — tick-107 gate battery, guard PASS full mode.
type Health struct {
    Build, Vet, Fmt, Gitleaks bool
    TestsPassed               int  // 10/10
    ReleaseHumanGated         bool // REL-003
    CooldownSec               int  // 900 => no PUT
}

// IdleTick reports whether the maintenance audit may make zero changes.
// Decision rule (guard PASS full mode):
//   any red gate       -> NOT idle, action required
//   release not gated  -> NOT idle, hold (human gate mandatory)
//   cooldown > 0       -> idle: run read-only audit, emit report, no PUT
func IdleTick(h Health) bool {
    if !(h.Build && h.Vet && h.Fmt && h.Gitleaks && h.TestsPassed >= 10) {
        return false // gate battery failed
    }
    if !h.ReleaseHumanGated {
        return false // never auto-trigger REL-003
    }
    return h.CooldownSec > 0 // quiet period: observe, verify, do not mutate
}

// OffByOne reports whether an E2E run index falls inside the next window.
// last=103, next window 108..113 => 104..107 are NOT in window (no overlap, no re-run).
func OffByOne(last, windowLo, windowHi int) bool { return last >= windowLo && last <= windowHi }
