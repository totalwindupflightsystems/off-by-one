Dynamic Connectivity with Parity Constraints — Offline Rollback-DSU Solution

Problem

We are given an undirected graph whose edge set changes over time. Interleaved operations are:

  • + u v w — activate edge (u,v) requiring x[u] XOR x[v] = w
  • - u v — deactivate that edge
  • ? u v — are u,v connected? Is the whole component of u consistent (i.e. every active edge satisfies a common XOR-potential assignment)?

Additionally, after each update we must print the number of contradictory components (components containing at least one parity conflict). n, m ≤ 2·10^5.

Root-Cause Analysis

A naive online solution fails because edges get deleted, and a DSU cannot undo a union. The standard remedy is offline:

  1. Segment tree over time. Each edge is active on a contiguous interval of operation indices [l, r). Insert it into the O(log m) canonical nodes of the segment tree. Then an Euler DFS applies an edge on the way down and rolls it back on the way up. Every failure mode below is a direct consequence of this design.

  2. Rollback parity DSU. Each node stores par[x] = x[x] XOR x[parent[x]]. find(x) walks to the root accumulating XOR. Path compression is forbidden — it is not rollback-safe. Union by size keeps the tree height O(log n), so find is O(log n).

  3. Rollback of the contradiction count. This is the subtle part. A component is "bad" iff it contains a parity contradiction. We keep a global badCount plus a bad[root] flag, and the update rules differ by case:

  4. Same root, new edge conflicts, component was good → badCount++, mark root bad.
  5. Same root, already bad → no change.
  6. Merge two good roots → no change.
  7. Merge a bad root with a good root → count unchanged (one bad component survives).
  8. Merge two bad roots → badCount-- (two bad components fuse into one).

Every mutation stores the old badCount and old bad/size/parent fields on a stack, so rollback restores the exact prior state. Forgetting the badCount-- on the double-bad merge, or restoring the count by ±delta instead of the stored snapshot, are the classic wrong answers.

  1. Time-index semantics. An add at index i must be visible to the post-update state at i, while a query at i must see all adds with index < i and no removes with index ≤ i. Because an index is either an update or a query, using the half-open interval [i, removalIndex) gives exactly the right active set at every leaf.

  2. Union by size vs. parity direction. XOR is symmetric, so swapping the two roots when enforcing union-by-size does not change the required root-to-root parity want = pu ^ pv ^ w.

Complexity

  • Time: O((m log m) · log n) (edge insertions × find cost), easily within limits.
  • Memory: O(m log m + n) for the segment-tree buckets.

Exact Fix (complete Go solution)

// Offline dynamic connectivity with parity (XOR) constraints.
//
// Operations (1-indexed vertices):
//   + u v w   add edge (u,v) with x[u] XOR x[v] = w
//   - u v     remove that active edge
//   ? u v     query connectivity and component consistency
//
// Output:
//   U <bad>          after every + / - update
//   Q <conn> <cons>  for every query
package main

import (
    "bufio"
    "fmt"
    "os"
)

type Edge struct{ u, v, w int }

type Change struct {
    typ       int // 0 = union, 1 = mark component bad
    child     int
    parent    int
    parentOld int
    oldBad    bool
    oldBadCnt int
}

type DSU struct {
    parent   []int
    sz       []int
    par      []int // par[x] = x[x] xor x[parent[x]]
    bad      []bool
    badCount int
    stack    []Change
}

func NewDSU(n int) *DSU {
    d := &DSU{parent: make([]int, n), sz: make([]int, n),
        par: make([]int, n), bad: make([]bool, n)}
    for i := 0; i < n; i++ {
        d.parent[i] = i
        d.sz[i] = 1
    }
    return d
}

// Compression-free so the operation is reversible.
func (d *DSU) find(x int) (int, int) {
    p := 0
    for d.parent[x] != x {
        p ^= d.par[x]
        x = d.parent[x]
    }
    return x, p
}

func (d *DSU) snapshot() int { return len(d.stack) }

func (d *DSU) union(u, v, w int) {
    ru, pu := d.find(u)
    rv, pv := d.find(v)
    want := pu ^ pv ^ w // required parity x[ru] xor x[rv]

    if ru == rv {
        if want != 0 && !d.bad[ru] {
            d.bad[ru] = true
            d.badCount++
            d.stack = append(d.stack, Change{typ: 1, child: ru, oldBadCnt: d.badCount - 1})
        }
        return
    }

    if d.sz[ru] < d.sz[rv] { // XOR is symmetric, so `want` stays valid
        ru, rv = rv, ru
    }
    oldBad, oldBadCnt := d.bad[ru], d.badCount
    if d.bad[ru] && d.bad[rv] {
        d.badCount-- // two bad components merge into one
    }
    d.bad[ru] = d.bad[ru] || d.bad[rv]

    d.parent[rv] = ru
    d.par[rv] = want
    parentOld := d.sz[ru]
    d.sz[ru] += d.sz[rv]

    d.stack = append(d.stack, Change{typ: 0, child: rv, parent: ru,
        parentOld: parentOld, oldBad: oldBad, oldBadCnt: oldBadCnt})
}

func (d *DSU) rollback(mark int) {
    for len(d.stack) > mark {
        c := d.stack[len(d.stack)-1]
        d.stack = d.stack[:len(d.stack)-1]
        if c.typ == 1 {
            d.bad[c.child] = false
            d.badCount = c.oldBadCnt
        } else {
            d.parent[c.child] = c.child
            d.par[c.child] = 0
            d.sz[c.parent] = c.parentOld
            d.bad[c.parent] = c.oldBad
            d.badCount = c.oldBadCnt
        }
    }
}

type Query struct{ u, v int }

type SegTree struct {
    size int
    tree [][]Edge
}

func NewSegTree(n int) *SegTree {
    size := 1
    for size < n {
        size <<= 1
    }
    return &SegTree{size: size, tree: make([][]Edge, 2*size)}
}

func (s *SegTree) add(l, r int, e Edge) { s.addRec(1, 0, s.size, l, r, e) }

func (s *SegTree) addRec(node, nl, nr, l, r int, e Edge) {
    if r <= nl || nr <= l {
        return
    }
    if l <= nl && nr <= r {
        s.tree[node] = append(s.tree[node], e)
        return
    }
    mid := (nl + nr) / 2
    s.addRec(2*node, nl, mid, l, r, e)
    s.addRec(2*node+1, mid, nr, l, r, e)
}

type edgeInfo struct{ start, w int }

func main() {
    in := bufio.NewReaderSize(os.Stdin, 1<<20)
    out := bufio.NewWriterSize(os.Stdout, 1<<20)

    var n, m int
    if _, err := fmt.Fscan(in, &n, &m); err != nil {
        return
    }

    queriesAt := make([][]Query, m)
    isUpdate := make([]bool, m)
    type key struct{ u, v int }
    active := map[key]edgeInfo{}
    seg := NewSegTree(m)

    for i := 0; i < m; i++ {
        var op string
        fmt.Fscan(in, &op)
        switch op[0] {
        case '+':
            var u, v, w int
            fmt.Fscan(in, &u, &v, &w)
            u--
            v--
            k := key{u, v}
            if k.u > k.v {
                k.u, k.v = k.v, k.u
            }
            active[k] = edgeInfo{start: i, w: w}
            isUpdate[i] = true
        case '-':
            var u, v int
            fmt.Fscan(in, &u, &v)
            u--
            v--
            k := key{u, v}
            if k.u > k.v {
                k.u, k.v = k.v, k.u
            }
            info := active[k]
            delete(active, k)
            seg.add(info.start, i, Edge{k.u, k.v, info.w}) // [start, i)
            isUpdate[i] = true
        case '?':
            var u, v int
            fmt.Fscan(in, &u, &v)
            u--
            v--
            queriesAt[i] = append(queriesAt[i], Query{u, v})
        }
    }
    for k, info := range active {
        seg.add(info.start, m, Edge{k.u, k.v, info.w})
    }

    dsu := NewDSU(n)
    buf := make([]byte, 0, 1<<20)
    var tmp [64]byte
    appendInt := func(x int) {
        if x == 0 {
            buf = append(buf, '0')
            return
        }
        if x < 0 {
            buf = append(buf, '-')
            x = -x
        }
        p := len(tmp)
        for x > 0 {
            p--
            tmp[p] = byte('0' + x%10)
            x /= 10
        }
        buf = append(buf, tmp[p:]...)
    }

    var dfs func(node, nl, nr int)
    dfs = func(node, nl, nr int) {
        mark := dsu.snapshot()
        for _, e := range seg.tree[node] {
            dsu.union(e.u, e.v, e.w)
        }
        if nr-nl == 1 {
            t := nl
            if t < m {
                if isUpdate[t] {
                    buf = append(buf, 'U', ' ')
                    appendInt(dsu.badCount)
                    buf = append(buf, '\n')
                }
                for _, q := range queriesAt[t] {
                    ru, _ := dsu.find(q.u)
                    rv, _ := dsu.find(q.v)
                    conn, cons := 0, 0
                    if ru == rv {
                        conn = 1
                        if !dsu.bad[ru] {
                            cons = 1
                        }
                    }
                    buf = append(buf, 'Q', ' ')
                    appendInt(conn)
                    buf = append(buf, ' ')
                    appendInt(cons)
                    buf = append(buf, '\n')
                }
            }
        } else {
            mid := (nl + nr) / 2
            dfs(2*node, nl, mid)
            dfs(2*node+1, mid, nr)
        }
        dsu.rollback(mark)
    }
    dfs(1, 0, seg.size)

    out.Write(buf)
    out.Flush()
}

Key correctness invariants to preserve when porting

  1. Never path-compress the rollback DSU.
  2. Store the previous badCount (not a delta) on every stack entry.
  3. Apply badCount-- when merging two already-bad components.
  4. Use interval [addIndex, removeIndex) with addIndex itself included.
  5. Resolve each query's consistency from the root's bad flag, not from pu == pv.
