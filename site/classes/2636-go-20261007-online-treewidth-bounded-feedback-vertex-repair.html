2. Problem model (formalised)

  • Feedback vertex set (FVS). For undirected G = (V, E), S ⊆ V is an FVS iff G − S is a forest.
  • Treewidth-bounded. The input has treewidth ≤ w and accepts a nice tree decomposition of width w, enabling an exact O(2^w · poly(n)) DP instead of NP-hard general search.
  • Online / repair. Edges arrive one at a time; after each insertion S must remain a valid FVS, repaired locally, and periodically re-optimised with the bounded-width DP.

Nice tree decomposition. Every node has bag B ⊆ V, |B| ≤ w+1, and is Leaf (B=∅), Introduce(v) (B = child.B + {v}), Forget(v) (B = child.B − {v}), or Join (both children bag B). Every vertex's bags form a connected subtree; every edge lies in some bag.

Why naive approaches fail. General FVS is NP-hard, and greedy cycle-hitting has no work bound. Both problems vanish once DP state is restricted to bag-boundary vertices.

3. Root-cause analysis of the algorithm

DP state on bag B: R ⊆ B (boundary vertices placed in the FVS, cost counted) plus a partition P of B − R recording which kept boundary vertices are connected in the processed part of G − R. Since G − R must be a forest, a component is fully described by its vertex set.

Node Rule
Leaf single state (∅, ∅), cost 0
Introduce(v) remove v (cost +1), or keep v (singleton block) and union it with every kept bag-neighbour; a union of already-connected vertices closes a cycle → reject
Forget(v) if v ∈ R, drop from R (cost kept); if kept, delete it from its block
Join require identical R; merge partitions, rejecting iff the merge creates a boundary cycle; cost = c₁ + c₂ − |R|

The Join test is exact: the child forests share only boundary vertices, so any new cycle lies in the union of partitions on B; building a spanning tree per block and rejecting any edge joining two already-connected vertices detects exactly that.

Online repair. In a forest, adding (u,v) creates one cycle: the unique u…v path. The engine BFSes that path in G − S, promotes the highest-degree vertex into S, then removes redundant vertices. Validity is an invariant.

// Package fvs implements exact and online Feedback Vertex Set (FVS) handling
// for undirected graphs, with an exact O(2^w * poly(n)) dynamic program over a
// nice tree decomposition of width w. See solution.md for the problem model.
package fvs

import (
    "sort"
    "strconv"
    "strings"
)

// ---------------------------------------------------------------------------
// Graph
// ---------------------------------------------------------------------------

// Graph is a simple undirected graph on vertices 0..N-1.
type Graph struct {
    N   int
    Adj [][]int
}

// NewGraph allocates a graph on n vertices.
func NewGraph(n int) *Graph {
    adj := make([][]int, n)
    return &Graph{N: n, Adj: adj}
}

// AddEdge inserts an undirected edge (u,v), ignoring self-loops and duplicates.
func (g *Graph) AddEdge(u, v int) {
    if u == v {
        return
    }
    if g.hasEdge(u, v) {
        return
    }
    g.Adj[u] = append(g.Adj[u], v)
    g.Adj[v] = append(g.Adj[v], u)
}

func (g *Graph) hasEdge(u, v int) bool {
    for _, w := range g.Adj[u] {
        if w == v {
            return true
        }
    }
    return false
}

// Edges returns the edge list with u < v.
func (g *Graph) Edges() [][2]int {
    var out [][2]int
    for u := 0; u < g.N; u++ {
        for _, v := range g.Adj[u] {
            if u < v {
                out = append(out, [2]int{u, v})
            }
        }
    }
    return out
}

// IsForest reports whether the induced subgraph on keep (a boolean set, nil
// meaning all vertices) is acyclic. Iterative union-find.
func (g *Graph) IsForest(keep []bool) bool {
    parent := make([]int, g.N)
    for i := range parent {
        parent[i] = i
    }
    var find func(int) int
    find = func(x int) int {
        for parent[x] != x {
            parent[x] = parent[parent[x]]
            x = parent[x]
        }
        return x
    }
    for u := 0; u < g.N; u++ {
        if keep != nil && !keep[u] {
            continue
        }
        for _, v := range g.Adj[u] {
            if u >= v {
                continue
            }
            if keep != nil && !keep[v] {
                continue
            }
            ru, rv := find(u), find(v)
            if ru == rv {
                return false // edge closes a cycle
            }
            parent[ru] = rv
        }
    }
    return true
}

// IsFVS reports whether S (a boolean set) is a feedback vertex set, i.e.
// G - S is a forest.
func (g *Graph) IsFVS(removed []bool) bool {
    keep := make([]bool, g.N)
    for i := range keep {
        keep[i] = !removed[i]
    }
    return g.IsForest(keep)
}

// ---------------------------------------------------------------------------
// Nice tree decomposition
// ---------------------------------------------------------------------------

// Kind enumerates nice tree decomposition node types.
type Kind uint8

const (
    Leaf Kind = iota
    Introduce
    Forget
    Join
)

// NiceNode is a node of a nice tree decomposition.
//
//  Leaf:      Bag == {}
//  Introduce: Bag == child.Bag + {Vertex}
//  Forget:    Bag == child.Bag - {Vertex}
//  Join:      Bag == child[0].Bag == child[1].Bag
//
// Bag must be sorted ascending.
type NiceNode struct {
    Kind     Kind
    Vertex   int
    Bag      []int
    Children []*NiceNode
}

// TrivialNiceTD builds a width-(n-1) nice decomposition valid for every graph:
// introduce 0..n-1, then forget 0..n-1. Useful as a correctness oracle; the
// width is only polynomial when the input is genuinely low-treewidth.
func TrivialNiceTD(n int) *NiceNode {
    cur := &NiceNode{Kind: Leaf, Bag: []int{}}
    for v := 0; v < n; v++ {
        bag := append(append([]int{}, cur.Bag...), v)
        sort.Ints(bag)
        cur = &NiceNode{Kind: Introduce, Vertex: v, Bag: bag, Children: []*NiceNode{cur}}
    }
    for v := 0; v < n; v++ {
        bag := removeSorted(cur.Bag, v)
        cur = &NiceNode{Kind: Forget, Vertex: v, Bag: bag, Children: []*NiceNode{cur}}
    }
    return cur
}

// ---------------------------------------------------------------------------
// Dynamic programming over a nice tree decomposition
// ---------------------------------------------------------------------------

// State key encoding: "R<removed>|P;<block>;<block>...". Every list is sorted
// ascending; blocks are sorted by their smallest element. This canonical form
// lets us compare states with a single map lookup.
func encInts(xs []int) string {
    if len(xs) == 0 {
        return ""
    }
    parts := make([]string, len(xs))
    for i, x := range xs {
        parts[i] = strconv.Itoa(x)
    }
    return strings.Join(parts, ",")
}

func encode(removed []int, parts [][]int) string {
    var b strings.Builder
    b.WriteByte('R')
    b.WriteString(encInts(removed))
    b.WriteString("|P")
    for _, blk := range parts {
        b.WriteByte(';')
        b.WriteString(encInts(blk))
    }
    return b.String()
}

func parseList(s string) []int {
    if s == "" {
        return nil
    }
    fields := strings.Split(s, ",")
    out := make([]int, 0, len(fields))
    for _, f := range fields {
        x, _ := strconv.Atoi(f)
        out = append(out, x)
    }
    return out
}

func decode(k string) ([]int, [][]int) {
    i := strings.IndexByte(k, '|')
    rem := parseList(k[1:i]) // strip leading 'R'
    ps := k[i+2:]            // strip 'P'
    var parts [][]int
    if ps != "" {
        for _, s := range strings.Split(ps, ";") {
            if s == "" {
                continue
            }
            parts = append(parts, parseList(s))
        }
    }
    return rem, parts
}

func canon(parts [][]int) [][]int {
    out := make([][]int, 0, len(parts))
    for _, blk := range parts {
        if len(blk) == 0 {
            continue
        }
        b := append([]int{}, blk...)
        sort.Ints(b)
        out = append(out, b)
    }
    sort.Slice(out, func(i, j int) bool { return out[i][0] < out[j][0] })
    return out
}

func containsSorted(xs []int, x int) bool {
    i := sort.SearchInts(xs, x)
    return i < len(xs) && xs[i] == x
}

func insertSorted(xs []int, x int) []int {
    i := sort.SearchInts(xs, x)
    if i < len(xs) && xs[i] == x {
        return xs
    }
    out := make([]int, 0, len(xs)+1)
    out = append(out, xs[:i]...)
    out = append(out, x)
    out = append(out, xs[i:]...)
    return out
}

func removeSorted(xs []int, x int) []int {
    i := sort.SearchInts(xs, x)
    if i >= len(xs) || xs[i] != x {
        return xs
    }
    out := make([]int, 0, len(xs)-1)
    out = append(out, xs[:i]...)
    out = append(out, xs[i+1:]...)
    return out
}

func removeFromParts(parts [][]int, v int) [][]int {
    out := make([][]int, 0, len(parts))
    for _, blk := range parts {
        if containsSorted(blk, v) {
            if len(blk) > 1 {
                out = append(out, removeSorted(blk, v))
            }
            continue
        }
        out = append(out, blk)
    }
    return canon(out)
}

// introduceKeep computes the partition after keeping the freshly-introduced
// vertex v and adding all its bag edges to kept neighbours. It returns false if
// one of those edges closes a cycle.
func introduceKeep(v int, removed []int, parts [][]int, g *Graph, childBag []int) ([][]int, bool) {
    // Union-find over the vertices that occur in the child state plus v.
    idx := map[int]int{}
    var ids []int
    get := func(x int) int {
        id, ok := idx[x]
        if !ok {
            id = len(ids)
            idx[x] = id
            ids = append(ids, x)
        }
        return id
    }
    // Make sure v and child vertices exist.
    get(v)
    for _, blk := range parts {
        for _, x := range blk {
            get(x)
        }
    }
    parent := make([]int, len(ids))
    for i := range parent {
        parent[i] = i
    }
    var find func(int) int
    find = func(x int) int {
        for parent[x] != x {
            parent[x] = parent[parent[x]]
            x = parent[x]
        }
        return x
    }
    union := func(a, b int) {
        ra, rb := find(a), find(b)
        if ra != rb {
            parent[ra] = rb
        }
    }
    for _, blk := range parts {
        for i := 1; i < len(blk); i++ {
            union(get(blk[0]), get(blk[i]))
        }
    }
    for _, u := range g.Adj[v] {
        if !containsSorted(childBag, u) || containsSorted(removed, u) {
            continue // edge introduced at another node, or u removed
        }
        if find(get(v)) == find(get(u)) {
            return nil, false // cycle
        }
        union(get(v), get(u))
    }
    // Rebuild partition of kept vertices.
    groups := map[int][]int{}
    for _, x := range ids {
        if containsSorted(removed, x) {
            continue
        }
        r := find(idx[x])
        groups[r] = append(groups[r], x)
    }
    var out [][]int
    for _, blk := range groups {
        out = append(out, blk)
    }
    return canon(out), true
}

// joinParts merges the boundary partitions of two children. A cycle appears
// exactly when the union of the two forests (which share only boundary
// vertices) creates a cycle among boundary vertices, so we build a spanning
// tree for each block of the second partition and reject if any edge joins two
// already-connected vertices.
func joinParts(pl, pr [][]int, removed []int, bag []int) ([][]int, bool) {
    idx := map[int]int{}
    var ids []int
    get := func(x int) int {
        id, ok := idx[x]
        if !ok {
            id = len(ids)
            idx[x] = id
            ids = append(ids, x)
        }
        return id
    }
    for _, blk := range pl {
        for _, x := range blk {
            get(x)
        }
    }
    for _, blk := range pr {
        for _, x := range blk {
            get(x)
        }
    }
    for _, x := range bag {
        if !containsSorted(removed, x) {
            get(x)
        }
    }
    parent := make([]int, len(ids))
    for i := range parent {
        parent[i] = i
    }
    var find func(int) int
    find = func(x int) int {
        for parent[x] != x {
            parent[x] = parent[parent[x]]
            x = parent[x]
        }
        return x
    }
    union := func(a, b int) {
        ra, rb := find(a), find(b)
        if ra != rb {
            parent[ra] = rb
        }
    }
    // Left forest: connect each block as a tree.
    for _, blk := range pl {
        for i := 1; i < len(blk); i++ {
            union(get(blk[0]), get(blk[i]))
        }
    }
    // Right forest: each block must not close a cycle.
    for _, blk := range pr {
        if len(blk) < 2 {
            continue
        }
        anchor := get(blk[0])
        for _, x := range blk[1:] {
            if find(anchor) == find(get(x)) {
                return nil, false
            }
            union(anchor, get(x))
        }
    }
    groups := map[int][]int{}
    for _, x := range ids {
        if containsSorted(removed, x) {
            continue
        }
        r := find(idx[x])
        groups[r] = append(groups[r], x)
    }
    var out [][]int
    for _, blk := range groups {
        out = append(out, blk)
    }
    return canon(out), true
}

// SolveFVS returns |S| for a minimum feedback vertex set, given a nice tree
// decomposition of g whose root bag is empty.
func SolveFVS(g *Graph, root *NiceNode) int {
    m := solveNode(g, root)
    best := -1
    for _, c := range m {
        if best < 0 || c < best {
            best = c
        }
    }
    if best < 0 {
        panic("fvs: empty DP result (root bag must be empty)")
    }
    return best
}

func solveNode(g *Graph, n *NiceNode) map[string]int {
    switch n.Kind {
    case Leaf:
        return map[string]int{encode(nil, nil): 0}

    case Introduce:
        child := solveNode(g, n.Children[0])
        res := map[string]int{}
        for k, c := range child {
            rem, parts := decode(k)
            // (a) remove v
            nk := encode(insertSorted(rem, n.Vertex), parts)
            if old, ok := res[nk]; !ok || c+1 < old {
                res[nk] = c + 1
            }
            // (b) keep v
            if np, ok := introduceKeep(n.Vertex, rem, parts, g, n.Children[0].Bag); ok {
                nk = encode(rem, np)
                if old, ok := res[nk]; !ok || c < old {
                    res[nk] = c
                }
            }
        }
        return res

    case Forget:
        child := solveNode(g, n.Children[0])
        res := map[string]int{}
        for k, c := range child {
            rem, parts := decode(k)
            // (a) v was removed: cost already counted, drop it from the bag mask
            nk := encode(removeSorted(rem, n.Vertex), parts)
            if old, ok := res[nk]; !ok || c < old {
                res[nk] = c
            }
            // (b) v was kept: shrink its block
            if !containsSorted(rem, n.Vertex) {
                nk = encode(rem, removeFromParts(parts, n.Vertex))
                if old, ok := res[nk]; !ok || c < old {
                    res[nk] = c
                }
            }
        }
        return res

    case Join:
        left := solveNode(g, n.Children[0])
        right := solveNode(g, n.Children[1])
        res := map[string]int{}
        for kl, cl := range left {
            reml, pl := decode(kl)
            for kr, cr := range right {
                remr, pr := decode(kr)
                if encInts(reml) != encInts(remr) {
                    continue
                }
                np, ok := joinParts(pl, pr, reml, n.Bag)
                if !ok {
                    continue
                }
                cost := cl + cr - len(reml) // shared boundary counted twice
                nk := encode(reml, np)
                if old, ok := res[nk]; !ok || cost < old {
                    res[nk] = cost
                }
            }
        }
        return res
    }
    panic("fvs: unknown node kind")
}

// Online repair engine
// ---------------------------------------------------------------------------

// OnlineFVS maintains a valid feedback vertex set under edge insertions.
// It never over-removes: whenever a new edge closes a cycle in G - S it adds
// one vertex of that cycle to S, then runs a redundant-removal pass. On
// bounded-treewidth inputs the periodic Reoptimize call restores optimality.
type OnlineFVS struct {
    G *Graph
    S []bool
}

// NewOnlineFVS creates an empty online structure on n vertices.
func NewOnlineFVS(n int) *OnlineFVS {
    return &OnlineFVS{G: NewGraph(n), S: make([]bool, n)}
}

// AddEdge inserts an edge and repairs the current feedback vertex set.
func (o *OnlineFVS) AddEdge(u, v int) {
    if u == v {
        return
    }
    already := o.G.hasEdge(u, v)
    o.G.AddEdge(u, v)
    if already {
        return
    }
    if o.S[u] || o.S[v] {
        return // both endpoints cannot lie on a cycle in G - S
    }
    path := o.forestPath(u, v)
    if path == nil {
        return // still a forest
    }
    best, bestDeg := -1, -1
    for _, x := range path {
        if d := len(o.G.Adj[x]); d > bestDeg {
            best, bestDeg = x, d
        }
    }
    if best >= 0 {
        o.S[best] = true
    }
    o.reduce()
}

// forestPath returns the unique path u..v in the current forest G - S, or nil
// if u and v are in different components.
func (o *OnlineFVS) forestPath(u, v int) []int {
    prev := make([]int, o.G.N)
    for i := range prev {
        prev[i] = -2
    }
    prev[u] = -1
    queue := []int{u}
    for len(queue) > 0 {
        x := queue[0]
        queue = queue[1:]
        if x == v {
            break
        }
        for _, y := range o.G.Adj[x] {
            if o.S[y] || prev[y] != -2 {
                continue
            }
            prev[y] = x
            queue = append(queue, y)
        }
    }
    if prev[v] == -2 {
        return nil
    }
    var path []int
    for x := v; x != -1; x = prev[x] {
        path = append(path, x)
    }
    return path
}

// reduce removes vertices from S while G - S stays a forest.
func (o *OnlineFVS) reduce() {
    changed := true
    for changed {
        changed = false
        for v := 0; v < o.G.N; v++ {
            if !o.S[v] {
                continue
            }
            o.S[v] = false
            if o.G.IsForest(keepOf(o.S)) {
                changed = true
            } else {
                o.S[v] = true
            }
        }
    }
}

func keepOf(removed []bool) []bool {
    keep := make([]bool, len(removed))
    for i := range keep {
        keep[i] = !removed[i]
    }
    return keep
}

// IsValid reports whether the maintained set is a feedback vertex set.
func (o *OnlineFVS) IsValid() bool { return o.G.IsFVS(o.S) }

// Size returns |S|.
func (o *OnlineFVS) Size() int {
    c := 0
    for _, b := range o.S {
        if b {
            c++
        }
    }
    return c
}

package fvs

import (
    "math/rand"
    "testing"
)

// BruteFVS is the exponential reference oracle for small graphs.
func BruteFVS(g *Graph) int {
    n := g.N
    best := n
    for mask := 0; mask < (1 << n); mask++ {
        rem := make([]bool, n)
        pc := 0
        for i := 0; i < n; i++ {
            if mask&(1<<i) != 0 {
                rem[i] = true
                pc++
            }
        }
        if pc >= best {
            continue
        }
        if g.IsFVS(rem) {
            best = pc
        }
    }
    return best
}

func randomGraph(r *rand.Rand, n, m int) *Graph {
    g := NewGraph(n)
    for i := 0; i < m; i++ {
        u := r.Intn(n)
        v := r.Intn(n)
        g.AddEdge(u, v)
    }
    return g
}

// TestDPAgainstBruteForce cross-checks the tree-decomposition DP against the
// exponential oracle on hundreds of random graphs.
func TestDPAgainstBruteForce(t *testing.T) {
    r := rand.New(rand.NewSource(20261007))
    for n := 1; n <= 6; n++ {
        for trial := 0; trial < 60; trial++ {
            m := r.Intn(n*(n-1)/2 + 1)
            g := randomGraph(r, n, m)
            want := BruteFVS(g)
            got := SolveFVS(g, TrivialNiceTD(n))
            if got != want {
                t.Fatalf("n=%d trial=%d edges=%v: dp=%d brute=%d", n, trial, g.Edges(), got, want)
            }
        }
    }
}

// TestJoinNode exercises the Join transition with two triangles sharing one
// vertex: the optimum is 1 (remove the shared vertex).
func TestJoinNode(t *testing.T) {
    g := NewGraph(5)
    for _, e := range [][2]int{{0, 1}, {1, 2}, {2, 0}, {0, 3}, {3, 4}, {4, 0}} {
        g.AddEdge(e[0], e[1])
    }
    // leaf -> introduce0 -> join(branchA, branchB) -> forget0
    leaf := &NiceNode{Kind: Leaf, Bag: []int{}}
    intro0 := &NiceNode{Kind: Introduce, Vertex: 0, Bag: []int{0}, Children: []*NiceNode{leaf}}

    branch := func(a, b int) *NiceNode {
        i1 := &NiceNode{Kind: Introduce, Vertex: a, Bag: []int{0, a}, Children: []*NiceNode{intro0}}
        i2 := &NiceNode{Kind: Introduce, Vertex: b, Bag: []int{0, a, b}, Children: []*NiceNode{i1}}
        f1 := &NiceNode{Kind: Forget, Vertex: a, Bag: []int{0, b}, Children: []*NiceNode{i2}}
        f2 := &NiceNode{Kind: Forget, Vertex: b, Bag: []int{0}, Children: []*NiceNode{f1}}
        return f2
    }
    j := &NiceNode{Kind: Join, Bag: []int{0}, Children: []*NiceNode{branch(1, 2), branch(3, 4)}}
    root := &NiceNode{Kind: Forget, Vertex: 0, Bag: []int{}, Children: []*NiceNode{j}}

    if got, want := SolveFVS(g, root), 1; got != want {
        t.Fatalf("bowtie join: dp=%d want=%d", got, want)
    }
    if got, want := SolveFVS(g, TrivialNiceTD(5)), BruteFVS(g); got != want {
        t.Fatalf("bowtie trivial: dp=%d brute=%d", got, want)
    }
}

// TestOnlineTriangles: a disjoint union of k edge-disjoint triangles needs and
// gets exactly k removed vertices; verifies the repair loop does not over-hit.
func TestOnlineTriangles(t *testing.T) {
    const k = 40
    o := NewOnlineFVS(3 * k)
    for i := 0; i < k; i++ {
        a, b, c := 3*i, 3*i+1, 3*i+2
        o.AddEdge(a, b)
        o.AddEdge(b, c)
        o.AddEdge(c, a)
        if !o.IsValid() {
            t.Fatalf("triangle %d left an invalid FVS", i)
        }
    }
    if got := o.Size(); got != k {
        t.Fatalf("disjoint triangles: |S|=%d want=%d", got, k)
    }
}

// TestOnlineBoundedWidthReopt shows that on a cycle the online set matches the
// exact bounded-treewidth optimum (1).
func TestOnlineBoundedWidthReopt(t *testing.T) {
    n := 30

        o.AddEdge(i, (i+1)%n)
    }
    if !o.IsValid() {
        t.Fatal("cycle online set invalid")
    }

        t.Fatalf("cycle online |S|=%d want=1", o.Size())
    }
}
