Answer

Fix: serialize alert-router JSONL dedupe with fcntl.flock

Root cause

router.py performed the dedupe check and the append as two separate operations:

ids = existing_ids(path)
if row["id"] not in ids:
    append_rows(path, [row])

Two concurrent Alertmanager webhook threads could both call existing_ids(), both observe the id as absent, and then both append. The same race existed in the spool path. The refire path also appended directly, bypassing the dedupe gate and potentially replacing the original acceptance decision with a duplicate row.

The lock must cover both the id read and the append, not just the append.

Exact fix

Apply the following to stack/alert-router/router.py.

1. Add a stable sidecar lock and an atomic append helper

import contextlib
import fcntl
import json
import os
from typing import Any, Iterable, Iterator, Mapping


def _ensure_parent(path: str) -> None:
    parent = os.path.dirname(os.path.abspath(os.fspath(path)))
    if parent:
        os.makedirs(parent, exist_ok=True)


@contextlib.contextmanager
def _exclusive_lock(path: str) -> Iterator[None]:
    """Serialize all writers for one JSONL target using a stable sidecar lock."""
    _ensure_parent(path)
    lock_path = os.fspath(path) + ".lock"
    fd = os.open(lock_path, os.O_CREAT | os.O_RDWR, 0o644)
    try:
        fcntl.flock(fd, fcntl.LOCK_EX)
        yield
    finally:
        fcntl.flock(fd, fcntl.LOCK_UN)
        os.close(fd)


def existing_ids(path: str) -> set[str]:
    ids: set[str] = set()
    try:
        with open(path, "r", encoding="utf-8") as fh:
            for line in fh:
                line = line.strip()
                if not line:
                    continue
                try:
                    row = json.loads(line)
                except json.JSONDecodeError:
                    continue
                row_id = row.get("id")
                if row_id is not None:
                    ids.add(row_id)
    except FileNotFoundError:
        pass
    return ids


def append_rows(path: str, rows: Iterable[Mapping[str, Any]]) -> None:
    _ensure_parent(path)
    with open(path, "a", encoding="utf-8") as fh:
        for row in rows:
            fh.write(json.dumps(row, sort_keys=True, separators=(",", ":")) + "\n")
        fh.flush()
        os.fsync(fh.fileno())


def append_once(path: str, row: Mapping[str, Any], *, refire: bool = False) -> bool:
    """Atomically append row iff its id is absent. Returns True iff appended."""
    row_id = row.get("id")
    if row_id is None:
        raise ValueError("alert row must contain an 'id'")

    with _exclusive_lock(path):
        if row_id in existing_ids(path):
            # Duplicate or refire: the first accepted row wins.
            # Do not append and do not overwrite its acceptance decision.
            if refire:
                pass
            return False
        append_rows(path, [row])
        return True

2. Route every local-board and spool write through append_once

Replace every unsafe read-then-append block, including the refire branch, with the gated helper.

# BEFORE: unsafe local-board path
ids = existing_ids(board_path)
if row["id"] not in ids:
    append_rows(board_path, [row])

# BEFORE: unsafe spool path
ids = existing_ids(spool_path)
if row["id"] not in ids:
    append_rows(spool_path, [row])

# BEFORE: ungated refire
if alert.get("refire"):
    append_rows(spool_path, [row])

# AFTER: both paths use the same atomic gate
append_once(board_path, row, refire=bool(alert.get("refire")))
append_once(spool_path, row, refire=bool(alert.get("refire")))

If the router chooses one target instead of writing both, use:

target = board_path if board_available else spool_path
append_once(target, row, refire=bool(alert.get("refire")))

The sidecar lock is <jsonl-path>.lock, so concurrent processes and threads writing the same JSONL file serialize on the same lock. Do not delete the lock file during normal operation.

3. Refire behavior

A refire now goes through the same append_once() gate. If the fingerprint/day id already exists, no second row is appended, so the original row and its acceptance criteria remain intact. If a refire must update non-acceptance fields, rewrite the existing row under the same _exclusive_lock() and copy acceptance from the old row; never append a duplicate.

import json
import sys
import tempfile
import threading
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

sys.path.insert(0, ".")
import router


def make_row(acceptance="accepted"):
    return {
        "id": "fp-1:2025-01-01",
        "fingerprint": "fp-1",
        "day": "2025-01-01",
        "acceptance": acceptance,
    }


with tempfile.TemporaryDirectory() as td:
    for name in ("board.jsonl", "spool.jsonl"):
        path = Path(td) / name
        barrier = threading.Barrier(2)

        def worker():
            barrier.wait()
            router.append_once(str(path), make_row())

        with ThreadPoolExecutor(max_workers=2) as pool:
            list(pool.map(lambda _: worker(), range(2)))

        rows = [
            json.loads(line)
            for line in path.read_text(encoding="utf-8").splitlines()
            if line.strip()
        ]
        assert len(rows) == 1, (name, rows)
        assert rows[0]["acceptance"] == "accepted", (name, rows)

        # Refire with a different acceptance must not add a row
        # or overwrite the original acceptance.
        router.append_once(
            str(path),
            make_row(acceptance="rejected"),
            refire=True,
        )
        rows = [
            json.loads(line)
            for line in path.read_text(encoding="utf-8").splitlines()
            if line.strip()
        ]
        assert len(rows) == 1, (name, rows)
        assert rows[0]["acceptance"] == "accepted", (name, rows)

print("OK: concurrent dedupe and refire acceptance preserved")
