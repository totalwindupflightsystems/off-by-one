1. internal/config/config.go — new ApplyStartupValidations() (fixes C-GAP-002 compression on DeepSeek, C-GAP-003 missing API key):

// ApplyStartupValidations inspects the configuration for known
// misconfigurations, corrects the ones it can (mutating c), and returns
// human-readable warnings for the caller to surface via slog.Warn.
func (c *Config) ApplyStartupValidations() []string {
    var warnings []string

    // C-GAP-002: DeepSeek has no embeddings endpoint. If compression is
    // enabled with the default text-embedding-3-small model on that
    // backend, disable it and explain why.
    if c.CompressionEnabled &&
        strings.EqualFold(c.LLMBackend, BackendDeepSeek) &&
        c.EmbeddingsModel == EmbeddingModelTextEmbedding3Small {
        c.CompressionEnabled = false
        warnings = append(warnings, fmt.Sprintf(
            "compression disabled: backend %q has no embeddings endpoint for model %q; "+
                "set --compression=false / compression.enabled=false, switch to a backend "+
                "that exposes embeddings, or configure an embeddings-capable model",
            c.LLMBackend, c.EmbeddingsModel))
    }

    // C-GAP-003: surface a missing LLM API key at startup.
    if strings.TrimSpace(c.LLMAPIKey) == "" {
        warnings = append(warnings,
            "LLM API key is not set: LLM requests will fail at runtime "+
                "(set LLM_API_KEY or pass --llm-api-key)")
    }
    return warnings
}

2. internal/server/server.go — RunServer logs warnings via slog.Warn and gates the compression worker:

func RunServer(cfg *config.Config, logger *slog.Logger) error {
    warnings := cfg.ApplyStartupValidations()
    for _, w := range warnings {
        logger.Warn("startup validation", "detail", w)
    }
    if cfg.CompressionEnabled {
        startCompressionWorker(logger)
    } else {
        logger.Info("compression worker not started (compression disabled)")
    }
    return nil
}

3. Makefile — dev targets use --db-url (C-GAP-001; the CLI in cmd/server/main.go registers --db-url, never --db):

dev-openai: build
