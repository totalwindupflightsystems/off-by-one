Fix: Wire the Existing Spec Selector into Worker Briefs

Root-cause analysis

The selector answer and the brief builder disagree about who owns block prose:

  1. The selector returns provenance only. Each block carries id, kind, status, source, and edge, but not the block text. The only prose in the answer is the task block itself. Therefore the brief builder must resolve selected blocks through the module's own corpus loader. Adding a --text/--include-body flag to the selector would duplicate the corpus and violate the existing API boundary.

  2. “No match” is a valid answer, not an error. A selection with status: "unscoped" and reason: "unknown-task" exits 0. The builder must:

  3. write exactly zero bytes for the spec section, with no heading, separator, placeholder, or newline;
  4. record unknown-task in the brief metadata;

  5. allow the tick to exit 0.

  6. Ambiguity is a hard failure. An ambiguous answer must raise/propagate and make the tick exit non-zero. It must not be mistaken for unknown-task, and a non-zero selector exit must not be swallowed.

The fix is an adapter at the brief boundary: it calls the existing selector, resolves text through the corpus loader, and distinguishes the three outcomes.

Exact fix

Add this module as brief_spec.py:

"""Wire an existing spec-selection verb into worker briefs.

The selector is the source of truth for *which* blocks apply.  The module's
own corpus loader is the source of truth for the *verbatim text* of those
blocks.  The selector payload only carries provenance.
"""

from __future__ import annotations

import json
import subprocess
import sys
from typing import Any, Mapping, Protocol, Sequence


class BriefError(RuntimeError):
    """A hard failure. The enclosing tick must report it and exit non-zero."""


class AmbiguousSelection(BriefError):
    """More than one usable spec selection exists; never choose silently."""


class CorpusLookupError(BriefError):
    """A selected block's provenance could not be resolved to block text."""


class Selector(Protocol):
    def select(self, task_id: str) -> Mapping[str, Any]:
        ...


class Corpus(Protocol):
    def get_block(
        self,
        block_id: str,
        *,
        source: str | None = None,
        edge: str | None = None,
    ) -> Any:
        ...


_AMBIGUOUS_REASONS = {
    "ambiguous",
    "ambiguous-task",
    "multiple-matches",
    "multiple-matching-blocks",
}
_SCOPED_STATUSES = {"scoped", "selected", "ok"}
_NO_MATCH_STATUS = "unscoped"
_NO_MATCH_REASON = "unknown-task"


def _norm(value: Any) -> str | None:
    if value is None:
        return None
    return str(value).strip().lower()


def _provenance(block: Mapping[str, Any]) -> dict[str, Any]:
    """Keep exactly the provenance fields the selector returned."""
    return {
        key: block[key]
        for key in ("id", "kind", "status", "source", "edge")
        if key in block
    }


def _loaded_text(loaded: Any) -> str | None:
    if loaded is None:
        return None
    if isinstance(loaded, str):
        return loaded
    if isinstance(loaded, Mapping):
        text = loaded.get("text")
        return text if isinstance(text, str) else None
    text = getattr(loaded, "text", None)
    return text if isinstance(text, str) else None


def build_spec_section(
    task_id: str,
    *,
    selector: Selector,
    corpus: Corpus,
) -> tuple[bytes, dict[str, Any]]:
    """Return ``(section_bytes, metadata)`` for one task.

    * A unique/scoped answer returns the selected blocks' text verbatim.
    * ``unscoped`` + ``unknown-task`` is an answer, not an error: it returns
      exactly zero bytes and records the reason in metadata.
    * An ambiguous answer raises, so the caller/tick can fail loudly.
    """

    answer = selector.select(task_id)
    if not isinstance(answer, Mapping):
        raise BriefError(f"spec selection for {task_id!r} was not an object")

    status = _norm(answer.get("status"))
    reason = _norm(answer.get("reason"))
    blocks = list(answer.get("blocks") or ())

    if status == "ambiguous" or reason in _AMBIGUOUS_REASONS:
        raise AmbiguousSelection(
            f"ambiguous spec selection for task {task_id!r}: "
            f"status={status!r} reason={reason!r}"
        )

    if status == _NO_MATCH_STATUS and reason == _NO_MATCH_REASON:
        return b"", {
            "status": _NO_MATCH_STATUS,
            "reason": _NO_MATCH_REASON,
            "blocks": [],
        }

    if status not in _SCOPED_STATUSES:
        raise BriefError(
            f"spec selection failed for task {task_id!r}: "
            f"status={status!r} reason={reason!r}"
        )
    if not blocks:
        raise BriefError(
            f"spec selection for task {task_id!r} was scoped but returned no blocks"
        )

    texts: list[str] = []
    provenance: list[dict[str, Any]] = []

    for block in blocks:
        prov = _provenance(block)
        block_id = prov.get("id")
        if not block_id:
            raise BriefError(
                f"selected block without id for task {task_id!r}: {block!r}"
            )

        # Do not read corpus files directly and do not expect/require a text
        # field in the selector answer. Resolve provenance with the module's
        # own loader/accessor.
        loaded = corpus.get_block(
            block_id,
            source=prov.get("source"),
            edge=prov.get("edge"),
        )
        text = _loaded_text(loaded)
        if text is None:
            raise CorpusLookupError(
                f"corpus has no text for block {block_id!r} "
                f"(source={prov.get('source')!r}, edge={prov.get('edge')!r})"
            )

        # Verbatim: no strip(), reflow, summarisation, or markdown escaping.
        texts.append(text)
        provenance.append(prov)

    # One selected block is emitted byte-for-byte. Multiple blocks are joined
    # with one newline; each block's own bytes are untouched.
    section = "\n".join(texts).encode("utf-8")
    return section, {
        "status": status,
        "reason": answer.get("reason"),
        "blocks": provenance,
    }


def render_brief(
    task_id: str,
    *,
    selector: Selector,
    corpus: Corpus,
    other_sections: Sequence[bytes] = (),
) -> tuple[bytes, dict[str, Any]]:
    """Render a brief and its metadata.

    Zero-byte spec sections contribute zero bytes: no heading, no separator,
    no newline. This is what makes a no-match a real empty section rather
    than a placeholder.
    """

    spec_bytes, spec_meta = build_spec_section(
        task_id, selector=selector, corpus=corpus
    )
    sections = list(other_sections)
    if spec_bytes:
        sections.append(b"## Spec context\n\n" + spec_bytes + b"\n")
    return b"".join(sections), {"spec": spec_meta}


class SpecCliSelector:
    """Adapter around the *existing* ``spec select`` verb.

    It adds no ``--text``/``--include-body`` flag. Ambiguity is converted to
    ``AmbiguousSelection`` even if the old verb accidentally exits 0; any
    other non-zero exit is also propagated as a hard failure.
    """

    def __init__(self, spec_cmd: Sequence[str]):
        self._cmd = list(spec_cmd)

    def select(self, task_id: str) -> Mapping[str, Any]:
        proc = subprocess.run(
            [*self._cmd, "select", "--task", task_id],
            capture_output=True,
            text=True,
        )
        try:
            answer = json.loads(proc.stdout) if proc.stdout.strip() else None
        except json.JSONDecodeError:
            answer = None

        status = _norm(answer.get("status")) if isinstance(answer, Mapping) else None
        reason = _norm(answer.get("reason")) if isinstance(answer, Mapping) else None

        if status == "ambiguous" or reason in _AMBIGUOUS_REASONS:
            raise AmbiguousSelection(
                f"ambiguous spec selection for task {task_id!r}: "
                f"status={status!r} reason={reason!r}"
            )
        if proc.returncode != 0:
            raise BriefError(
                f"spec select failed for task {task_id!r} "
                f"(exit {proc.returncode}): {proc.stderr.strip()}"
            )
        if not isinstance(answer, Mapping):
            raise BriefError(
                f"spec select for task {task_id!r} did not return a JSON object"
            )
        return answer


def run_tick(
    task_id: str,
    *,
    selector: Selector,
    corpus: Corpus,
    write_brief,
) -> int:
    """Minimal tick integration: report hard failures, write empty no-match."""

    try:
        brief, metadata = render_brief(
            task_id, selector=selector, corpus=corpus
        )
    except AmbiguousSelection as exc:
        print(f"tick: {exc}", file=sys.stderr)
        return 2
    except BriefError as exc:
        print(f"tick: {exc}", file=sys.stderr)
        return 1

    write_brief(brief, metadata)
    return 0

Wire it into the existing brief/tick path

Replace the old spec handling with a call to this adapter. Do not add a selector flag.

from brief_spec import SpecCliSelector, run_tick

selector = SpecCliSelector([sys.executable, "-m", "your_spec_module"])
rc = run_tick(
    task_id,
    selector=selector,
    corpus=your_spec_module.corpus,   # the module's existing corpus loader
    write_brief=write_brief,
)
raise SystemExit(rc)

If the corpus loader uses a different method name, adapt only the one-method shim:

class ModuleCorpus:
    def __init__(self, loader):
        self.loader = loader

    def get_block(self, block_id, *, source=None, edge=None):
        return self.loader.read_block(
            id=block_id,
            source=source,
            edge=edge,
        )

Verified cases:

Case Verified behavior
status: "scoped" Selected block text comes from the corpus loader, byte-for-byte, including trailing newlines. The selector payload has no text field.
status: "unscoped", reason: "unknown-task" Returns exactly b""; records {"status": "unscoped", "reason": "unknown-task", "blocks": []}; tick exits 0; no spec heading/separator is emitted.
status: "ambiguous" / ambiguous reason Raises AmbiguousSelection; tick exits non-zero (2); brief is not written.
CLI ambiguous answer that exits 0 Still raises AmbiguousSelection; ambiguity is never accepted silently.
CLI non-zero exit Propagates as BriefError, causing the tick to report it.
Selected block missing from corpus Raises CorpusLookupError instead of silently emitting an empty or partial section.
