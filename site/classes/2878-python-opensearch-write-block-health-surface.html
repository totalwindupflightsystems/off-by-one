Surfacing per-index read_only_allow_delete write blocks in the health summary

Problem class: python-opensearch-write-block-health-surface

Root cause

A green cluster status only means all primary/replica shards are allocated — it does not report index-level write blocks.

When a data node's disk usage crosses cluster.routing.allocation.disk.flood_stage (default 95%), OpenSearch automatically sets index.blocks.read_only_allow_delete: true on the affected indices. Those indices stay allocated (health remains green) but reject every document write with cluster_block_exception / FORBIDDEN/12. The block persists until disk pressure clears and is explicitly cleared.

The existing health summary calls only client.cluster.health(), so it is blind to this state. The fix adds a best-effort per-index write-block probe: read the block settings for the write alias and for the concrete targets returned by cat.indices, never letting a missing index/setting fail the health check.

Exact fix

Add this module (or merge the helpers into the existing health module). It preserves the original cluster-health payload and adds write_blocks (per index: True / False / None) and a roll-up write_blocked.

# opensearch_health/health.py
"""Extend the cluster health summary with per-index write-block state."""

from __future__ import annotations

import logging
from typing import Any, Dict, Optional, Set

from opensearchpy.exceptions import NotFoundError, TransportError

log = logging.getLogger(__name__)

# read_only_allow_delete = flood-stage block; read_only = manual block.
WRITE_BLOCK_SETTINGS = (
    "index.blocks.read_only_allow_delete",
    "index.blocks.read_only",
)

DEFAULT_WRITE_ALIAS = "datapoints-write"


def _is_true(value: Any) -> bool:
    return str(value).strip().lower() in ("true", "1", "yes", "on")


def _is_not_found(exc: BaseException) -> bool:
    return isinstance(exc, NotFoundError) or getattr(exc, "status_code", None) == 404


def block_from_settings(index_entry: Dict[str, Any]) -> Optional[bool]:
    """True=blocked, False=explicitly unblocked, None=setting absent."""
    if not isinstance(index_entry, dict):
        return None
    settings = index_entry.get("settings", index_entry)
    if not isinstance(settings, dict):
        return None
    index = settings.get("index", {})
    blocks = index.get("blocks", {}) if isinstance(index, dict) else {}
    if not isinstance(blocks, dict):
        blocks = {}
    for short_key in ("read_only_allow_delete", "read_only"):
        if short_key in blocks:
            return _is_true(blocks[short_key])
    for dotted in WRITE_BLOCK_SETTINGS:          # flat dotted form
        if dotted in settings:
            return _is_true(settings[dotted])
    return None


def _get_settings(client: Any, index: str) -> Dict[str, Any]:
    """Fetch settings for *index*; return {} on 404/not-found."""
    try:
        resp = client.indices.get_settings(
            index=index,
            name=",".join(WRITE_BLOCK_SETTINGS),
            allow_no_indices=True,
            expand_wildcards="open",
        )
    except TransportError as exc:                # NotFoundError is a TransportError
        if _is_not_found(exc):
            return {}
        raise
    return resp or {}


def _cat_indices(client: Any, pattern: str) -> Set[str]:
    """Concrete index names from cat.indices; empty on 404/not-found."""
    try:
        rows = client.cat.indices(
            index=pattern, format="json", h="index", expand_wildcards="open"
        )
    except TransportError as exc:
        if _is_not_found(exc):
            return set()
        raise
    return {row["index"] for row in (rows or []) if isinstance(row, dict) and row.get("index")}


def write_block_state(client: Any, index: str) -> Optional[bool]:
    """Best-effort write-block state for one concrete index."""
    settings = _get_settings(client, index)
    entry = settings.get(index)
    if entry is None and settings:
        entry = next(iter(settings.values()))
    return None if entry is None else block_from_settings(entry)


def get_health_summary(
    client: Any, write_alias: str = DEFAULT_WRITE_ALIAS, **health_kwargs: Any
) -> Dict[str, Any]:
    """Cluster health summary + per-index write blocks.

    Targets come from two sources:
      * indices.get_settings(index=<write_alias>) -> concrete alias indices + settings
      * cat.indices(index=<write_alias>*)          -> concrete cat targets
    Missing settings/index -> None, never an exception.
    """
    summary: Dict[str, Any] = dict(client.cluster.health(**health_kwargs))

    alias_settings = _get_settings(client, write_alias)
    targets: Set[str] = set(alias_settings)
    targets |= _cat_indices(client, f"{write_alias}*")

    write_blocks: Dict[str, Optional[bool]] = {}
    for index in sorted(targets):
        if index in alias_settings:
            write_blocks[index] = block_from_settings(alias_settings[index])
        else:
            write_blocks[index] = write_block_state(client, index)

    summary["write_blocks"] = write_blocks
    summary["write_blocked"] = any(state is True for state in write_blocks.values())
    return summary

The call site is unchanged; the summary just gains two keys:

summary = get_health_summary(client)
# summary["write_blocked"] is True while cluster status is still "green"

Clearing a flood-stage block (after freeing disk space)

# Inspect
curl -s 'http://localhost:9200/_cat/indices/datapoints*?h=index&format=json'
curl -s 'http://localhost:9200/datapoints-write/_settings?name=index.blocks.read_only_allow_delete&flat_settings=true'

# Clear it on the affected concrete indices
curl -X PUT 'http://localhost:9200/datapoints-*/_settings' \
  -H 'Content-Type: application/json' \
  -d '{"index.blocks.read_only_allow_delete": null}'

Verification

Mocked tests cover blocked / unblocked / missing, alias+cat merge, and best-effort 404s.

# opensearch_health/test_health.py
from unittest import mock
from opensearchpy.exceptions import NotFoundError
from opensearch_health.health import get_health_summary


def _client(settings=None, cat=None, settings_by_index=None):
    client = mock.Mock()
    client.cluster.health.return_value = {"status": "green", "cluster_name": "test"}
    client.indices.get_settings.return_value = {} if settings is None else settings
    client.cat.indices.return_value = cat if cat is not None else []
    by_index = settings_by_index or {}

    def _get_settings(index, **kwargs):
        value = by_index.get(index, client.indices.get_settings.return_value)
        if isinstance(value, Exception):
            raise value
        return value

    client.indices.get_settings.side_effect = _get_settings
    return client


def _settings(value):
    return {"datapoints-000001": {"settings": {"index": {"blocks": {"read_only_allow_delete": value}}}}}


def test_blocked_index_is_surfaced_while_health_is_green():
    client = _client(settings=_settings("true"), cat=[{"index": "datapoints-000001"}])
    summary = get_health_summary(client)
    assert summary["status"] == "green"          # cluster health alone is misleading
    assert summary["write_blocked"] is True
    assert summary["write_blocks"] == {"datapoints-000001": True}


def test_unblocked_index_reports_false():
    client = _client(settings=_settings("false"), cat=[{"index": "datapoints-000001"}])
    summary = get_health_summary(client)
    assert summary["write_blocked"] is False
    assert summary["write_blocks"] == {"datapoints-000001": False}


def test_missing_settings_are_non_fatal():
    client = _client(settings=NotFoundError(404, "index_not_found_exception", {}), cat=[])
    summary = get_health_summary(client)
    assert summary["status"] == "green"
    assert summary["write_blocks"] == {}
    assert summary["write_blocked"] is False


def test_missing_setting_key_reports_none():
    client = _client(
        settings={"datapoints-000001": {"settings": {"index": {}}}},
        cat=[{"index": "datapoints-000001"}],
    )
    summary = get_health_summary(client)
    assert summary["write_blocks"] == {"datapoints-000001": None}
    assert summary["write_blocked"] is False


def test_cat_target_missing_settings_is_non_fatal():
    client = _client(
        settings={}, cat=[{"index": "datapoints-000001"}],
        settings_by_index={"datapoints-000001": NotFoundError(404, "index_not_found_exception", {})},
    )
    summary = get_health_summary(client)
    assert summary["write_blocks"] == {"datapoints-000001": None}
    assert summary["write_blocked"] is False


def test_alias_and_cat_targets_are_merged():
    client = _client(
        settings=_settings("true"),
        cat=[{"index": "datapoints-000001"}, {"index": "datapoints-000002"}],
        settings_by_index={"datapoints-000002": {
            "datapoints-000002": {"settings": {"index": {"blocks": {"read_only": "true"}}}}}},
    )
    summary = get_health_summary(client)
    assert summary["write_blocks"] == {"datapoints-000001": True, "datapoints-000002": True}
    assert summary["write_blocked"] is True


def test_alias_lookup_is_best_effort_when_it_raises_404():
    client = _client(
        settings=NotFoundError(404, "index_not_found_exception", {}),
        cat=[{"index": "datapoints-000001"}],
        settings_by_index={"datapoints-000001": {
            "datapoints-000001": {"settings": {"index": {"blocks": {"read_only_allow_delete": "true"}}}}}},
    )
    summary = get_health_summary(client)
    assert summary["write_blocks"] == {"datapoints-000001": True}
    assert summary["write_blocked"] is True

Edge cases

  • Both read_only_allow_delete and read_only are checked; the flat dotted form is handled.
  • An entry with no blocks key yields None (unknown), not False, so "unblocked" and "unknown" remain distinguishable.
  • Only 404/not-found is swallowed; any other TransportError propagates, so genuine connectivity failures still surface.
