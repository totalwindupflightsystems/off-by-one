Fix: digest show miss should print a search hint and exit 1

Root cause

cmd_show had a single fall-through miss branch:

if path is not None:
    sys.stdout.write(path.read_text()); return 0
print("no data")
return 0          # miss indistinguishable from hit

Two bugs: (1) miss exited 0, so scripts couldn't detect it; (2) only no data was printed, hiding the 3 roots × 6 name patterns actually searched. The candidate list lived only inside _show_find, so a separately-written hint would drift from the real search.

Exact fix

Promote search inputs to module tables and derive both search and hint from them:

SHOW_ROOTS = (Path("archive"), Path.home() / ".digest", Path("/var/lib/digest"))
SHOW_NAME_PATTERNS = ("{stem}.md", "{stem}.txt", "{stem}.json",
                      "{stem}.yaml", "{stem}.rst", "{stem}.log")

def _show_stem(args):
    return f"{args.project}-{args.date}" if getattr(args, "project", None) else str(args.date)

def _show_names(stem, patterns=SHOW_NAME_PATTERNS):
    return [p.format(stem=stem) for p in patterns]

def _show_miss_hint(args):
    stem = _show_stem(args)
    roots = "\n".join(f"  {r}" for r in SHOW_ROOTS)
    names = "\n".join(f"  {n}" for n in _show_names(stem))
    return f"searched roots:\n{roots}\nsearched names:\n{names}"

def _show_find(args):
    for root in SHOW_ROOTS:
        for name in _show_names(_show_stem(args)):
            if (root / name).is_file():
                return root / name
    return None

def cmd_show(args):
    path = _show_find(args)
    if path is not None:
        sys.stdout.write(path.read_text())
        return 0
    print("no data")
    print(_show_miss_hint(args))   # same tables as the search
    return 1                       # miss != hit

Plus main() returns args.func(args) and raise SystemExit(main()).
