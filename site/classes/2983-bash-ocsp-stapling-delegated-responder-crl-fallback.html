Root-cause analysis

The failure class is "OCSP says good, therefore good." On OpenSSL 3.5.5 a naive wrapper is unsound:

  • Expired responses don't fail. openssl ocsp exits 0 and only prints WARNING: ... status expired ... when nextUpdate is in the past — freshness must be enforced by the caller.
  • -VAfile disables authorization. A self-signed responder and a responder without id-kp-OCSPSigning both report Response verify OK when listed in -VAfile. Chain and EKU must be checked separately with openssl verify / openssl x509 -text.
  • A valid signature only proves some key signed the bytes — not that it is an authorized responder, nor that the CertID is about the leaf. Responder identity and CertID serial/issuer hashes must be compared explicitly.
  • Nonce echo is unchecked unless the original request is supplied.
  • Unsuccessful responses surface as Responder Error: unauthorized (6) (exit 1), not as an OCSP Response Status: line.
  • CRL fallback is easy to abuse — it must require an issuer-DN match, AKI/SKI match, CRL signature, its own freshness window, and CRL.lastUpdate > OCSP.thisUpdate.

Exact fix — check_revocation.sh

Every decision prints exactly REASON=<reason> and exits with a fixed code (0 good, 1 revoked, 2 unknown, >=10 fail-closed).

#!/usr/bin/env bash
#
# check_revocation.sh -- fail-closed TLS revocation checker (bash + openssl)
set -u
LC_ALL=C
export LC_ALL

declare -A REASON_CODE=(
  [GOOD]=0 [GOOD_CRL_FALLBACK]=0 [REVOKED]=1 [UNKNOWN]=2
  [OCSP_MISSING]=10 [OCSP_MALFORMED]=11 [OCSP_RESPONSE_STATUS_ERROR]=12
  [OCSP_SIGNATURE_INVALID]=13 [RESPONDER_SELF_SIGNED]=14
  [RESPONDER_NO_OCSP_EKU]=15 [RESPONDER_NOT_ISSUED_BY_CA]=16
  [RESPONDER_UNTRUSTED]=17 [RESPONDER_EXPIRED]=18 [RESPONDER_NOT_YET_VALID]=19
  [RESPONDER_WRONG]=20 [OCSP_STALE_NEXTUPDATE]=21 [OCSP_STALE_THISUPDATE]=22
  [OCSP_NEXTUPDATE_MISSING]=23 [NONCE_MISSING_IN_REQUEST]=24
  [NONCE_MISSING_IN_RESPONSE]=25 [NONCE_MISMATCH]=26 [SERIAL_MISMATCH]=27
  [OCSP_ISSUER_MISMATCH]=28 [LEAF_ISSUER_MISMATCH]=29 [CRL_STALE]=30
  [CRL_ISSUER_MISMATCH]=31 [CRL_AKI_MISMATCH]=32 [CRL_SIGNATURE_INVALID]=33
  [CRL_NOT_FRESHER]=34 [INPUT_ERROR]=35
)
die() { printf 'REASON=%s\n' "$1"; exit "${REASON_CODE[$1]:-99}"; }

LEAF=""; ISSUER=""; ROOT=""; RESPONDER=""; REQUEST=""; RESPONSE=""; CRL=""
NOW=""; SKEW=300
while [ $# -gt 0 ]; do
  case "$1" in
    --leaf) LEAF="${2:-}"; shift 2 ;;   --issuer) ISSUER="${2:-}"; shift 2 ;;
    --root) ROOT="${2:-}"; shift 2 ;;   --responder) RESPONDER="${2:-}"; shift 2 ;;
    --request) REQUEST="${2:-}"; shift 2 ;; --response) RESPONSE="${2:-}"; shift 2 ;;
    --crl) CRL="${2:-}"; shift 2 ;;     --now) NOW="${2:-}"; shift 2 ;;
    --skew) SKEW="${2:-}"; shift 2 ;;   -h|--help) exit 0 ;;
    *) die INPUT_ERROR ;;
  esac
done
[ -n "$LEAF$ISSUER$ROOT$RESPONDER$RESPONSE" ] || die INPUT_ERROR
for f in "$LEAF" "$ISSUER" "$ROOT" "$RESPONDER"; do [ -f "$f" ] || die INPUT_ERROR; done
[ -n "$NOW" ] || NOW="$(date +%s)"

norm_hex() { if [ "$#" -gt 0 ]; then printf '%s' "$1"; else cat; fi | tr -d ' :' | tr 'a-f' 'A-F'; }
to_epoch() { [ -n "$1" ] && date -u -d "$1" +%s 2>/dev/null || printf ''; }
text_field() { printf '%s\n' "$1" | sed -n "s/.*$2: *//p" | head -1 | sed 's/[[:space:]]*$//'; }
extract_nonce() {
  printf '%s\n' "$1" | awk '
    /OCSP Nonce:/ {
      if (match($0, /OCSP Nonce:[ \t]*[0-9A-Fa-f]/)) { s=$0; sub(/.*OCSP Nonce:[ \t]*/,"",s); print s; exit }
      if ((getline line) > 0) { gsub(/[ \t\r]/,"",line); print line; exit } }'
}

# 1. leaf <-> issuer
leaf_issuer="$(openssl x509 -in "$LEAF" -noout -issuer 2>/dev/null | sed 's/^issuer=//')"
ica_subject="$(openssl x509 -in "$ISSUER" -noout -subject 2>/dev/null | sed 's/^subject=//')"
[ "$leaf_issuer" = "$ica_subject" ] || die LEAF_ISSUER_MISMATCH
openssl verify -attime "$NOW" -partial_chain -CAfile "$ISSUER" "$LEAF" >/dev/null 2>&1 || die LEAF_ISSUER_MISMATCH

# 2. delegated-responder authorization
resp_subject="$(openssl x509 -in "$RESPONDER" -noout -subject 2>/dev/null | sed 's/^subject=//')"
resp_issuer="$(openssl x509 -in "$RESPONDER" -noout -issuer 2>/dev/null | sed 's/^issuer=//')"
[ "$resp_subject" = "$resp_issuer" ] && die RESPONDER_SELF_SIGNED
openssl x509 -in "$RESPONDER" -noout -text 2>/dev/null | grep -q 'OCSP Signing' || die RESPONDER_NO_OCSP_EKU
resp_na_e="$(to_epoch "$(openssl x509 -in "$RESPONDER" -noout -enddate | sed 's/^notAfter=//')")"
[ -n "$resp_na_e" ] && [ "$resp_na_e" -lt "$NOW" ] && die RESPONDER_EXPIRED
[ "$resp_issuer" = "$ica_subject" ] || die RESPONDER_NOT_ISSUED_BY_CA
openssl verify -attime "$NOW" -partial_chain -CAfile "$ISSUER" "$RESPONDER" >/dev/null 2>&1 || die RESPONDER_NOT_ISSUED_BY_CA
openssl verify -attime "$NOW" -CAfile "$ROOT" -untrusted "$ISSUER" "$RESPONDER" >/dev/null 2>&1 || die RESPONDER_UNTRUSTED

# 3. status / parse
[ -f "$RESPONSE" ] || die OCSP_MISSING
ocsp_text="$(openssl ocsp -respin "$RESPONSE" -resp_text -noverify 2>&1)"
printf '%s\n' "$ocsp_text" | grep -q 'OCSP Response Status: successful' || die OCSP_RESPONSE_STATUS_ERROR
cert_status="$(printf '%s\n' "$ocsp_text" | awk -F': ' '/Cert Status:/{print $2; exit}' | tr -d ' ')"
ocsp_serial="$(text_field "$ocsp_text" 'Serial Number')"
ocsp_this="$(text_field "$ocsp_text" 'This Update')"
ocsp_next="$(text_field "$ocsp_text" 'Next Update')"
ocsp_this_e="$(to_epoch "$ocsp_this")"; ocsp_next_e="$(to_epoch "$ocsp_next")"

# 4. signer identity + signature
resp_id="$(text_field "$ocsp_text" 'Responder Id' | tr -d ' ')"
auth_keyhash="$(openssl x509 -in "$RESPONDER" -noout -ocspid | sed -n 's/.*Public key OCSP hash: *//p' | norm_hex)"
auth_subject_norm="$(openssl x509 -in "$RESPONDER" -noout -subject | sed 's/^subject=//' | tr -d ' ')"
[ -n "$resp_id" ] && [ "$resp_id" != "$auth_keyhash" ] && [ "$resp_id" != "$auth_subject_norm" ] && die RESPONDER_WRONG
openssl ocsp -respin "$RESPONSE" -issuer "$ISSUER" -VAfile "$RESPONDER" -no_nonce 2>&1 | grep -q 'Response verify OK' || die OCSP_SIGNATURE_INVALID

# 5. serial + issuer-hash binding
leaf_serial="$(openssl x509 -in "$LEAF" -noout -serial | sed 's/^serial=//')"
[ "$(norm_hex "$ocsp_serial" | sed 's/^0*//')" = "$(norm_hex "$leaf_serial" | sed 's/^0*//')" ] || die SERIAL_MISMATCH

# 6. nonce binding
req_nonce="$(extract_nonce "$(openssl ocsp -reqin "$REQUEST" -req_text -noverify 2>&1)")"
resp_nonce="$(extract_nonce "$ocsp_text")"
[ -n "$req_nonce" ] || die NONCE_MISSING_IN_REQUEST
[ -n "$resp_nonce" ] || die NONCE_MISSING_IN_RESPONSE
[ "$(norm_hex "$req_nonce")" = "$(norm_hex "$resp_nonce")" ] || die NONCE_MISMATCH

# 7. OCSP freshness
ocsp_fresh=1; ocsp_stale_reason=""
if [ -z "$ocsp_next_e" ]; then ocsp_fresh=0; ocsp_stale_reason=OCSP_NEXTUPDATE_MISSING
elif [ "$ocsp_next_e" -lt $((NOW - SKEW)) ]; then ocsp_fresh=0; ocsp_stale_reason=OCSP_STALE_NEXTUPDATE
elif [ "$ocsp_this_e" -gt $((NOW + SKEW)) ]; then ocsp_fresh=0; ocsp_stale_reason=OCSP_STALE_THISUPDATE; fi

# 8. CRL evaluation
evaluate_crl() {
  CRL_REASON=""; CRL_REVOKED=0
  local t; t="$(openssl crl -in "$CRL" -noout -text 2>&1)" || { CRL_REASON=CRL_SIGNATURE_INVALID; return 1; }
  [ "$(openssl crl -in "$CRL" -noout -issuer | sed 's/^issuer=//')" = "$ica_subject" ] || { CRL_REASON=CRL_ISSUER_MISMATCH; return 1; }
  local aki ski
  aki="$(printf '%s\n' "$t" | awk '/X509v3 Authority Key Identifier:/{getline v; print v; exit}' | norm_hex)"
  ski="$(openssl x509 -in "$ISSUER" -noout -ext subjectKeyIdentifier | tail -1 | norm_hex)"
  [ -n "$aki" ] && [ "$aki" = "$ski" ] || { CRL_REASON=CRL_AKI_MISMATCH; return 1; }
  openssl crl -in "$CRL" -noout -verify -CAfile "$ISSUER" >/dev/null 2>&1 || { CRL_REASON=CRL_SIGNATURE_INVALID; return 1; }
  local lu nu; lu="$(to_epoch "$(openssl crl -in "$CRL" -noout -lastupdate | sed 's/^lastUpdate=//')")"
  nu="$(to_epoch "$(openssl crl -in "$CRL" -noout -nextupdate | sed 's/^nextUpdate=//')")"
  [ -z "$nu" ] || [ "$nu" -lt $((NOW - SKEW)) ] || [ "$lu" -gt $((NOW + SKEW)) ] && { CRL_REASON=CRL_STALE; return 1; }
  [ "$lu" -gt "$ocsp_this_e" ] || { CRL_REASON=CRL_NOT_FRESHER; return 1; }
  printf '%s\n' "$t" | grep -qiE "Serial Number:[[:space:]]*$(norm_hex "$leaf_serial")" && CRL_REVOKED=1
  return 0
}

# 9. decision
[ "$cert_status" = "revoked" ] && die REVOKED
if [ "$cert_status" = "good" ] && [ "$ocsp_fresh" -eq 1 ]; then
  [ -n "$CRL" ] && evaluate_crl && [ "$CRL_REVOKED" -eq 1 ] && die REVOKED
  die GOOD
fi
if [ "$cert_status" = "unknown" ] && [ "$ocsp_fresh" -eq 1 ]; then
  if [ -n "$CRL" ] && evaluate_crl; then [ "$CRL_REVOKED" -eq 1 ] && die REVOKED; die GOOD_CRL_FALLBACK; fi
  die UNKNOWN
fi
[ "$ocsp_stale_reason" = "OCSP_STALE_THISUPDATE" ] && die OCSP_STALE_THISUPDATE
if [ -n "$CRL" ]; then
  if evaluate_crl; then [ "$CRL_REVOKED" -eq 1 ] && die REVOKED; die GOOD_CRL_FALLBACK; else die "$CRL_REASON"; fi
fi
die "$ocsp_stale_reason"
