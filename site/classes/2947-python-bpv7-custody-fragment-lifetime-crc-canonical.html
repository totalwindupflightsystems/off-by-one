The complete implementation is written to /workspace/bpv7.py, /workspace/scenario.py, /workspace/test_bpv7.py, /workspace/sample_scenario.json, and the assembled document /workspace/SOLUTION.md.

1. Root-cause analysis

This is a from-scratch RFC 9171 subset, and the golden vectors are byte-exact, so each of these subtle points is an independent way to fail:

# Failure mode Correct rule
1 Wrong outer container A bundle is a CBOR indefinite-length array: 0x9f + blocks + 0xff (RFC 9171 §4.1).
2 Wrong primary array length The primary block has 8/9/10/11 items depending on the fragment flag and CRC type. Fragment offset/total length are appended only when IS_FRAGMENT is set; the CRC only when crc_type != 0 (RFC 9171 §4.3.1). A fixed 11-element array is wrong.
3 Non-canonical CBOR Integer and length arguments must use the shortest form; no indefinite-length items inside blocks (RFC 8949 §4.2).
4 Wrong CRC coverage Each block's CRC is computed over that block's own CBOR encoding including the CRC field itself, temporarily set to zero bytes (e.g. 42 00 00), then substituted (RFC 9171 §4.3.1/§4.3.2). Hashing the whole bundle, hashing without the CRC field, or using a b"" placeholder all give wrong bytes.
5 Wrong CRC algorithm CRC type 1 is CRC-16/X25 (0x1021 reflected = 0x8408, init 0xFFFF, xorout 0xFFFF, network byte order); type 2 is CRC-32C. Known answers: crc16_x25(b"123456789")==0x906E, crc32c(b"123456789")==0xE3069283.
6 Wrong time base Lifetimes are on DTN time = ms since 2000-01-01T00:00:00Z, not Unix time (RFC 9171 §4.2.6). Creation time 0 means unknown and must not be treated as the year 2000.
7 Fragment extent bugs Offsets are byte offsets; reassembly must accept out-of-order and overlapping fragments and complete only when [0, total) is contiguously covered (RFC 9171 §5.9).
8 Hop-count/custody/dedup Hop count is block type 10 encoded [limit, count]; dedup keys on source + creation time (+ fragment offset/length); custody refusal triggers retransmission, not silent drop.

bpv7.py — canonical codec (stdlib only)

"""Canonical BPv7 (RFC 9171) codec."""

from __future__ import annotations

import struct
from dataclasses import dataclass, field
from datetime import datetime, timezone

DTN_EPOCH = datetime(2000, 1, 1, tzinfo=timezone.utc)

CRC_NONE = 0
CRC_16 = 1  # CRC-16/X25
CRC_32 = 2  # CRC-32C (Castagnoli)
CRC_LEN = {CRC_NONE: 0, CRC_16: 2, CRC_32: 4}

FLAG_IS_FRAGMENT = 1 << 0
FLAG_ADMIN = 1 << 1
FLAG_NO_FRAGMENT = 1 << 2

BLOCK_PAYLOAD = 1
BLOCK_PREV_NODE = 6
BLOCK_BUNDLE_AGE = 7
BLOCK_HOP_COUNT = 10


class CBORError(ValueError): ...
class CBORDecodeError(CBORError): ...
class CRCError(ValueError): ...
class ReassemblyError(ValueError): ...


def dtn_now() -> int:
    delta = datetime.now(timezone.utc) - DTN_EPOCH
    return delta.days * 86_400_000 + delta.seconds * 1000 + delta.microseconds // 1000


# --- canonical CBOR --------------------------------------------------------- #

def _head(major: int, arg: int) -> bytes:
    if arg < 0:
        raise CBORError("negative CBOR argument")
    if arg < 24:
        return bytes([(major << 5) | arg])
    if arg < 1 << 8:
        return bytes([(major << 5) | 24, arg])
    if arg < 1 << 16:
        return bytes([(major << 5) | 25]) + arg.to_bytes(2, "big")
    if arg < 1 << 32:
        return bytes([(major << 5) | 26]) + arg.to_bytes(4, "big")
    if arg < 1 << 64:
        return bytes([(major << 5) | 27]) + arg.to_bytes(8, "big")
    raise CBORError("integer too large for CBOR")


def cbor_dumps(obj) -> bytes:
    out = bytearray()
    _emit(obj, out)
    return bytes(out)


def _emit(obj, out: bytearray) -> None:
    if obj is None:
        out.append(0xF6)
    elif obj is True:
        out.append(0xF5)
    elif obj is False:
        out.append(0xF4)
    elif isinstance(obj, int):
        out += _head(0, obj) if obj >= 0 else _head(1, -1 - obj)
    elif isinstance(obj, (bytes, bytearray, memoryview)):
        buf = bytes(obj)
        out += _head(2, len(buf)) + buf
    elif isinstance(obj, str):
        buf = obj.encode("utf-8")
        out += _head(3, len(buf)) + buf
    elif isinstance(obj, (list, tuple)):
        out += _head(4, len(obj))
        for item in obj:
            _emit(item, out)
    elif isinstance(obj, dict):
        keys = sorted(obj.keys(), key=lambda k: cbor_dumps(k))  # RFC 8949 bytewise
        out += _head(5, len(keys))
        for key in keys:
            _emit(key, out)
            _emit(obj[key], out)
    else:
        raise CBORError(f"cannot CBOR-encode {type(obj)!r}")


class _Decoder:
    def __init__(self, data: bytes):
        self.data, self.pos = data, 0

    def _byte(self) -> int:
        if self.pos >= len(self.data):
            raise CBORDecodeError("truncated CBOR")
        b = self.data[self.pos]; self.pos += 1; return b

    def _read(self, n: int) -> bytes:
        if self.pos + n > len(self.data):
            raise CBORDecodeError("truncated CBOR")
        chunk = self.data[self.pos:self.pos + n]; self.pos += n; return chunk

    def _arg(self, info: int):
        if info < 24: return info
        if info == 24: return self._byte()
        if info == 25: return int.from_bytes(self._read(2), "big")
        if info == 26: return int.from_bytes(self._read(4), "big")
        if info == 27: return int.from_bytes(self._read(8), "big")
        if info == 31: return None
        raise CBORDecodeError(f"invalid additional information {info}")

    def item(self):
        ib = self._byte(); major, info = ib >> 5, ib & 0x1F
        if major == 0:
            return self._arg(info)
        if major == 1:
            return -1 - self._arg(info)
        if major in (2, 3):
            arg = self._arg(info)
            if arg is None:
                chunks = []
                while self.data[self.pos] != 0xFF:
                    chunks.append(self.item())
                self.pos += 1
                return b"".join(chunks) if major == 2 else "".join(chunks)
            raw = self._read(arg)
            return raw if major == 2 else raw.decode("utf-8")
        if major == 4:
            arg = self._arg(info)
            if arg is None:
                items = []
                while self.data[self.pos] != 0xFF:
                    items.append(self.item())
                self.pos += 1
                return items
            return [self.item() for _ in range(arg)]
        if major == 5:
            arg = self._arg(info)
            result = {}
            if arg is None:
                while self.data[self.pos] != 0xFF:
                    result[self.item()] = self.item()
                self.pos += 1
            else:
                for _ in range(arg):
                    result[self.item()] = self.item()
            return result
        if major == 6:
            self._arg(info); return self.item()
        if major == 7:
            if info == 20: return False
            if info == 21: return True
            if info in (22, 23): return None
            raise CBORDecodeError(f"unsupported simple value {info}")
        raise CBORDecodeError(f"unsupported major type {major}")


def cbor_loads(data: bytes):
    return _Decoder(data).item()


# --- CRC -------------------------------------------------------------------- #

def crc16_x25(data: bytes, crc: int = 0xFFFF) -> int:
    for byte in data:
        crc ^= byte
        for _ in range(8):
            crc = (crc >> 1) ^ 0x8408 if crc & 1 else crc >> 1
    return crc ^ 0xFFFF


def crc32c(data: bytes, crc: int = 0xFFFFFFFF) -> int:
    poly = 0x82F63B78
    for byte in data:
        crc ^= byte
        for _ in range(8):
            crc = (crc >> 1) ^ poly if crc & 1 else crc >> 1
    return crc ^ 0xFFFFFFFF


def crc_bytes(data: bytes, crc_type: int) -> bytes:
    if crc_type == CRC_NONE: return b""
    if crc_type == CRC_16: return struct.pack(">H", crc16_x25(data))
    if crc_type == CRC_32: return struct.pack(">L", crc32c(data))
    raise ValueError(f"unknown CRC type {crc_type!r}")


# --- EIDs ------------------------------------------------------------------- #

def parse_eid(text: str):
    if text in ("dtn:none", "none", ""): return [1, 0]
    if text.startswith("dtn:"): return [1, text[4:]]
    if text.startswith("ipn:"):
        node, _, service = text[4:].partition(".")
        if not node or not service: raise ValueError(f"bad ipn EID {text!r}")
        return [2, [int(node), int(service)]]
    raise ValueError(f"unsupported EID {text!r}")


def format_eid(eid) -> str:
    scheme, ssp = eid[0], eid[1]
    if scheme == 1: return "dtn:none" if ssp == 0 else f"dtn:{ssp}"
    if scheme == 2: return f"ipn:{ssp[0]}.{ssp[1]}"
    return f"eid:{scheme}:{ssp}"


def eid_to_cbor(eid):
    scheme, ssp = eid[0], eid[1]
    if scheme == 2: return [2, [int(ssp[0]), int(ssp[1])]]
    return [1, 0] if ssp == 0 else [1, str(ssp)]


def eid_from_cbor(value):
    if not isinstance(value, list) or len(value) != 2:
        raise CBORDecodeError("EID must be a 2-element array")
    scheme, ssp = value
    if scheme == 2:
        if not isinstance(ssp, list) or len(ssp) != 2:
            raise CBORDecodeError("ipn SSP must be [node, service]")
        return [2, [int(ssp[0]), int(ssp[1])]]
    return [1, ssp if ssp == 0 else str(ssp)]


def _eid_key(eid):
    scheme, ssp = eid[0], eid[1]
    return (2, (int(ssp[0]), int(ssp[1]))) if scheme == 2 else (1, ssp)


# --- blocks ----------------------------------------------------------------- #

@dataclass
class PrimaryBlock:
    destination: list
    source: list
    report_to: list
    creation_time: int
    sequence: int
    lifetime: int
    flags: int = 0
    crc_type: int = CRC_16
    fragment_offset: int = 0
    total_adu_length: int = 0
    crc: bytes = b""

    @property
    def is_fragment(self) -> bool:
        return bool(self.flags & FLAG_IS_FRAGMENT)

    def _array(self) -> list:
        arr = [
            7, self.flags, self.crc_type,
            eid_to_cbor(self.destination), eid_to_cbor(self.source),
            eid_to_cbor(self.report_to),
            [self.creation_time, self.sequence], self.lifetime,
        ]
        if self.is_fragment:
            arr.append(self.fragment_offset)
            arr.append(self.total_adu_length)
        if self.crc_type != CRC_NONE:
            arr.append(b"\x00" * CRC_LEN[self.crc_type])
        return arr

    def encode(self) -> bytes:
        arr = self._array()
        if self.crc_type != CRC_NONE:
            self.crc = crc_bytes(cbor_dumps(arr), self.crc_type)
            arr[-1] = self.crc
        return cbor_dumps(arr)


@dataclass
class CanonicalBlock:
    block_type: int
    block_number: int
    flags: int = 0
    crc_type: int = CRC_16
    data: bytes = b""
    crc: bytes = b""

    def _array(self) -> list:
        arr = [self.block_type, self.block_number, self.flags,
               self.crc_type, bytes(self.data)]
        if self.crc_type != CRC_NONE:
            arr.append(b"\x00" * CRC_LEN[self.crc_type])
        return arr

    def encode(self) -> bytes:
        arr = self._array()
        if self.crc_type != CRC_NONE:
            self.crc = crc_bytes(cbor_dumps(arr), self.crc_type)
            arr[-1] = self.crc
        return cbor_dumps(arr)


@dataclass
class Bundle:
    primary: PrimaryBlock
    blocks: list = field(default_factory=list)

    @property
    def identity(self):
        p = self.primary
        return (_eid_key(p.source), p.creation_time, p.sequence,
                p.fragment_offset if p.is_fragment else 0,
                p.total_adu_length if p.is_fragment else 0)

    def encode(self) -> bytes:
        out = bytearray(b"\x9f")
        out += self.primary.encode()
        for block in self.blocks:
            out += block.encode()
        out += b"\xff"
        return bytes(out)

    def payload_block(self):
        return self.block(BLOCK_PAYLOAD)

    def block(self, block_type: int):
        for b in self.blocks:
            if b.block_type == block_type:
                return b
        return None


def _verify_block_crc(arr, crc_type, actual, label):
    if crc_type == CRC_NONE:
        return
    placeholder = arr[:]
    placeholder[-1] = b"\x00" * CRC_LEN[crc_type]
    expected = crc_bytes(cbor_dumps(placeholder), crc_type)
    if expected != actual:
        raise CRCError(f"{label} CRC mismatch: expected {expected.hex()} got {actual.hex()}")


def decode_bundle(data: bytes, verify_crc: bool = True) -> Bundle:
    top = _Decoder(data).item() if (not data or data[0] != 0x9F) else cbor_loads(data)
    if not isinstance(top, list) or len(top) < 2:
        raise CBORDecodeError("bundle must be an array of at least two blocks")
    parr = top[0]
    if not isinstance(parr, list) or len(parr) < 8:
        raise CBORDecodeError("primary block too short")
    if parr[0] != 7:
        raise CBORDecodeError(f"unsupported BP version {parr[0]!r}")

    flags, crc_type = int(parr[1]), int(parr[2])
    idx, foff, tlen = 8, 0, 0
    if flags & FLAG_IS_FRAGMENT:
        if len(parr) < 10: raise CBORDecodeError("fragmented primary too short")
        foff, tlen, idx = int(parr[8]), int(parr[9]), 10
    crc = b""
    if crc_type != CRC_NONE:
        if len(parr) <= idx: raise CBORDecodeError("primary missing CRC")
        crc = bytes(parr[idx])
        if verify_crc: _verify_block_crc(parr, crc_type, crc, "primary")

    creation = parr[6]
    if not isinstance(creation, list) or len(creation) != 2:
        raise CBORDecodeError("creation timestamp must be 2 elements")
    primary = PrimaryBlock(
        destination=eid_from_cbor(parr[3]), source=eid_from_cbor(parr[4]),
        report_to=eid_from_cbor(parr[5]), creation_time=int(creation[0]),
        sequence=int(creation[1]), lifetime=int(parr[7]), flags=flags,
        crc_type=crc_type, fragment_offset=foff, total_adu_length=tlen, crc=crc)

    blocks = []
    for barr in top[1:]:
        if not isinstance(barr, list) or len(barr) < 5:
            raise CBORDecodeError("canonical block too short")
        bcrc_type = int(barr[3])
        bcrc = b""
        if bcrc_type != CRC_NONE:
            if len(barr) < 6: raise CBORDecodeError("canonical block missing CRC")
            bcrc = bytes(barr[5])
            if verify_crc:
                _verify_block_crc(barr, bcrc_type, bcrc, f"block {barr[0]}")
        blocks.append(CanonicalBlock(int(barr[0]), int(barr[1]), int(barr[2]),
                                     bcrc_type, bytes(barr[4]), bcrc))
    return Bundle(primary=primary, blocks=blocks)


# --- hop count -------------------------------------------------------------- #

def get_hop_count(bundle):
    block = bundle.block(BLOCK_HOP_COUNT)
    if block is None: return None
    limit, count = cbor_loads(block.data)
    return int(limit), int(count)


def set_hop_count(bundle, limit, count):
    block = bundle.block(BLOCK_HOP_COUNT)
    if block is not None:
        block.data = cbor_dumps([limit, count]); block.encode()


def increment_hop_count(bundle) -> bool:
    current = get_hop_count(bundle)
    if current is None: return True
    limit, count = current
    if count + 1 > limit: return False
    set_hop_count(bundle, limit, count + 1)
    return True


# --- fragmentation / reassembly -------------------------------------------- #

def fragment_bundle(bundle: Bundle, fragment_size: int) -> list:
    if fragment_size <= 0: raise ValueError("fragment_size must be positive")
    payload = bundle.payload_block()
    if payload is None: raise ValueError("bundle has no payload block")
    if bundle.primary.flags & FLAG_NO_FRAGMENT:
        raise ValueError("bundle is marked do-not-fragment")
    adu = payload.data
    chunks = ([b""] if not adu
              else [adu[i:i + fragment_size] for i in range(0, len(adu), fragment_size)])
    fragments = []
    for offset, chunk in enumerate(chunks):
        p = bundle.primary
        new_primary = PrimaryBlock(
            destination=[p.destination[0], p.destination[1]],
            source=[p.source[0], p.source[1]],
            report_to=[p.report_to[0], p.report_to[1]],
            creation_time=p.creation_time, sequence=p.sequence,
            lifetime=p.lifetime, flags=p.flags | FLAG_IS_FRAGMENT,
            crc_type=p.crc_type, fragment_offset=offset * fragment_size,
            total_adu_length=len(adu))
        new_blocks = []
        for block in bundle.blocks:
            data = chunk if block.block_type == BLOCK_PAYLOAD else bytes(block.data)
            new_blocks.append(CanonicalBlock(block.block_type, block.block_number,
                                             block.flags, block.crc_type, data))
        fragments.append(Bundle(new_primary, new_blocks))
    return fragments


class Reassembler:
    def __init__(self):
        self._pending = {}

    def add(self, bundle: Bundle):
        p = bundle.primary
        if not p.is_fragment:
            payload = bundle.payload_block()
            return payload.data if payload else b""
        key = (_eid_key(p.source), p.creation_time, p.sequence)
        entry = self._pending.setdefault(key, {"total": p.total_adu_length, "extents": []})
        if entry["total"] != p.total_adu_length:
            raise ReassemblyError("inconsistent total ADU length")
        payload = bundle.payload_block()
        data = payload.data if payload else b""
        if p.fragment_offset + len(data) > entry["total"]:
            raise ReassemblyError("fragment extends past total ADU length")
        entry["extents"].append((p.fragment_offset, data))
        coverage = self._coverage(entry)
        if coverage is None:
            return None
        self._pending.pop(key, None)
        return coverage

    @staticmethod
    def _coverage(entry):
        total = entry["total"]
        if total == 0: return b""
        out, cursor = bytearray(), 0
        for start, data in sorted(entry["extents"]):
            if start > cursor: return None
            if start + len(data) <= cursor: continue
            out += data[cursor - start:]
            cursor = start + len(data)
            if cursor >= total: break
        return bytes(out[:total]) if cursor == total else None


# --- lifetime --------------------------------------------------------------- #

def is_expired(bundle: Bundle, now_dtn: int) -> bool:
    creation = bundle.primary.creation_time
    if creation == 0: return False
    return (now_dtn - creation) > bundle.primary.lifetime

scenario.py — deterministic store-and-forward runner

The runner keeps one record per originating bundle, fragments it if fragment_size is set, walks each Packet along the route over active contacts, applies bit-flip injections to the wire bytes, verifies CRCs on receive, retransmits on CRC failure or custody refusal, dedups by bundle.identity, reassembles fragments at the destination, and emits JSONL.

Key logic (full file in /workspace/scenario.py):

def _transmit(self, bid, record, packet, now, bnode, contact_key) -> bool:
    wire = packet.bundle.encode()
    wire = self._fire_injection(bid, bnode, "bitflip", wire)

    received = None
    try:
        received = B.decode_bundle(wire, verify_crc=True)
    except (B.CRCError, B.CBORError):
        pass

    if received is None:                       # integrity failure
        if record["custody"] and record["retransmissions"] < self.max_retransmits:
            record["retransmissions"] += 1
            packet.last_contact = None         # retry on next contact
            return True
        self._finish(bid, "CRC_REJECTED", now)
        return False

    if not B.increment_hop_count(received):    # hop limit exceeded
        self._finish(bid, "EXPIRED", now)
        return False

    identity = received.identity
    seen = record["received"].setdefault(bnode, set())
    if identity in seen:                       # replay
        if record["terminal"] is None:
            self._finish(bid, "DEDUP_DROPPED", now)
        return False
    seen.add(identity)

    if record["custody"] and bnode in self.refuse:   # custody refusal
        if record["retransmissions"] < self.max_retransmits:
            record["retransmissions"] += 1
            packet.last_contact = None
            return True
        return False

    packet.bundle, packet.hop_index, packet.holder = received, packet.hop_index + 1, bnode
    if bnode == record["dest_node"]:
        self._deliver(bid, record, received, now)
    self._fire_injection(bid, bnode, "replay", wire)
    return True


def _deliver(self, bid, record, received, now) -> None:
    if not received.primary.is_fragment:
        self._finish(bid, "DELIVERED", now); return
    payload = received.payload_block()
    assembled = record["reassembly"].add(
        received.primary.fragment_offset, payload.data if payload else b"")
    if assembled is not None:
        self._finish(bid, "DELIVERED", now)

Expiry is checked before each event with B.is_expired (DTN time), the destination path is chosen from the bundle's route, and custody retries are bounded by custody.max_retransmits.
