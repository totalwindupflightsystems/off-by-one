Verkle IPA Pedersen Multiproof — Diagnosis and Fix

Root Cause Analysis

The common failure modes for this problem class are:

  1. The IPA does not bind the claimed evaluation. A plain Pedersen/IPA reduction over <a, G> only proves knowledge of a. To bind y = <a, b>, the verifier must use the Bulletproofs relation with an extra generator Q: P = <a,G> + <b,H> + yQ. Omitting the yQ term lets an attacker prove an unrelated opening.

  2. The Fiat–Shamir transcript omits part of the statement. Challenges must be chained over the commitment, sorted indices, values, and every L_j/R_j tape entry. Otherwise proofs can be replayed against a different commitment or evaluation point.

  3. Decoders are permissive. Accepting scalars >= r, non-48-byte points, uncompressed/hybrid G1 points, infinity where disallowed, off-curve points, or cofactor-component points breaks binding and canonical byte agreement.

  4. Subgroup check scalar reduction bug. Calling mulCT(P, r) where mulCT reduces r modulo the scalar field computes [0]P = O, so every point passes. The subgroup test must use raw scalar multiplication, or the fast BLS12-381 G1 endomorphism check.

  5. Folding orientation/sign mistakes. Generator folding uses the highest index bit first. The verifier’s s_i coefficients and the <b,H> term in the final MSM must match the prover’s fold order and signs.

  6. Naïve prover performance. Folding the G/H basis with scalar multiplication every round is too slow. The fix uses coefficient lists over the original basis plus sparse Pippenger MSMs with mixed affine/Jacobian additions.

'use strict';
/*
 * Dependency-free Verkle / Pedersen vector commitment + IPA multiproofs
 * over BLS12-381 G1 for Node 20.
 *
 * Public points are always canonical 48-byte compressed encodings.  The
 * internal point representation is Jacobian projective coordinates.
 */
const crypto = require('node:crypto');

// ---------------------------------------------------------------------------
// BLS12-381 parameters
// ---------------------------------------------------------------------------
const P = 0x1a0111ea397fe69a4b1ba7b6434bacd764774b84f38512bf6730d2a0f6b0f6241eabfffeb153ffffb9feffffffffaaabn;
const R = 0x73eda753299d7d483339d80809a1d80553bda402fffe5bfeffffffff00000001n;
const CURVE_B = 4n;
const H_COFACTOR = 0x396c8c005555e1568c00aaab0000aaabn;
const G1X = 0x17f1d3a73197d7942695638c4fa9ac0fc3688c4f9774b905a14e3a3f171bac586c55e83ff97a1aeffb3af00adb22c6bbn;
const G1Y = 0x08b3f481e3aaa0f1a09e30ed741d8ae4fcf5e095d5d00af600db18cb2c04b3edd03cc744a2888ae40caa232946c5e7e1n;
const MASK = (1n << 512n) - 1n;

const mod = (a, m = P) => { a %= m; return a < 0n ? a + m : a; };
const modR = (a) => mod(a, R);
const inv = (a, m = P) => {
  a = mod(a, m);
  if (a === 0n) throw new Error('division by zero');
  let lm = 1n, hm = 0n, low = a, high = m;
  while (low > 1n) {
    const q = high / low;
    [lm, hm] = [hm - lm * q, lm];
    [low, high] = [high - low * q, low];
  }
  return mod(lm, m);
};
const powMod = (base, exp, m = P) => {
  base = mod(base, m);
  let out = 1n;
  while (exp > 0n) {
    if (exp & 1n) out = mod(out * base, m);
    base = mod(base * base, m);
    exp >>= 1n;
  }
  return out;
};

// ---------------------------------------------------------------------------
// Jacobian G1 arithmetic. No scalar-dependent branches are used in mulCT().
// ---------------------------------------------------------------------------
const INF = Object.freeze({ X: 0n, Y: 1n, Z: 0n });
const isInf = (a) => a.Z === 0n;
const pt = (X, Y, Z) => ({ X, Y, Z });

function jDouble(a) {
  if (a.Z === 0n || a.Y === 0n) return INF;
  const A = mod(a.X * a.X);
  const B = mod(a.Y * a.Y);
  const C = mod(B * B);
  const t = mod((a.X + B) * (a.X + B) - A - C);
  const D = mod(2n * t);
  const E = mod(3n * A);
  const F = mod(E * E);
  const X3 = mod(F - 2n * D);
  const Y3 = mod(E * (D - X3) - 8n * C);
  const Z3 = mod(2n * a.Y * a.Z);
  return pt(X3, Y3, Z3);
}

function jAdd(a, b) {
  if (a.Z === 0n) return b;
  if (b.Z === 0n) return a;
  const Z1Z1 = mod(a.Z * a.Z);
  const Z2Z2 = mod(b.Z * b.Z);
  const U1 = mod(a.X * Z2Z2);
  const U2 = mod(b.X * Z1Z1);
  const S1 = mod(a.Y * b.Z * Z2Z2);
  const S2 = mod(b.Y * a.Z * Z1Z1);
  const H = mod(U2 - U1);
  const rr = mod(2n * (S2 - S1));
  if (H === 0n) return rr === 0n ? jDouble(a) : INF;
  const I = mod((2n * H) * (2n * H));
  const J = mod(H * I);
  const V = mod(U1 * I);
  const X3 = mod(rr * rr - J - 2n * V);
  const Y3 = mod(rr * (V - X3) - 2n * S1 * J);
  const Z3 = mod(((a.Z + b.Z) * (a.Z + b.Z) - Z1Z1 - Z2Z2) * H);
  return pt(X3, Y3, Z3);
}

function cselect(a, b, mask) {
  const nm = MASK ^ mask;
  return pt((a.X & nm) | (b.X & mask),
            (a.Y & nm) | (b.Y & mask),
            (a.Z & nm) | (b.Z & mask));
}

// Fast BLS12-381 G1 subgroup check: phi(P) == [-x^2]P.
const BLS_X = 0xd201000000010000n;
const BETA = 0x5f19672fdf76ce51ba69c6076a0f77eaddb3a93be6f89688de17d813620a00022e01fffffffefffen;
function mulCTBits(a, k, bits) {
  let out = INF;
  for (let i = bits - 1; i >= 0; i--) {
    out = jDouble(out);
    const sum = jAdd(out, a);
    const bit = (k >> BigInt(i)) & 1n;
    out = cselect(out, sum, bit ? MASK : 0n);
  }
  return out;
}

// Mixed Jacobian + affine addition.
function jAddMixed(a, b) {
  if (a.Z === 0n) return pt(mod(b.x), mod(b.y), 1n);
  const z1z1 = mod(a.Z * a.Z);
  const u2 = mod(b.x * z1z1);
  const s2 = mod(b.y * a.Z * z1z1);
  const h = mod(u2 - a.X);
  const rr = mod(2n * (s2 - a.Y));
  if (h === 0n) return rr === 0n ? jDouble(a) : INF;
  const hh = mod(h * h);
  const i = mod(4n * hh);
  const j = mod(h * i);
  const v = mod(a.X * i);
  const x3 = mod(rr * rr - j - 2n * v);
  const y3 = mod(rr * (v - x3) - 2n * a.Y * j);
  const z3 = mod((a.Z + h) * (a.Z + h) - z1z1 - hh);
  return pt(x3, y3, z3);
}

// Fixed 255-bit double-and-add with masked select.
function mulCTRaw(a, k) {
  let out = INF;
  for (let i = 254; i >= 0; i--) {
    out = jDouble(out);
    const sum = jAdd(out, a);
    const bit = (k >> BigInt(i)) & 1n;
    out = cselect(out, sum, bit ? MASK : 0n);
  }
  return out;
}
function mulCT(a, k) {
  return mulCTRaw(a, modR(k));
}

function toAffine(a) {
  if (a.Z === 0n) return null;
  const zi = inv(a.Z);
  const zi2 = mod(zi * zi);
  return { x: mod(a.X * zi2), y: mod(a.Y * zi2 * zi) };
}

function fromAffine(x, y) {
  if (mod(y * y) !== mod(x * x * x + CURVE_B) || x >= P || y >= P) {
    throw new Error('point is not on BLS12-381 G1');
  }
  return pt(mod(x), mod(y), 1n);
}

function eqPoint(a, b) {
  if (a.Z === 0n || b.Z === 0n) return a.Z === 0n && b.Z === 0n;
  const aX = mod(a.X * b.Z * b.Z);
  const bX = mod(b.X * a.Z * a.Z);
  const aY = mod(a.Y * b.Z * b.Z * b.Z);
  const bY = mod(b.Y * a.Z * a.Z * a.Z);
  return aX === bX && aY === bY;
}

const BASE = fromAffine(G1X, G1Y);

// ---------------------------------------------------------------------------
// Canonical encodings
// ---------------------------------------------------------------------------
function bytesToBig(b) {
  let x = 0n;
  for (const v of b) x = (x << 8n) | BigInt(v);
  return x;
}
function bigToBytes(x, len) {
  const out = new Uint8Array(len);
  let v = x;
  for (let i = len - 1; i >= 0; i--) { out[i] = Number(v & 255n); v >>= 8n; }
  if (v !== 0n) throw new Error('integer does not fit');
  return out;
}
function u32(n) {
  if (!Number.isInteger(n) || n < 0 || n > 0xffffffff) throw new Error('bad u32');
  return bigToBytes(BigInt(n), 4);
}
function readU32(b, o) {
  if (o + 4 > b.length) throw new Error('truncated u32');
  return ((b[o] << 24) | (b[o+1] << 16) | (b[o+2] << 8) | b[o+3]) >>> 0;
}

function encodeScalar(s) {
  s = modR(s);
  return bigToBytes(s, 32);
}
function decodeScalar(b) {
  if (!(b instanceof Uint8Array) || b.length !== 32) throw new Error('scalar must be 32 bytes');
  const s = bytesToBig(b);
  if (s >= R) throw new Error('non-canonical scalar');
  return s;
}
function canonicalScalar(v) {
  if (typeof v === 'bigint') {
    if (v < 0n || v >= R) throw new Error('non-canonical scalar');
    return v;
  }
  if (typeof v === 'number') {
    if (!Number.isSafeInteger(v) || v < 0) throw new Error('non-canonical scalar');
    const x = BigInt(v);
    if (x >= R) throw new Error('non-canonical scalar');
    return x;
  }
  if (typeof v === 'string') {
    if (!/^[0-9a-fA-F]{1,64}$/.test(v)) throw new Error('non-canonical scalar');
    const x = BigInt('0x' + v);
    if (x >= R) throw new Error('non-canonical scalar');
    return x;
  }
  throw new Error('non-canonical scalar');
}

function encodePoint(p) {
  const a = toAffine(p);
  if (!a) {
    const out = new Uint8Array(48);
    out[0] = 0xc0;
    return out;
  }
  const xb = bigToBytes(a.x, 48);
  let flags = 0x80;
  if (a.y > P - a.y) flags |= 0x20;
  if (xb[0] & 0x80) throw new Error('x coordinate does not fit 381 bits');
  xb[0] |= flags;
  return xb;
}

function isInSubgroup(p) {
  if (isInf(p)) return true;
  const phi = pt(mod(p.X * BETA), p.Y, p.Z);
  const xP = mulCTBits(p, BLS_X, 64);
  const neg = pt(xP.X, mod(-xP.Y), xP.Z);
  return eqPoint(mulCTBits(neg, BLS_X, 64), phi);
}

function decodePoint(b, { allowInfinity = false } = {}) {
  if (!(b instanceof Uint8Array) || b.length !== 48) {
    throw new Error('G1 point must be a canonical 48-byte compressed encoding');
  }
  const raw = new Uint8Array(b);
  const compression = (raw[0] & 0x80) !== 0;
  const infinity = (raw[0] & 0x40) !== 0;
  const sort = (raw[0] & 0x20) !== 0;
  if (!compression) throw new Error('uncompressed / hybrid G1 encoding rejected');
  raw[0] &= 0x1f;
  const x = bytesToBig(raw);
  if (infinity) {
    if (x !== 0n || sort || raw.some((v, i) => i > 0 && v !== 0)) {
      throw new Error('non-canonical infinity encoding');
    }
    if (!allowInfinity) throw new Error('infinity point rejected');
    return INF;
  }
  if (x >= P) throw new Error('non-canonical field element in G1 encoding');
  const y2 = mod(x * x * x + CURVE_B);
  let y = powMod(y2, (P + 1n) / 4n);
  if (mod(y * y) !== y2) throw new Error('G1 encoding is not on the curve');
  if (y === 0n) throw new Error('invalid zero y');
  const yOther = P - y;
  if ((y > yOther) !== sort) y = yOther;
  const out = fromAffine(x, y);
  if (!isInSubgroup(out)) throw new Error('G1 point is not in the prime-order subgroup');
  return out;
}

// ---------------------------------------------------------------------------
// MSM (Pippenger with a size-selected window).
// ---------------------------------------------------------------------------
function msm(points, scalars, forceC) {
  if (points.length !== scalars.length) throw new Error('msm length mismatch');
  if (points.length === 0) return INF;
  const N = points.length;
  let C = 4;
  if (N > 3000) C = 9;
  else if (N > 1000) C = 8;
  else if (N > 400) C = 7;
  else if (N > 70) C = 5;
  if (forceC !== undefined) C = forceC;
  const W = 1 << C;
  const windows = Math.ceil(255 / C);
  let result = INF;
  for (let w = windows - 1; w >= 0; w--) {
    for (let i = 0; i < C; i++) result = jDouble(result);
    const buckets = new Array(W).fill(INF);
    const shift = BigInt(w * C);
    for (let i = 0; i < points.length; i++) {
      const d = Number((scalars[i] >> shift) & BigInt(W - 1));
      if (d !== 0) {
        const q = points[i];
        buckets[d] = (q.x !== undefined) ? jAddMixed(buckets[d], q) : jAdd(buckets[d], q);
      }
    }
    let running = INF;
    let total = INF;
    for (let j = W - 1; j >= 1; j--) {
      running = jAdd(running, buckets[j]);
      total = jAdd(total, running);
    }
    result = jAdd(result, total);
  }
  return result;
}

// ---------------------------------------------------------------------------
// Deterministic hash-to-curve basis derivation.
// ---------------------------------------------------------------------------
function sha256(...parts) {
  const h = crypto.createHash('sha256');
  for (const p of parts) h.update(p);
  return h.digest();
}

function hashToField(seed) {
  const a = sha256(seed, Uint8Array.of(1));
  const b = sha256(seed, Uint8Array.of(2));
  return mod(bytesToBig(Buffer.concat([a, b])));
}

function hashToG1(label) {
  for (let ctr = 0; ; ctr++) {
    const seed = Buffer.concat([Buffer.from('verkle-ipa-basis-v1', 'utf8'), label, u32(ctr)]);
    const x = hashToField(seed);
    if (x >= P) continue;
    const y2 = mod(x * x * x + CURVE_B);
    let y = powMod(y2, (P + 1n) / 4n);
    if (mod(y * y) !== y2) continue;
    let q = fromAffine(x, y);
    q = mulCT(q, H_COFACTOR);
    if (isInf(q)) continue;
    return q;
  }
}

let DEFAULT_CTX = null;
class Context {
  constructor(n = 256, domain = 'verkle-ipa-v1') {
    if (!Number.isInteger(n) || n <= 0 || (n & (n - 1)) !== 0) throw new Error('n must be a power of two');
    this.n = n;
    this.logN = Math.log2(n);
    this.domain = domain;
    this.G = new Array(n);
    this.H = new Array(n);
    for (let i = 0; i < n; i++) this.G[i] = hashToG1(Buffer.concat([Buffer.from('G'), u32(i)]));
    for (let i = 0; i < n; i++) this.H[i] = hashToG1(Buffer.concat([Buffer.from('H'), u32(i)]));
    this.Q = hashToG1(Buffer.from('Q'));
    this.B = hashToG1(Buffer.from('blind'));
    this.Ga = this.G.map(toAffine);
    this.Ha = this.H.map(toAffine);
    this.Qa = toAffine(this.Q);
    this.Ba = toAffine(this.B);
  }
}
function defaultContext() {
  if (!DEFAULT_CTX) DEFAULT_CTX = new Context(256, 'verkle-ipa-v1');
  return DEFAULT_CTX;
}

// ---------------------------------------------------------------------------
// Transcript: every challenge hashes the entire ordered transcript.
// ---------------------------------------------------------------------------
class Transcript {
  constructor(label) {
    this.state = sha256(Buffer.from('verkle-transcript-v1'), Buffer.from(label, 'utf8'));
  }
  append(label, data) {
    const l = Buffer.from(label, 'utf8');
    const d = Buffer.from(data);
    this.state = sha256(this.state, u32(l.length), l, u32(d.length), d);
    return this;
  }
  challenge(label) {
    const l = Buffer.from(label, 'utf8');
    const c = sha256(this.state, Buffer.from('challenge'), u32(l.length), l);
    this.state = sha256(this.state, Buffer.from('challenge'), u32(l.length), l, c);
    return modR(bytesToBig(c));
  }
}

// ---------------------------------------------------------------------------
// Proof object and canonical serialization
// ---------------------------------------------------------------------------
const MAGIC = Buffer.from('VKIP');
const VERSION = 1;

function serializeProof(proof) {
  const n = proof.n;
  const logN = Math.log2(n);
  if (!Number.isInteger(logN) || proof.indices.length !== proof.values.length) throw new Error('bad proof shape');
  if (proof.L.length !== logN || proof.R.length !== logN) throw new Error('bad proof tape length');
  const parts = [MAGIC, Uint8Array.of(VERSION), u32(n), u32(proof.indices.length), Buffer.from(proof.commitment)];
  for (let i = 0; i < proof.indices.length; i++) {
    parts.push(u32(proof.indices[i]), encodeScalar(proof.values[i]));
  }
  for (let i = 0; i < logN; i++) {
    parts.push(Buffer.from(proof.L[i]), Buffer.from(proof.R[i]));
  }
  parts.push(encodeScalar(proof.a), encodeScalar(proof.b), encodeScalar(proof.blind || 0n));
  return Buffer.concat(parts);
}

function deserializeProof(bytes) {
  const b = Buffer.from(bytes);
  let o = 0;
  const take = (n) => {
    if (o + n > b.length) throw new Error('truncated proof');
    const x = b.subarray(o, o + n); o += n; return x;
  };
  if (!b.subarray(0, 4).equals(MAGIC)) throw new Error('bad proof magic');
  o = 4;
  const ver = take(1)[0];
  if (ver !== VERSION) throw new Error('unsupported proof version');
  const n = readU32(take(4), 0);
  if (!Number.isInteger(n) || n <= 0 || (n & (n - 1)) !== 0 || n > 65536) throw new Error('bad vector length');
  const logN = Math.log2(n);
  const m = readU32(take(4), 0);
  if (m === 0 || m > n) throw new Error('bad opening count');
  const commitment = new Uint8Array(take(48));
  decodePoint(commitment, { allowInfinity: true });
  const indices = new Array(m);
  const values = new Array(m);
  let last = -1;
  for (let i = 0; i < m; i++) {
    const idx = readU32(take(4), 0);
    if (idx >= n || idx <= last) throw new Error('indices must be strictly increasing');
    last = idx;
    indices[i] = idx;
    values[i] = decodeScalar(new Uint8Array(take(32)));
  }
  const L = new Array(logN), RR = new Array(logN);
  for (let i = 0; i < logN; i++) {
    L[i] = new Uint8Array(take(48)); decodePoint(L[i], { allowInfinity: true });
    RR[i] = new Uint8Array(take(48)); decodePoint(RR[i], { allowInfinity: true });
  }
  const a = decodeScalar(new Uint8Array(take(32)));
  const bscalar = decodeScalar(new Uint8Array(take(32)));
  const blind = decodeScalar(new Uint8Array(take(32)));
  if (o !== b.length) throw new Error('trailing bytes in proof');
  return { n, indices, values, commitment, L, R: RR, a, b: bscalar, blind };
}

// ---------------------------------------------------------------------------
// Multiproof
// ---------------------------------------------------------------------------
function validateValues(ctx, values) {
  if (!Array.isArray(values) || values.length !== ctx.n) throw new Error(`expected ${ctx.n} values`);
  const out = new Array(ctx.n);
  for (let i = 0; i < ctx.n; i++) out[i] = modR(BigInt(values[i]));
  return out;
}

function commit(values, blind = 0n, ctx = defaultContext()) {
  const a = validateValues(ctx, values);
  let C = msm(ctx.G, a);
  blind = modR(blind);
  if (blind !== 0n) C = jAdd(C, mulCT(ctx.B, blind));
  return { commitment: encodePoint(C), blind };
}

function selectorWeights(indices, rho, n) {
  const b = new Array(n).fill(0n);
  let pow = 1n;
  for (const idx of indices) {
    b[idx] = modR(b[idx] + pow);
    pow = modR(pow * rho);
  }
  return b;
}

function buildPublic(ctx, commitmentBytes, indices, values) {
  const cb = Uint8Array.from(commitmentBytes);
  const t = new Transcript(ctx.domain);
  t.append('commitment', cb);
  t.append('openings', u32(indices.length));
  for (let i = 0; i < indices.length; i++) {
    t.append('index', u32(indices[i]));
    t.append('value', encodeScalar(values[i]));
  }
  const rho = t.challenge('aggregation');
  const b = selectorWeights(indices, rho, ctx.n);
  let y = 0n;
  let pow = 1n;
  for (const v of values) { y = modR(y + pow * v); pow = modR(pow * rho); }
  t.append('ipa', Buffer.from('start'));
  return { t, b, y, rho };
}

function foldScalarList(list, s) {
  if (s === 1n) return list;
  if (s === 0n) return [];
  return list.map(([i, c]) => [i, modR(c * s)]);
}

function prove(values, indices, opts = {}) {
  const ctx = opts.ctx || defaultContext();
  const a0 = validateValues(ctx, values);
  if (!Array.isArray(indices) || indices.length === 0) throw new Error('at least one opening required');
  const seen = new Set();
  for (const i of indices) {
    if (!Number.isInteger(i) || i < 0 || i >= ctx.n || seen.has(i)) throw new Error('invalid / duplicate opening index');
    seen.add(i);
  }
  indices = [...indices].sort((x, y) => x - y);
  const vals = indices.map(i => a0[i]);
  const blind = modR(opts.blind || 0n);
  let Cpt = msm(ctx.G, a0);
  if (blind !== 0n) Cpt = jAdd(Cpt, mulCT(ctx.B, blind));
  const commitment = encodePoint(Cpt);
  const pub = buildPublic(ctx, commitment, indices, vals);
  const t = pub.t;
  const n = ctx.n;
  const logN = ctx.logN;

  let a = a0.slice();
  let b = pub.b;
  let gcoef = new Array(n);
  let hcoef = new Array(n);
  for (let i = 0; i < n; i++) { gcoef[i] = [[i, 1n]]; hcoef[i] = [[i, 1n]]; }
  const L = [], RR = [];
  let m = n;
  for (let round = 0; round < logN; round++) {
    const h = m >> 1;
    const aL = a.slice(0, h), aR = a.slice(h);
    const bL = b.slice(0, h), bR = b.slice(h);
    const Lpts = [], Lsc = [], Rpts = [], Rsc = [];
    let qL = 0n, qR = 0n;
    for (let i = 0; i < h; i++) {
      const al = aL[i], ar = aR[i], bl = bL[i], br = bR[i];
      for (const [j, c] of gcoef[i + h]) { const s = modR(c * al); if (s !== 0n) { Lpts.push(ctx.Ga[j]); Lsc.push(s); } }
      for (const [j, c] of hcoef[i])     { const s = modR(c * br); if (s !== 0n) { Lpts.push(ctx.Ha[j]); Lsc.push(s); } }
      for (const [j, c] of gcoef[i])     { const s = modR(c * ar); if (s !== 0n) { Rpts.push(ctx.Ga[j]); Rsc.push(s); } }
      for (const [j, c] of hcoef[i + h]) { const s = modR(c * bl); if (s !== 0n) { Rpts.push(ctx.Ha[j]); Rsc.push(s); } }
      qL = modR(qL + al * br);
      qR = modR(qR + ar * bl);
    }
    Lpts.push(ctx.Qa); Lsc.push(qL);
    Rpts.push(ctx.Qa); Rsc.push(qR);
    const Lpt = msm(Lpts, Lsc);
    const Rpt = msm(Rpts, Rsc);
    const Lb = encodePoint(Lpt), Rb = encodePoint(Rpt);
    L.push(Lb); RR.push(Rb);
    t.append('L', Lb); t.append('R', Rb);
    const x = t.challenge(`round-${round}`);
    const xi = inv(x, R);
    const aNext = new Array(h), bNext = new Array(h);
    for (let i = 0; i < h; i++) {
      aNext[i] = modR(aL[i] * x + aR[i] * xi);
      bNext[i] = modR(bL[i] * xi + bR[i] * x);
    }
    const ng = new Array(h), nh = new Array(h);
    for (let i = 0; i < h; i++) {
      ng[i] = foldScalarList(gcoef[i], xi).concat(foldScalarList(gcoef[i + h], x));
      nh[i] = foldScalarList(hcoef[i], x).concat(foldScalarList(hcoef[i + h], xi));
    }
    a = aNext; b = bNext; gcoef = ng; hcoef = nh; m = h;
  }
  return {
    n, indices, values: vals, commitment,
    L, R: RR, a: modR(a[0]), b: modR(b[0]), blind
  };
}

// ---------------------------------------------------------------------------
// Verification
// ---------------------------------------------------------------------------
function verify(proof, opts = {}) {
  try {
    const ctx = opts.ctx || defaultContext();
    if (!proof || proof.n !== ctx.n) return false;
    const n = ctx.n, logN = ctx.logN;
    if (!Array.isArray(proof.indices) || !Array.isArray(proof.values)) return false;
    if (proof.indices.length === 0 || proof.indices.length !== proof.values.length) return false;
    if (!Array.isArray(proof.L) || !Array.isArray(proof.R) || proof.L.length !== logN || proof.R.length !== logN) return false;
    const indices = proof.indices.slice();
    const values = proof.values.map(v => canonicalScalar(v));
    let last = -1;
    for (const i of indices) {
      if (!Number.isInteger(i) || i < 0 || i >= n || i <= last) return false;
      last = i;
    }
    const C = decodePoint(Uint8Array.from(proof.commitment), { allowInfinity: true });
    const pub = buildPublic(ctx, proof.commitment, indices, values);
    const t = pub.t;
    let b = pub.b;
    let m = n;
    let sumLR = INF;
    const challenges = [];
    const Lpts = [], Rpts = [];
    for (let round = 0; round < logN; round++) {
      const L = decodePoint(Uint8Array.from(proof.L[round]), { allowInfinity: true });
      const Rr = decodePoint(Uint8Array.from(proof.R[round]), { allowInfinity: true });
      Lpts.push(L); Rpts.push(Rr);
      t.append('L', proof.L[round]); t.append('R', proof.R[round]);
      const x = t.challenge(`round-${round}`);
      const xi = inv(x, R);
      challenges.push(x);
      const h = m >> 1;
      const bNext = new Array(h);
      for (let i = 0; i < h; i++) bNext[i] = modR(b[i] * xi + b[i + h] * x);
      b = bNext;
      const x2 = modR(x * x), xi2 = modR(xi * xi);
      sumLR = jAdd(sumLR, jAdd(mulCT(L, x2), mulCT(Rr, xi2)));
      m = h;
    }
    const bFinal = modR(b[0]);
    if (canonicalScalar(proof.b) !== bFinal) return false;
    const aFinal = canonicalScalar(proof.a);

    const gScalars = new Array(n);
    const hScalars = new Array(n).fill(0n);
    const xis = challenges.map(x => inv(x, R));
    for (let i = 0; i < n; i++) {
      let sg = 1n, sh = 1n;
      for (let j = 0; j < logN; j++) {
        const bit = (i >> (logN - 1 - j)) & 1;
        const x = challenges[j];
        const xi = xis[j];
        if (bit === 0) { sg = modR(sg * xi); sh = modR(sh * x); }
        else           { sg = modR(sg * x);  sh = modR(sh * xi); }
      }
      gScalars[i] = modR(aFinal * sg);
      hScalars[i] = modR(bFinal * sh - pub.b[i]);
    }
    const pts = [], sc = [];
    for (let i = 0; i < n; i++) { if (gScalars[i]) { pts.push(ctx.Ga[i]); sc.push(gScalars[i]); } }
    for (let i = 0; i < n; i++) { if (hScalars[i]) { pts.push(ctx.Ha[i]); sc.push(hScalars[i]); } }
    const ab = modR(aFinal * bFinal);
    const qCoeff = modR(ab - pub.y);
    if (qCoeff) { pts.push(ctx.Qa); sc.push(qCoeff); }
    const blind = canonicalScalar(proof.blind || 0n);
    if (blind) { pts.push(ctx.Ba); sc.push(blind); }
    for (let j = 0; j < logN; j++) {
      const x = challenges[j], xi = xis[j];
      const x2 = modR(x * x), xi2 = modR(xi * xi);
      pts.push(Lpts[j]); sc.push(modR(-x2));
      pts.push(Rpts[j]); sc.push(modR(-xi2));
    }
    pts.push(C); sc.push(modR(-1n));
    const diff = msm(pts, sc);
    return isInf(diff);
  } catch (e) {
    return false;
  }
}
