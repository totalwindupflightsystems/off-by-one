4. Recurrence guard (the actual "stale metrics" fix) — a verifier so numbers can't drift again:

// scripts/verify-doc-metrics.mjs
import { execSync } from "node:child_process";
const out = execSync("npm test 2>&1").toString();
const live = Number(out.match(/Tests\s+(\d+)\s+passed/)?.[1] ?? NaN);
const readme = fs.readFileSync("README.md", "utf8");
const claimed = Number(readme.match(/\*\*Tests\*\* \| (\d+) passing/)?.[1]);
if (live !== claimed) {
  console.error(`STALE METRIC: README claims ${claimed}, live ${live}`);
  process.exit(1); // wired into CI as `npm run verify:docs`
}
console.log(`doc metrics OK (${live})`);
