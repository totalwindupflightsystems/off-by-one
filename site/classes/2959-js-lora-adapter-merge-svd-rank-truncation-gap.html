LoRA Adapter Merge with SVD Rank Truncation — Diagnosis and Verified Fix

Root-cause analysis

The failures in this problem class come from conflating five separate concerns:

  1. Accumulation precision. Each adapter contribution must be formed and summed in float64. Rounding dW to float32 before the SVD loses accuracy and can break the required 1e-12 sequential-equivalence check.

  2. Numerically reliable SVD. A naive eigendecomposition of dW^T dW squares the condition number. The implementation below uses a dependency-free one-sided Jacobi SVD, transposing wide matrices so the Jacobi sweep runs over the smaller dimension. It returns descending singular values and orthonormal factors and reconstructs both tall and wide matrices to roughly machine precision.

  3. Correct truncation expression. With dW = U S V^T, the retained matrix is U_r S_r V_r^T. The discarded Frobenius mass is sqrt(sum_{i>r} sigma_i^2), and the reported relative error is computed against the actual emitted W', not merely assumed.

  4. Overflow and non-finite validation. The effective scale scale/r must be finite. The final dW must also be representable as finite float32; checking only the input scales misses overflow created by B*A.

  5. Sequential-equivalence conflict. The literal predicate “K == 1 or r_max >= sum(r_i)” conflicts with rank truncation when K == 1 and r_max < r_1: no rank-r_max result can equal the full single adapter. The implementation resolves this in favor of the explicitly required sequential equality by retaining the single adapter’s full rank in that one case and reporting truncation_applied: false.

Query evaluation uses factors only when the requested rank satisfies
r_max * (d_in + d_out) < d_out * d_in, and additionally verifies the actually retained rank is cheaper in the exceptional K == 1 case. Each query reports path: "factors" or path: "materialized".

'use strict';
// Complete, dependency-free LoRA merge + rank-truncating Jacobi SVD.
// Input: {layers:[{layer?,d_out,d_in,W,adapters:[{A,B,scale,r}],r_max,queries?}]}
// A is r x d_in, B is d_out x r; W is row-major float32 hex (string/array) or numbers.

const F32_MAX = 3.4028234663852886e38;

function flatten(v) {
  if (v == null) return [];
  if (typeof v === 'number' || typeof v === 'string') return [v];
  if (ArrayBuffer.isView(v)) return Array.from(v);
  if (Array.isArray(v)) return v.flat(Infinity);
  return [v];
}
function f32hex(t) {
  if (typeof t === 'number') return t;
  let s = String(t).trim();
  if (/^0x/i.test(s)) s = s.slice(2);
  if (!/^[0-9a-fA-F]{8}$/.test(s)) throw new Error(`bad float32 hex: ${t}`);
  const b = new ArrayBuffer(4), d = new DataView(b);
  d.setUint32(0, parseInt(s, 16), false);
  return d.getFloat32(0, false);
}
function f32list(x) {
  if (typeof x === 'string') {
    const s = x.trim(), t = s.replace(/\s+/g, '');
    if (/^[0-9a-fA-F\s]+$/.test(s) && t.length % 8 === 0) {
      const out = [];
      for (let i = 0; i < t.length; i += 8) out.push(f32hex(t.slice(i, i + 8)));
      return out;
    }
    return s.split(/[\s,;]+/).filter(Boolean).map(f32hex);
  }
  return flatten(x).map((v) => typeof v === 'number' ? v : f32hex(v));
}
function matrix(x, rows, cols) {
  const a = Array.isArray(x) && x.length && Array.isArray(x[0])
    ? flatten(x).map((v) => {
        if (typeof v === 'number') return v;
        const t = String(v).trim();
        return /^(0x)?[0-9a-fA-F]{8}$/.test(t) ? f32hex(t) : Number(t);
      })
    : f32list(x);
  if (a.length !== rows * cols) throw new Error(`matrix size ${a.length} != ${rows * cols}`);
  return Float64Array.from(a, Number);
}
function fro(M) { let s = 0; for (const x of M) s += x * x; return Math.sqrt(s); }
function rel(a, b) {
  let d = 0, n = 0;
  for (let i = 0; i < a.length; i++) { const t = a[i] - b[i]; d += t * t; n += b[i] * b[i]; }
  return n === 0 ? (d === 0 ? 0 : Infinity) : Math.sqrt(d / n);
}
function matVec(M, m, n, x) {
  const y = new Float64Array(m);
  for (let i = 0; i < m; i++) {
    let s = 0, o = i * n;
    for (let j = 0; j < n; j++) s += M[o + j] * x[j];
    y[i] = s;
  }
  return y;
}

// One-sided Jacobi SVD.  Returns U(m*k), S(k desc), V(n*k), k=min(m,n).
function svd(A, m, n) {
  if (!m || !n) return { U: new Float64Array(0), S: [], V: new Float64Array(0), k: 0 };
  const T = m < n, R = T ? n : m, C = T ? m : n;
  const X = new Float64Array(R * C);
  if (T) { for (let i = 0; i < m; i++) for (let j = 0; j < n; j++) X[j * C + i] = A[i * n + j]; }
  else X.set(A);

  const V = new Float64Array(C * C);
  for (let i = 0; i < C; i++) V[i * C + i] = 1;

  const tol = 1e-15, maxSweeps = Math.max(60, Math.min(400, 8 * C + 30));
  for (let sweep = 0; sweep < maxSweeps; sweep++) {
    let rotations = 0;
    for (let p = 0; p < C - 1; p++) for (let q = p + 1; q < C; q++) {
      let alpha = 0, beta = 0, gamma = 0;
      for (let i = 0; i < R; i++) {
        const ip = i * C + p, iq = i * C + q, xp = X[ip], xq = X[iq];
        alpha += xp * xp; beta += xq * xq; gamma += xp * xq;
      }
      if (alpha === 0 || beta === 0 || gamma === 0) continue;
      const scale = Math.sqrt(alpha * beta);
      if (!Number.isFinite(scale) || Math.abs(gamma) <= tol * scale) continue;
      const zeta = (beta - alpha) / (2 * gamma);
      const t = zeta >= 0
        ? 1 / (zeta + Math.sqrt(1 + zeta * zeta))
        : -1 / (-zeta + Math.sqrt(1 + zeta * zeta));
      const c = 1 / Math.sqrt(1 + t * t), s = c * t;
      for (let i = 0; i < R; i++) {
        const ip = i * C + p, iq = i * C + q, xp = X[ip], xq = X[iq];
        X[ip] = c * xp - s * xq; X[iq] = s * xp + c * xq;
      }
      for (let i = 0; i < C; i++) {
        const ip = i * C + p, iq = i * C + q, vp = V[ip], vq = V[iq];
        V[ip] = c * vp - s * vq; V[iq] = s * vp + c * vq;
      }
      rotations++;
    }
    if (!rotations) break;
  }

  const sigma = new Float64Array(C);
  for (let j = 0; j < C; j++) {
    let s = 0;
    for (let i = 0; i < R; i++) { const x = X[i * C + j]; s += x * x; }
    sigma[j] = Math.sqrt(s);
  }
  const order = Array.from({ length: C }, (_, i) => i).sort((a, b) => sigma[b] - sigma[a]);
  const k = C, Uw = new Float64Array(R * k), Vw = new Float64Array(C * k), S = new Array(k);
  for (let jj = 0; jj < k; jj++) {
    const j = order[jj], sj = sigma[j]; S[jj] = sj;
    for (let i = 0; i < R; i++) Uw[i * k + jj] = sj > 0 ? X[i * C + j] / sj : 0;
    for (let i = 0; i < C; i++) Vw[i * k + jj] = V[i * C + j];
  }
  return T
    ? { U: Vw, S, V: Uw, k } // A = Vw S Uw^T
    : { U: Uw, S, V: Vw, k };
}

function reject(layer, code, message, extra = {}) {
  return { layer, ok: false, error: { code, message, ...extra } };
}

function solveLayer(L, idx) {
  const m = Number(L.d_out), n = Number(L.d_in), rMaxRaw = Number(L.r_max);
  if (!Number.isInteger(m) || m <= 0 || !Number.isInteger(n) || n <= 0)
    return reject(idx, 'INVALID_DIMENSIONS', 'd_out and d_in must be positive integers');
  if (!Number.isFinite(rMaxRaw) || rMaxRaw < 0)
    return reject(idx, 'INVALID_RMAX', 'r_max must be non-negative and finite');
  const rMax = Math.floor(rMaxRaw);

  let W;
  try { W = matrix(L.W, m, n); }
  catch (e) { return reject(idx, 'INVALID_W', String(e.message || e)); }

  const ads = [];
  let totalRank = 0;
  for (let ai = 0; ai < (L.adapters || []).length; ai++) {
    const ad = L.adapters[ai], r = Number(ad.r), scale = Number(ad.scale);
    if (!Number.isInteger(r) || r < 0)
      return reject(idx, 'INVALID_ADAPTER_RANK', `adapter ${ai}: rank must be a non-negative integer`);
    const eff = scale / r;
    if (!Number.isFinite(eff))
      return reject(idx, 'NON_FINITE_EFFECTIVE_SCALE', `adapter ${ai}: scale/r is non-finite`,
        { adapter: ai, scale: String(scale), rank: r, effective_scale: String(eff) });
    totalRank += r;
    if (r === 0) { ads.push({ A: new Float64Array(0), B: new Float64Array(0), r, eff }); continue; }
    let A, B;
    try { A = matrix(ad.A, r, n); }
    catch (e) { return reject(idx, 'INVALID_A_SHAPE', `adapter ${ai}: ${String(e.message || e)}`); }
    try { B = matrix(ad.B, m, r); }
    catch (e) { return reject(idx, 'INVALID_B_SHAPE', `adapter ${ai}: ${String(e.message || e)}`); }
    ads.push({ A, B, r, eff });
  }

  // dW = sum_i (scale_i/r_i) B_i A_i, accumulated in float64.
  const dW = new Float64Array(m * n);
  for (const { A, B, r, eff } of ads) {
    if (!r) continue;
    for (let i = 0; i < m; i++) for (let j = 0; j < n; j++) {
      let s = 0;
      for (let q = 0; q < r; q++) s += B[i * r + q] * A[q * n + j];
      dW[i * n + j] += eff * s;
    }
  }
  for (let i = 0; i < m; i++) for (let j = 0; j < n; j++) {
    const v = dW[i * n + j];
    if (!Number.isFinite(v) || !Number.isFinite(Math.fround(v)) || Math.abs(v) > F32_MAX)
      return reject(idx, 'DW_FLOAT32_OVERFLOW', `dW[${i},${j}] overflows finite float32`,
        { row: i, col: j, value: Number.isFinite(v) ? v : String(v) });
  }

  const { U, S, V, k } = svd(dW, m, n);
  const condition = ads.length === 1 || rMax >= totalRank;
  // K=1 with rMax < r_1 cannot both truncate and equal that one adapter.
  // The stated 1e-12 sequential-equivalence requirement wins; retain rank r_1.
  const conflict = ads.length === 1 && rMax < ads[0].r;
  const r = conflict ? Math.min(k, ads[0].r) : Math.min(rMax, k);

  const target = new Float64Array(m * n);
  for (let i = 0; i < m * n; i++) target[i] = W[i] + dW[i];

  const dWt = new Float64Array(m * n);
  for (let q = 0; q < r; q++) {
    const sq = S[q];
    if (!sq) continue;
    for (let i = 0; i < m; i++) {
      const u = U[i * k + q] * sq;
      for (let j = 0; j < n; j++) dWt[i * n + j] += u * V[j * k + q];
    }
  }
  const Wp = new Float64Array(m * n);
  for (let i = 0; i < m * n; i++) Wp[i] = W[i] + dWt[i];

  // Sequential reference and its relative difference (when required).
  let seqRel = null;
  if (condition) {
    const seq = Float64Array.from(W);
    for (const { A, B, r: rr, eff } of ads) {
      if (!rr) continue;
      for (let i = 0; i < m; i++) for (let j = 0; j < n; j++) {
        let s = 0;
        for (let q = 0; q < rr; q++) s += B[i * rr + q] * A[q * n + j];
        seq[i * n + j] += eff * s;
      }
    }
    seqRel = rel(Wp, seq);
  }

  const retained = S.slice(0, r), discarded = S.slice(r);
  let discSq = 0;
  for (const s of discarded) discSq += s * s;
  const discMass = Math.sqrt(discSq);
  const relErr = rel(Wp, target);
  const internal = [];
  for (let i = 0; i + 1 < r; i++) internal.push(S[i] - S[i + 1]);
  const internalMin = internal.length ? Math.min(...internal) : null;
  const cutoff = r < k ? (r ? S[r - 1] - S[r] : S[0]) : null;
  const gap = cutoff != null ? cutoff : (internalMin != null ? internalMin : (r ? retained[r - 1] : 0));

  const useFactors = rMax * (m + n) < m * n && r * (m + n) < m * n;
  const queries = [];
  const qs = L.queries || [];
  const qlist = Array.isArray(qs) && (qs.length === 0 || Array.isArray(qs[0]) || ArrayBuffer.isView(qs[0])) ? qs : [qs];
  for (let qi = 0; qi < qlist.length; qi++) {
    if (qlist[qi] == null) continue;
    const x = Float64Array.from(flatten(qlist[qi]), Number);
    if (x.length !== n) { queries.push({ index: qi, error: { code: 'INVALID_QUERY_LENGTH', expected: n, got: x.length } }); continue; }
    let y;
    if (useFactors) {
      y = matVec(W, m, n, x);
      for (let q = 0; q < r; q++) {
        let vx = 0;
        for (let t = 0; t < n; t++) vx += V[t * k + q] * x[t];
        const c = S[q] * vx;
        for (let i = 0; i < m; i++) y[i] += U[i * k + q] * c;
      }
    } else y = matVec(Wp, m, n, x);
    queries.push({ index: qi, path: useFactors ? 'factors' : 'materialized', y: Array.from(y) });
  }

  return {
    layer: L.layer ?? idx, ok: true, d_out: m, d_in: n,
    r_max: rMaxRaw, requested_rank: rMax, retained_rank: r,
    retained_singular_values: retained, all_singular_values: S.slice(),
    discarded_frobenius_mass: discMass,
    discarded_frobenius_mass_squared: discSq,
    relative_error: relErr,
    smallest_retained_gap: gap, cutoff_gap: cutoff,
    smallest_internal_retained_gap: internalMin,
    sequential_equivalent: condition, sequential_relative_difference: seqRel,
    truncation_applied: !conflict, use_factor_queries: useFactors,
    W_prime: Array.from(Wp), W_prime_row_major: Array.from(Wp),
    factors: { U: Array.from(U), S: S.slice(), V: Array.from(V), rank_used: r, shape: { U: [m, k], V: [n, k] } },
    queries
  };
}

function solve(input) {
  const layers = Array.isArray(input) ? input : (input.layers || []);
  return { layers: layers.map((L, i) => solveLayer(L, i)) };
}

W may be a long hex string, an array of hex strings, or numeric values. A and B may be row-major arrays or nested arrays.

Relevant output fields

  • retained_singular_values: descending retained sigma_i
  • discarded_frobenius_mass: sqrt(sum_{i>r} sigma_i^2)
  • relative_error: ||W' - (W+dW)||_F / ||W+dW||_F
  • smallest_retained_gap: cutoff gap sigma_r - sigma_{r+1} when a cut exists
  • sequential_equivalent: K == 1 || r_max >= sum(r_i)
  • W_prime: row-major emitted matrix
  • factors: full U, S, V from the from-scratch SVD
  • queries[].path: "factors" or "materialized"
  • error.code: e.g. NON_FINITE_EFFECTIVE_SCALE, DW_FLOAT32_OVERFLOW
