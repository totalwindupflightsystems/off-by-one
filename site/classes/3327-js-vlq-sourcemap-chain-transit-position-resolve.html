Fixing ECMA-426 source-map chaining, VLQ decode, and nearest-predecessor resolution

Problem class: js-vlq-sourcemap-chain-transit-position-resolve · Env: Node 20 Deliverable: a dependency-free transcoder that decodes base64 VLQ, chains minified → transpiled → original, and resolves originalPositionFor per ECMA-426 on adversarial inputs.

Full worked copy of this solution (with the fuzz harness and the official-suite runner) is saved at /workspace/SOLUTION.md.

1. Root-cause analysis

Every listed edge maps to one line of the spec. These are the failure modes that make the difference "bit-for-bit":

# Failure Root cause Spec anchor
1 Wrong columns after line 1 generatedColumn must reset to 0 at each ;; source/line/column/name must not ECMA-426 LineList ("Set state.[[GeneratedColumn]] to 0")
2 Positions drift Segment arity is 1, 4 or 5; only 4/5-field updates source/line/col, only 5-field updates name Mapping productions
3 Lookup picks wrong segment Segments may be out of order; the sort key is only the generated position and must be stable 9.1.2 ParseDecodedSourceMap + ECMAScript stable sort
4 Wrong duplicate-column result Find the nearest predecessor generated position, then take the first stable-ordered segment there — i.e. GetOriginalPositions(...)[0] 12.1 GetOriginalPositions
5 Chain off-by-one / stale carry-over Public API is 1-based line / 0-based column; intermediate result feeds the next map unchanged; a miss on the requested line returns null, never a previous line 12.1 + consumer line check
6 1-field segment gets an original 1-field mapping has [[OriginalPosition]] = null Table 5 /
7 VLQ mis-decoded Sign is the LSB, 5-bit little-endian groups, bit 0x20 = continuation, deltas can be negative ECMA-426 VLQ

The decisive semantic: ECMA-426 9.1.2 sorts by generated position only (ComparePositions compares line then column), and ECMAScript sort is stable, so duplicate generated columns keep segment order. GetOriginalPositions then returns all mappings at that position in forward order; the single-value API returns element 0.

Mozilla source-map 0.6/0.7 re-sorts ties by (source, originalLine, originalColumn, name), and @jridgewell/trace-mapping returns the first tie on an exact hit but the last tie on a miss. Those are implementation artifacts; the stable rule below is what the spec text and the official TC39 suite prescribe.

2. Exact fix — sourcemap-transcoder.js

"use strict";
/*
 * Dependency-free ECMA-426 source-map decoder + position resolver,
 * transitive chaining and composition.
 */
const BASE64_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
const BASE64_LOOKUP = (() => {
  const table = new Int32Array(128).fill(-1);
  for (let i = 0; i < BASE64_CHARS.length; i++) table[BASE64_CHARS.charCodeAt(i)] = i;
  return table;
})();
const VLQ_SHIFT = 5, VLQ_BASE_MASK = (1 << VLQ_SHIFT) - 1, VLQ_CONTINUATION = 1 << VLQ_SHIFT;

function decodeVLQ(str, state) {
  let result = 0, shift = 0, digit;
  do {
    if (state.i >= str.length) throw new Error("Invalid VLQ: unexpected end of segment");
    const code = str.charCodeAt(state.i++);
    digit = code < 128 ? BASE64_LOOKUP[code] : -1;
    if (digit === -1) throw new Error("Invalid base64 VLQ character: " + JSON.stringify(str[state.i - 1]));
    result += (digit & VLQ_BASE_MASK) * Math.pow(2, shift); // avoid 32-bit wrap
    shift += VLQ_SHIFT;
  } while (digit & VLQ_CONTINUATION);
  const negative = result % 2 === 1;
  result = Math.floor(result / 2);
  return negative ? -result : result;
}

function encodeVLQ(value) {
  let vlq = value < 0 ? -value * 2 + 1 : value * 2;
  let out = "";
  do {
    let digit = vlq % 32;
    vlq = Math.floor(vlq / 32);
    if (vlq > 0) digit |= VLQ_CONTINUATION;
    out += BASE64_CHARS[digit];
  } while (vlq > 0);
  return out;
}

/*
 * Decoded lines: one of [gcol] / [gcol, src, oline, ocol] /
 * [gcol, src, oline, ocol, name]. Values are absolute.
 * Running fields: gcol resets per ';', the other four persist.
 */
function decodeMappings(rawMappings) {
  if (rawMappings == null || rawMappings === "") return [[]];
  const lines = [];
  let sourceIndex = 0, originalLine = 0, originalColumn = 0, nameIndex = 0;
  const lineChunks = String(rawMappings).split(";");
  for (let li = 0; li < lineChunks.length; li++) {
    let generatedColumn = 0; // reset per generated line
    const segments = [];
    for (const chunk of lineChunks[li].split(",")) {
      if (chunk === "") continue;
      const state = { i: 0 }, fields = [];
      while (state.i < chunk.length) fields.push(decodeVLQ(chunk, state));
      if (fields.length !== 1 && fields.length !== 4 && fields.length !== 5) {
        throw new Error("Invalid mapping segment: expected 1, 4 or 5 fields but got " +
          fields.length + " in " + JSON.stringify(chunk));
      }
      generatedColumn += fields[0];
      if (fields.length === 1) {
        segments.push([generatedColumn]);
      } else {
        sourceIndex += fields[1];
        originalLine += fields[2];
        originalColumn += fields[3];
        if (fields.length === 5) {
          nameIndex += fields[4];
          segments.push([generatedColumn, sourceIndex, originalLine, originalColumn, nameIndex]);
        } else {
          segments.push([generatedColumn, sourceIndex, originalLine, originalColumn]);
        }
      }
    }
    lines.push(segments);
  }
  return lines;
}

function encodeMappings(lines) {
  let sourceIndex = 0, originalLine = 0, originalColumn = 0, nameIndex = 0;
  const outLines = [];
  for (const line of lines) {
    let generatedColumn = 0;
    const parts = [];
    for (const seg of line) {
      let encoded = encodeVLQ(seg[0] - generatedColumn);
      generatedColumn = seg[0];
      if (seg.length > 1) {
        encoded += encodeVLQ(seg[1] - sourceIndex); sourceIndex = seg[1];
        encoded += encodeVLQ(seg[2] - originalLine); originalLine = seg[2];
        encoded += encodeVLQ(seg[3] - originalColumn); originalColumn = seg[3];
        if (seg.length === 5) { encoded += encodeVLQ(seg[4] - nameIndex); nameIndex = seg[4]; }
      }
      parts.push(encoded);
    }
    outLines.push(parts.join(","));
  }
  return outLines.join(";");
}

const GREATEST_LOWER_BOUND = 1, LEAST_UPPER_BOUND = 2;
const NULL_POSITION = Object.freeze({ source: null, line: null, column: null, name: null });

function joinSourceRoot(sourceRoot, source) {
  if (!sourceRoot) return source;
  let root = sourceRoot;
  if (root[root.length - 1] !== "/" && source && source[0] !== "/") root += "/";
  return root + source;
}

class SourceMap {
  constructor(raw) {
    if (typeof raw === "string") raw = JSON.parse(raw);
    if (!raw || typeof raw !== "object") throw new Error("SourceMap requires an object or JSON string");
    this.raw = raw;
    this.sourceRoot = raw.sourceRoot || "";
    this.sources = Array.isArray(raw.sources) ? raw.sources : [];
    this.names = Array.isArray(raw.names) ? raw.names : [];
    this._decoded = decodeMappings(raw.mappings || "");

    // Per line, stably sort by generatedColumn only (ECMA-426 9.1.2).
    this._lines = new Array(this._decoded.length);
    this._all = [];
    for (let line = 0; line < this._decoded.length; line++) {
      const records = this._decoded[line].map((seg) => ({
        generatedLine: line,
        generatedColumn: seg[0],
        sourceIndex: seg.length === 1 ? -1 : seg[1],
        originalLine: seg.length === 1 ? -1 : seg[2],
        originalColumn: seg.length === 1 ? -1 : seg[3],
        nameIndex: seg.length === 5 ? seg[4] : -1,
      }));
      records.sort((a, b) => a.generatedColumn - b.generatedColumn); // stable
      this._lines[line] = records;
      for (const r of records) this._all.push(r);
    }
  }

  _resolveMapping(m) {
    if (m.sourceIndex < 0) return NULL_POSITION;
    const source = this.sources[m.sourceIndex];
    return {
      source: source == null ? null : joinSourceRoot(this.sourceRoot, source),
      line: m.originalLine + 1, // 0-based -> 1-based
      column: m.originalColumn,
      name: m.nameIndex >= 0 ? this.names[m.nameIndex] : null,
    };
  }

  originalPositionFor(args) {
    const { line, column } = args;
    if (line == null || line < 1) throw new Error("Line numbers must be >= 1, got " + line);
    if (column == null || column < 0) throw new Error("Column numbers must be >= 0, got " + column);
    const bias = args.bias == null ? GREATEST_LOWER_BOUND : args.bias;
    const mappings = this._lines[line - 1];
    if (!mappings || mappings.length === 0) return NULL_POSITION;
    let index = bias === LEAST_UPPER_BOUND
      ? this._findLeastUpperBound(mappings, column)
      : this._findGreatestLowerBound(mappings, column);
    if (index === -1) return NULL_POSITION;
    const matchedColumn = mappings[index].generatedColumn;
    while (index > 0 && mappings[index - 1].generatedColumn === matchedColumn) index--; // first tie
    return this._resolveMapping(mappings[index]);
  }

  _findGreatestLowerBound(mappings, column) {
    let lo = 0, hi = mappings.length - 1, found = -1;
    while (lo <= hi) {
      const mid = (lo + hi) >> 1;
      if (mappings[mid].generatedColumn <= column) { found = mid; lo = mid + 1; }
      else hi = mid - 1;
    }
    return found;
  }

  _findLeastUpperBound(mappings, column) {
    let lo = 0, hi = mappings.length - 1, found = -1;
    while (lo <= hi) {
      const mid = (lo + hi) >> 1;
      if (mappings[mid].generatedColumn >= column) { found = mid; hi = mid - 1; }
      else lo = mid + 1;
    }
    return found;
  }

  // Faithful ECMA-426 12.1 GetOriginalPositions (all ties, stable order).
  originalPositionsFor(args) {
    const { line, column } = args;
    if (line == null || line < 1) throw new Error("Line numbers must be >= 1");
    if (column == null || column < 0) throw new Error("Column numbers must be >= 0");
    const targetLine = line - 1, all = this._all;
    if (all.length === 0) return [];
    let lo = 0, hi = all.length - 1, index = -1;
    while (lo <= hi) {
      const mid = (lo + hi) >> 1, m = all[mid];
      if (m.generatedLine < targetLine || (m.generatedLine === targetLine && m.generatedColumn <= column)) {
        index = mid; lo = mid + 1;
      } else hi = mid - 1;
    }
    if (index === -1) return [];
    const target = all[index];
    let start = index;
    while (start > 0 && all[start - 1].generatedLine === target.generatedLine &&
           all[start - 1].generatedColumn === target.generatedColumn) start--;
    const result = [];
    for (let i = start; i < all.length; i++) {
      const m = all[i];
      if (m.generatedLine !== target.generatedLine || m.generatedColumn !== target.generatedColumn) break;
      result.push(m.sourceIndex < 0 ? null : this._resolveMapping(m)); // 1-field => null
    }
    return result;
  }
}

function originalPositionForChain(maps, position) {
  let current = position;
  for (const map of maps) {
    current = map.originalPositionFor(current);
    if (current.source == null && current.line == null) return NULL_POSITION;
  }
  return current;
}

function transcode(outer, inner) {
  outer = outer instanceof SourceMap ? outer : new SourceMap(outer);
  inner = inner instanceof SourceMap ? inner : new SourceMap(inner);
  const sources = [], sourceIndexByName = new Map();
  const names = [], nameIndexByName = new Map();
  const internSource = (n) => {
    if (sourceIndexByName.has(n)) return sourceIndexByName.get(n);
    const i = sources.length; sources.push(n); sourceIndexByName.set(n, i); return i;
  };
  const internName = (n) => {
    if (n == null) return -1;
    if (nameIndexByName.has(n)) return nameIndexByName.get(n);
    const i = names.length; names.push(n); nameIndexByName.set(n, i); return i;
  };
  const outLines = [];
  for (const records of outer._lines) {
    const outSegments = [];
    for (const m of records) {
      if (m.sourceIndex < 0) { outSegments.push([m.generatedColumn]); continue; }
      const p = inner.originalPositionFor({ line: m.originalLine + 1, column: m.originalColumn });
      if (p.source == null && p.line == null) { outSegments.push([m.generatedColumn]); continue; }
      const si = internSource(p.source), ni = internName(p.name);
      outSegments.push(ni < 0
        ? [m.generatedColumn, si, p.line - 1, p.column]
        : [m.generatedColumn, si, p.line - 1, p.column, ni]);
    }
    outLines.push(outSegments);
  }
  return new SourceMap({ version: 3, sources, names, mappings: encodeMappings(outLines) });
}

module.exports = {
  decodeVLQ, encodeVLQ, decodeMappings, encodeMappings, SourceMap,
  originalPositionForChain, transcode, GREATEST_LOWER_BOUND, LEAST_UPPER_BOUND,
};

Key correctness guarantees: generatedColumn resets per line while other running fields persist; only present fields advance; each line is stably sorted by generated column; the nearest predecessor is the greatest (line,column) ≤ query on the same generated line; duplicate columns resolve to the first stable-ordered segment (matching GetOriginalPositions()[0]); 1-field segments stay unmapped; chains feed the 1-based line / 0-based column straight into the next map and short-circuit on null.
