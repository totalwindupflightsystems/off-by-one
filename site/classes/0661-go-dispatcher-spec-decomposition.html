Answer

The bug (GAP-007): DecomposeSpec only recognized ## Phase / ## Feature H2 headings as task sections, but the repo's own specs use # Helix Feature N H1 titles whose H2 sections (## Overview, ## Requirements, ## Acceptance Criteria) contain neither keyword — so helix dispatcher list-tasks --spec specs/*.md found zero sections and errored on every spec.

The fix (in decompose.go) adds one case-insensitive H1 prefix matcher and one switch branch; existing H2 detection and the ErrNoSections path are untouched:

var (
    // h2Section matches existing "## Phase N" / "## Feature X" H2 headings.
    h2Section = regexp.MustCompile(`(?i)^##\s+.*(?:phase|feature).*$`)
    // h1Feature matches the repo's "# Helix Feature N" H1 title convention
    // (case-insensitive, e.g. "# HELIX FEATURE 1 — Persistent Job Queue").
    h1Feature = regexp.MustCompile(`(?i)^#\s+helix\s+feature\b`)
)

func DecomposeSpec(markdown string) ([]Task, error) {
    var tasks []Task
    var cur *Task

    start := func(heading string) {
        if cur != nil {
            cur.Body = strings.TrimSpace(cur.Body)
            tasks = append(tasks, *cur)
        }
        cur = &Task{Title: heading, Priority: len(tasks) + 1}
    }

    for _, line := range strings.Split(markdown, "\n") {
        trimmed := strings.TrimSpace(line)
        switch {
        case h1Feature.MatchString(trimmed):
            start(strings.TrimSpace(trimmed[1:])) // drop leading '#'
        case h2Section.MatchString(trimmed):
            start(strings.TrimSpace(trimmed[2:])) // drop leading '##'
        default:
            if cur != nil {
                cur.Body += line + "\n"
            }
        }
    }
    if cur != nil {
        cur.Body = strings.TrimSpace(cur.Body)
        tasks = append(tasks, *cur)
    }

    if len(tasks) == 0 {
        return nil, ErrNoSections
    }
    return tasks, nil
}

Semantics: each # Helix Feature… H1 (matched case-insensitively, word-boundary so # Helix Features doesn't match) starts one task; priority is the 1-based document order; H1 and H2 markers can be mixed and interleave in order; content before the first marker is skipped; original heading case is preserved in Task.Title.
