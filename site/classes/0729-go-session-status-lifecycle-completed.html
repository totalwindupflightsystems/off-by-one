go-session-status-lifecycle-completed

Answer

Root cause (GAP-DOG-003): sessionEntry.Status was only ever assigned active/cancelled in processHandler/resultHandler, so an end decision left the persisted status as active and GET /v1/sessions/{id} reported "status": "active". The protocol.SessionCompleted constant existed but was never written.

Fix: in both processHandler and resultHandler, set e.Status = protocol.SessionCompleted when d.Decision == protocol.DecisionEnd, inside the same update block that records CurrentDecision (GAP-009), so status and decision commit atomically and never diverge. sessionEntry.Status stays a plain string (consistent with existing code); getSessionHandler already normalizes it via protocol.SessionStatus().

// protocol/protocol.go — constants already present; SessionCompleted was unused
const (
    SessionActive    = "active"
    SessionCancelled = "cancelled"
    SessionCompleted = "completed" // now actually persisted on end decisions
)

const (
    DecisionEnd   = "end"
    DecisionCancel = "cancel"
)

// SessionStatus normalizes the stored plain string for the wire.
func SessionStatus(s string) string {
    switch s {
    case SessionCancelled:
        return SessionCancelled
    case SessionCompleted:
        return SessionCompleted
    default:
        return SessionActive // backward-compatible fallback
    }
}

// processHandler: one atomic update block (GAP-009 + GAP-DOG-003)
func processHandler(s *store, d Decision) error {
    e := s.get(d.SessionID)
    if e == nil {
        return ErrNotFound
    }
    e.CurrentDecision = d.Decision                    // GAP-009
    if d.Decision == protocol.DecisionEnd {          // GAP-DOG-003
        e.Status = protocol.SessionCompleted
    }
    return s.update(e)                               // single update/transaction
}

// resultHandler: identical lifecycle rules for end decisions on step results
func resultHandler(s *store, d Decision) error {
    e := s.get(d.SessionID)
    if e == nil {
        return ErrNotFound
    }
    e.CurrentDecision = d.Decision                    // GAP-009
    if d.Decision == protocol.DecisionEnd {          // GAP-DOG-003
        e.Status = protocol.SessionCompleted
    }
    return s.update(e)
}

getSessionHandler needs no change:

func getSessionHandler(s *store, id string) (*sessionResponse, error) {
    e := s.get(id)
    if e == nil {
        return nil, ErrNotFound
    }
    return &sessionResponse{
        ID:              e.ID,
        Status:          protocol.SessionStatus(e.Status), // maps "completed" -> "completed"
        CurrentDecision: e.CurrentDecision,
    }, nil
}
