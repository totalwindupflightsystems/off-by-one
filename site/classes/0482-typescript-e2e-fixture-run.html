// tests/e2e/perf-gate.spec.ts — enforces p95 < 25ms
import { test, expect } from '@playwright/test';

const GATE_MS = 25;

test('perf p95 stays under the 25ms gate', async ({ request }) => {
  const latencies: number[] = [];
  for (let i = 0; i < 120; i++) {
    const t0 = performance.now();
    const res = await request.get('/api/ping');
    latencies.push(performance.now() - t0);
    expect(res.ok()).toBeTruthy();
  }
  const sorted = [...latencies].sort((a, b) => a - b);
  const p95 = sorted[Math.floor(sorted.length * 0.95)] ?? 0;
  expect(p95, `p95 ${p95.toFixed(2)}ms exceeded gate ${GATE_MS}ms`).toBeLessThan(GATE_MS);
});

// tests/e2e/visual.spec.ts — refreshes the 8 evidence screenshots each tick
import { test, expect } from '@playwright/test';

const EVIDENCE = ['home', 'login', 'dashboard', 'checkout', 'empty', 'error', 'mobile', 'a11y'];

test('8 evidence screenshots refreshed', async ({ page }) => {
  await page.goto('/');
  expect(EVIDENCE).toHaveLength(8); // count gate
  for (const name of EVIDENCE) {
    await page.screenshot({ path: `artifacts/evidence/${name}.png`, fullPage: true });
  }
});
