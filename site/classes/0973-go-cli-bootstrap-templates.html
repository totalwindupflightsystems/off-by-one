C-GAP-013 — .memory-bank/ was gitignored yet referenced by AGENTS.md, so fresh checkouts never contained it. Fix: bootstrap the skeleton in consensus init with idempotent O_CREATE|O_EXCL writes; templates live as consts in internal/bootstrap/memory_bank.go; spec 016 gained the one-line requirement.

internal/bootstrap/memory_bank.go (templates as consts — identical skeleton on every checkout):

const memoryBankRoot = ".memory-bank"

const (
    memoryBankIndexTemplate = `# Memory Bank - Index
...` // + _prompt.md and findings/_index.md consts
)

// skeleton: files relative to the bank root (order-independent of dirs).
var skeleton = []entry{
    {path: "_index.md", tmpl: memoryBankIndexTemplate},
    {path: "_prompt.md", tmpl: memoryBankPromptTemplate},
    {path: filepath.Join("findings", "_index.md"), tmpl: memoryBankFindingsIndexTemplate},
}
var skeletonDirs = []string{"", "findings", "work-items"}

func BootstrapMemoryBank(root string) (created []string, err error) {
    bank := filepath.Join(root, memoryBankRoot)
    for _, d := range skeletonDirs {
        p := filepath.Join(bank, d)
        if _, err := os.Stat(p); err == nil {
            continue // already bootstrapped
        }
        if err := os.MkdirAll(p, 0o755); err != nil {
            return created, fmt.Errorf("bootstrap: mkdir %s: %w", p, err)
        }
        created = append(created, relPath(d))
    }
    for _, e := range skeleton {
        p := filepath.Join(bank, e.path)
        f, err := os.OpenFile(p, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o644)
        if err != nil {
            if errors.Is(err, os.ErrExist) {
                continue // idempotent: never clobber user edits
            }
            return created, fmt.Errorf("bootstrap: create %s: %w", p, err)
        }
        if _, werr := f.WriteString(e.tmpl); werr != nil {
            _ = f.Close()
            return created, fmt.Errorf("bootstrap: write %s: %w", p, werr)
        }
        _ = f.Close()
        created = append(created, filepath.Join(memoryBankRoot, e.path))
    }
    return created, nil
}
