Answer

The bug: helix-estimate's default --pricing/--known-friends values were CWD-relative (pkg/estimate/testdata/*.yaml joined against the current working directory). Invoked from outside the checkout root, those paths don't exist → CONFIG_ERROR.

The fix, in /home/kara/helix/pkg/estimate/estimate.go: testdataPath() anchors to the repo root by (1) walking up from CWD to the first go.mod, (2) falling back to the runtime.Caller source location when CWD is outside the repo, (3) returning "" otherwise so the caller emits the normal CONFIG_ERROR. It never returns a relative path — only an absolute path to an existing file under a discovered repo root, or "".

// testdataPath resolves an absolute path to rel under the repo root.
// 1) go.mod walk-up from CWD; 2) runtime.Caller source fallback;
// 3) else "" so the caller emits CONFIG_ERROR. Never a broken relative path.
func testdataPath(rel string) string {
    root := repoRoot()
    if root == "" {
        return ""
    }
    p := filepath.Join(root, rel)
    fi, err := os.Stat(p)
    if err != nil || fi.IsDir() {
        return ""
    }
    return p
}

func repoRoot() string {
    if cwd, err := os.Getwd(); err == nil {
        if root := findRepoRootFromDir(cwd); root != "" {
            return root
        }
    }
    return findRepoRootFromSource()
}

// findRepoRootFromDir walks up from dir to the first directory with go.mod.
func findRepoRootFromDir(dir string) string {
    d := dir
    if !filepath.IsAbs(d) {
        abs, err := filepath.Abs(d)
        if err != nil {
            return ""
        }
        d = abs
    }
    d = filepath.Clean(d)
    for {
        if _, err := os.Stat(filepath.Join(d, "go.mod")); err == nil {
            return d
        }
        parent := filepath.Dir(d)
        if parent == d {
            return ""
        }
        d = parent
    }
}

// findRepoRootFromSource locates the repo root from the compiled-in source
// location of this file, independent of the process working directory.
func findRepoRootFromSource() string {
    for i := 0; i < 8; i++ { // frame 0 is estimate.go; loop guards against inlining
        _, file, _, ok := runtime.Caller(i)
        if !ok {
            break
        }
        if root := findRepoRootFromDir(filepath.Dir(file)); root != "" {
            return root
        }
    }
    return ""
}

LoadConfig/resolvePath then feed empty flag values through testdataPath; a "" result becomes a ConfigError whose message contains CONFIG_ERROR, so the CLI reports it identically to before:

func resolvePath(explicit, rel string) string {
    if explicit != "" {
        abs, err := filepath.Abs(explicit)
        if err != nil {
            return ""
        }
        return abs
    }
    return testdataPath(rel)
}
