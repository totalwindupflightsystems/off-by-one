type SourceKind string
const (
    SourceBoard SourceKind = "board"    // Hilo dispatch board state
    SourceEnv   SourceKind = "env"      // environment variables
    SourceSpec  SourceKind = "spec"     // implementation spec documents
    SourceMessage  SourceKind = "message"  // session / message log
    SourceToolSpec SourceKind = "toolspec" // tool schema definitions
)

type Source interface {
    ID() string
    Kind() SourceKind
    Raw(ctx context.Context) (io.Reader, error) // must be concurrency-safe
}

type Options struct {
    IncludeEnv    bool
    MaxDepth      int           // ref resolution cap; default 8
    Timeout       time.Duration // per-compile budget; default 5s
    Required      []SourceKind  // fail fast if absent
    SecretPattern string        // regex; matched env values redacted
}

type Section struct {
    ID, SourceID, Body, Digest string
    Kind       SourceKind
    Refs       []string
    Depth      int
    Provenance map[string]string
}

type Compiled struct {
    Sections    []Section // stable order: source order, then ID
    Digest      string    // sha256 over ordered section digests
    CompiledAt  time.Time
    Elapsed     time.Duration
    SourceCount int
}

type Validator interface{ Validate(c *Compiled) error }
type Hook      interface{ AfterCompile(ctx context.Context, c *Compiled) error }
type Store     interface {
    Save(ctx context.Context, c *Compiled) error
    Load(ctx context.Context, digest string) (*Compiled, error)
}
type Compiler interface {
    Compile(ctx context.Context, sources []Source, opts Options) (*Compiled, error)
}

func New(store Store, validators []Validator, hooks []Hook) Compiler

Ordered algorithm (deterministic): 1. Preflight — defaults (Timeout 5s, MaxDepth 8, secret regex); validate opts. 2. Fail fast — every Required kind must have ≥1 source → else ErrSourceMissing (never a partial compile). 3. Normalize — iterate sources in caller order; Raw(ctx) capped at 1 MiB; per-section sha256(ID+Kind+SourceID+Body). 4. Resolve refs — DFS with visiting set: cycle → ErrRefCycle, dangling → ErrRefDangling, depth > MaxDepth → ErrTooDeep; record Depth. 5. Validate — run Validators in order (unique IDs, required bodies non-empty). 6. Compose — final order = source order, then ID ascending; overall digest over ordered section digests. 7. Persist — store.Save best-effort (logged, not fatal). 8. Hooks — fire in order; errors logged, not fatal. Deadline at any step → ErrTimeout.

Error catalog (10 codes): SRC_MISSING, SRC_READ, PARSE, REF_CYCLE, REF_DANGLING, VAL, OP_TIMEOUT, DEPTH (all 422, OP_TIMEOUT 504), NOT_FOUND (404), INTERNAL (500) — via ErrorCode(err) sentinel mapping.
