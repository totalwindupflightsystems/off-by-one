export function isToken(value: unknown): value is Token {
  return typeof value === 'object' && value !== null
    && 'kind' in value && 'start' in value && 'end' in value;
}
export function tokenAt(tokens: Token[], offset: number): Token | undefined {
  return tokens.find(t => offset >= t.start && offset < t.end);
}

3. extract.ts — pure extract* functions converted to explicit args (previously private methods reading this.source):

import { ReferenceImage, Token } from './types';
import { tokenize } from './token-utils';

export function extractImports(source: string): Token[] {
  return tokenize(source).filter(t => t.kind === 'import');
}
export function extractSelectors(source: string): string[] {
  return tokenize(source).filter(t => t.kind === 'selector').map(t => t.value);
}
export function extractReferences(source: string): ReferenceImage[] {
  return tokenize(source)
    .filter(t => t.kind === 'image-ref')
    .map(t => ({ path: t.value, hash: '', width: 0, height: 0 }));
}

5. gather-reference-images.ts — db-dependent fn takes explicit db param (no module-level singleton, no class field):

import { Db, ReferenceImage } from './types';

export async function gatherReferenceImages(refs: ReferenceImage[], db: Db): Promise<ReferenceImage[]> {
  const out: ReferenceImage[] = [];
  for (const ref of refs) {
    const meta = await db.getImage(ref.path);
    out.push(meta ? { ...ref, ...meta } : ref);
  }
  return out;
}

6. oop-pipeline.ts — the class keeps ONLY stateful traversal + 3 public generate* methods:

import {
  Db, GenerateReportOptions, LocationHierarchyOptions, LocationNode,
  ReferenceImage, StyleHierarchyOptions, StyleNode, Token,
} from './types';
import { tokenize } from './token-utils';
import { extractReferences } from './extract';
import { gatherReferenceImages } from './gather-reference-images';

export class OopPipeline {
  private readonly source: string;
  private readonly tokens: Token[];                      // stateful: tokenized once
  private readonly styleCache = new Map<string, StyleNode>();
  private readonly locCache = new Map<string, LocationNode>();

  constructor(source: string) {
    this.source = source;
    this.tokens = tokenize(source);                      // pure helper, explicit arg
  }

  /** Stateful traversal: memoized walk over this.tokens. */
  buildStyleHierarchy(options: StyleHierarchyOptions = {}): StyleNode {
    const key = `${this.tokens.length}:${options.maxDepth ?? -1}`;
    const hit = this.styleCache.get(key);
    if (hit) return hit;
    const root: StyleNode = { id: 'root', selector: '', rules: [], children: [] };
    // ... depth-limited walk over this.tokens, populating root ...
    this.styleCache.set(key, root);
    return root;
  }

  /** Stateful traversal: memoized walk, keeps baseDir state. */
  buildLocationHierarchy(options: LocationHierarchyOptions = {}): LocationNode {
    const key = `${options.baseDir ?? ''}:${options.followSymlinks ?? false}`;
    const hit = this.locCache.get(key);
    if (hit) return hit;
    const root: LocationNode = { id: 'root', path: '', line: 0, children: [] };
    // ... walk using this.source, populating root ...
    this.locCache.set(key, root);
    return root;
  }

  generateStyleHierarchy(options: StyleHierarchyOptions = {}): StyleNode {
    return this.buildStyleHierarchy(options);
  }
  generateLocationHierarchy(options: LocationHierarchyOptions = {}): LocationNode {
    return this.buildLocationHierarchy(options);
  }
  async generateReport(options: GenerateReportOptions, db: Db): Promise<string> {
    const refs: ReferenceImage[] = extractReferences(this.source);
    const resolved = options.includeImages ? await gatherReferenceImages(refs, db) : refs;
    return options.format === 'json' ? JSON.stringify(resolved) : resolved.map(r => r.path).join('\n');
  }
}

  | grep -oE '^export (declare )?(interface|class|function|const|type) [A-Za-z0-9_]+' \
  | awk '{print $NF}' | sort -u > /tmp/expected.txt

grep -rhoE '^export (interface|class|function|const|type) [A-Za-z0-9_]+' src/oop-pipeline/ \
  | awk '{print $NF}' | sort -u > /tmp/actual.txt
diff -u /tmp/expected.txt /tmp/actual.txt        # must be empty
test "$(wc -l < /tmp/actual.txt)" -eq 11         # exactly 11 symbols

**Checks the verification procedure exercises (and their edge cases):**

1. **Export parity vs `git show HEAD~1`** — the diff between the committed pre-split surface and the new module dir's exported declarations must be empty. Comparing against `HEAD~1` (not the working tree) is essential: a working-tree file edited during the refactor would give false parity and hide a lost symbol.
2. **Exactly 11 public symbols** — 9 interfaces + `OopPipeline` + `createOopPipeline`. Because no parent barrel does `export *`, nothing masks omissions: every symbol must be explicitly re-exported by the barrel.
3. **`tsc --noEmit` passes** — types import cleanly across the 8 files; no `any` leaks from the helper extraction.
4. **Shim smoke test** — importing from the original path `./oop-pipeline` (file) resolves through `./oop-pipeline/index.ts` and exposes all 11 names; deep imports (`./oop-pipeline/types`) still work because the dir holds real files.
5. **Collision guard** — `export *` silently drops names exported from two modules; the check fails on any duplicate exported name (the layout above has none).
6. **Signature/state tests** — `buildStyleHierarchy`/`buildLocationHierarchy` remain stateful (memoized via `this.styleCache`/`this.locCache`, reading `this.tokens`); the 3 `generate*` methods are the only public class API besides the ctor; every `extract*`/token util is a standalone function with explicit args (`extractImports(source)`), and `gatherReferenceImages` takes an explicit `db: Db` param with no module-level singleton.

**Edge cases handled:** file-vs-dir name collision (`.ts` file + same-named dir coexist and resolve deterministically); uncommitted working-tree edits during the split (defeated by `HEAD~1` comparison); silent `export *` ambiguity (defeated by the duplicate-name check); public-surface shrinkage (defeated by the exact 11-symbol count); refactoring the factory/class away from private helpers (defeated by the tsc + smoke checks).
