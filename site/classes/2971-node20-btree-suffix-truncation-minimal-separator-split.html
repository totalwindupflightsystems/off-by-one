Answer

B+tree Prefix Compression + Minimal Separator Split

Root cause

The failure class comes from treating an internal separator as “the first key of the right page” (or as a copied full key) instead of a value in the legal interval

last_key(left) < separator <= first_key(right)

That causes several related failures:

  1. No true suffix truncation
  2. A separator that copies the entire right key is legal but not the shortest separator.

  3. A truncated separator that is not strictly greater than the left page’s last key breaks routing and ordering.

  4. All-equal-prefix split is mishandled

  5. If leftLast is a prefix of rightFirst, no proper prefix of rightFirst is greater than leftLast.
  6. The shortest legal separator is leftLast + 0x00.

  7. Using leftLast directly violates the strict lower bound. Using rightFirst is legal but not minimal.

  8. Stale separators after boundary changes

  9. Inserting/deleting a new minimum or maximum in a child changes the boundary represented by its parent separator.

  10. Recomputing only during splits leaves “legal but not shortest” or even out-of-interval separators after insertion/deletion/borrow/merge.

  11. Rightmost/sibling metadata not repaired

  12. Splitting the rightmost leaf must clear the old leaf’s rightmost flag, set it on the new leaf, and update the tree’s rightmostLeaf.
  13. Splitting the root must create a new root and assign rightmost correctly to the left/right pages.

  14. Splitting an internal rightmost page must propagate the rightmost flag to the new right sibling.

  15. Decode-side validation too weak

  16. Reconstructing pagePrefix + keySuffix must be followed by a strict ordering check.
  17. Truncated separators that reconstruct to duplicate or decreasing keys must be rejected.
  18. Lengths, child counts, and trailing bytes must be checked to avoid accepting corrupt pages.

Exact fix

The complete verified implementation is below.

Key points:

  • minimalSeparator computes the shortest legal separator and handles the prefix/sentinel case.
  • Pages store one prefixLen plus one truncLen per key.
  • decodePage reconstructs keys and rejects non-increasing ordering.
  • _refreshSeparator recomputes a parent separator from the current child boundaries.
  • _refreshAncestors propagates boundary repairs only along leftmost/rightmost paths.
  • Splits repair next/prev, rightmost, and the root/rightmost-leaf pointers.
  • Deletes implement borrow/merge at both leaf and internal levels while maintaining separators.

btree.js

'use strict';

/*
 * btree.js -- B+tree with prefix-compressed pages and shortest legal
 * (suffix-truncated) internal separators.
 *
 * Page codec format:
 *
 *   magic         4 bytes  "BTPC"
 *   kind          1 byte   0 = leaf, 1 = internal
 *   flags         1 byte   bit 0 = rightmost page flag
 *   count         2 bytes  number of keys
 *   pageId        4 bytes
 *   prefixLen     4 bytes  page-level shared prefix length
 *   prefix        prefixLen bytes
 *   entries       count times:
 *                   truncLen 4 bytes
 *                   suffix   truncLen bytes
 *                   if leaf: valueLen 4 bytes, value bytes
 *   childIds      only for internal: (count + 1) * 4 bytes
 *
 * `truncLen` is the per-key truncation length after the page prefix.
 * Leaves retain complete keys; internal keys are shortest legal separators.
 */

const MAGIC = Buffer.from('BTPC');
const LEAF = 0;
const INTERNAL = 1;
const HEADER_SIZE = 4 + 1 + 1 + 2 + 4 + 4; // 16

let NEXT_PAGE_ID = 1;

class CodecError extends Error {
  constructor(message) {
    super(message);
    this.name = 'CodecError';
  }
}

function toBuffer(value) {
  if (Buffer.isBuffer(value)) return Buffer.from(value);
  if (typeof value === 'string') return Buffer.from(value, 'utf8');
  if (value instanceof Uint8Array) return Buffer.from(value);
  throw new TypeError('key/value must be Buffer, Uint8Array, or string');
}

function compareBytes(a, b) {
  return Buffer.compare(a, b);
}

function commonPrefixLength(a, b) {
  const n = Math.min(a.length, b.length);
  let i = 0;
  while (i < n && a[i] === b[i]) i++;
  return i;
}

function pagePrefix(keys) {
  if (keys.length === 0) return Buffer.alloc(0);
  let p = keys[0];
  for (let i = 1; i < keys.length && p.length; i++) {
    p = p.subarray(0, commonPrefixLength(p, keys[i]));
  }
  return Buffer.from(p);
}

/*
 * Shortest key S with left < S <= right.
 *
 * If the keys first differ at byte i, right[0..i] is the shortest key in
 * the allowed interval. If left is a prefix of right, no proper prefix of
 * right is > left, so append the smallest sentinel byte (0x00). This is the
 * all-equal-prefix case.
 */
function minimalSeparator(left, right) {
  if (compareBytes(left, right) >= 0) {
    throw new Error('minimalSeparator requires left < right');
  }
  const n = Math.min(left.length, right.length);
  let i = 0;
  while (i < n && left[i] === right[i]) i++;

  if (i < left.length && i < right.length) {
    // First differing byte. right[i] > left[i].
    return Buffer.from(right.subarray(0, i + 1));
  }

  // left is a proper prefix of right. left + 0x00 is strictly > left and
  // is <= right (if right is exactly left + 0x00 it is equal, which is legal).
  if (i === left.length && i < right.length) {
    const out = Buffer.alloc(left.length + 1);
    left.copy(out, 0);
    out[left.length] = 0x00;
    return out;
  }

  throw new Error('unreachable minimalSeparator state');
}

function writeU32(buf, off, v) {
  buf.writeUInt32BE(v >>> 0, off);
}
function readU32(buf, off) {
  return buf.readUInt32BE(off);
}

function encodePage(page) {
  const keys = page.keys;
  const prefix = pagePrefix(keys);
  const kind = page.leaf ? LEAF : INTERNAL;
  const count = keys.length;
  if (count > 0xffff) throw new CodecError('too many keys on page');

  const items = [];
  let bodyLen = 0;
  for (let i = 0; i < count; i++) {
    const suffix = keys[i].subarray(prefix.length);
    if (page.leaf) {
      const value = page.values[i];
      const size = 4 + suffix.length + 4 + value.length;
      const item = Buffer.allocUnsafe(size);
      writeU32(item, 0, suffix.length);
      suffix.copy(item, 4);
      writeU32(item, 4 + suffix.length, value.length);
      value.copy(item, 8 + suffix.length);
      items.push(item);
      bodyLen += item.length;
    } else {
      const size = 4 + suffix.length;
      const item = Buffer.allocUnsafe(size);
      writeU32(item, 0, suffix.length);
      suffix.copy(item, 4);
      items.push(item);
      bodyLen += item.length;
    }
  }

  const childBytes = page.leaf ? 0 : (count + 1) * 4;
  const out = Buffer.allocUnsafe(HEADER_SIZE + prefix.length + bodyLen + childBytes);
  MAGIC.copy(out, 0);
  out[4] = kind;
  out[5] = page.rightmost ? 1 : 0;
  out.writeUInt16BE(count, 6);
  writeU32(out, 8, page.id || 0);
  writeU32(out, 12, prefix.length);
  prefix.copy(out, HEADER_SIZE);

  let off = HEADER_SIZE + prefix.length;
  for (const item of items) {
    item.copy(out, off);
    off += item.length;
  }

  if (!page.leaf) {
    if (page.children.length !== count + 1) {
      throw new CodecError('internal page must have count + 1 children');
    }
    off = HEADER_SIZE + prefix.length + bodyLen;
    for (const child of page.children) {
      writeU32(out, off, child.id || 0);
      off += 4;
    }
  }

  return out;
}

function decodePage(buf) {
  if (!Buffer.isBuffer(buf)) buf = Buffer.from(buf);
  if (buf.length < HEADER_SIZE) throw new CodecError('truncated page header');
  if (!buf.subarray(0, 4).equals(MAGIC)) throw new CodecError('bad page magic');

  const kind = buf[4];
  if (kind !== LEAF && kind !== INTERNAL) throw new CodecError('bad page kind');
  const flags = buf[5];
  if (flags & ~1) throw new CodecError('unknown page flags');
  const count = buf.readUInt16BE(6);
  const id = readU32(buf, 8);
  const prefixLen = readU32(buf, 12);
  let off = HEADER_SIZE;
  if (off + prefixLen > buf.length) throw new CodecError('prefix overruns page');
  const prefix = Buffer.from(buf.subarray(off, off + prefixLen));
  off += prefixLen;

  const keys = [];
  const values = [];
  const truncLens = [];

  for (let i = 0; i < count; i++) {
    if (off + 4 > buf.length) throw new CodecError('truncated truncation length');
    const truncLen = readU32(buf, off);
    off += 4;
    if (off + truncLen > buf.length) throw new CodecError('truncated key suffix');
    const key = Buffer.concat([prefix, buf.subarray(off, off + truncLen)]);
    off += truncLen;
    keys.push(key);
    truncLens.push(truncLen);

    if (kind === LEAF) {
      if (off + 4 > buf.length) throw new CodecError('truncated value length');
      const valueLen = readU32(buf, off);
      off += 4;
      if (off + valueLen > buf.length) throw new CodecError('truncated value');
      values.push(Buffer.from(buf.subarray(off, off + valueLen)));
      off += valueLen;
    }
  }

  let childIds = null;
  if (kind === INTERNAL) {
    const nChildren = count + 1;
    if (off + nChildren * 4 > buf.length) throw new CodecError('truncated child ids');
    childIds = [];
    for (let i = 0; i < nChildren; i++) {
      childIds.push(readU32(buf, off));
      off += 4;
    }
  }
  if (off !== buf.length) throw new CodecError('trailing bytes after page');

  // Decode-side ordering check. A shortened separator that became equal to
  // (or less than) its predecessor makes the reconstructed page illegal.
  for (let i = 1; i < keys.length; i++) {
    if (compareBytes(keys[i - 1], keys[i]) >= 0) {
      throw new CodecError('reconstructed keys are not strictly increasing');
    }
  }

  return {
    id,
    leaf: kind === LEAF,
    rightmost: (flags & 1) !== 0,
    prefixLen,
    keys,
    truncLens,
    values,
    childIds,
  };
}

class Page {
  constructor(leaf, rightmost = false, id = NEXT_PAGE_ID++) {
    this.id = id;
    this.leaf = leaf;
    this.rightmost = rightmost;
    this.keys = [];
    this.values = leaf ? [] : undefined;
    this.children = leaf ? undefined : [];
    this.next = null;
    this.prev = null;
    this.parent = null;
  }
}

class BPlusTree {
  constructor({
    maxKeys = 8,
    requireMinimalSeparators = true,
    truncateSeparators = true,
  } = {}) {
    if (maxKeys < 3) throw new RangeError('maxKeys must be >= 3');
    this.maxKeys = maxKeys;
    this.minLeafKeys = Math.ceil(maxKeys / 2);
    this.minInternalKeys = Math.ceil((maxKeys + 1) / 2) - 1;
    this.requireMinimalSeparators = requireMinimalSeparators;
    this.truncateSeparators = truncateSeparators;
    this.root = new Page(true, true);
    this.firstLeaf = this.root;
    this.rightmostLeaf = this.root;
    this.size = 0;
  }

  _findLeaf(key) {
    let node = this.root;
    while (!node.leaf) {
      let i = 0;
      while (i < node.keys.length && compareBytes(key, node.keys[i]) >= 0) i++;
      node = node.children[i];
    }
    return node;
  }

  _lowerBound(keys, key) {
    let lo = 0;
    let hi = keys.length;
    while (lo < hi) {
      const mid = (lo + hi) >>> 1;
      if (compareBytes(keys[mid], key) < 0) lo = mid + 1;
      else hi = mid;
    }
    return lo;
  }

  _minKey(node) {
    while (!node.leaf) node = node.children[0];
    return node.keys.length ? node.keys[0] : null;
  }

  _maxKey(node) {
    while (!node.leaf) node = node.children[node.children.length - 1];
    return node.keys.length ? node.keys[node.keys.length - 1] : null;
  }

  _makeSeparator(leftLast, rightFirst) {
    return this.truncateSeparators
      ? minimalSeparator(leftLast, rightFirst)
      : Buffer.from(rightFirst);
  }

  _refreshSeparator(parent, i) {
    if (!parent || i < 0 || i >= parent.keys.length) return;
    const left = parent.children[i];
    const right = parent.children[i + 1];
    const lmax = this._maxKey(left);
    const rmin = this._minKey(right);
    if (lmax === null || rmin === null) return;
    if (compareBytes(lmax, rmin) < 0) {
      parent.keys[i] = this._makeSeparator(lmax, rmin);
    }
  }

  // Recompute separators immediately around `node`, and propagate up only
  // while the changed subtree is its parent's leftmost/rightmost child.
  _refreshAncestors(node, minChanged, maxChanged) {
    while (node.parent) {
      const parent = node.parent;
      const ci = parent.children.indexOf(node);
      if (ci < 0) return;
      if (minChanged && ci > 0) this._refreshSeparator(parent, ci - 1);
      if (maxChanged && ci < parent.children.length - 1) {
        this._refreshSeparator(parent, ci);
      }
      const nextMin = minChanged && ci === 0;
      const nextMax = maxChanged && ci === parent.children.length - 1;
      if (!nextMin && !nextMax) return;
      minChanged = nextMin;
      maxChanged = nextMax;
      node = parent;
    }
  }

  _insertIntoParent(left, separator, right) {
    const parent = left.parent;
    if (!parent) {
      const root = new Page(false, true);
      root.keys = [separator];
      root.children = [left, right];
      left.parent = root;
      right.parent = root;
      left.rightmost = false;
      right.rightmost = true;
      this.root = root;
      return;
    }

    const ci = parent.children.indexOf(left);
    if (ci < 0) throw new Error('split child is not attached to its parent');
    parent.keys.splice(ci, 0, separator);
    parent.children.splice(ci + 1, 0, right);
    right.parent = parent;
    if (parent.keys.length > this.maxKeys) this._splitInternal(parent);
  }

  _splitLeaf(node) {
    const mid = Math.ceil(node.keys.length / 2);
    const right = new Page(true, node.rightmost);
    right.keys = node.keys.splice(mid);
    right.values = node.values.splice(mid);
    right.next = node.next;
    right.prev = node;
    if (node.next) node.next.prev = right;
    node.next = right;

    if (node.rightmost) {
      node.rightmost = false;
      right.rightmost = true;
      this.rightmostLeaf = right;
    }

    const separator = this._makeSeparator(
      node.keys[node.keys.length - 1],
      right.keys[0]
    );
    this._insertIntoParent(node, separator, right);
    return right;
  }

  _splitInternal(node) {
    const mid = Math.floor(node.keys.length / 2);
    const up = node.keys[mid];
    const right = new Page(false, node.rightmost);
    right.keys = node.keys.slice(mid + 1);
    right.children = node.children.slice(mid + 1);
    node.keys = node.keys.slice(0, mid);
    node.children = node.children.slice(0, mid + 1);
    for (const child of right.children) child.parent = right;

    if (node.rightmost) {
      node.rightmost = false;
      right.rightmost = true;
    }

    this._insertIntoParent(node, up, right);
  }

  insert(key, value) {
    key = toBuffer(key);
    value = toBuffer(value);
    if (key.length === 0) throw new RangeError('empty keys are not supported');

    const leaf = this._findLeaf(key);
    const i = this._lowerBound(leaf.keys, key);
    if (i < leaf.keys.length && compareBytes(leaf.keys[i], key) === 0) {
      leaf.values[i] = value;
      return false;
    }

    const oldLen = leaf.keys.length;
    leaf.keys.splice(i, 0, key);
    leaf.values.splice(i, 0, value);
    this.size++;
    if (leaf.keys.length > this.maxKeys) {
      const right = this._splitLeaf(leaf);
      // Insertion may also have changed the old leaf's min and/or the new
      // right leaf's max. Recompute affected outer separators.
      this._refreshAncestors(leaf, true, true);
      this._refreshAncestors(right, true, true);
    } else {
      this._refreshAncestors(leaf, i === 0, i === oldLen);
    }
    return true;
  }

  get(key) {
    key = toBuffer(key);
    const leaf = this._findLeaf(key);
    const i = this._lowerBound(leaf.keys, key);
    if (i < leaf.keys.length && compareBytes(leaf.keys[i], key) === 0) {
      return leaf.values[i];
    }
    return undefined;
  }

  delete(key) {
    key = toBuffer(key);
    const leaf = this._findLeaf(key);
    const i = this._lowerBound(leaf.keys, key);
    if (i >= leaf.keys.length || compareBytes(leaf.keys[i], key) !== 0) {
      return false;
    }

    const oldLen = leaf.keys.length;
    leaf.keys.splice(i, 1);
    leaf.values.splice(i, 1);
    this.size--;

    const minChanged = i === 0;
    const maxChanged = i === oldLen - 1;

    let survivor = leaf;
    if (leaf !== this.root && leaf.keys.length < this.minLeafKeys) {
      survivor = this._rebalanceLeaf(leaf);
    } else {
      this._refreshAncestors(leaf, minChanged, maxChanged);
    }
    return true;
  }

  _rebalanceLeaf(node) {
    const parent = node.parent;
    if (!parent) return node;
    const ci = parent.children.indexOf(node);

    if (ci > 0) {
      const left = parent.children[ci - 1];
      if (left.keys.length > this.minLeafKeys) {
        node.keys.unshift(left.keys.pop());
        node.values.unshift(left.values.pop());
        this._refreshSeparator(parent, ci - 1);
        this._refreshAncestors(node, true, true);
        return node;
      }
    }

    if (ci < parent.children.length - 1) {
      const right = parent.children[ci + 1];
      if (right.keys.length > this.minLeafKeys) {
        node.keys.push(right.keys.shift());
        node.values.push(right.values.shift());
        this._refreshSeparator(parent, ci);
        this._refreshAncestors(node, true, true);
        return node;
      }
    }

    let survivor;
    if (ci > 0) {
      const left = parent.children[ci - 1];
      left.keys.push(...node.keys);
      left.values.push(...node.values);
      parent.keys.splice(ci - 1, 1);
      parent.children.splice(ci, 1);
      left.next = node.next;
      if (node.next) node.next.prev = left;
      if (node.rightmost) {
        left.rightmost = true;
        this.rightmostLeaf = left;
      }
      survivor = left;
    } else {
      const right = parent.children[ci + 1];
      node.keys.push(...right.keys);
      node.values.push(...right.values);
      parent.keys.splice(ci, 1);
      parent.children.splice(ci + 1, 1);
      node.next = right.next;
      if (right.next) right.next.prev = node;
      if (right.rightmost) {
        node.rightmost = true;
        this.rightmostLeaf = node;
      }
      survivor = node;
    }

    this._refreshAncestors(survivor, true, true);
    this._rebalanceInternal(parent);
    return survivor;
  }

  _rebalanceInternal(node) {
    if (node === this.root) {
      if (node.keys.length === 0) {
        const child = node.children[0];
        this.root = child;
        child.parent = null;
        child.rightmost = true;
      }
      return;
    }

    if (node.keys.length >= this.minInternalKeys) return;
    const parent = node.parent;
    if (!parent) return;
    const ci = parent.children.indexOf(node);

    if (ci > 0) {
      const left = parent.children[ci - 1];
      if (left.keys.length > this.minInternalKeys) {
        this._refreshSeparator(parent, ci - 1);
        node.keys.unshift(parent.keys[ci - 1]);
        const moved = left.children.pop();
        moved.parent = node;
        node.children.unshift(moved);
        parent.keys[ci - 1] = left.keys.pop();
        this._refreshSeparator(parent, ci - 1);
        this._refreshSeparator(node, 0);
        this._refreshAncestors(node, true, true);
        return;
      }
    }

    if (ci < parent.children.length - 1) {
      const right = parent.children[ci + 1];
      if (right.keys.length > this.minInternalKeys) {
        this._refreshSeparator(parent, ci);
        node.keys.push(parent.keys[ci]);
        const moved = right.children.shift();
        moved.parent = node;
        node.children.push(moved);
        parent.keys[ci] = right.keys.shift();
        this._refreshSeparator(parent, ci);
        this._refreshSeparator(node, node.keys.length - 1);
        this._refreshAncestors(node, true, true);
        return;
      }
    }

    if (ci > 0) {
      const left = parent.children[ci - 1];
      this._refreshSeparator(parent, ci - 1);
      const leftOldKeyCount = left.keys.length;
      left.keys.push(parent.keys[ci - 1], ...node.keys);
      left.children.push(...node.children);
      for (const child of node.children) child.parent = left;
      parent.keys.splice(ci - 1, 1);
      parent.children.splice(ci, 1);
      left.rightmost = node.rightmost;
      this._refreshSeparator(left, leftOldKeyCount);
      this._refreshAncestors(left, true, true);
      this._rebalanceInternal(parent);
    } else {
      const right = parent.children[ci + 1];
      this._refreshSeparator(parent, ci);
      const nodeOldKeyCount = node.keys.length;
      node.keys.push(parent.keys[ci], ...right.keys);
      node.children.push(...right.children);
      for (const child of right.children) child.parent = node;
      parent.keys.splice(ci, 1);
      parent.children.splice(ci + 1, 1);
      node.rightmost = right.rightmost;
      this._refreshSeparator(node, nodeOldKeyCount);
      this._refreshAncestors(node, true, true);
      this._rebalanceInternal(parent);
    }
  }

  range(start, end, { includeEnd = true } = {}) {
    start = start == null ? null : toBuffer(start);
    end = end == null ? null : toBuffer(end);
    const out = [];
    let node = start === null ? this.firstLeaf : this._findLeaf(start);
    while (node) {
      const from = start === null ? 0 : this._lowerBound(node.keys, start);
      for (let i = from; i < node.keys.length; i++) {
        const k = node.keys[i];
        if (end !== null) {
          const c = compareBytes(k, end);
          if (c > 0 || (c === 0 && !includeEnd)) return out;
        }
        out.push([Buffer.from(k), Buffer.from(node.values[i])]);
      }
      node = node.next;
    }
    return out;
  }

  entries() {
    return this.range(null, null);
  }

  /*
   * Structural + separator validation. With throughCodec=true every page is
   * encoded and decoded as it would be on disk, so decode-side ordering and
   * truncation checks are exercised too.
   */
  validate({ throughCodec = false } = {}) {
    if (this.size < 0) throw new Error('negative size');

    const leaves = [];
    let n = this.firstLeaf;
    let prev = null;
    let count = 0;
    let lastKey = null;
    while (n) {
      if (!n.leaf) throw new Error('non-leaf in leaf chain');
      if (n.prev !== prev) throw new Error('bad prev sibling pointer');
      if (n.parent && n.parent.children.indexOf(n) < 0) {
        throw new Error('leaf missing from parent child list');
      }
      if (throughCodec) {
        const decoded = decodePage(encodePage(n));
        if (!decoded.leaf || decoded.keys.length !== n.keys.length) {
          throw new Error('leaf codec round-trip mismatch');
        }
        for (let i = 0; i < n.keys.length; i++) {
          if (!decoded.keys[i].equals(n.keys[i])) {
            throw new Error('leaf key round-trip mismatch');
          }
          if (!decoded.values[i].equals(n.values[i])) {
            throw new Error('leaf value round-trip mismatch');
          }
        }
        if (decoded.rightmost !== n.rightmost) {
          throw new Error('leaf flag round-trip mismatch');
        }
      }
      for (let i = 0; i < n.keys.length; i++) {
        if (i > 0 && compareBytes(n.keys[i - 1], n.keys[i]) >= 0) {
          throw new Error('leaf keys not strictly increasing');
        }
        if (lastKey !== null && compareBytes(lastKey, n.keys[i]) >= 0) {
          throw new Error('global key order violation');
        }
        lastKey = n.keys[i];
        count++;
      }
      leaves.push(n);
      prev = n;
      n = n.next;
    }
    if (count !== this.size) throw new Error(`size mismatch: ${count} != ${this.size}`);
    if (this.rightmostLeaf !== prev) throw new Error('rightmostLeaf pointer is stale');
    for (const leaf of leaves) {
      if (leaf.rightmost !== (leaf === prev)) {
        throw new Error('leaf rightmost flag is wrong');
      }
    }

    const seen = new Set();
    const check = (node, isRoot) => {
      if (seen.has(node)) throw new Error('cycle in tree');
      seen.add(node);
      if (throughCodec) {
        const decoded = decodePage(encodePage(node));
        if (decoded.leaf !== node.leaf || decoded.keys.length !== node.keys.length) {
          throw new Error('page codec round-trip mismatch');
        }
        if (decoded.childIds && node.children) {
          for (let i = 0; i < node.children.length; i++) {
            if (decoded.childIds[i] !== node.children[i].id) {
              throw new Error('child id round-trip mismatch');
            }
          }
        }
      }
      for (let i = 1; i < node.keys.length; i++) {
        if (compareBytes(node.keys[i - 1], node.keys[i]) >= 0) {
          throw new Error('page keys not strictly increasing');
        }
      }
      if (node.leaf) {
        if (node.keys.length > this.maxKeys) throw new Error('leaf overflow');
        if (!isRoot && node.keys.length < this.minLeafKeys) {
          throw new Error('leaf underflow');
        }
        return;
      }
      if (node.keys.length > this.maxKeys) throw new Error('internal overflow');
      if (!isRoot && node.keys.length < this.minInternalKeys) {
        throw new Error('internal underflow');
      }
      if (!isRoot && node.keys.length === 0) throw new Error('empty internal page');
      if (node.children.length !== node.keys.length + 1) {
        throw new Error('internal child/key count mismatch');
      }
      for (const child of node.children) {
        if (child.parent !== node) throw new Error('bad child parent pointer');
      }
      for (let i = 0; i < node.keys.length; i++) {
        const lmax = this._maxKey(node.children[i]);
        const rmin = this._minKey(node.children[i + 1]);
        if (lmax === null || rmin === null) throw new Error('empty child subtree');
        const sep = node.keys[i];
        if (!(compareBytes(lmax, sep) < 0 && compareBytes(sep, rmin) <= 0)) {
          throw new Error('separator outside (leftLast, rightFirst]');
        }
        if (this.requireMinimalSeparators && this.truncateSeparators) {
          const shortest = minimalSeparator(lmax, rmin);
          if (!sep.equals(shortest)) {
            throw new Error(
              `separator is legal but not shortest: ` +
              `sep=${sep.toString('hex')} shortest=${shortest.toString('hex')} ` +
              `lmax=${lmax.toString('hex')} rmin=${rmin.toString('hex')}`
            );
          }
        }
      }
      for (const child of node.children) check(child, false);
    };
    check(this.root, true);

    let leftmost = this.root;
    while (!leftmost.leaf) leftmost = leftmost.children[0];
    if (leftmost !== this.firstLeaf) throw new Error('firstLeaf pointer is stale');
    return true;
  }
}

'use strict';

/*
 * 100k-operation differential fuzz.
 *
 *   compressed  : B+tree, prefix-compressed pages, shortest truncated separators
 *   reference   : same B+tree, but full uncompressed separators (right first key)
 *   oracle      : Map + sorted key projection, catches bugs shared by the trees
 */

const assert = require('assert');
const {
  BPlusTree,
  Page,
  encodePage,
  decodePage,
  minimalSeparator,
  compareBytes,
  CodecError,
} = require('./btree');

const OPS = Number(process.env.OPS || 100000);
const MAX_KEYS = Number(process.env.MAX_KEYS || 8);
const VERBOSE = process.env.VERBOSE === '1';

function mulberry32(a) {
  return function () {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
const rnd = mulberry32(0x51ed270b);
const randInt = (n) => Math.floor(rnd() * n);

function val(s) {
  return Buffer.from(s, 'utf8');
}

// Deliberately create many prefix relationships, including keys that are
// prefixes of other keys and keys differing only after a long common prefix.
let serial = 0;
function freshKey() {
  const n = serial++;
  const base = (n % 4096).toString(36).padStart(4, '0');
  const variant = n % 11;
  const prefixes = ['', 'a', 'aa', 'a\u0000', 'prefix-', 'prefix-\u0000'];
  const suffixes = ['', '\u0000', '\u0000\u0000', 'z', '\u00ff', '-tail'];
  const p = prefixes[(n >>> 4) % prefixes.length];
  const s = suffixes[variant % suffixes.length];
  return Buffer.concat([
    Buffer.from(p, 'latin1'),
    Buffer.from(base, 'latin1'),
    Buffer.from(s, 'latin1'),
  ]);
}

const pool = [];
for (let i = 0; i < 3500; i++) pool.push(freshKey());
for (let i = 0; i < 700; i++) pool.push(freshKey());

function keySignature(k) {
  return k.toString('hex');
}

function sortedOracleEntries(oracle) {
  return Array.from(oracle.entries())
    .map(([hex, value]) => [Buffer.from(hex, 'hex'), value])
    .sort((a, b) => compareBytes(a[0], b[0]));
}

function samePairs(a, b, label) {
  assert.strictEqual(a.length, b.length, `${label}: length`);
  for (let i = 0; i < a.length; i++) {
    assert(a[i][0].equals(b[i][0]), `${label}: key at ${i}`);
    assert(a[i][1].equals(b[i][1]), `${label}: value at ${i}`);
  }
}

function checkAll(compressed, reference, oracle, throughCodec, where) {
  compressed.validate({ throughCodec });
  reference.validate();

  assert.strictEqual(compressed.size, reference.size, `${where}: size`);
  assert.strictEqual(compressed.size, oracle.size, `${where}: oracle size`);

  const c = compressed.entries();
  const r = reference.entries();
  const o = sortedOracleEntries(oracle);
  samePairs(c, r, `${where}: compressed vs uncompressed tree`);
  samePairs(c, o, `${where}: compressed vs oracle`);
}

function testCodecRejectsBadOrdering() {
  const p = new Page(true, true);
  p.keys = [Buffer.from('dup'), Buffer.from('dup')];
  p.values = [Buffer.from('1'), Buffer.from('2')];
  const buf = encodePage(p);
  assert.throws(() => decodePage(buf), (e) => e instanceof CodecError);
  console.log('  codec rejects reconstructed duplicate/out-of-order keys');
}

function testSentinelCase() {
  const left = Buffer.from('abc');
  const right = Buffer.from('abcd');
  const sep = minimalSeparator(left, right);
  assert(compareBytes(left, sep) < 0);
  assert(compareBytes(sep, right) <= 0);
  assert(sep.equals(Buffer.from('abc\u0000', 'latin1')));
  assert(!(compareBytes(sep.subarray(0, sep.length - 1), left) > 0));
  console.log('  all-equal-prefix sentinel separator is strict and minimal');
}

function main() {
  console.log(`B+tree differential fuzz: ${OPS} operations, maxKeys=${MAX_KEYS}`);
  testSentinelCase();
  testCodecRejectsBadOrdering();

  const compressed = new BPlusTree({ maxKeys: MAX_KEYS });
  const reference = new BPlusTree({
    maxKeys: MAX_KEYS,
    truncateSeparators: false,
    requireMinimalSeparators: false,
  });
  const oracle = new Map();

  const counts = { insert: 0, delete: 0, get: 0, range: 0 };

  for (let i = 0; i < OPS; i++) {
    const key = pool[randInt(pool.length)];
    const hx = keySignature(key);
    const x = rnd();

    if (x < 0.55) {
      const value = val(`v:${i}:${randInt(1e9)}`);
      compressed.insert(key, value);
      reference.insert(key, value);
      oracle.set(hx, value);
      counts.insert++;
    } else if (x < 0.77) {
      const a = compressed.delete(key);
      const b = reference.delete(key);
      const c = oracle.delete(hx);
      assert.strictEqual(a, b, `delete return mismatch at op ${i}`);
      assert.strictEqual(a, c, `oracle delete return mismatch at op ${i}`);
      counts.delete++;
    } else if (x < 0.93) {
      const gc = compressed.get(key);
      const gr = reference.get(key);
      const go = oracle.get(hx);
      assert.strictEqual(gc === undefined, gr === undefined, `get presence mismatch at op ${i}`);
      if (gc !== undefined) {
        assert(gc.equals(gr), `get value mismatch tree/tree at op ${i}`);
        assert(gc.equals(go), `get value mismatch tree/oracle at op ${i}`);
      }
      counts.get++;
    } else {
      let start = pool[randInt(pool.length)];
      let end = pool[randInt(pool.length)];
      if (compareBytes(start, end) > 0) [start, end] = [end, start];
      const includeEnd = rnd() < 0.5;
      const rc = compressed.range(start, end, { includeEnd });
      const rr = reference.range(start, end, { includeEnd });
      samePairs(rc, rr, `range tree/tree at op ${i}`);

      const ro = sortedOracleEntries(oracle).filter(([k]) => {
        if (compareBytes(k, start) < 0) return false;
        const c = compareBytes(k, end);
        return includeEnd ? c <= 0 : c < 0;
      });
      samePairs(rc, ro, `range tree/oracle at op ${i}`);
      counts.range++;
    }

    if ((i + 1) % 1000 === 0) {
      const throughCodec = (i + 1) % 10000 === 0;
      checkAll(compressed, reference, oracle, throughCodec, `checkpoint ${i + 1}`);
      if (VERBOSE) {
        console.log(
          `  ${i + 1}: size=${compressed.size} height=${height(compressed)}` +
            (throughCodec ? ' [codec round-trip]' : '')
        );
      }
    }
  }

  checkAll(compressed, reference, oracle, true, 'final');
  console.log('  counts:', counts);
  console.log(`  final size: ${compressed.size}, height: ${height(compressed)}`);
  console.log(`PASS: ${OPS} differential fuzz`);
}

function height(tree) {
  let h = 1;
  let n = tree.root;
  while (!n.leaf) {
    h++;
    n = n.children[0];
  }
  return h;
}

try {
  main();
} catch (err) {
  console.error('FAIL:', err && err.stack ? err.stack : err);
  process.exitCode = 1;
}
