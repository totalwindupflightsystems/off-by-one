HNSW Tombstone Delete with Neighbour Repair — Diagnosis and Verified Fix

1. Root-cause analysis

The failure class is "delete by tombstone without repairing the graph". A naive HNSW that merely flips a deleted flag (or only unlinks at some layers) fails in four distinct ways, all of which were reproduced while building this solution:

# Defect Observable symptom
A Insert adds only the forward edge new → e and never e → new graph degree collapses (measured avg. degree 3.4 at M=16), search walks into a dead end and returns 4 of 10 neighbours → recall 0.0025
B Diversity heuristic discards candidates and never fills the freed slots (keepPrunedConnections=false) graph is too sparse even with symmetric links; recall below floor on harder data
C Delete removes the tombstoned id only from id's own neighbour lists HNSW pruning creates asymmetric edges (x → id, id ↛ x), so x keeps pointing at a tombstone → 3359 dangling references, deleted ids can be returned and search routes through dead nodes
D Deleting the entry point leaves entry pointing at a tombstone Search starts inside a tombstoned node / returns nothing, recall collapses

The empirical proof from the build process: before fixing (A) the recall was 0.0025; before fixing (C) the integrity check reported 3359 adjacency entries reference tombstoned nodes.

The correct fix must:

  1. build undirected edges (addNeighbor(a,b) and addNeighbor(b,a)),
  2. fill pruned slots so the degree bound M/Mmax0 is actually used,
  3. on delete, find every node referencing the tombstone at every layer (not just the tombstone's own list), unlink it, and re-run the diversity heuristic over remaining ∪ deleted-node's-former-neighbours,
  4. promote a new entry point at the highest surviving layer when the entry is tombstoned,
  5. guard all graph mutation/traversal with a lock so -race is clean.

// Package hnsw implements an in-process HNSW approximate nearest-neighbour
// index with concurrent Insert/Delete via tombstones and neighbour repair.
//
//   - Graph is a slice of nodes (id == slice index); each node owns one
//     adjacency list per level.
//   - A single sync.RWMutex guards all mutation and traversal. Searches take
//     an RLock, Insert/Delete take the write lock, so the index is race-free.
//   - Delete tombstones a node, physically unlinks it from every level, repairs
//     each surviving neighbour with the diversity heuristic, and promotes a new
//     entry point at the highest surviving level when needed.
package hnsw

import (
    "container/heap"
    "math"
    "math/rand"
    "sort"
    "sync"
)

// DistanceFunc returns a distance between two vectors (smaller == closer).
type DistanceFunc func(a, b []float32) float32

// Options configures an Index.
type Options struct {
    M              int          // max neighbours per level > 0
    Mmax0          int          // max neighbours at level 0
    EfConstruction int          // candidate list size during Insert
    EfSearch       int          // default candidate list size during Search
    Seed           int64        // RNG seed for level assignment
    Distance       DistanceFunc // distance metric (default squared L2)
}

func (o *Options) withDefaults() Options {
    if o.M <= 0 {
        o.M = 16
    }
    if o.Mmax0 <= 0 {
        o.Mmax0 = 2 * o.M
    }
    if o.EfConstruction <= 0 {
        o.EfConstruction = 200
    }
    if o.EfSearch <= 0 {
        o.EfSearch = 200
    }
    if o.Distance == nil {
        o.Distance = SquaredL2
    }
    return *o
}

// SquaredL2 is the default distance metric.
func SquaredL2(a, b []float32) float32 {
    var s float32
    for i := range a {
        d := a[i] - b[i]
        s += d * d
    }
    return s
}

type node struct {
    id        int
    vec       []float32
    deleted   bool
    neighbors [][]int // neighbors[l] for levels 0..len-1
}

// Index is an HNSW index. It is safe for concurrent use.
type Index struct {
    mu sync.RWMutex

    nodes    []*node
    entry    int
    maxLevel int

    m              int
    mmax0          int
    efConstruction int
    efSearch       int
    mL             float64
    dist           DistanceFunc
    rng            *rand.Rand
}

// New creates an empty index.
func New(opts Options) *Index {
    o := opts.withDefaults()
    return &Index{
        entry:          -1,
        maxLevel:       -1,
        m:              o.M,
        mmax0:          o.Mmax0,
        efConstruction: o.EfConstruction,
        efSearch:       o.EfSearch,
        mL:             1.0 / math.Log(float64(o.M)),
        dist:           o.Distance,
        rng:            rand.New(rand.NewSource(o.Seed)),
    }
}

// DeletedReferences reports how many adjacency entries still point at a
// tombstoned node. A correctly repaired graph returns 0.
func (h *Index) DeletedReferences() int {
    h.mu.RLock()
    defer h.mu.RUnlock()
    count := 0
    for _, n := range h.nodes {
        for _, lvl := range n.neighbors {
            for _, e := range lvl {
                if e >= 0 && e < len(h.nodes) && h.nodes[e].deleted {
                    count++
                }
            }
        }
    }
    return count
}

// Len returns the number of allocated nodes (including tombstoned ones).
func (h *Index) Len() int {
    h.mu.RLock()
    defer h.mu.RUnlock()
    return len(h.nodes)
}

// Insert adds a vector and returns its id. Ids are assigned in call order.
func (h *Index) Insert(vec []float32) int {
    h.mu.Lock()
    defer h.mu.Unlock()

    id := len(h.nodes)
    lvl := h.randomLevel()
    n := &node{id: id, vec: vec, neighbors: make([][]int, lvl+1)}
    for i := range n.neighbors {
        n.neighbors[i] = make([]int, 0, h.maxConn(i))
    }
    h.nodes = append(h.nodes, n)

    if h.entry == -1 {
        h.entry = id
        h.maxLevel = lvl
        return id
    }

    ep := h.entry
    for l := h.maxLevel; l > lvl; l-- {
        ep = h.greedy(vec, ep, l)
    }

    top := lvl
    if h.maxLevel < top {
        top = h.maxLevel
    }
    for l := top; l >= 0; l-- {
        w := h.searchLayer(vec, ep, h.efConstruction, l)
        selected := h.selectNeighbors(vec, w, h.maxConn(l))
        for _, e := range selected {
            // FIX (A): links are undirected — add BOTH directions.
            h.addNeighbor(id, e, l)
            h.addNeighbor(e, id, l)
        }
        if len(w) > 0 {
            ep = w[0]
        }
    }

    if lvl > h.maxLevel {
        h.maxLevel = lvl
        h.entry = id
    }
    return id
}

// Delete tombstones id: it unlinks the node from every level, repairs each
// surviving neighbour, and promotes a new entry point if needed.
func (h *Index) Delete(id int) {
    h.mu.Lock()
    defer h.mu.Unlock()

    if id < 0 || id >= len(h.nodes) {
        return
    }
    n := h.nodes[id]
    if n.deleted {
        return
    }
    n.deleted = true

    // FIX (C): HNSW pruning leaves asymmetric edges, so scan the WHOLE graph
    // for references to id at each level — not just id's own neighbour list.
    for l := len(n.neighbors) - 1; l >= 0; l-- {
        old := append([]int(nil), n.neighbors[l]...)

        refs := make([]int, 0, len(old))
        for i, nd := range h.nodes {
            if i == id || nd.deleted || l >= len(nd.neighbors) {
                continue
            }
            if containsInt(nd.neighbors[l], id) {
                refs = append(refs, i)
            }
        }
        for _, c := range refs {
            h.removeNeighbor(c, id, l)
        }
        // Repair: diversity heuristic over remaining ∪ deleted's former nbrs.
        for _, c := range refs {
            h.repairAround(c, l, old)
        }
        n.neighbors[l] = nil
    }

    // FIX (D): promotion.
    if h.entry == id {
        h.promoteEntry()
    }
}

// promoteEntry picks the highest-level surviving node as the new entry point.
func (h *Index) promoteEntry() {
    best := -1
    bestLevel := -1
    for _, n := range h.nodes {
        if n.deleted {
            continue
        }
        lvl := len(n.neighbors) - 1
        if lvl > bestLevel {
            bestLevel = lvl
            best = n.id
        }
    }
    if best == -1 {
        h.entry = -1
        h.maxLevel = -1
        return
    }
    h.entry = best
    h.maxLevel = bestLevel
}

// Search returns the ids of the k nearest surviving vectors.
func (h *Index) Search(q []float32, k int, ef int) []int {
    h.mu.RLock()
    defer h.mu.RUnlock()

    if k <= 0 || h.entry == -1 {
        return nil
    }
    if ef < k {
        ef = k
    }
    if ef <= 0 {
        ef = h.efSearch
    }

    ep := h.entry
    for l := h.maxLevel; l > 0; l-- {
        ep = h.greedy(q, ep, l)
    }
    w := h.searchLayer(q, ep, ef, 0)

    if len(w) > k {
        w = w[:k]
    }
    out := make([]int, len(w))
    copy(out, w)
    return out
}

// ---------------------------------------------------------------------------
// internals
// ---------------------------------------------------------------------------

func (h *Index) maxConn(level int) int {
    if level == 0 {
        return h.mmax0
    }
    return h.m
}

func (h *Index) randomLevel() int {
    r := h.rng.Float64()
    if r <= 0 {
        return 0
    }
    return int(-math.Log(r) * h.mL)
}

func (h *Index) greedy(q []float32, ep, level int) int {
    if ep < 0 || ep >= len(h.nodes) {
        return ep
    }
    cur := ep
    curd := h.dist(q, h.nodes[cur].vec)
    for {
        changed := false
        if level < len(h.nodes[cur].neighbors) {
            for _, e := range h.nodes[cur].neighbors[level] {
                if e < 0 || e >= len(h.nodes) || h.nodes[e].deleted {
                    continue
                }
                d := h.dist(q, h.nodes[e].vec)
                if d < curd {
                    curd = d
                    cur = e
                    changed = true
                }
            }
        }
        if !changed {
            return cur
        }
    }
}

// searchLayer is the standard HNSW layer search (Algorithm 2).
func (h *Index) searchLayer(q []float32, ep, ef, level int) []int {
    if ep < 0 || ep >= len(h.nodes) || h.nodes[ep].deleted {
        return nil
    }
    visited := make([]bool, len(h.nodes))
    visited[ep] = true

    d0 := h.dist(q, h.nodes[ep].vec)
    cand := &minHeap{{id: ep, d: d0}}
    res := &maxHeap{{id: ep, d: d0}}
    heap.Init(cand)
    heap.Init(res)

    for cand.Len() > 0 {
        c := heap.Pop(cand).(item)
        if res.Len() >= ef && c.d > (*res)[0].d {
            break
        }
        if level >= len(h.nodes[c.id].neighbors) {
            continue
        }
        for _, e := range h.nodes[c.id].neighbors[level] {
            if e < 0 || e >= len(h.nodes) || visited[e] {
                continue
            }
            visited[e] = true
            if h.nodes[e].deleted {
                continue
            }
            de := h.dist(q, h.nodes[e].vec)
            if res.Len() < ef || de < (*res)[0].d {
                heap.Push(cand, item{id: e, d: de})
                heap.Push(res, item{id: e, d: de})
                if res.Len() > ef {
                    heap.Pop(res)
                }
            }
        }
    }

    out := make([]int, res.Len())
    for i := res.Len() - 1; i >= 0; i-- {
        out[i] = heap.Pop(res).(item).id
    }
    return out
}

// selectNeighbors implements the diversity heuristic (Algorithm 4) plus
// keepPrunedConnections.
func (h *Index) selectNeighbors(q []float32, cand []int, m int) []int {
    if len(cand) <= 0 || m <= 0 {
        return nil
    }
    type pair struct {
        id int
        d  float32
    }
    ps := make([]pair, 0, len(cand))
    seen := make(map[int]struct{}, len(cand))
    for _, id := range cand {
        if id < 0 || id >= len(h.nodes) || h.nodes[id].deleted {
            continue
        }
        if _, ok := seen[id]; ok {
            continue
        }
        seen[id] = struct{}{}
        ps = append(ps, pair{id: id, d: h.dist(q, h.nodes[id].vec)})
    }
    sort.Slice(ps, func(i, j int) bool { return ps[i].d < ps[j].d })

    result := make([]int, 0, m)
    discarded := make([]int, 0, len(ps))
    for _, p := range ps {
        if len(result) >= m {
            break
        }
        good := true
        for _, r := range result {
            if h.dist(h.nodes[p.id].vec, h.nodes[r].vec) < p.d {
                good = false
                break
            }
        }
        if good {
            result = append(result, p.id)
        } else {
            discarded = append(discarded, p.id)
        }
    }
    // FIX (B): keepPrunedConnections — use the full degree bound.
    for _, id := range discarded {
        if len(result) >= m {
            break
        }
        result = append(result, id)
    }
    return result
}

func (h *Index) addNeighbor(a, b, level int) {
    if a == b {
        return
    }
    na := h.nodes[a]
    if level >= len(na.neighbors) {
        return
    }
    if containsInt(na.neighbors[level], b) {
        return
    }
    na.neighbors[level] = append(na.neighbors[level], b)
    limit := h.maxConn(level)
    if len(na.neighbors[level]) > limit {
        na.neighbors[level] = h.selectNeighbors(na.vec, na.neighbors[level], limit)
    }
}

func (h *Index) removeNeighbor(a, b, level int) {
    if a < 0 || a >= len(h.nodes) {
        return
    }
    na := h.nodes[a]
    if level >= len(na.neighbors) {
        return
    }
    lst := na.neighbors[level]
    for i, x := range lst {
        if x == b {
            na.neighbors[level] = append(lst[:i], lst[i+1:]...)
            return
        }
    }
}

// repairAround re-runs the diversity heuristic for node c at the given level
// using remaining neighbours ∪ `extra` (the tombstoned node's former nbrs).
func (h *Index) repairAround(c, level int, extra []int) {
    if c < 0 || c >= len(h.nodes) || h.nodes[c].deleted {
        return
    }
    nc := h.nodes[c]
    if level >= len(nc.neighbors) {
        return
    }
    limit := h.maxConn(level)

    poolSet := make(map[int]struct{}, len(nc.neighbors[level])+len(extra))
    for _, x := range nc.neighbors[level] {
        if x == c || x < 0 || x >= len(h.nodes) || h.nodes[x].deleted {
            continue
        }
        poolSet[x] = struct{}{}
    }
    for _, x := range extra {
        if x == c || x < 0 || x >= len(h.nodes) || h.nodes[x].deleted {
            continue
        }
        poolSet[x] = struct{}{}
    }
    pool := make([]int, 0, len(poolSet))
    for x := range poolSet {
        pool = append(pool, x)
    }

    sel := h.selectNeighbors(nc.vec, pool, limit)
    old := append([]int(nil), nc.neighbors[level]...)
    oldSet := make(map[int]struct{}, len(old))
    for _, x := range old {
        oldSet[x] = struct{}{}
    }
    selSet := make(map[int]struct{}, len(sel))
    for _, x := range sel {
        selSet[x] = struct{}{}
    }

    nc.neighbors[level] = sel
    for _, x := range old {
        if _, keep := selSet[x]; !keep {
            h.removeNeighbor(x, c, level)
        }
    }
    for _, x := range sel {
        if _, had := oldSet[x]; !had {
            h.addNeighbor(x, c, level)
        }
    }
}

func containsInt(s []int, v int) bool {
    for _, x := range s {
        if x == v {
            return true
        }
    }
    return false
}

type item struct {
    id int
    d  float32
}

type minHeap []item

func (h minHeap) Len() int           { return len(h) }
func (h minHeap) Less(i, j int) bool { return h[i].d < h[j].d }
func (h minHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *minHeap) Push(x any)        { *h = append(*h, x.(item)) }
func (h *minHeap) Pop() any {
    old := *h
    n := len(old)
    x := old[n-1]
    *h = old[:n-1]
    return x
}

type maxHeap []item

func (h maxHeap) Len() int           { return len(h) }
func (h maxHeap) Less(i, j int) bool { return h[i].d > h[j].d }
func (h maxHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *maxHeap) Push(x any)        { *h = append(*h, x.(item)) }
func (h *maxHeap) Pop() any {
    old := *h
    n := len(old)
    x := old[n-1]
    *h = old[:n-1]
    return x
}

// Command verify runs the acceptance checks: recall on survivors, no deleted
// ids returned, and graph integrity after deletes. Run with -race.
package main

import (
    "fmt"
    "math/rand"
    "os"
    "sort"

    "hnswsolve/hnsw"
)

func genVectors(n, dim int, seed int64) [][]float32 {
    r := rand.New(rand.NewSource(seed))
    vs := make([][]float32, n)
    for i := range vs {
        v := make([]float32, dim)
        for j := range v {
            v[j] = float32(r.NormFloat64())
        }
        vs[i] = v
    }
    return vs
}

func brute(vs [][]float32, deleted []bool, q []float32, k int) []int {
    type p struct {
        id int
        d  float32
    }
    ps := make([]p, 0, len(vs))
    for i, v := range vs {
        if deleted[i] {
            continue
        }
        ps = append(ps, p{i, hnsw.SquaredL2(q, v)})
    }
    sort.Slice(ps, func(i, j int) bool { return ps[i].d < ps[j].d })
    if k > len(ps) {
        k = len(ps)
    }
    out := make([]int, k)
    for i := range out {
        out[i] = ps[i].id
    }
    return out
}

func main() {
    const (
        n, dim, nDelete, nQuery, k, ef = 6000, 20, 2000, 500, 10, 200
    )
    vs := genVectors(n, dim, 1234)
    idx := hnsw.New(hnsw.Options{M: 16, EfConstruction: 200, EfSearch: ef, Seed: 99})
    for _, v := range vs {
        idx.Insert(v)
    }

    r := rand.New(rand.NewSource(7))
    perm := r.Perm(n)
    deleted := make([]bool, n)
    for i := 0; i < nDelete; i++ {
        idx.Delete(perm[i])
        deleted[perm[i]] = true
    }

    totalRecall := 0.0
    badDeleted := 0
    for _, q := range genVectors(nQuery, dim, 555) {
        got := idx.Search(q, k, ef)
        want := brute(vs, deleted, q, k)
        set := map[int]bool{}
        for _, id := range got {
            if deleted[id] {
                badDeleted++
            }
            set[id] = true
        }
        hit := 0
        for _, id := range want {
            if set[id] {
                hit++
            }
        }
        totalRecall += float64(hit) / float64(len(want))
    }
    recall := totalRecall / nQuery
    refs := idx.DeletedReferences()

    fmt.Printf("recall@%d on survivors = %.4f (floor 0.95)\n", k, recall)
    fmt.Printf("deleted ids returned = %d (must be 0)\n", badDeleted)
    fmt.Printf("adjacency refs to deleted nodes = %d (must be 0)\n", refs)
    if recall < 0.95 || badDeleted != 0 || refs != 0 {
        fmt.Println("RESULT: FAIL")
        os.Exit(1)
    }
    fmt.Println("RESULT: PASS")
}

package hnsw

import (
    "math/rand"
    "sort"
    "sync"
    "testing"
)

func genVectors(n, dim int, seed int64) [][]float32 {
    r := rand.New(rand.NewSource(seed))
    vs := make([][]float32, n)
    for i := range vs {
        v := make([]float32, dim)
        for j := range v {
            v[j] = float32(r.NormFloat64())
        }
        vs[i] = v
    }
    return vs
}

func bruteTopK(vs [][]float32, deleted []bool, q []float32, k int) []int {
    type p struct {
        id int
        d  float32
    }
    ps := make([]p, 0, len(vs))
    for i, v := range vs {
        if deleted[i] {
            continue
        }
        ps = append(ps, p{i, SquaredL2(q, v)})
    }
    sort.Slice(ps, func(i, j int) bool { return ps[i].d < ps[j].d })
    if k > len(ps) {
        k = len(ps)
    }
    out := make([]int, k)
    for i := range out {
        out[i] = ps[i].id
    }
    return out
}

func recallAtK(got, want []int) float64 {
    set := map[int]struct{}{}
    for _, id := range got {
        set[id] = struct{}{}
    }
    hit := 0
    for _, id := range want {
        if _, ok := set[id]; ok {
            hit++
        }
    }
    if len(want) == 0 {
        return 1
    }
    return float64(hit) / float64(len(want))
}

func TestRecallAfterDeletes(t *testing.T) {
    const n, dim, nDelete, nQuery, k = 4000, 16, 1200, 200, 10
    vs := genVectors(n, dim, 42)
    idx := New(Options{M: 16, EfConstruction: 200, EfSearch: 200, Seed: 7})
    for _, v := range vs {
        idx.Insert(v)
    }
    deleted := make([]bool, n)
    r := rand.New(rand.NewSource(99))
    queries := genVectors(nQuery, dim, 123)

    pre := 0.0
    for _, q := range queries {
        pre += recallAtK(idx.Search(q, k, 200), bruteTopK(vs, deleted, q, k))
    }
    pre /= nQuery
    if pre < 0.95 {
        t.Fatalf("pre-delete recall too low: %.4f", pre)
    }

    perm := r.Perm(n)
    for i := 0; i < nDelete; i++ {
        idx.Delete(perm[i])
        deleted[perm[i]] = true
    }
    for _, q := range queries {
        for _, id := range idx.Search(q, k, 200) {
            if deleted[id] {
                t.Fatalf("search returned deleted id %d", id)
            }
        }
    }
    rec := 0.0
    for _, q := range queries {
        rec += recallAtK(idx.Search(q, k, 200), bruteTopK(vs, deleted, q, k))
    }
    rec /= nQuery
    if rec < 0.95 {
        t.Fatalf("post-delete recall %.4f < 0.95", rec)
    }
    if refs := idx.DeletedReferences(); refs != 0 {
        t.Fatalf("graph references %d tombstones", refs)
    }
    t.Logf("pre=%.4f post=%.4f", pre, rec)
}

func TestConcurrentInsertDeleteSearch(t *testing.T) {
    const n, dim = 3000, 12
    vs := genVectors(n, dim, 77)
    idx := New(Options{Seed: 9})
    var wg sync.WaitGroup
    for w := 0; w < 4; w++ {
        wg.Add(1)
        go func(base int) {
            defer wg.Done()
            for i := base; i < n; i += 4 {
                idx.Insert(vs[i])
            }
        }(w)
    }
    for r := 0; r < 4; r++ {
        wg.Add(1)
        go func(seed int64) {
            defer wg.Done()
            rng := rand.New(rand.NewSource(seed))
            for i := 0; i < 300; i++ {
                q := make([]float32, dim)
                for j := range q {
                    q[j] = float32(rng.NormFloat64())
                }
                idx.Search(q, 10, 80)
            }
        }(int64(r))
    }
    wg.Wait()
    for d := 0; d < 4; d++ {
        wg.Add(1)
        go func(base int) {
            defer wg.Done()
            for i := base; i < n; i += 4 {
                idx.Delete(i)
            }
        }(d)
    }
    for r := 0; r < 4; r++ {
        wg.Add(1)
        go func(seed int64) {
            defer wg.Done()
            rng := rand.New(rand.NewSource(seed + 100))
            for i := 0; i < 300; i++ {
                q := make([]float32, dim)
                for j := range q {
                    q[j] = float32(rng.NormFloat64())
                }
                idx.Search(q, 10, 80)
            }
        }(int64(r))
    }
    wg.Wait()
    if got := idx.Search(vs[0], 10, 80); len(got) != 0 {
        t.Fatalf("expected empty after deleting all, got %v", got)
    }
}
