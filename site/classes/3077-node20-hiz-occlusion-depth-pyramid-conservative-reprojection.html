Root-cause analysis

Four independent bugs make a Hi-Z occlusion pass non-conservative:

  1. Reversed-Z inversion In reversed-Z raw depth, near = 1, far = 0. A tile's conservative bound must be the farthest sample: min(raw). Using max(raw) stores the nearest sample, so a thin near occluder can cause visible geometry behind it to be culled when the tile also contains far background.

  2. Non-power-of-two mip truncation floor(w/2) drops the last row/column when w or h is odd. A parent tile can then contain fewer samples than it should, making it look nearer than the real farthest sample and causing false culling. Use ceil(w/2) and clamp child lookups.

  3. Stale screen-border texels after camera motion The previous frame's depth buffer is undefined outside the previous viewport. Clamping reprojected coordinates to [0, 1] reuses the previous border depth for newly exposed screen regions. The fix is to fail open: if an AABB's previous-screen footprint leaves the previous viewport, do not cull.

  4. Perspective and coplanar edge cases AABB corners behind the camera (w <= 0) must not be projected. Coplanar geometry must not be treated as strictly behind the occluder. Use strict depth comparison and reject non-valid projections.

The key invariant is:

If the culler reports occluded, then for every base depth sample the AABB can touch, the AABB's nearest depth is farther than that sample's depth.

Exact fix

The complete implementation is below. It stores raw depth buffers, handles both reversedZ = true / false, handles ceil-sized mip chains, queries the pyramid exactly over a rectangle, converts OpenGL -1..1 clip depth when needed, and fails open on borders.

'use strict';

/**
 * Conservative Hi-Z (hierarchical Z) helper for CPU-side reprojection tests.
 *
 * Depth convention
 * ----------------
 * A depth buffer value is the *raw* depth stored by the renderer.  This module
 * works with two common conventions:
 *
 *   reversedZ = true   near = 1, far = 0  (WebGPU / reversed-Z style)
 *   reversedZ = false  near = 0, far = 1  (classic OpenGL/D3D style)
 *
 * The pyramid is called a "max depth" pyramid because semantically it stores a
 * conservative bound on the *farthest* depth in each tile.  For reversed-Z raw
 * depth, "farthest" is the minimum raw value, so the reduction is min(); for
 * classic depth it is max().  Using max(raw) on a reversed-Z buffer is the
 * classic bug: it stores the nearest sample and can cull visible geometry.
 */

const EPS = 1e-12;

function defaultFarDepth(reversedZ) {
  return reversedZ ? 0 : 1;
}

function sanitizeDepth(value, reversedZ, farDepth) {
  if (!Number.isFinite(value)) return farDepth;
  return value;
}

function reduceValues(a, b, c, d, reversedZ) {
  if (reversedZ) return Math.min(a, b, c, d);
  return Math.max(a, b, c, d);
}

function assertPositiveInt(value, name) {
  if (!Number.isInteger(value) || value <= 0) {
    throw new RangeError(`${name} must be a positive integer, got ${value}`);
  }
}

/**
 * Build a conservative mip pyramid from a width x height depth buffer.
 *
 * Non-power-of-two handling
 * -------------------------
 * A naive `floor(w/2)` mip chain drops the last row/column and can make a
 * parent tile look nearer (reversed-Z) than the pixels it should cover.  This
 * implementation uses `ceil(w/2)` and clamps child coordinates.  That is a
 * conservative over-approximation: every base pixel contributes to every
 * ancestor, and a parent never becomes nearer than the farthest base sample it
 * covers.
 *
 * @param {ArrayLike<number>} depth
 * @param {number} width
 * @param {number} height
 * @param {{reversedZ?: boolean, farDepth?: number}} [options]
 * @returns {{width:number,height:number,reversedZ:boolean,farDepth:number,levels:Array, maxLevel:number}}
 */
function buildDepthPyramid(depth, width, height, options = {}) {
  assertPositiveInt(width, 'width');
  assertPositiveInt(height, 'height');

  const reversedZ = options.reversedZ !== false;
  const farDepth = options.farDepth !== undefined
    ? options.farDepth
    : defaultFarDepth(reversedZ);

  const count = width * height;
  if (!depth || depth.length < count) {
    throw new RangeError(`depth buffer has ${depth ? depth.length : 0} elements; expected at least ${count}`);
  }

  const base = new Float64Array(count);
  for (let i = 0; i < count; i++) {
    base[i] = sanitizeDepth(Number(depth[i]), reversedZ, farDepth);
  }

  const levels = [{ width, height, data: base }];
  let w = width;
  let h = height;

  while (w > 1 || h > 1) {
    const nw = Math.max(1, Math.ceil(w / 2));
    const nh = Math.max(1, Math.ceil(h / 2));
    const src = levels[levels.length - 1];
    const dst = new Float64Array(nw * nh);

    for (let y = 0; y < nh; y++) {
      const y0 = Math.min(2 * y, h - 1);
      const y1 = Math.min(2 * y + 1, h - 1);
      for (let x = 0; x < nw; x++) {
        const x0 = Math.min(2 * x, w - 1);
        const x1 = Math.min(2 * x + 1, w - 1);
        const a = src.data[y0 * w + x0];
        const b = src.data[y0 * w + x1];
        const c = src.data[y1 * w + x0];
        const d = src.data[y1 * w + x1];
        dst[y * nw + x] = reduceValues(a, b, c, d, reversedZ);
      }
    }

    levels.push({ width: nw, height: nh, data: dst });
    w = nw;
    h = nh;
  }

  return {
    width,
    height,
    reversedZ,
    farDepth,
    levels,
    maxLevel: levels.length - 1,
  };
}

/**
 * Exact conservative reduction over an inclusive base-pixel rectangle.
 *
 * The traversal recurses through the ceil()-sized pyramid so that it does not
 * rely on power-of-two dimensions.  Every visited node's covered base interval
 * is derived from its level and index, and a node is used directly only when
 * that whole interval lies inside the query rectangle.  That makes the result
 * exactly min(raw) for reversed-Z and max(raw) for classic depth over the
 * requested base pixels.
 *
 * @returns {number} min raw depth (reversedZ) or max raw depth (classic)
 */
function queryDepthPyramid(pyramid, x0, y0, x1, y1) {
  const { width, height, reversedZ, levels, maxLevel } = pyramid;

  if (x0 > x1 || y0 > y1) {
    return reversedZ ? Infinity : -Infinity;
  }

  // Clamp to the valid base range.  Callers should have already rejected
  // rectangles outside the previous screen; this is only defensive.
  x0 = Math.max(0, x0);
  y0 = Math.max(0, y0);
  x1 = Math.min(width - 1, x1);
  y1 = Math.min(height - 1, y1);
  if (x0 > x1 || y0 > y1) {
    return reversedZ ? Infinity : -Infinity;
  }

  let best = reversedZ ? Infinity : -Infinity;
  const stack = [[maxLevel, 0, 0]];

  while (stack.length > 0) {
    const [level, ix, iy] = stack.pop();
    const lvl = levels[level];
    const size = 2 ** level;

    const bx0 = ix * size;
    const by0 = iy * size;
    const bx1 = Math.min((ix + 1) * size - 1, width - 1);
    const by1 = Math.min((iy + 1) * size - 1, height - 1);

    if (bx0 > x1 || bx1 < x0 || by0 > y1 || by1 < y0) continue;

    if (bx0 >= x0 && bx1 <= x1 && by0 >= y0 && by1 <= y1) {
      const value = lvl.data[iy * lvl.width + ix];
      if (reversedZ) {
        if (value < best) best = value;
      } else if (value > best) {
        best = value;
      }
      continue;
    }

    if (level === 0) {
      const value = levels[0].data[iy * width + ix];
      if (reversedZ) {
        if (value < best) best = value;
      } else if (value > best) {
        best = value;
      }
      continue;
    }

    const prev = levels[level - 1];
    const cx0 = Math.min(2 * ix, prev.width - 1);
    const cx1 = Math.min(2 * ix + 1, prev.width - 1);
    const cy0 = Math.min(2 * iy, prev.height - 1);
    const cy1 = Math.min(2 * iy + 1, prev.height - 1);

    stack.push([level - 1, cx0, cy0]);
    stack.push([level - 1, cx1, cy0]);
    stack.push([level - 1, cx0, cy1]);
    stack.push([level - 1, cx1, cy1]);
  }

  return best;
}

function transformPoint(m, x, y, z) {
  // Column-major 4x4: m[col * 4 + row].
  const cx = m[0] * x + m[4] * y + m[8] * z + m[12];
  const cy = m[1] * x + m[5] * y + m[9] * z + m[13];
  const cz = m[2] * x + m[6] * y + m[10] * z + m[14];
  const cw = m[3] * x + m[7] * y + m[11] * z + m[15];
  return { x: cx, y: cy, z: cz, w: cw };
}

/**
 * Project an AABB into the supplied clip-space matrix.
 *
 * @param {{min:[number,number,number], max:[number,number,number]}} aabb
 * @param {ArrayLike<number>} viewProj column-major 4x4 matrix
 * @param {number} width target width in pixels
 * @param {number} height target height in pixels
 * @param {{reversedZ?: boolean, depthRange?: '0..1'|'-1..1'}} [options]
 *        depthRange describes the clip-space NDC depth convention produced by
 *        `viewProj`: '0..1' (WebGPU style, the default) or '-1..1' (OpenGL
 *        style, where a value of 0.5 is the middle of the depth buffer).
 * @returns {{valid:boolean, reason?:string, ...}}
 */
function projectAabb(aabb, viewProj, width, height, options = {}) {
  assertPositiveInt(width, 'width');
  assertPositiveInt(height, 'height');

  const reversedZ = options.reversedZ !== false;
  const depthRange = options.depthRange || '0..1';
  if (depthRange !== '0..1' && depthRange !== '-1..1') {
    throw new RangeError(`depthRange must be '0..1' or '-1..1', got ${depthRange}`);
  }
  const min = aabb && aabb.min;
  const max = aabb && aabb.max;
  if (!min || !max || min.length < 3 || max.length < 3) {
    throw new TypeError('aabb must be {min:[x,y,z], max:[x,y,z]}');
  }

  const corners = [];
  for (let i = 0; i < 8; i++) {
    corners.push([
      (i & 1) ? max[0] : min[0],
      (i & 2) ? max[1] : min[1],
      (i & 4) ? max[2] : min[2],
    ]);
  }

  let minX = Infinity;
  let maxX = -Infinity;
  let minY = Infinity;
  let maxY = -Infinity;
  let minDepth = Infinity;
  let maxDepth = -Infinity;

  for (const [x, y, z] of corners) {
    const clip = transformPoint(viewProj, x, y, z);
    if (!Number.isFinite(clip.w) || clip.w <= EPS) {
      return { valid: false, reason: 'behind-camera-or-on-near-plane' };
    }
    const invW = 1 / clip.w;
    const ndcX = clip.x * invW;
    const ndcY = clip.y * invW;
    const ndcZ = clip.z * invW;
    if (!Number.isFinite(ndcX) || !Number.isFinite(ndcY) || !Number.isFinite(ndcZ)) {
      return { valid: false, reason: 'non-finite-projection' };
    }

    // Convert clip NDC depth to the renderer's raw depth buffer value.  The
    // rest of this module, including the pyramid reduction, always compares
    // raw depth values.
    const rawDepth = depthRange === '-1..1' ? ndcZ * 0.5 + 0.5 : ndcZ;

    const screenX = (ndcX * 0.5 + 0.5) * width;
    const screenY = (1 - (ndcY * 0.5 + 0.5)) * height;

    if (screenX < minX) minX = screenX;
    if (screenX > maxX) maxX = screenX;
    if (screenY < minY) minY = screenY;
    if (screenY > maxY) maxY = screenY;
    if (rawDepth < minDepth) minDepth = rawDepth;
    if (rawDepth > maxDepth) maxDepth = rawDepth;
  }

  return {
    valid: true,
    minX,
    maxX,
    minY,
    maxY,
    depthMin: minDepth,
    depthMax: maxDepth,
    nearDepth: reversedZ ? maxDepth : minDepth,
    farDepth: reversedZ ? minDepth : maxDepth,
  };
}

/**
 * Test whether an AABB is *provably* occluded by the previous-frame pyramid.
 *
 * Conservative rules:
 *  - AABB behind the previous camera / on the near plane -> not occluded.
 *  - AABB projection leaves the previous viewport -> not occluded.  Previous
 *    frame depth is undefined there; clamping to the border would reuse stale
 *    texels and cull visible geometry after camera motion.
 *  - Zero-area screen projection -> not occluded.
 *  - Depth comparison is strict (object must be behind the farthest occluder).
 *
 * @param {object} aabb
 * @param {object} pyramid result of buildDepthPyramid
 * @param {ArrayLike<number>} viewProj previous frame's column-major view-projection matrix
 * @param {{reversedZ?: boolean, depthRange?: '0..1'|'-1..1'}} [options]
 * @returns {{occluded:boolean, reason:string, ...}}
 */
function isAabbOccluded(aabb, pyramid, viewProj, options = {}) {
  const reversedZ = options.reversedZ !== undefined ? options.reversedZ : pyramid.reversedZ;
  const projected = projectAabb(aabb, viewProj, pyramid.width, pyramid.height, {
    reversedZ,
    depthRange: options.depthRange,
  });

  if (!projected.valid) {
    return { occluded: false, reason: projected.reason, projected };
  }

  const { minX, maxX, minY, maxY, nearDepth, farDepth } = projected;

  if (maxX <= minX || maxY <= minY) {
    return { occluded: false, reason: 'zero-screen-area', projected };
  }

  // No stale border texels: if the footprint touches outside the previous
  // viewport, this frame's camera sees a region that previous depth cannot
  // prove is occluded.
  if (minX < 0 || maxX > pyramid.width || minY < 0 || maxY > pyramid.height) {
    return { occluded: false, reason: 'previous-screen-border', projected };
  }

  let x0 = Math.floor(minX);
  let y0 = Math.floor(minY);
  let x1 = Math.floor(maxX);
  let y1 = Math.floor(maxY);
  x0 = Math.max(0, Math.min(pyramid.width - 1, x0));
  y0 = Math.max(0, Math.min(pyramid.height - 1, y0));
  x1 = Math.max(0, Math.min(pyramid.width - 1, x1));
  y1 = Math.max(0, Math.min(pyramid.height - 1, y1));

  if (x0 > x1 || y0 > y1) {
    return { occluded: false, reason: 'empty-pixel-footprint', projected };
  }

  const bound = queryDepthPyramid(pyramid, x0, y0, x1, y1);
  const pixelRect = { x0, y0, x1, y1 };

  // bound is the farthest depth in the query rectangle:
  //   reversed-Z: min raw depth
  //   classic:    max raw depth
  const occluded = reversedZ
    ? nearDepth < bound - EPS
    : farDepth > bound + EPS;

  return {
    occluded,
    reason: occluded ? 'behind-farthest-occluder' : 'not-provable-visible',
    bound,
    pixelRect,
    projected,
  };
}

module.exports = {
  buildDepthPyramid,
  queryDepthPyramid,
  projectAabb,
  isAabbOccluded,
  defaultFarDepth,
};

The fuzz test is the actual proof of the conservativeness invariant:

if (result.occluded) {
  for every pixel in result.pixelRect:
    reversed-Z:  projected.nearDepth < depth[pixel]
    classic:     projected.farDepth  > depth[pixel]
}

If this ever failed, the culler would have culled visible geometry. It passes for both depth conventions and all tested NPOT sizes.
