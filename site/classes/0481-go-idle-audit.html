// Action decides what a tick must do, in priority order.
// Cooldown always wins: it prevents thrashed retries of known blockers.
type Audit struct {
    CooldownRemaining time.Duration
    BuildOK, VetOK, GofmtClean, GuardOK, CIGreen bool
    LintIssues int
    DockerBlocked bool // human-blocked, non-actionable
}

func (a Audit) Action() string {
    switch {
    case a.CooldownRemaining > 0:
        return "noop:cooldown" // 7200s confirmed via scheduler API; do NOT PUT
    case !a.BuildOK || !a.VetOK || !a.GofmtClean || a.LintIssues > 0:
        return "fix:code"
    case !a.GuardOK:
        return "fix:gitreins"
    case !a.CIGreen:
        return "fix:ci"
    case a.DockerBlocked:
        return "noop:blocked-human(DOCKER-GHCR-001)" // escalate, never retry
    default:
        return "noop:idle" // tick 49: this branch; emit no commit
    }
}
