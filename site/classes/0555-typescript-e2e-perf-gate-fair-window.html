Answer

The fair-window problem: the E2E perf smoke (frame-count / p95) is only trustworthy when the host is quiet. Under contention (git pack-objects storm, lingering workerd, load1 ≥ 8) the smoke starves and flakes — not a product regression. Fix: a fair-window gate that (1) refuses to run the full battery unless the host is quiet, (2) on smoke failure disambiguates environment-starvation from real regression via an isolate-probe, and (3) only then re-runs once.

1. Fair-window pre-check (TypeScript)

// fair-window.ts — pre-flight gate for the E2E perf battery
export interface FairWindow {
  ok: boolean;
  load1: number;
  workerd: number;
  packObjects: number;
  reason?: string;
}

const LOAD1_MAX = 8;
export const FRAME_GATE = 90;     // min frames in perf smoke
export const P95_GATE_MS = 25;    // hard p95 perf gate
export const MAX_RUNS = 2;        // full-battery attempts (fair + one re-run)

export async function checkFairWindow(sh: Shell): Promise<FairWindow> {
  const load1 = Number((await sh("cat /proc/loadavg")).split(/\s+/)[0]);
  const workerd = Number(await sh("pgrep -x workerd | wc -l"));
  const packObjects = Number(await sh("pgrep -af 'git pack-objects' | wc -l"));

  const reasons: string[] = [];
  if (workerd > 0) reasons.push(`workerd procs running (${workerd})`);
  if (packObjects > 0) reasons.push(`git pack-objects storm (${packObjects})`);
  if (load1 >= LOAD1_MAX) reasons.push(`load1 ${load1} >= ${LOAD1_MAX}`);

  return { ok: reasons.length === 0, load1, workerd, packObjects, reason: reasons.join("; ") };
}

2. Orchestrator with isolate-probe on smoke starvation

// run-battery.ts — single fair attempt; starved smoke → isolate-probe → re-run
export async function runBattery(): Promise<BatteryResult> {
  const win = await checkFairWindow(sh);
  if (!win.ok) {
    log(`✗ fair-window blocked (${win.reason}); isolate-probe & re-run only`);
    return { status: "skipped-unfair", attempts: 0 };
  }

  for (let attempt = 1; attempt <= MAX_RUNS; attempt++) {
    const res = await fullBattery();          // typecheck, validate:configs, unit, E2E
    const smoke = res.e2e.perfSmoke;          // { frames, p95Ms }

    if (smoke.frames >= FRAME_GATE && smoke.p95Ms < P95_GATE_MS) {
      await refreshScreenshots(8);            // evidence refresh
      await boardAppend(res.summary);
      await parityProbe();                    // parity check vs last window
      return { status: "passed", attempts: attempt, ...res };
    }

    // Starvation signal: low frame-count on an otherwise healthy suite
    if (smoke.frames < FRAME_GATE) {
      const iso = await isolateProbe();       // one smoke in quiet isolation
      if (iso.ok) {                           // env flake, not a regression
        log(`attempt ${attempt}: env-starvation (frames ${smoke.frames}); re-running`);
        await settle(30_000);                 // wait for pack storm / load to drain
        continue;                             // exactly one re-run
      }
    }
    return { status: "failed", attempts: attempt, ...res }; // real regression → fail gate
  }
  return { status: "failed", attempts: MAX_RUNS };
}

async function isolateProbe(): Promise<PerfProbe> {
  // Same perf smoke, dedicated worker, host otherwise idle; measures true ceiling.
  return runPerfSmoke({ isolated: true });
}
