1. config.go — default changed in BOTH places (struct tag + Load() literal) from 8080 → 8081:

// DefaultPort is the host-first default. The krakend gateway owns :8080 on
// the host, so host-mode dev uses 8081.
const DefaultPort = 8081

type Config struct {
    Port int `env:"PORT" default:"8081"` // host-first; krakend owns :8080
}

func Load() (Config, error) {
    cfg := Config{
        Port: 8081, // Load() literal — mirrors tag default:"8081"
    }
    if raw := os.Getenv("PORT"); raw != "" {
        v, err := strconv.Atoi(raw)
        if err != nil {
            return cfg, fmt.Errorf("PORT: invalid value %q: %w", raw, err)
        }
        cfg.Port = v
    }
    return cfg, nil
}

2. .env.example — host-first value:
