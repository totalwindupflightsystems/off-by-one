Answer

Root cause: hilo graph warm calls inventory::append_edges_deduped, which is append-only (dedup, never removal). After the warpfs-* → hilo-* crate rename, the old crate names were never re-emitted by the scan, so their edges stayed in the cache forever → 161 stale edges and 55 phantom files in hilo graph untested.

Fix: add a hilo graph clean subcommand that deletes the entire graph cache; the next warm rebuilds edges.jsonl, graph.db, and .last_warm from a fresh scan.

src/main.rs — new clap enum variant (the whole reset surface is one variant + dispatch):

#[derive(Subcommand)]
enum GraphCmd {
    /// Scan the crate inventory and append edges to the graph cache (deduped)
    Warm { #[arg(long)] base_dir: Option<PathBuf> },
    /// List edges whose target file is missing (phantom files)
    Untested { #[arg(long)] base_dir: Option<PathBuf> },
    /// Remove the graph cache (edges.jsonl, graph.db, .last_warm); next warm rebuilds from scratch
    Clean { #[arg(long)] base_dir: Option<PathBuf> },
}
// dispatch:
GraphCmd::Clean { base_dir } => commands::graph::run_clean(&base_dir.unwrap_or_default()),

src/commands/graph.rs — removal helper takes base_dir as a parameter (no chdir), so it is unit-testable with tempdirs and safe under parallel tests:

pub const GRAPH_SUBDIR: &str = ".vfs/graph";
const CACHE_ARTIFACTS: [&str; 3] = ["edges.jsonl", "graph.db", ".last_warm"];

/// Remove the graph cache under `base_dir`. Missing files are not an error,
/// so this is idempotent and safe on a fresh checkout.
pub fn clean_graph_dir(base_dir: &Path) -> io::Result<()> {
    let dir = base_dir.join(GRAPH_SUBDIR);
    for name in CACHE_ARTIFACTS {
        match fs::remove_file(dir.join(name)) {
            Ok(()) => {}
            Err(e) if e.kind() == io::ErrorKind::NotFound => {}
            Err(e) => return Err(e),
        }
    }
    Ok(())
}

pub fn run_clean(base_dir: &Path) -> io::Result<()> {
    clean_graph_dir(base_dir)?;
    Ok(())
}

Unit test (tempdir, no chdir — parallel-test-safe):

#[test]
fn clean_then_warm_rebuilds_all_artifacts() {
    let base = tempdir();
    seed_cache(&base);                      // writes all 3 artifacts
    clean_graph_dir(&base).unwrap();
    assert!(!base.join(GRAPH_SUBDIR).join("edges.jsonl").exists()); // + graph.db, .last_warm
    run_warm(&base, &inventory).unwrap();   // append_edges_deduped + rebuild db/.last_warm
    let edges = fs::read_to_string(base.join(GRAPH_SUBDIR).join("edges.jsonl")).unwrap();
    assert!(!edges.contains("old"), "stale cache content must be gone");
    assert!(edges.contains("hilo-"));
}
