DF-CONSENSUS-49 — Session tool-call visibility endpoint

Root cause

The executor and staging subsystems persist tool activity in two internal tables:

  • tool_requests — executor-owned rows (id, session_id, tool_name, arguments, status, result, error, tool_call_ref, created_at, updated_at).
  • staging_buffer — pre-dispatch rows (id, session_id, tool_call_ref, tool_name, arguments, status, created_at).

Neither table is reachable through the HTTP API, so a client cannot reconcile what it asked for against what the executor queued/ran, nor see a tool call that is still staged. The change that exposes them was only partially landed, causing three coupled failures:

  1. No handler / route — nothing serves a session-scoped, merged view.
  2. No authz — a naive implementation would read these tables by session, so it must be gated by checkSessionAccess (per-session authz), exactly like the handlers in internal/api/tools.go. Skipping it leaks another session's tool activity (IDOR).
  3. Spec drift — specs/openapi and the generated bundled.yaml are not updated, and the path-count test still asserts the old route count, so go test ./... (or the spec check) fails. All three must land in the same commit.

The correct fix is a single merged endpoint that unions executor rows and staging rows, stamps each with a status field and a source, de-duplicates staged rows already promoted into tool_requests, and is authorized per session.

Fix

1. Types + store queries

internal/store/tool_calls.go (new)

package store

import (
    "context"
    "database/sql"
    "encoding/json"
    "time"
)

// ToolCallStatus is the normalized status surfaced to clients.
type ToolCallStatus string

const (
    ToolCallStatusStaged    ToolCallStatus = "staged"
    ToolCallStatusQueued    ToolCallStatus = "queued"
    ToolCallStatusRunning   ToolCallStatus = "running"
    ToolCallStatusCompleted ToolCallStatus = "completed"
    ToolCallStatusFailed    ToolCallStatus = "failed"
)

// ToolRequest mirrors a row in the internal executor table.
type ToolRequest struct {
    ID          string
    SessionID   string
    ToolName    string
    Arguments   json.RawMessage
    Status      string
    Result      json.RawMessage
    Error       string
    ToolCallRef string
    CreatedAt   time.Time
    UpdatedAt   time.Time
}

// StagedToolCall mirrors a row in the staging buffer.
type StagedToolCall struct {
    ID          string
    SessionID   string
    ToolCallRef string
    ToolName    string
    Arguments   json.RawMessage
    Status      string
    CreatedAt   time.Time
}

func (s *Store) ListToolRequestsBySession(ctx context.Context, sessionID string) ([]ToolRequest, error) {
    const q = `
        SELECT id, session_id, tool_name, arguments, status,
               result, error, tool_call_ref, created_at, updated_at
        FROM tool_requests
        WHERE session_id = $1
        ORDER BY created_at ASC`
    rows, err := s.db.QueryContext(ctx, q, sessionID)
    if err != nil {
        return nil, err
    }
    defer rows.Close()

    var out []ToolRequest
    for rows.Next() {
        var tr ToolRequest
        if err := rows.Scan(
            &tr.ID, &tr.SessionID, &tr.ToolName, &tr.Arguments, &tr.Status,
            &tr.Result, &tr.Error, &tr.ToolCallRef, &tr.CreatedAt, &tr.UpdatedAt,
        ); err != nil {
            return nil, err
        }
        out = append(out, tr)
    }
    return out, rows.Err()
}

func (s *Store) ListStagedToolCallsBySession(ctx context.Context, sessionID string) ([]StagedToolCall, error) {
    const q = `
        SELECT id, session_id, tool_call_ref, tool_name, arguments, status, created_at
        FROM staging_buffer
        WHERE session_id = $1
        ORDER BY created_at ASC`
    rows, err := s.db.QueryContext(ctx, q, sessionID)
    if err != nil {
        return nil, err
    }
    defer rows.Close()

    var out []StagedToolCall
    for rows.Next() {
        var sc StagedToolCall
        var ref sql.NullString
        if err := rows.Scan(&sc.ID, &sc.SessionID, &ref, &sc.ToolName,
            &sc.Arguments, &sc.Status, &sc.CreatedAt); err != nil {
            return nil, err
        }
        sc.ToolCallRef = ref.String
        out = append(out, sc)
    }
    return out, rows.Err()
}

2. Handler in internal/api/tools.go

Add the DTO and handler next to the existing tool handlers. Reuse the package's existing writeJSON / error helpers and the checkSessionAccess helper.

// ToolCallView is the merged, client-facing representation of a tool call.
type ToolCallView struct {
    ID          string          `json:"id"`
    SessionID   string          `json:"session_id"`
    ToolName    string          `json:"tool_name"`
    Status      string          `json:"status"`
    Source      string          `json:"source"` // "executor" | "staging"
    ToolCallRef *string         `json:"tool_call_ref,omitempty"`
    Arguments   json.RawMessage `json:"arguments,omitempty"`
    Result      json.RawMessage `json:"result,omitempty"`
    Error       string          `json:"error,omitempty"`
    CreatedAt   time.Time       `json:"created_at"`
    UpdatedAt   time.Time       `json:"updated_at"`
}

// handleListSessionToolCalls serves GET /api/v1/sessions/{sessionID}/tool-calls.
// It returns a merged, session-scoped view of executor rows (tool_requests) and
// staged rows (staging_buffer). Access is enforced per session.
func (s *Server) handleListSessionToolCalls(w http.ResponseWriter, r *http.Request) {
    sessionID := chi.URLParam(r, "sessionID")
    if sessionID == "" {
        writeError(w, http.StatusBadRequest, "missing session id", nil)
        return
    }

    // Per-session authz — same pattern used by every other handler in this file.
    if _, err := s.checkSessionAccess(r.Context(), sessionID); err != nil {
        writeError(w, http.StatusForbidden, "session access denied", err)
        return
    }

    ctx := r.Context()
    requests, err := s.store.ListToolRequestsBySession(ctx, sessionID)
    if err != nil {
        writeError(w, http.StatusInternalServerError, "list tool requests", err)
        return
    }
    staged, err := s.store.ListStagedToolCallsBySession(ctx, sessionID)
    if err != nil {
        writeError(w, http.StatusInternalServerError, "list staged tool calls", err)
        return
    }

    // Index executor rows by ref so a staged row that was already promoted is
    // not returned twice. Executor status wins; staging only adds rows whose
    // ref has not yet reached the executor.
    promoted := make(map[string]struct{}, len(requests))
    for _, tr := range requests {
        if tr.ToolCallRef != "" {
            promoted[tr.ToolCallRef] = struct{}{}
        }
    }

    views := make([]ToolCallView, 0, len(requests)+len(staged))
    for _, tr := range requests {
        v := ToolCallView{
            ID:        tr.ID,
            SessionID: tr.SessionID,
            ToolName:  tr.ToolName,
            Status:    normalizeToolStatus(tr.Status),
            Source:    "executor",
            Arguments: tr.Arguments,
            Result:    tr.Result,
            Error:     tr.Error,
            CreatedAt: tr.CreatedAt,
            UpdatedAt: tr.UpdatedAt,
        }
        if tr.ToolCallRef != "" {
            ref := tr.ToolCallRef
            v.ToolCallRef = &ref
        }
        views = append(views, v)
    }
    for _, sc := range staged {
        if sc.ToolCallRef != "" {
            if _, ok := promoted[sc.ToolCallRef]; ok {
                continue // already represented by an executor row
            }
        }
        v := ToolCallView{
            ID:        sc.ID,
            SessionID: sc.SessionID,
            ToolName:  sc.ToolName,
            Status:    normalizeToolStatus(sc.Status),
            Source:    "staging",
            Arguments: sc.Arguments,
            CreatedAt: sc.CreatedAt,
            UpdatedAt: sc.CreatedAt,
        }
        if sc.ToolCallRef != "" {
            ref := sc.ToolCallRef
            v.ToolCallRef = &ref
        }
        views = append(views, v)
    }

    sort.SliceStable(views, func(i, j int) bool {
        return views[i].CreatedAt.Before(views[j].CreatedAt)
    })

    writeJSON(w, http.StatusOK, map[string]any{"tool_calls": views})
}

// normalizeToolStatus maps internal/executor state strings to the public set.
func normalizeToolStatus(s string) string {
    switch strings.ToLower(strings.TrimSpace(s)) {
    case "staged":
        return string(ToolCallStatusStaged)
    case "pending", "queued":
        return string(ToolCallStatusQueued)
    case "running", "in_progress":
        return string(ToolCallStatusRunning)
    case "completed", "succeeded", "success", "done":
        return string(ToolCallStatusCompleted)
    case "failed", "error":
        return string(ToolCallStatusFailed)
    default:
        if s == "" {
            return string(ToolCallStatusStaged)
        }
        return strings.ToLower(s)
    }
}

4. OpenAPI spec (specs/openapi)

Append the path (bump nothing else structurally):

/api/v1/sessions/{sessionID}/tool-calls:
  get:
    summary: List merged tool calls for a session
    description: >
      Returns executor tool requests (tool_requests) merged with staged tool
      calls (staging_buffer). Staged rows already promoted to the executor are
      omitted. Access is restricted to the caller's session.
    operationId: listSessionToolCalls
    tags: [tools]
    parameters:

        in: path

    responses:

        description: Merged list of tool calls
        content:
          application/json:

              type: object

                tool_calls:
                  type: array

      "403":
        description: Caller cannot access this session

        description: Session not found
