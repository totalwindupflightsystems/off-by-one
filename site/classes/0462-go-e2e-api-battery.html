No server fix required — the live fixture on :18999 is green (battery 10/10, zero 501 stubs, server continuous). The actionable deliverable is a reproducible Go battery harness that encodes the fixture contract and can run at the first tick of every due window, so the "all PASS" state is provable rather than assumed. It fixes two real gaps found while validating:

  1. Flaky 501 false alarm — the first live run caught a transient 501 marker in /metrics that never reappeared (288-series endpoint regenerates per request). A naive scan would fail the battery spuriously. Fix: confirm-retry after 300 ms; a persistent marker fails and dumps the offending lines.
  2. Honest result reporting — the summary previously claimed "zero 501 stubs" even when the battery failed. Now it prints PASSED/FAILED with correct exit codes.
// battery.go (core probes, condensed) — go-e2e-api-battery, 10 probes
var base = defaultBase // overridable: GO_E2E_BASE=http://127.0.0.1:18999

// P08: zero 501 stubs — transient-tolerant, diagnostic on persistence
add("P08 zero 501 stubs on live routes", func() error {
    for _, p := range []string{"/health", "/status", "/metrics"} {
        find := func() (string, []string) {
            _, body, err := get(p)
            if err != nil { return "", nil }
            low := strings.ToLower(string(body))
            for _, m := range []string{"501", "not implemented", "stub"} {
                if strings.Contains(low, m) {
                    var lines []string
                    for _, l := range strings.Split(string(body), "\n") {
                        if strings.Contains(strings.ToLower(l), m) { lines = append(lines, l) }
                    }
                    return m, lines
                }
            }
            return "", nil
        }
        m, lines := find()
        if m != "" { time.Sleep(300 * time.Millisecond); m, lines = find() } // confirm vs transient
        if m != "" { return fmt.Errorf("%s persistently contains stub marker %q\n  %s", p, m, strings.Join(lines, "\n  ")) }
    }
    return nil
})
