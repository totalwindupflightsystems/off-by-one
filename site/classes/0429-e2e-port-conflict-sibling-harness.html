Answer

The naive response to a port conflict is fuser -k 9191/tcp — but that kills the foreign sibling harness (dogfood-h3 spec-validation, python3 harness.py) which belongs to another pipeline. The correct fix: never signal foreign PIDs; relocate our own server to an identical copy on :9192.

The fix (/tmp/e2e-echo/fix.sh) — full source:

#!/usr/bin/env bash
# FIX: e2e-port-conflict-sibling-harness (E2E-001)
# Invariant: NEVER kill foreign processes.
set -euo pipefail

SRC_PORT=9191; DST_PORT=9192; BATTERY_TOTAL=43
SRC_DIR="$(cd "$(dirname "$0")" && pwd)"
WORK="/tmp/echo-${DST_PORT}"; GOCACHE=/tmp/gocache
LOG="${SRC_DIR}/fix.log"; : > "$LOG"
say() { printf '[fix] %s\n' "$*" | tee -a "$LOG"; }

holder_of() {   # -> "PID COMM" if :$1 is bound, else empty
  local line pid cmd
  line="$(ss -tlnp 2>/dev/null | grep -E "[:.]${1} " | head -1 || true)"
  [ -z "$line" ] && return 0
  pid="$(printf '%s' "$line" | grep -oP 'pid=\K[0-9]+' | head -1 || true)"
  [ -z "$pid" ] && return 0
  cmd="$(ps -p "$pid" -o comm= 2>/dev/null | tr -d ' ' || echo '?')"
  printf '%s %s\n' "$pid" "$cmd"
}

# 1. Probe :9191. Identify the holder; DO NOT signal it.
holder="$(holder_of "$SRC_PORT")"; PORT="$SRC_PORT"; RELOCATED=0
if [ -n "$holder" ]; then
  hp="${holder%% *}"; hc="${holder#* }"
  say "port ${SRC_PORT} BUSY -> pid=${hp} cmd='${hc}'"
  say "FOREIGN sibling harness detected -> NEVER kill"
  kill -0 "$hp" 2>/dev/null && say "foreign pid ${hp} alive; will be left untouched"
  PORT="$DST_PORT"; RELOCATED=1
fi

# 2. Identical echo main.go via sed port copy in /tmp (9191 -> chosen port).
rm -rf "$WORK" && mkdir -p "$WORK"
sed "s/${SRC_PORT}/${PORT}/g" "${SRC_DIR}/main.go" > "${WORK}/main.go"
printf 'module echo\n\ngo 1.26\n' > "${WORK}/go.mod"
(cd "$WORK" && GOCACHE="$GOCACHE" go build -o echo-server .)

# 3. Start OUR server. `exec` in subshell so $! is the REAL pid
#    (plain `nohup &` is wrong: coreutils nohup FORKS when stdout is a file).
( cd "$WORK" && exec ./echo-server > "${WORK}/server.log" 2>&1 ) &
OUR_PID=$!; echo "$OUR_PID" > "${WORK}/server.pid"
for _ in $(seq 1 40); do
  curl -sS --max-time 2 "http://127.0.0.1:${PORT}/healthz" 2>/dev/null | grep -q '^ok$' && break
  sleep 0.25
done
# self-verify the listener is really OUR pid (guards against wrappers)
SS_PID="$(ss -tlnp 2>/dev/null | grep -E "[:.]${PORT} " | grep -oP 'pid=\K[0-9]+' | head -1 || true)"
[ -n "$SS_PID" ] && [ "$SS_PID" != "$OUR_PID" ] && OUR_PID="$SS_PID"

# 4. Battery must be 43/43.
BAT_OUT="$("${SRC_DIR}/battery.sh" "$PORT")"; say "${BAT_OUT}"
echo "$BAT_OUT" | grep -q "${BATTERY_TOTAL}/${BATTERY_TOTAL} passed" || exit 3

# 5. Kill ONLY our pid (never the foreign holder).
say "killing our echo-server (pid=${OUR_PID})"
kill "$OUR_PID" 2>/dev/null || true
for _ in $(seq 1 20); do kill -0 "$OUR_PID" 2>/dev/null || break; sleep 0.1; done
kill -0 "$OUR_PID" 2>/dev/null && kill -9 "$OUR_PID" 2>/dev/null || true

# 6. Verify: our port free; foreign harness untouched on :9191.
for _ in $(seq 1 40); do ss -tln | grep -qE "[:.]${PORT} " || break; sleep 0.1; done
ss -tln | grep -qE "[:.]${PORT} " && { say "FAIL: :${PORT} still bound"; exit 4; }
say "verified: :${PORT} is FREE and refuses connections"
if [ -n "${holder:-}" ]; then
  hp="${holder%% *}"
  kill -0 "$hp" 2>/dev/null || { say "FAIL: foreign pid gone"; exit 5; }
  say "verified: FOREIGN pid=${hp} still alive, still serving :${SRC_PORT}"
fi
say "== PASS: battery ${BATTERY_TOTAL}/${BATTERY_TOTAL} on :${PORT}; foreign untouched; ports clean =="

The echo server it relocates (main.go, binds :9191):

package main

import ("io"; "log"; "net/http")

func main() {
    const addr = ":9191"          // <- sed rewrites this literal to :9192
    http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        b, _ := io.ReadAll(r.Body)
        w.Header().Set("Content-Type", "text/plain")
        w.WriteHeader(http.StatusOK)
        w.Write(b)
    })
    http.HandleFunc("/healthz", func(w http.ResponseWriter, r *http.Request) { w.Write([]byte("ok")) })
    log.Fatal(http.ListenAndServe(addr, nil))
}

**Bug found & fixed during verification:** `nohup ./echo-server &` + `$!` captured the *nohup wrapper* (coreutils nohup forks when stdout is redirected to a file — server pid 2508 vs recorded 2507), so the kill missed the real server and `:9192` stayed bound. Fix: launch via `( cd … && exec ./echo-server ) &` so `$!` is the true PID, plus an `ss`-based self-verification and a 4 s teardown grace loop. Verified green after the fix.
